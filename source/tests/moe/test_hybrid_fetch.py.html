1"""Hybrid decode's bandwidth-matched fetch split.
2
3Covers the profile reader of --moe-hybrid-max-fetch auto, which turns the `ft bench bw`
4row of each MoE layer kind into a fetch fraction.
5"""
6
7import json
8import os
9
10import pytest
11
12from freetoken.moe.bench_profile import PROFILE_VERSION, default_profile_path, fetch_fraction, layer_kind_key, load_layer_kinds
13
14
15def test_layer_kind_rows_give_fetch_fractions(tmp_path):
16 bf16 = layer_kind_key("bf16", {}, 2048, 768, 8, 12)
17 gguf = layer_kind_key("gguf", {"gate_up_type": 12, "down_type": 14}, 2048, 768, 8, 12)
18 assert gguf == "gguf[down_type=14,gate_up_type=12]:2048x768:top8:t12"
19 prof = {
20 "version": PROFILE_VERSION,
21 "gpu": {"name": "FAKE GPU"},
22 "layer_kinds": {
23 bf16: {"cpu_moe_gbs": 100.0, "pcie_gather_gbs": 40.0},
24 # overlapped (contended) pair wins over the standalone numbers when present
25 gguf: {"cpu_moe_gbs": 100.0, "pcie_gather_gbs": 40.0, "cpu_moe_overlap_gbs": 90.0, "pcie_gather_overlap_gbs": 30.0},
26 },
27 }
28 path = tmp_path / "benchbw.json"
29 path.write_text(json.dumps(prof))
30 rows = load_layer_kinds(path=str(path))
31 # standalone fallback: full-contention assumption -> pcie / cpu
32 assert fetch_fraction(rows[bf16]) == pytest.approx(0.4)
33 # overlapped pair preferred: pcie_ov / (pcie_ov + cpu_ov)
34 assert fetch_fraction(rows[gguf]) == pytest.approx(0.25)
35 # another thread count is another kind
36 assert layer_kind_key("bf16", {}, 2048, 768, 8, 6) not in rows
37 # a profile from different hardware is ignored
38 assert load_layer_kinds(gpu_name="OTHER", path=str(path)) is None
39 # so is one of another version
40 path.write_text(json.dumps({**prof, "version": PROFILE_VERSION - 1}))
41 assert load_layer_kinds(path=str(path)) is None

44def test_profile_lookup_prefers_the_gpu_uuid_file(tmp_path, monkeypatch):
45 monkeypatch.setenv("XDG_CACHE_HOME", str(tmp_path))
46 monkeypatch.delenv("FREETOKEN_BENCHBW_PATH", raising=False)
47 uuid = "GPU-2f3a9b1c-0000-1111-2222-333344445555"
48
49 def write(path, name, row):
50 os.makedirs(os.path.dirname(path), exist_ok=True)
51 with open(path, "w") as f:
52 json.dump({"version": PROFILE_VERSION, "gpu": {"name": name}, "layer_kinds": {"k": row}}, f)
53
54 # legacy single file only: used when the name matches, ignored otherwise
55 write(default_profile_path(), "FAKE GPU", "legacy")
56 assert load_layer_kinds(gpu_name="FAKE GPU", gpu_uuid=uuid) == {"k": "legacy"}
57 assert load_layer_kinds(gpu_name="OTHER", gpu_uuid=uuid) is None
58 # this card's own file wins over the legacy one
59 write(default_profile_path(uuid), "FAKE GPU", "card")
60 assert load_layer_kinds(gpu_name="FAKE GPU", gpu_uuid=uuid) == {"k": "card"}
