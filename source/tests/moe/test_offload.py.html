535def test_nvfp4_materialize_keeps_bookkeeping_consistent_across_requests():
536 """Regression: a full-layer prefill loads the layer's experts into slots [0, E).
537 If that overwrite does not invalidate the previous owners' mappings, a later
538 decode "hits" a stale slot_for_id entry and silently reads another expert's
539 weights. materialize_layer must keep bookkeeping == slot contents."""
540 import pytest
541 from freetoken.moe.offload_cache import OffloadMoeCache
542
543 if not torch.cuda.is_available():
544 pytest.skip("CUDA is required for the GPU offload cache kernel")
545
546 L, E, S = 2, 8, 8
547 OUT, IN = 64, 512 # keep rows >= 128B so the fast_index_copy JIT has a kernel
548 dev = torch.device("cuda")
549
550 def bank(out, inner, dtype):
551 # one independently allocated [E, out, inner] tensor per layer (the per-layer host
552 # bank contract); row idx within layer l keeps the old flat fingerprint l*E+idx.
553 layers = []
554 for l in range(L):
555 t = torch.zeros(E, out, inner, dtype=dtype)
556 for e in range(E):
557 t[e].view(torch.uint8).fill_(l * E + e)
558 layers.append(t)
559 return layers
560
561 def pinned(layers):
562 return [t.pin_memory() for t in layers]
563
564 cache = OffloadMoeCache(
565 num_layers=L, num_experts=E, cache_size=S, device=dev, quant_format="nvfp4"
566 )
567 cache.set_bank_sources(
568 {
569 "gate_up_packed": pinned(bank(OUT, IN // 2, torch.uint8)),
570 "gate_up_scale": pinned(bank(OUT, IN // 16, torch.float8_e4m3fn)),
571 "gate_up_global": pinned([t.squeeze(-1).contiguous() for t in bank(OUT, 1, torch.float16)]),
572 "down_packed": pinned(bank(OUT, IN // 2, torch.uint8)),
573 "down_scale": pinned(bank(OUT, IN // 16, torch.float8_e4m3fn)),
574 "down_global": pinned([t.squeeze(-1).contiguous() for t in bank(OUT, 1, torch.float16)]),
575 }
576 )
577 cache.reset()
578
579 def fingerprint(slot): # which source row's bytes live in this slot?
580 return int(cache.bank_caches["gate_up_packed"][slot].view(torch.uint8).flatten()[0].item())
581
582 # Request A, decode: layer 0 loads experts 3 and 5 somewhere in the cache.
583 ids = torch.tensor([3, 5], dtype=torch.int32, device=dev)
584 cache.ensure_experts(0, ids)
585 cache.copy_missing()
586 torch.cuda.synchronize()
587 assert [fingerprint(s) for s in ids.tolist()] == [3, 5]
588
589 # Request B, prefill: layer 1 is materialized into slots [0, E), overwriting
590 # every slot (S == E), including the ones decode A used.
