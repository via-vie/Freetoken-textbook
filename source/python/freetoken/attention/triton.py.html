68class TritonAttentionBackend(BaseAttnBackend):
69 def __init__(self, config: ModelConfig):
70 self.config = config
71 self.kvcache = get_global_ctx().kv_cache
72 self.device = self.kvcache.device
73 self.capture: TritonCaptureData | None = None
74 self.capture_bs: List[int] = []
75 self.max_graph_bs = 0
76 self.prefill_tile_min_q = 128
77 self.num_q_heads = int(getattr(config, "num_qo_heads", 1))
78 kv_groups = getattr(config, "kv_cache_group_specs", lambda: ())()
79 self.max_head_dim = max(
80 (group.head_dim for group in kv_groups),
81 default=int(getattr(config, "head_dim", 1)),
82 )
83 self.kv_heads = sorted({group.num_kv_heads for group in kv_groups}) or [int(getattr(config, "num_kv_heads", 1))]

85 def _split_cap(self, bs: int) -> int:
86 """The most decode KV splits any layer group takes at this batch size."""
87 return max(decode_kv_split_cap(bs, max(1, self.num_q_heads), heads, self.device.index) for heads in self.kv_heads)

89 def _ensure_decode_scratch(
90 self,
91 metadata: TritonMetadata,
92 bs: int,
93 num_q_heads: int,
94 head_dim: int,
95 ) -> None:
96 splits = self._split_cap(bs)
97 if (
98 metadata.attn_logits is not None
99 and metadata.attn_lse is not None
100 and metadata.attn_logits.shape[0] >= bs
101 and metadata.attn_logits.shape[1] >= num_q_heads
102 and metadata.attn_logits.shape[2] >= splits
103 and metadata.attn_logits.shape[3] >= head_dim
104 ):
105 return
106 scratch_head_dim = max(self.max_head_dim, head_dim)
107 metadata.attn_logits = torch.empty(
108 (bs, num_q_heads, splits, scratch_head_dim),
109 dtype=torch.float32,
110 device=self.device,
111 )
112 metadata.attn_lse = torch.empty(
113 (bs, num_q_heads, splits),
114 dtype=torch.float32,
115 device=self.device,
116 )

118 def forward(
119 self,
120 q: torch.Tensor,
121 k: torch.Tensor,
122 v: torch.Tensor,
123 layer_id: int,
124 batch: Batch,
125 attn_spec: AttentionSpec | None = None,
126 ) -> torch.Tensor:
127 from freetoken.kernel.triton.attention import (
128 decode_paged_attention,
129 extend_paged_attention,
130 paged_attention,
131 )
132
133 metadata = batch.attn_metadata
134 assert isinstance(metadata, TritonMetadata)
135 self.kvcache.store_kv(k, v, batch.out_loc, layer_id)
136
137 k_raw = self.kvcache.k_cache(layer_id)
138 v_raw = self.kvcache.v_cache(layer_id)
139 kv_heads, head_dim = k_raw.shape[-2], k_raw.shape[-1]
140 assert head_dim == q.shape[-1]
141 k_cache = k_raw.view(-1, kv_heads, head_dim)
142 v_cache = v_raw.view(-1, kv_heads, head_dim)
143
144 spec = attn_spec or AttentionSpec()
145 block_ends = batch.mm_block_ends if spec.bidirectional_mm_blocks else None
146 indices = metadata.indices
147 if spec.sliding_window is not None and metadata.swa_indices is not None:
148 indices = metadata.swa_indices
149 scale = spec.sm_scale if spec.sm_scale is not None else q.shape[-1] ** -0.5
150 if metadata.is_decode and q.dtype in (torch.float16, torch.bfloat16):
151 bs = metadata.indptr.numel() - 1
152 self._ensure_decode_scratch(metadata, bs, q.shape[1], q.shape[-1])
153 assert metadata.attn_logits is not None
154 assert metadata.attn_lse is not None
155 return decode_paged_attention(
156 q=q,
157 k_cache=k_cache,
158 v_cache=v_cache,
159 indptr=metadata.indptr,
160 indices=indices,
161 q_positions=metadata.q_positions,
162 attn_logits=metadata.attn_logits[:bs],
163 attn_lse=metadata.attn_lse[:bs],
164 sm_scale=scale,
165 sliding_window=spec.sliding_window,
166 sinks=spec.sinks,
167 )
168 if (
169 (not metadata.is_decode)
170 and q.dtype in (torch.float16, torch.bfloat16)
171 and (q.shape[-1] <= 256 or metadata.max_q_len >= self.prefill_tile_min_q)
172 ):
173 return extend_paged_attention(
174 q=q,
175 k_cache=k_cache,
176 v_cache=v_cache,
177 qo_indptr=metadata.cu_seqlens_q_gpu,
178 kv_indptr=metadata.indptr,
179 kv_indices=indices,
180 prefix_lens=metadata.prefix_lens,
181 max_q_len=metadata.max_q_len,
182 sm_scale=scale,
183 sliding_window=spec.sliding_window,
184 sinks=spec.sinks,
185 k_extend=k.view(q.shape[0], kv_heads, head_dim),
186 v_extend=v.view(q.shape[0], kv_heads, head_dim),
187 block_ends=block_ends,
188 )
189 if block_ends is not None:
190 raise NotImplementedError("bidirectional multimodal blocks need the extend kernel path")
191 return paged_attention(
192 q=q,
193 k_cache=k_cache,
194 v_cache=v_cache,
195 indptr=metadata.indptr,
196 indices=indices,
197 q_to_req=metadata.q_to_req,
198 q_positions=metadata.q_positions,
199 sm_scale=scale,
200 sliding_window=spec.sliding_window,
201 sinks=spec.sinks,
202 )

204 def prepare_metadata(self, batch: Batch) -> None:
205 reqs = batch.padded_reqs
206 device = self.device
207 ctx = get_global_ctx()
208 page_table = ctx.page_table
209 padded_size = len(reqs)
210 seqlens_q = [req.extend_len for req in reqs]
211 seqlens_k = [req.device_len for req in reqs]
212 cached_lens = [req.cached_len for req in reqs]
213 num_query_tokens = sum(seqlens_q)
214 is_decode = max(seqlens_q) == 1
215 CPU_KWARGS = {"device": "cpu", "dtype": torch.int32, "pin_memory": device.type == "cuda"}
216 prefix_lens = torch.tensor(cached_lens, **CPU_KWARGS).to(device, non_blocking=True)
217
218 indptr = torch.tensor([0] + seqlens_k, **CPU_KWARGS).cumsum_(0).to(device, non_blocking=True)
219 if is_decode:
220 cu_seqlens_q_gpu = torch.arange(0, padded_size + 1, device=device, dtype=torch.int32)
221 elif all(l == 0 for l in cached_lens):
222 cu_seqlens_q_gpu = indptr
223 else:
224 cu_seqlens_q_gpu = torch.tensor([0] + seqlens_q, **CPU_KWARGS).cumsum_(0).to(device, non_blocking=True)
225 indices = torch.cat([page_table[req.table_idx, : req.device_len] for req in reqs])
226 swa_indices = None
227 if getattr(self.kvcache, "swa_paged", False):
228 # Global-paged SWA (naive + radix): the swa-layer gather reads swa-pool slots = full->swa
229 # map of the full page-table slots (live for in-window tokens; out-of-window -> sentinel 0,
230 # masked by the sliding window). Recomputed each step (incl. graph replay via
231 # _point_to_capture) since the page table grows during decode.
232 swa_indices = self.kvcache.translate_loc_from_full_to_swa(indices)
233
234 q_to_req = torch.empty(num_query_tokens, dtype=torch.int32, device=device)
235 offset = 0
236 for req_idx, q_len in enumerate(seqlens_q):
237 q_to_req[offset : offset + q_len].fill_(req_idx)
238 offset += q_len
239
240 q_positions = getattr(batch, "positions", None)
241 if q_positions is None:
242 q_positions = torch.zeros(num_query_tokens, dtype=torch.int64, device=device)
243
244 batch.attn_metadata = TritonMetadata(
245 cu_seqlens_q_gpu=cu_seqlens_q_gpu,
246 indptr=indptr,
247 indices=indices,
248 q_to_req=q_to_req,
249 q_positions=q_positions,
250 is_decode=is_decode,
251 prefix_lens=prefix_lens,
252 max_q_len=max(seqlens_q),
253 swa_indices=swa_indices,
254 )

256 def init_capture_graph(self, max_seq_len: int, bs_list: List[int]) -> None:
257 assert self.capture is None, "Capture already initialized."
258 max_bs = max(bs_list)
259 self.capture = TritonCaptureData.create(
260 max_bs,
261 max_seq_len,
262 self.device,
263 num_q_heads=max(1, self.num_q_heads),
264 max_head_dim=max(1, self.max_head_dim),
265 scratch_rows=max(bs * self._split_cap(bs) for bs in bs_list),
266 )
267 if self._swa_capture_enabled():
268 self.capture.swa_page_table = torch.zeros(
269 (max_bs, max_seq_len),
270 dtype=torch.int32,
271 device=self.device,
272 )
273 self.capture_bs = sorted(bs_list)
274 self.max_graph_bs = max_bs

276 def _swa_capture_enabled(self) -> bool:
277 # A persistent swa-index capture buffer is needed for the global-paged SWA mode
278 # (full->swa mapping translate, recomputed each replay).
279 return getattr(self.kvcache, "swa_paged", False)

281 def _capture_swa_indices(self) -> torch.Tensor | None:
282 assert self.capture is not None
283 if not self._swa_capture_enabled():
284 return None
285 assert self.capture.swa_page_table is not None
286 return self.capture.swa_page_table.view(-1)

288 def _point_to_capture(self, metadata: TritonMetadata, bs: int) -> None:
289 assert self.capture is not None
290 indices = self.capture.page_table.view(-1)
291 self.capture.cu_seqlens_q[: bs + 1].copy_(metadata.cu_seqlens_q_gpu)
292 self.capture.cu_seqlens_k[: bs + 1].copy_(metadata.indptr)
293 total = metadata.indices.numel()
294 indices[:total].copy_(metadata.indices)
295 if metadata.swa_indices is not None:
296 swa_indices = self._capture_swa_indices()
297 assert swa_indices is not None
298 swa_indices[:total].copy_(metadata.swa_indices)
299 metadata.swa_indices = swa_indices
300 else:
301 metadata.swa_indices = None
302 q_tokens = metadata.q_positions.numel()
303 self.capture.positions[:q_tokens].copy_(metadata.q_positions)
304 metadata.cu_seqlens_q_gpu = self.capture.cu_seqlens_q[: bs + 1]
305 metadata.indptr = self.capture.cu_seqlens_k[: bs + 1]
306 metadata.indices = indices
307 metadata.q_to_req = self.capture.q_to_req[: metadata.q_to_req.numel()]
308 metadata.q_positions = self.capture.positions[:q_tokens]
309 metadata.attn_logits, metadata.attn_lse = self._capture_scratch(bs)
310
311 def _capture_scratch(self, bs: int) -> tuple[torch.Tensor, torch.Tensor]:
312 assert self.capture is not None
313 splits, heads, dim = self._split_cap(bs), max(1, self.num_q_heads), max(1, self.max_head_dim)
314 logits = self.capture.attn_logits[: bs * heads * splits * dim].view(bs, heads, splits, dim)
315 lse = self.capture.attn_lse[: bs * heads * splits].view(bs, heads, splits)
316 return logits, lse

318 def prepare_for_capture(self, batch: Batch) -> None:
319 bs = batch.size
320 assert bs in self.capture_bs and self.capture is not None
321 capture = self.capture
322 attn_logits, attn_lse = self._capture_scratch(bs)
323 batch.attn_metadata = TritonMetadata(
324 cu_seqlens_q_gpu=capture.cu_seqlens_q[: bs + 1],
325 indptr=capture.cu_seqlens_k[: bs + 1],
326 indices=capture.page_table.view(-1),
327 q_to_req=capture.q_to_req[:bs],
328 q_positions=capture.positions[:bs],
329 is_decode=True,
330 prefix_lens=capture.positions[:bs],
331 max_q_len=1,
332 attn_logits=attn_logits,
333 attn_lse=attn_lse,
334 swa_indices=self._capture_swa_indices(),
335 )

337 def prepare_for_replay(self, batch: Batch) -> None:
338 metadata, bs = batch.attn_metadata, batch.padded_size
339 assert isinstance(metadata, TritonMetadata)
340 assert self.capture is not None and bs in self.capture_bs
341 self._point_to_capture(metadata, bs)
