10def _is_chinese_char(cp: int):
11 """Checks whether CP is the codepoint of a CJK character."""
12 # This defines a "chinese character" as anything in the CJK Unicode block:
13 # https://en.wikipedia.org/wiki/CJK_Unified_Ideographs_(Unicode_block)
14 #
15 # Note that the CJK Unicode block is NOT all Japanese and Korean characters,
16 # despite its name. The modern Korean Hangul alphabet is a different block,
17 # as is Japanese Hiragana and Katakana. Those alphabets are used to write
18 # space-separated words, so they are not treated specially and handled
19 # like the all of the other languages.
20 if (
21 (cp >= 0x4E00 and cp <= 0x9FFF)
22 or (cp >= 0x3400 and cp <= 0x4DBF) #
23 or (cp >= 0x20000 and cp <= 0x2A6DF) #
24 or (cp >= 0x2A700 and cp <= 0x2B73F) #
25 or (cp >= 0x2B740 and cp <= 0x2B81F) #
26 or (cp >= 0x2B820 and cp <= 0x2CEAF) #
27 or (cp >= 0xF900 and cp <= 0xFAFF)
28 or (cp >= 0x2F800 and cp <= 0x2FA1F) #
29 ): #
30 return True
31
32 return False

35def find_printable_text(text: str):
36 """Returns the longest printable substring of text that contains only entire words."""
37 # Borrowed from https://github.com/huggingface/transformers/blob/061580c82c2db1de9139528243e105953793f7a2/src/transformers/generation/streamers.py#L99
38
39 # After the symbol for a new line, we flush the cache.
40 if text.endswith("\n"):
41 return text
42 # If the last token is a CJK character, we print the characters.
43 elif len(text) > 0 and _is_chinese_char(ord(text[-1])):
44 return text
45 # Otherwise if the penultimate token is a CJK character, we print the characters except for the last one.
46 elif len(text) > 1 and _is_chinese_char(ord(text[-2])):
47 return text[:-1]
48 # Otherwise, prints until the last space char (simple heuristic to avoid printing incomplete words,
49 # which may change with the subsequent token -- there are probably smarter ways to do this!)
50 else:
51 return text[: text.rfind(" ") + 1]

54def _stop_prefix_holdback(text: str, stop_strs: list[str]) -> int:
55 """Length of the longest trailing suffix of ``text`` that is a proper prefix of some
56 stop string. Those chars are withheld until a later token resolves whether the stop
57 completes, so a partial stop is never streamed and then needs retracting."""
58 hold = 0
59 for stop in stop_strs:
60 for i in range(min(len(stop) - 1, len(text)), 0, -1):
61 if text.endswith(stop[:i]):
62 hold = max(hold, i)
63 break
64 return hold

67@dataclass
68class DecodeStatus:
69 decoded_ids: List[int]
70 decoded_str: str
71 read_offset: int # length of read ids
72 surr_offset: int # length of surr ids
73 sent_offset: int # length of sent out string

76class DetokenizeManager:
77 def __init__(
78 self, tokenizer: PreTrainedTokenizerBase, eos_token_ids: FrozenSet[int] | None = None
79 ) -> None:
80 # uid -> DecodeStatus
81 self.decode_map: Dict[int, DecodeStatus] = {}
82 self.tokenizer = tokenizer
83 self.eos_token_ids = (
84 eos_token_ids if eos_token_ids is not None else frozenset({tokenizer.eos_token_id})
85 )
86
87 def discard(self, uid: int) -> None:
88 """Drop a uid's decode state without a finished DetokenizeMsg. An aborted or errored
89 request never sends one (its terminal reply is an ErrorReplyMsg), so without this its
90 accumulated ids and text stay in ``decode_map`` for the life of the worker."""
91 self.decode_map.pop(uid, None)
92
93 def detokenize(self, msgs: List[DetokenizeMsg]) -> List[str]:
94 read_ids: List[List[int]] = []
95 surr_ids: List[List[int]] = []
96 for msg in msgs:
97 if msg.uid not in self.decode_map:
98 self.decode_map[msg.uid] = DecodeStatus(
99 decoded_ids=[],
100 decoded_str="",
101 read_offset=0,
102 surr_offset=0,
103 sent_offset=0,
104 )
105 s = self.decode_map[msg.uid]
106 if not (msg.finished and msg.next_token in self.eos_token_ids):
107 s.decoded_ids.append(msg.next_token)
108 read_ids.append(s.decoded_ids[s.surr_offset :])
109 surr_ids.append(s.decoded_ids[s.surr_offset : s.read_offset])
110
111 read_texts = self.tokenizer.batch_decode(read_ids)
112 surr_texts = self.tokenizer.batch_decode(surr_ids)
113
114 incremental_strs: List[str] = []
115 for msg, read_str, surr_str in zip(msgs, read_texts, surr_texts, strict=True):
116 s = self.decode_map[msg.uid]
117 new_text = read_str[len(surr_str) :]
118 # Streaming chunk: update the decode status
119 if len(new_text) > 0 and not new_text.endswith("�"):
120 output_str = s.decoded_str + new_text
121 s.decoded_str = output_str
122 s.surr_offset = s.read_offset
123 s.read_offset = len(s.decoded_ids)
124 else:
125 new_text = find_printable_text(new_text)
126 output_str = s.decoded_str + new_text
127
128 prev_sent = s.sent_offset
129 if msg.finished:
130 # Generation is over: flush everything, trimming at the matched stop string.
131 if msg.matched_stop:
132 cut = output_str.find(msg.matched_stop)
133 emit_end = cut if cut >= 0 else len(output_str)
134 else:
135 emit_end = len(output_str)
136 elif msg.stop_strs:
137 # Hold back a trailing suffix that could still grow into a stop string.
138 emit_end = len(output_str) - _stop_prefix_holdback(output_str, msg.stop_strs)
139 else:
140 emit_end = len(output_str)
141 incremental_output = output_str[prev_sent:emit_end] if emit_end > prev_sent else ""
142 s.sent_offset = max(prev_sent, emit_end)
143 incremental_strs.append(incremental_output)
144 if msg.finished:
145 del self.decode_map[msg.uid]
146
147 return incremental_strs
