31def resolve_thinking_mode(chat_template_kwargs: dict[str, Any] | None, tools: Any | None) -> str:
32 """Resolve the thinking mode (``"thinking"`` or ``"chat"``) for a chat request.
33
34 The single source of truth for this decision: the encode side
35 (``_apply_dsv4_chat_encoder`` below) uses it to pick the prompt the model
36 sees, and the frontend parse side (``server/openai_api.py``) imports it to
37 decide whether the model's output begins inside a reasoning block. Keeping
38 one implementation prevents the two sides from disagreeing. Thinking is on
39 when tools are offered (dsv4 only emits well-formed tool calls in thinking
40 mode) or when the caller requests it via ``chat_template_kwargs``.
41 """
42 ctk = chat_template_kwargs or {}
43 mode = str(ctk.get("thinking_mode") or "chat")
44 if tools or ctk.get("enable_thinking") or ctk.get("thinking"):
45 mode = "thinking"
46 if mode not in ("chat", "thinking"):
47 mode = "chat"
48 return mode

54class TokenizeManager:
55 def __init__(self, tokenizer: PreTrainedTokenizerBase, mm_processor: MMProcessor | None = None) -> None:
56 self.tokenizer = tokenizer
57 self.mm_processor = mm_processor # None: the model takes no images
58 self._dsv4_encoder = _load_dsv4_encoder_if_needed(tokenizer)
59 self._effort_profile: EffortProfile | None = None
60 self._thinking_profile: ThinkingProfile | None = None
61 self._effort_lock = threading.Lock()
62 self._logged_effort_maps: set[tuple[Any, str | int | None]] = set()
63 self._inline_system_lock = threading.Lock()
64 self._inline_system_modes: dict[str, str] = {}

66 def tokenize(self, msgs: List[TokenizeMsg]) -> List[UserMsg]:
67 results: List[UserMsg] = []
68 # TODO: batch tokenization
69 for msg in msgs:
70 prompt = self.render_prompt(msg)
71 # A jinja chat template owns every special token (HF's apply_chat_template
72 # tokenizes with add_special_tokens=False for the same reason): tokenizers
73 # that auto-add bos (muse-glimmer's, llama's) would otherwise double it --
74 # the template already rendered one. Raw-string prompts and the dsv4
75 # encoder path keep the default.
76 templated = isinstance(msg.text, list) and self._dsv4_encoder is None
77 input_ids: torch.Tensor = ( # type: ignore
78 self.tokenizer.encode(
79 prompt, return_tensors="pt", add_special_tokens=not templated
80 )
81 )
82 input_ids = input_ids.view(-1).to(torch.int32)
83 if msg.images:
84 if self.mm_processor is None:
85 raise ValueError("image input is not supported for this model")
86 mm = self.mm_processor.apply(input_ids, msg.images)
87 results.append(
88 UserMsg(
89 uid=msg.uid,
90 input_ids=mm.input_ids,
91 sampling_params=msg.sampling_params,
92 mm_items=mm.mm_items,
93 mrope_positions=mm.mrope_positions,
94 mrope_delta=mm.mrope_delta,
95 )
96 )
97 else:
98 results.append(
99 UserMsg(uid=msg.uid, input_ids=input_ids, sampling_params=msg.sampling_params)
100 )
101 return results

103 def render_prompt(self, msg: TokenizeMsg) -> str:
104 """The template/encoder half of ``tokenize``, exposed so the frontend can
105 validate a request before committing an SSE stream. Sanitizes
106 ``reasoning_effort`` first: every render path (worker, frontend
107 validation, count_tokens) must quantize identically."""
108 if not isinstance(msg.text, list):
109 return msg.text
110 kwargs = self._sanitize_effort(msg.chat_template_kwargs or {})
111 messages = msg.text
112 if msg.inline_system_policy is not None:
113 if msg.inline_system_policy not in ("auto", "preserve", "fold"):
114 raise ValueError(f"invalid inline system policy: {msg.inline_system_policy}")
115 if has_inline_system(messages):
116 mode = msg.inline_system_policy
117 if mode == "auto":
118 mode = self.inline_system_mode(msg.tools, kwargs)
119 messages = normalize_inline_system(messages, mode)
120 return self._render(messages, msg.tools, kwargs, images=msg.images)
121
122 def inline_system_mode(self, tools: list[dict] | None, kwargs: dict[str, Any]) -> str:
123 key = json.dumps([tools, kwargs], sort_keys=True)
124 with self._inline_system_lock:
125 if key not in self._inline_system_modes:
126 mode, reason = probe_inline_system(
127 lambda messages: self._render(messages, deepcopy(tools), deepcopy(kwargs)), self.tokenizer
128 )
129 if len(self._inline_system_modes) >= 128:
130 self._inline_system_modes.pop(next(iter(self._inline_system_modes)))
131 self._inline_system_modes[key] = mode
132 logger.info("Anthropic inline system policy: %s (%s)", mode, reason)
133 return self._inline_system_modes[key]

135 def _render(
136 self,
137 messages: list[dict[str, Any]],
138 tools: list[dict[str, Any]] | None,
139 chat_template_kwargs: dict[str, Any],
140 *, images: list[bytes] | None = None,
141 ) -> str:
142 """Raw render, no effort sanitation — the probe needs unsupported values
143 to actually reach the template so rejection is observable."""
144 if self._dsv4_encoder is not None:
145 return _apply_dsv4_chat_encoder(
146 self._dsv4_encoder, messages, tools, chat_template_kwargs, images=images
147 )
148 # Broadcast the effort in every spelling the ecosystem's templates read
149 # (muse-glimmer grades ``reasoning_strength``; Jinja ignores undeclared
150 # variables) -- the same rule the thinking toggles use. An explicit
151 # caller-provided spelling wins over the broadcast.
152 if "reasoning_effort" in chat_template_kwargs:
153 chat_template_kwargs = dict(chat_template_kwargs)
154 chat_template_kwargs.setdefault(
155 "reasoning_strength", chat_template_kwargs["reasoning_effort"]
156 )
157 if tools is not None:
158 chat_template_kwargs = {**chat_template_kwargs, "tools": tools}
159 prompt = self.tokenizer.apply_chat_template(
160 messages,
161 tokenize=False,
162 add_generation_prompt=True,
163 **chat_template_kwargs,
164 )
165 assert isinstance(prompt, str)
166 return prompt

168 def effort_profile(self) -> EffortProfile:
169 """The checkpoint's effort vocabulary, probed on first use and cached
170 for the process lifetime."""
171 with self._effort_lock:
172 if self._effort_profile is None:
173 self._effort_profile = probe_effort_profile(self._probe_render)
174 logger.info(
175 "reasoning-effort profile: supported=%s default=%s",
176 sorted(self._effort_profile.supported) or "(none)",
177 self._effort_profile.default,
178 )
179 return self._effort_profile

181 def thinking_profile(self) -> ThinkingProfile:
182 """The checkpoint's thinking controls (toggle behavior + effort
183 vocabulary), probed on first use and cached for the process lifetime.
184 Feeds the /v1/cache/status gear derivation."""
185 efforts = self.effort_profile()
186 with self._effort_lock:
187 if self._thinking_profile is None:
188 self._thinking_profile = probe_thinking_profile(self._probe_render, efforts)
189 return self._thinking_profile

191 def _probe_render(
192 self, kwargs: dict[str, Any], tools: list[dict[str, Any]] | None
193 ) -> str:
194 return self._render(_EFFORT_PROBE_MESSAGES, tools, kwargs)

196 def _sanitize_effort(self, chat_template_kwargs: dict[str, Any]) -> dict[str, Any]:
197 if "reasoning_effort" not in chat_template_kwargs:
198 return chat_template_kwargs
199 raw = chat_template_kwargs.get("reasoning_effort")
200 mapped = quantize_effort(raw, self.effort_profile())
201 if mapped == raw:
202 return chat_template_kwargs
203 # raw is client-controlled and may be unhashable (a JSON list/dict).
204 key = (raw if isinstance(raw, str) else repr(raw), mapped)
205 if key not in self._logged_effort_maps:
206 self._logged_effort_maps.add(key)
207 logger.info(
208 "reasoning_effort %r is not supported by this checkpoint; using %s",
209 raw,
210 mapped if mapped is not None else "the template default",
211 )
212 sanitized = dict(chat_template_kwargs)
213 if mapped is None:
214 del sanitized["reasoning_effort"]
215 else:
216 sanitized["reasoning_effort"] = mapped
217 return sanitized

220def _load_dsv4_encoder_if_needed(tokenizer: PreTrainedTokenizerBase) -> ModuleType | None:
221 if getattr(tokenizer, "chat_template", None):
222 return None
223 model_path = getattr(tokenizer, "name_or_path", None) or getattr(tokenizer, "_name_or_path", "")
224 if not model_path:
225 return None
226 # DeepSeek-V4 ships encoding/encoding_dsv4.py, V4.1 encoding/encoding.py; same encode_messages contract
227 encoder_path = next(
228 (p for p in (os.path.join(str(model_path), "encoding", name) for name in ("encoding_dsv4.py", "encoding.py")) if os.path.isfile(p)),
229 None,
230 )
231 if encoder_path is None:
232 return None
233 spec = importlib.util.spec_from_file_location("encoding_dsv4", encoder_path)
234 if spec is None or spec.loader is None:
235 return None
236 module = importlib.util.module_from_spec(spec)
237 spec.loader.exec_module(module)
238 if not hasattr(module, "encode_messages"):
239 return None
240 return module

243def _apply_dsv4_chat_encoder(
244 encoder: ModuleType,
245 messages: list[dict],
246 tools: list[dict] | None,
247 chat_template_kwargs: dict,
248 *, images: list[bytes] | None = None,
249) -> str:
250 rendered_messages = [dict(message) for message in messages]
251 image_data = iter(images) if images is not None else None
252 for message in rendered_messages:
253 if isinstance(message.get("content"), list):
254 content = []
255 for part in message["content"]:
256 if isinstance(part, dict) and part.get("type") == "image":
257 # Prerender still has refs; the tokenizer worker receives fetched bytes.
258 # The checkpoint encoder validates sources even when rendering only text.
259 ref = part.get("freetoken_ref")
260 if image_data is not None:
261 try:
262 part = {"type": "image", "data": next(image_data)}
263 except StopIteration:
264 raise ValueError("image blocks outnumber fetched images") from None
265 elif ref is not None:
266 key = "data" if ref["kind"] == "b64" else "url"
267 part = {"type": "image", key: ref["data"]}
268 content.append(part)
269 message["content"] = content
270 if message.get("tool_calls"):
271 message["tool_calls"] = _dsv4_tool_calls(message["tool_calls"])
272 if tools:
273 _attach_tools_to_dsv4_messages(rendered_messages, tools)
274
275 # No effort filtering here: the caller sanitized already, and the probe
276 # needs raw values to reach the encoder's own validation.
277 return encoder.encode_messages(
278 rendered_messages,
279 thinking_mode=resolve_thinking_mode(chat_template_kwargs, tools),
280 reasoning_effort=chat_template_kwargs.get("reasoning_effort"),
281 )

284def _dsv4_tool_calls(tool_calls: list[dict]) -> list[dict]:
285 """The dsv4 encoder's contract is ``function.arguments`` = JSON-object STRING
286 (it json.loads then iterates .items()); a dict (what ``render_messages``
287 produces for Jinja templates) trips its bare-except fallback, which wraps the
288 whole payload in a bogus parameter literally named ``arguments``. Re-serialize
289 here. Copies each tool-call dict: the outer message copy is shallow, so these
290 are shared with the caller."""
291 rendered = []
292 for tc in tool_calls:
293 tc = dict(tc)
294 fn = dict(tc.get("function") or {})
295 fn["arguments"] = _dsv4_arguments_str(fn.get("arguments"))
296 tc["function"] = fn
297 rendered.append(tc)
298 return rendered

301def _dsv4_arguments_str(arguments: Any) -> str:
302 """Missing/empty means no arguments (vLLM parity); anything else that is not
303 a JSON object is rejected -- ValueError becomes a per-request "could not
304 encode request" error, never a worker crash -- matching sglang's
305 validate-then-400. A non-object would otherwise raise uncaught in the
306 encoder's .items() or be wrapped as garbage."""
307 if arguments is None or (isinstance(arguments, str) and not arguments.strip()):
308 return "{}"
309 if isinstance(arguments, dict):
310 return json.dumps(arguments, ensure_ascii=False)
311 shown = f"{arguments!r:.200}"
312 if isinstance(arguments, str):
313 try:
314 parsed = json.loads(arguments)
315 except json.JSONDecodeError as err:
316 raise ValueError(
317 f"tool call function.arguments must be valid JSON, got {shown}"
318 ) from err
319 if isinstance(parsed, dict):
320 return arguments
321 raise ValueError(f"tool call function.arguments must be a JSON object, got {shown}")

324def _attach_tools_to_dsv4_messages(messages: list[dict], tools: list[dict]) -> None:
325 for message in messages:
326 if message.get("role") == "system":
