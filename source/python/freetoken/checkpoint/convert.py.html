1"""Convert an HF safetensors checkpoint into a FreeToken Weight (FTW) checkpoint.
2
3Model-agnostic: it drives the *existing* per-model loaders once and stores their output, so
4no per-model conversion code is needed.
5
6* dense weights = exactly what ``load_weight(include_moe_experts=...)`` yields (post
7 fusion/TP-shard) -> ``kind="weight"``; at load they feed ``model.load_state_dict``.
8* offload experts = exactly what ``load_expert_banks(parallel=True)`` produces (post
9 backend-repack pinned banks + alpha scale vectors) -> ``kind="experts_bank"`` (alphas are
10 told apart at load by their reserved names, so they need no separate kind).
11
12The output directory is a self-contained checkpoint (config + tokenizer copied), so you can
13point ``--model`` straight at it; the load path auto-detects the FTW and reads it (FTW).
14"""
15

28# Machine-readable convert progress for a supervising process (e.g. a GUI frontend parses
29# these `FTCONVERT <phase> <done> <total>` stdout lines to drive its convert bar). Gated by
30# FREETOKEN_CONVERT_PROGRESS=1 so plain CLI use isn't spammed; the human tqdm bars stay on
31# stderr. Phases: `dense` (indeterminate, done=total=0), `experts` (byte totals), `finalize`.
32_EMIT_PROGRESS = os.environ.get("FREETOKEN_CONVERT_PROGRESS") == "1"
33
34
35def _progress(phase: str, done: int = 0, total: int = 0) -> None:
36 if _EMIT_PROGRESS:
37 print(f"FTCONVERT {phase} {done} {total}", flush=True)

40def _source_fingerprint(model_path: str, model_config, *, device) -> str:
41 """Identity of (checkpoint + quant + GPU capability), stored in the FTW index so
42 it's clear what an FTW was built from. Cheap (stat only)."""
43 h = hashlib.sha256()
44 h.update(f"quant={getattr(model_config, 'expert_quant', None)}|".encode())
45 h.update(f"arch={getattr(model_config, 'architectures', None)}|".encode())
46 try: # nvfp4 marlin/b12x layout depends on compute capability
47 h.update(f"cc={torch.cuda.get_device_capability(device)}|".encode())
48 except Exception:
49 pass
50 files = sorted(
51 glob.glob(os.path.join(model_path, "*.safetensors")) + glob.glob(os.path.join(model_path, "*.gguf"))
52 )
53 for f in files:
54 st = os.stat(f)
55 h.update(f"{os.path.basename(f)}:{st.st_size}:{int(st.st_mtime)}|".encode())
56 return h.hexdigest()[:16]

58# Checkpoint metadata to carry over so the FTW dir is a usable checkpoint on its own.
59# (Weight shards + the safetensors index are intentionally NOT copied.)
60# Everything that is NOT a weight shard is metadata we carry over verbatim. A whitelist
61# misses model-specific layouts (e.g. DSV4's inference/config.json + encoding/ live in
62# subdirs), so we copy every non-weight file preserving its relative path instead.
63_WEIGHT_SUFFIXES = (".safetensors", ".gguf", ".ftw") # .ftw: a nested FTW, not source
64_SKIP_NAMES = ("model.safetensors.index.json",) # indexes shards the FTW replaces
65# .freetoken_expert_cache: the legacy per-bank cache (can be tens of GB of stale .bin)
66_SKIP_DIRS = (".git", ".cache", ".freetoken_expert_cache")

69def _copy_metadata(model_path: str, out_dir: str) -> list[str]:
70 """Copy all non-weight files (config, tokenizer, remote-code, nested model configs)
71 preserving directory structure, so the FTW dir is a self-contained checkpoint."""
72 if os.path.isfile(model_path):
73 # A single-file source has no sibling metadata to walk: a .gguf carries its config
74 # AND tokenizer in its own KV section. Emit a metadata-only copy (header + KV, no
75 # weight data) the FTW dir resolves those from. We deliberately do NOT sweep the
76 # file's parent dir -- an HF gguf snapshot dir can hold unrelated blobs.
77 from freetoken.models.gguf.reader import (
78 FTW_METADATA_GGUF,
79 is_gguf_path,
80 write_metadata_gguf,
81 )
82
83 if is_gguf_path(model_path):
84 os.makedirs(out_dir, exist_ok=True)
85 write_metadata_gguf(model_path, os.path.join(out_dir, FTW_METADATA_GGUF))
86 return [FTW_METADATA_GGUF]
87 return []
88
89 out_abs = os.path.abspath(out_dir)
90 copied = []
91 for root, dirs, files in os.walk(model_path):
92 dirs[:] = [d for d in dirs
93 if d not in _SKIP_DIRS and os.path.abspath(os.path.join(root, d)) != out_abs]
94 for name in files:
95 if name.endswith(_WEIGHT_SUFFIXES) or name in _SKIP_NAMES:
96 continue
97 src = os.path.join(root, name)
98 rel = os.path.relpath(src, model_path)
99 dst = os.path.join(out_dir, rel)
100 os.makedirs(os.path.dirname(dst), exist_ok=True)
101 shutil.copy2(src, dst)
102 copied.append(rel)
103 return copied

106class _ConvertSink:
107 """Layer-completion sink for ``load_expert_banks(layer_sink=...)``: writes each
108 completed layer's banks as their own FTW entries immediately (name
109 ``f"{bank_name}#L{layer_id:05d}"``, kind ``"experts_bank"``) and releases them, so
110 conversion RAM peaks at ~in-flight layers instead of the whole bank set.
111
112 Only engaged for streamable formats -- ``ExpertBanks.streamed`` reports whether this
113 actually fired; if not, the caller falls back to the materialize-and-write path
114 instead. The progress bar is created lazily on the first call, so a format that never
115 streams never shows one.
116
117 ``FTWWriter`` buffers file/shard state and is not thread-safe; completion callbacks
118 can fire from the loader's own reader threads, so the write+release is serialized
119 under one lock (disk-bound anyway).
120 """
121
122 def __init__(self, writer: FTWWriter, desc: str = "Converting expert banks", *, names: dict[str, str] | None = None) -> None:
123 self._writer = writer
124 self._desc = desc
125 # canonical role -> the bank name the file stores
126 self._names = names or {}
127 self._bar = None
128 self._lock = threading.Lock()
129 self._seen: set[int] = set()
130 self.n_written = 0
131 self.n_bytes = 0
132
133 def __call__(self, layer_id: int, banks: dict) -> None:
134 with self._lock:
135 assert layer_id not in self._seen, f"layer {layer_id} streamed to the sink twice"
136 self._seen.add(layer_id)
137 if self._bar is None:
138 from freetoken.utils.progress import byte_bar
139
140 self._bar = byte_bar(0, self._desc) # total unknown up front (streamed)
141 nbytes = 0
142 for bank_name, bank in banks.items():
143 self._writer.add_tensor(
144 layer_bank_entry_name(self._names.get(bank_name, bank_name), layer_id), bank.tensor, kind="experts_bank"
145 )
146 nbytes += bank.nbytes
147 bank.release()
148 self.n_written += 1
149 self.n_bytes += nbytes
150 self._bar.update(nbytes)
151 # Cumulative BYTES (not the bank count): the supervisor maps this against the
152 # known expert-pool size for a smooth phase-budgeted bar. Total stays 0 (unknown
153 # up front while streaming); the materialize path below emits a real total.
154 _progress("experts", self.n_bytes, 0)
155
156 def close(self) -> None:
157 if self._bar is not None:
158 self._bar.close()
159
160 @property
161 def num_layers(self) -> int:
162 return len(self._seen)
163

165def convert_checkpoint(
166 model_path: str,
167 out_dir: str,
168 *,
169 dtype: torch.dtype = torch.bfloat16,
170 moe_backend: str = "offload",
171 quant_backend: str | None = None,
172 shard_limit: int = DEFAULT_SHARD_LIMIT,
173 device: str | None = None,
174) -> dict:
175 """Write ``model_path`` as an FTW checkpoint at ``out_dir``. Returns the index dict.
176
177 The FTW format is TP-agnostic and conversion runs single-process, so the resulting
178 checkpoint records no TP layout and loads independently of the runtime TP setting."""
179 from freetoken.distributed import DistributedInfo, set_tp_info, try_get_tp_info
180 from freetoken.engine.config import EngineConfig
181 from freetoken.models.weight import load_weight
182 from freetoken.moe.expert_banks import load_expert_banks
183 from .ftw import is_ftw_checkpoint
184
185 if is_ftw_checkpoint(model_path):
186 raise SystemExit(f"{model_path} is already an FTW checkpoint")
187 tp = try_get_tp_info()
188 if tp is None:
189 set_tp_info(rank=0, size=1)
190 tp = try_get_tp_info()
191 elif tp.size != 1:
192 raise SystemExit(
193 f"FTW conversion runs single-process and the format records no TP layout, "
194 f"but TP is already set to size={tp.size}"
195 )
196 dev = torch.device(device or "cuda:0")
197 torch.cuda.set_device(dev)
198 torch.zeros(1, device=dev) # init CUDA context (needed by nvfp4 backend pick / pinning)
199
200 cfg = EngineConfig(model_path=model_path, tp_info=DistributedInfo(tp.rank, tp.size),
201 dtype=dtype, moe_strategy=moe_backend, quant_backend=quant_backend)
202 mc = cfg.model_config
203 # every MoE strategy reads the routed experts from the banks, fused included
204 offload = getattr(mc, "is_moe", False)
205 include_moe_experts = not offload
206 method = layer_methods = None
207 if offload:
208 from freetoken.engine.engine import offload_expert_methods
209 from freetoken.layers import set_rope_device
210
211 set_rope_device(dev)
212 object.__setattr__(cfg, "moe_strategy", "offload")
213 method, layer_methods = offload_expert_methods(cfg)
214
215 from freetoken.utils.progress import byte_bar, count_bar
216
217 writer = FTWWriter(out_dir, shard_limit=shard_limit)
218 n_weight = n_bank = n_alpha = 0
219
220 # 1) dense weights (host tensors; load straight to CPU to avoid GPU pressure)
221 _progress("dense", 0, 0) # phase start; per-tensor cumulative bytes follow (total unknown)
222 dense_bytes = 0
223 for name, tensor in count_bar(load_weight(model_path, torch.device("cpu"),
224 include_moe_experts=include_moe_experts),
225 "Converting dense weights"):
226 writer.add_tensor(name, tensor, kind="weight")
227 n_weight += 1
228 dense_bytes += tensor.numel() * tensor.element_size()
229 _progress("dense", dense_bytes, 0)
230
231 # files the model reads directly from the checkpoint dir, not through FTW entries (Qwen3.8-Flash-Next PLE table)
232 from freetoken.models.register import _load_attr, get_model_spec
233
234 try:
235 side_hook = _load_attr(get_model_spec(mc.architectures[0]).module, "ftw_side_files")
236 except (AttributeError, KeyError, ValueError):
237 side_hook = None
238 side_files = side_hook(model_path, out_dir) if side_hook is not None else []
239
240 # 2) offload expert banks (post-repack) + alpha scales (slow path auto-picks parallel/serial)
241 quant_format = None
242 layer_quant_formats = None
243 num_layers = None
244 if offload:
245 # every method-packed format streams each layer to its own FTW entry as it completes (via the sink); the GGUF provider reports through ExpertBanks.streamed whether it engaged the sink or materialized the whole bank set first
246 from freetoken.moe.legacy_format import legacy_bank_names, legacy_format_for
247
248 names = legacy_bank_names(legacy_format_for(method.kind, method.kernel.name)) if method is not None else {}
249 sink = _ConvertSink(writer, names=names)
250 banks = load_expert_banks(
251 model_path, mc, method=method, device=dev, layer_sink=sink, layer_methods=layer_methods
252 )
253 quant_format = banks.quant_format
254 layer_quant_formats = banks.layer_formats
255 assert layer_quant_formats is None or banks.streamed, "experts of several formats always stream to the sink"
256 if banks.streamed:
257 sink.close()
258 num_layers = sink.num_layers # however many distinct layers the sink actually saw
259 n_bank = sink.n_written
260 assert num_layers > 0, (
261 "provider reported streamed=True but the sink never fired -- the FTW "
262 "would silently have no expert banks"
263 )
264 # Formats that fold their global scales (nvfp4 marlin/b12x) stream the weight
265 # banks per layer but keep the alphas as flat [L*E] GPU vectors; write those as
266 # flat reserved-name entries (same kind + names the materialize branch uses, so
267 # the reader's reserved-name path reconstructs them identically).
268 for an in ("gate_up_alpha", "down_alpha"):
269 alpha = getattr(banks, an, None)
270 if alpha is not None:
271 writer.add_tensor(an, alpha, kind="experts_bank")
272 n_alpha += 1
273 else:
274 # The on-disk format keeps one contiguous region per bank and the writer only
275 # has whole-tensor add_tensor, so the per-layer sources reassemble into one
276 # flat tensor (a per-bank host RAM spike during conversion).
277 items = []
278 names = legacy_bank_names(quant_format)
279 for name, per_layer in banks.sources.items():
280 if num_layers is None:
281 num_layers = len(per_layer)
282 else:
283 assert len(per_layer) == num_layers, (name, len(per_layer), num_layers)
284 items.append((names.get(name, name), torch.cat(per_layer, dim=0) if len(per_layer) > 1 else per_layer[0]))
285 for an in ("gate_up_alpha", "down_alpha"):
286 if getattr(banks, an, None) is not None:
287 items.append((an, getattr(banks, an)))
288 total_bytes = sum(t.numel() * t.element_size() for _, t in items)
289 bar = byte_bar(total_bytes, "Converting expert banks")
290 done_bytes = 0
291 _progress("experts", 0, total_bytes)
292 for name, tensor in items:
293 writer.add_tensor(name, tensor, kind="experts_bank")
294 nbytes = tensor.numel() * tensor.element_size()
295 bar.update(nbytes)
296 done_bytes += nbytes
297 _progress("experts", done_bytes, total_bytes)
298 n_bank += name not in ("gate_up_alpha", "down_alpha")
299 n_alpha += name in ("gate_up_alpha", "down_alpha")
300 bar.close()
301
302 _progress("finalize") # writing shard index + copying config/tokenizer
303 copied = _copy_metadata(model_path, out_dir)
304
305 try:
306 fingerprint = _source_fingerprint(model_path, mc, device=dev)
307 except Exception:
308 fingerprint = None
309
310 index = writer.finalize({
311 "source_model_path": os.path.abspath(model_path),
312 "fingerprint": fingerprint,
313 # quant_format records the actual on-disk bank layout (e.g. nvfp4_marlin vs
314 # nvfp4_b12x): the suffix is a runtime backend pick (GPU capability / env), NOT in
315 # config, and the stored bytes are physically repacked into it -- so it's kept and
316 # read back at load (ftw.load_ftw_banks). dtype/moe_backend were dropped: each
317 # tensor already carries its own dtype, and nothing reads a model-level backend.
318 "quant_format": quant_format,
319 # Experts of several formats: each MoE layer's quant_format (quant_format above is the most common). A layer only
320 # stores the banks its kernel reads; the reader gives it zero-byte placeholders for the others.
321 "layer_quant_formats": layer_quant_formats,
322 # The reader takes num_layers from the model config (copied into this
323 # checkpoint); recording it here too gives load_ftw_banks a cross-check that
324 # the banks match the config they ship with. None for non-offload checkpoints.
325 "expert_bank_num_layers": num_layers,
326 "counts": {"weight": n_weight, "experts_bank": n_bank + n_alpha},
327 "copied_metadata": copied,
328 "side_files": side_files,
329 })
330 return index
