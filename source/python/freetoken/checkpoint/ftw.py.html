1"""FreeToken Weight (FTW) checkpoint: one O_DIRECT-friendly on-disk format for a whole model.
2
3The format is a single *logical contiguous byte region* of all tensors, sliced *physically*
4into shard files of at most ``shard_limit`` bytes (default 8 GiB, for HF/filesystem
5friendliness). It exists because reading the original safetensors back fast is awkward:
6tensors are packed with no alignment, so an individual tensor can't be O_DIRECT-read at an
7arbitrary offset; and the earlier per-bank cache prototype worked around that by giving every expert bank
8its own file -- which doesn't cover dense weights and turns a model's long tail of tiny
9tensors (norms, biases, router) into hundreds of tiny I/Os.
10
11FTW fixes both:
12
13* **Aligned.** Every tensor starts at a 4096-aligned region offset and is padded to 4096;
14 shards are cut at 4096-aligned boundaries. So any tensor (or any shard-local slice of one)
15 is read with offset, length (rounded up to 4096), and destination all block-aligned --
16 exactly what O_DIRECT requires. A tensor larger than a shard simply spans shards; because
17 both its start and the shard boundary are aligned, each piece stays aligned.
18* **Unified.** It holds dense weights as ``kind="weight"`` (exactly what a model's
19 ``iter_weights`` yields -- post fusion/TP-shard, fed straight to ``load_state_dict``) and
20 the offload expert state as ``kind="experts_bank"`` (post backend-repack -- the per-expert
21 weight banks plus, distinguished only by their reserved names, the alpha scale vectors;
22 the FTW content). The converter runs the per-model loaders once; this reader is
23 model-agnostic.
24
25Layout on disk::
26
27 <dir>/freetoken_weight.json # index: tensors[] + shards[] + meta
28 <dir>/freetoken-00000.ftw # the byte region, sliced <= shard_limit
29 <dir>/freetoken-00001.ftw
30 <dir>/config.json, tokenizer*, ...# copied so the dir is a self-contained checkpoint
31"""
32

70def _pread_into(fd: int, mv: memoryview, offset: int) -> None:
71 """POSIX positional read into ``mv`` at ``offset``, looping over any short preadv.
72
73 preadv may return short (a signal, or the EOF-adjacent tail); the loop resumes
74 at the running offset, which stays O_DIRECT-legal: the writer pads every tensor
75 to ALIGN and cuts shards at ALIGN boundaries, so direct-IO short reads land on
76 block boundaries. EOF before the buffer is filled raises ``OSError`` — a
77 truncated shard must not silently load garbage weights."""
78 done = 0
79 total = len(mv)
80 while done < total:
81 n = os.preadv(fd, [mv[done:]], offset + done)
82 if n == 0:
83 raise OSError(
84 f"unexpected EOF reading FTW: got {done}/{total} bytes at offset {offset}"
85 )
86 done += n

125# ============================== writer ==============================
126class FTWWriter:
127 """Stream tensors into the FTW, rolling shard files at ``shard_limit``.
128
129 Tensors are written in call order into one logical byte stream; each is padded to
130 ``ALIGN`` so the next starts aligned. A tensor that doesn't fit the current shard's
131 remaining room is split across shards (the split point is the shard boundary, which is
132 aligned). Call :meth:`add_tensor` for each tensor, then :meth:`finalize`.
133 """
134
135 def __init__(self, out_dir: str, *, shard_limit: int = DEFAULT_SHARD_LIMIT):
136 assert shard_limit % ALIGN == 0, "shard_limit must be a multiple of ALIGN"
137 os.makedirs(out_dir, exist_ok=True)
138 self.out_dir = out_dir
139 self.shard_limit = shard_limit
140 self._tensors: list[dict] = []
141 self._shards: list[dict] = []
142 self._global = 0 # running FTW offset (incl. padding)
143 self._f = None # current shard file handle
144 self._shard_idx = -1
145 self._shard_start = 0 # FTW offset where the current shard began
146 self._cur = 0 # bytes written to the current shard
147
148 def _roll(self) -> None:
149 if self._f is not None:
150 self._shards.append({"file": _SHARD_FMT.format(self._shard_idx),
151 "global_off": self._shard_start, "nbytes": self._cur})
152 self._f.close()
153 self._shard_idx += 1
154 self._shard_start = self._global
155 self._cur = 0
156 self._f = open(os.path.join(self.out_dir, _SHARD_FMT.format(self._shard_idx)), "wb")
157
158 def _write_raw(self, data: memoryview) -> None:
159 """Write ``data`` into the FTW byte stream, splitting across shards at the limit."""
160 if self._f is None:
161 self._roll()
162 off = 0
163 n = len(data)
164 while off < n:
165 if self._cur == self.shard_limit:
166 self._roll()
167 take = min(n - off, self.shard_limit - self._cur)
168 self._f.write(data[off:off + take])
169 off += take
170 self._cur += take
171 self._global += take
172
173 def add_tensor(self, name: str, tensor: torch.Tensor, kind: str = "weight") -> None:
174 t = tensor.detach().cpu().contiguous()
175 raw = t.reshape(-1).view(torch.uint8)
176 nbytes = int(raw.numel())
177 # A small tensor (<= shard) never splits: roll early so it lands whole in one shard.
178 if self._f is None or (nbytes <= self.shard_limit
179 and self._cur + nbytes > self.shard_limit):
180 self._roll()
181 global_off = self._global
182 assert global_off % ALIGN == 0, "tensor start must be aligned (invariant)"
183 self._write_raw(memoryview(raw.numpy()))
184 self._tensors.append({"name": name, "kind": kind, "dtype": _dtype_str(t.dtype),
185 "shape": list(t.shape), "global_off": global_off, "nbytes": nbytes})
186 # pad to ALIGN so the next tensor starts aligned
187 pad = _align_up(self._global) - self._global
188 if pad:
189 self._write_raw(memoryview(bytes(pad)))
190
191 def finalize(self, meta: dict) -> dict:
192 if self._f is not None:
193 self._shards.append({"file": _SHARD_FMT.format(self._shard_idx),
194 "global_off": self._shard_start, "nbytes": self._cur})
195 self._f.close()
196 self._f = None
197 index = {"format": FORMAT_TAG, "version": FORMAT_VERSION, "align": ALIGN,
198 "shard_limit": self.shard_limit, "total_bytes": self._global,
199 "tensors": self._tensors, "shards": self._shards, **meta}
200 tmp = os.path.join(self.out_dir, INDEX_NAME + ".tmp")
201 with open(tmp, "w") as f:
202 json.dump(index, f)
203 os.replace(tmp, os.path.join(self.out_dir, INDEX_NAME))
204 return index

207# ============================== reader ==============================
208class FTWReader:
209 """Random-access reader over an FTW checkpoint.
210
211 Maps a tensor's logical byte range to one-or-more shard-file ranges (split at shard
212 boundaries) and reads each piece with chunked multi-threaded O_DIRECT directly into the
213 destination buffer. Offsets/lengths are all 4096-aligned (lengths rounded up into the
214 rounded-up destination), so O_DIRECT is always legal -- including the tail of a tensor
215 (the rounding reads into the region's padding, which is discarded by the tensor view)."""
216
217 def __init__(self, path: str):
218 with open(os.path.join(path, INDEX_NAME)) as f:
219 self.index = json.load(f)
220 assert self.index.get("format") == FORMAT_TAG, f"not a {FORMAT_TAG}: {path}"
221 self.dir = path
222 self.shards = sorted(self.index["shards"], key=lambda s: s["global_off"])
223 self.tensors = {t["name"]: t for t in self.index["tensors"]}
224 self._fds: dict[str, int] = {}
225 self._maps: dict[str, tuple[mmap.mmap, memoryview]] = {}
226 # O_DIRECT (DMA straight from disk, bypassing the page cache) is the fast path but a
227 # perf choice, not a correctness one. Some filesystems reject it at open with EINVAL
228 # (tmpfs, many overlay/network mounts) and the flag is Linux-only; when it's absent
229 # we fall back to mmap (below), NOT to chunked buffered preadv -- a whole-shard
230 # mapping + kernel readahead copies far faster than per-chunk page-cache reads.
231 # 0 here means "O_DIRECT unavailable -> use the mmap path".
232 self._direct = getattr(os, "O_DIRECT", 0)
233 self._probed = False
234 self._lock = threading.Lock() # load_ftw_banks calls read_into concurrently
235
236 def meta(self, key: str, default=None):
237 return self.index.get(key, default)
238
239 def entries(self, *kinds: str) -> list[dict]:
240 keep = set(kinds)
241 return [t for t in self.index["tensors"] if not keep or t["kind"] in keep]
242
243 def _ensure_mode(self) -> None:
244 """Resolve the read backend once: keep O_DIRECT if the filesystem accepts it, else
245 drop to the mmap fallback. Thread-safe -- ``_probed`` is published only after
246 ``_direct`` is final, so a concurrent reader never races onto a stale direct path."""
247 if self._probed:
248 return
249 with self._lock:
250 if self._probed:
251 return
252 if self._direct and self.shards:
253 try:
254 os.close(os.open(os.path.join(self.dir, self.shards[0]["file"]),
255 os.O_RDONLY | self._direct))
256 except OSError:
257 self._direct = 0
258 logger.warning("O_DIRECT unsupported on %s; using mmap fallback for "
259 "FTW load", self.dir)
260 self._probed = True
261
262 def _fd(self, file: str) -> int:
263 fd = self._fds.get(file)
264 if fd is None:
265 with self._lock: # first-open only; chunk reads reuse the cached fd lock-free
266 fd = self._fds.get(file)
267 if fd is None:
268 fd = os.open(os.path.join(self.dir, file), os.O_RDONLY | self._direct)
269 self._fds[file] = fd
270 return fd
271
272 def _map(self, file: str) -> memoryview:
273 entry = self._maps.get(file)
274 if entry is None:
275 with self._lock:
276 entry = self._maps.get(file)
277 if entry is None:
278 fd = os.open(os.path.join(self.dir, file), os.O_RDONLY)
279 try:
280 m = mmap.mmap(fd, 0, prot=mmap.PROT_READ)
281 finally:
282 os.close(fd) # the mapping keeps its own reference to the file
283 try:
284 m.madvise(mmap.MADV_SEQUENTIAL) # kernel readahead for streaming
285 except (AttributeError, OSError):
286 pass
287 entry = (m, memoryview(m))
288 self._maps[file] = entry
289 return entry[1]
290
291 def close(self) -> None:
292 for fd in self._fds.values():
293 os.close(fd)
294 self._fds.clear()
295 for m, mv in self._maps.values():
296 mv.release()
297 m.close()
298 self._maps.clear()
299
300 def _pieces(self, global_off: int, nbytes: int):
301 """Yield (file, file_off, dest_off, length) covering [global_off, +nbytes),
302 split at shard boundaries. All file_off/dest_off are ALIGN-aligned."""
303 dest_off = 0
304 remaining = nbytes
305 pos = global_off
306 for sh in self.shards:
307 s0, s1 = sh["global_off"], sh["global_off"] + sh["nbytes"]
308 if pos >= s1 or remaining <= 0:
309 continue
310 if pos < s0: # regions are contiguous; a gap means a corrupt index
311 raise ValueError("FTW gap / misordered shards")
312 take = min(remaining, s1 - pos)
313 yield sh["file"], pos - s0, dest_off, take
314 pos += take
315 dest_off += take
316 remaining -= take
317 if remaining:
318 raise ValueError("tensor range exceeds FTW shards")
319
320 def read_into(self, dest: memoryview, entry: dict, *, workers: int = 8,
321 chunk: int = _DEFAULT_CHUNK) -> None:
322 """Read one tensor's bytes into ``dest`` (length >= entry nbytes rounded to ALIGN)."""
323 self._ensure_mode()
324 jobs = [] # (file, file_off, dest_off, length) all ALIGN-aligned
325 for file, file_off, dest_off, length in self._pieces(entry["global_off"], entry["nbytes"]):
326 rlen = _align_up(length) # round the tail up; padding is in-region, harmless
327 for c in range(0, rlen, chunk):
328 jobs.append((file, file_off + c, dest_off + c, min(chunk, rlen - c)))
329
330 # Open/map each distinct shard once, single-threaded, so the pool only reuses handles.
331 touch = self._fd if self._direct else self._map
332 for file in {j[0] for j in jobs}:
333 touch(file)
334
335 if self._direct:
336 def rd(job):
337 file, fo, do, ln = job
338 try:
339 _pread_into(self._fd(file), dest[do:do + ln], fo)
340 except OSError as e:
341 raise OSError(f"shard {file}: {e}") from e
342 else:
343 def rd(job):
344 file, fo, do, ln = job
345 mv = self._map(file)
346 if fo + ln > len(mv):
347 raise OSError(
348 f"unexpected EOF reading FTW: shard {file} has "
349 f"{len(mv)} bytes, need {ln} at offset {fo}"
350 )
351 dest[do:do + ln] = mv[fo:fo + ln]
352
353 if len(jobs) <= 1:
354 for j in jobs:
355 rd(j)
356 else:
357 with ThreadPoolExecutor(workers) as ex:
358 list(ex.map(rd, jobs))
359

365def iter_ftw_weights(path: str, *, kinds=("weight",), keep: Callable[[str], bool] | None = None,
366 workers: int = 8, chunk: int = _DEFAULT_CHUNK, prefetch: int = 2):
367 """Yield ``(name, host_tensor)`` for the requested kinds, reading each tensor via
368 chunked O_DIRECT. A background thread prefetches the next ``prefetch`` tensors so the
369 disk stays busy while the consumer copies the current one to the GPU. Transient buffers
370 are freed as the consumer advances (peak host mem ~ prefetch+1 tensors). An entry whose
371 name ``keep`` rejects is dropped before any of its bytes are read."""
372 import queue
373 import threading
374
375 from freetoken.utils.progress import byte_bar
376
377 reader = FTWReader(path)
378 entries = reader.entries(*kinds)
379 if keep is not None:
380 entries = [e for e in entries if keep(e["name"])]
381 q: queue.Queue = queue.Queue(maxsize=max(1, prefetch))
382 _DONE = object()
383 err: list[BaseException] = []
384 cancel = threading.Event()
385
386 def _put(item) -> bool:
387 # A plain q.put would deadlock teardown: if the consumer stops with the queue
388 # full (early break out of the generator, or an exception mid-load), close()
389 # runs the finally below, which joins this thread while it waits for queue
390 # space forever. Poll the cancel flag instead of blocking indefinitely.
391 while not cancel.is_set():
392 try:
393 q.put(item, timeout=0.1)
394 return True
395 except queue.Full:
396 continue
397 return False
398
399 def _producer():
400 try:
401 for e in entries:
402 buf = _transient_buffer(e["nbytes"])
403 reader.read_into(memoryview(buf), e, workers=workers, chunk=chunk)
404 dt = _dtype_of(e["dtype"])
405 t = torch.frombuffer(buf, dtype=dt, count=e["nbytes"] // _elsize(dt))
406 # a 0-d entry (a per-tensor scale) comes back 0-d, not [1]
407 if not _put((e["name"], t.view(*e["shape"]) if e["shape"] else t.view(()), buf, e["nbytes"])):
408 return
409 except BaseException as ex: # surface to consumer
410 err.append(ex)
411 finally:
412 _put(_DONE)
413
414 th = threading.Thread(target=_producer, name="FTW-prefetch", daemon=True)
415 th.start()
416 bar = byte_bar(sum(e["nbytes"] for e in entries), "Loading weights (FTW)")
417 try:
418 while True:
419 item = q.get()
420 if item is _DONE:
421 break
422 name, tensor, buf, nbytes = item
423 yield name, tensor
424 bar.update(nbytes)
425 del tensor, buf # buffer reclaimable once the consumer drops the tensor
426 finally:
427 bar.close()
428 cancel.set()
429 th.join()
430 reader.close()
431 if err:
432 raise err[0]

435def _split_bank_entries(reader: FTWReader, path: str, num_layers: int):
436 """``(all, alphas, flat regions, {base: {layer: entry}})`` of the ``experts_bank`` entries; None when there are none."""
437 bank_entries = reader.entries("experts_bank")
438 if not bank_entries:
439 return None
440
441 alpha_entries = [e for e in bank_entries if e["name"] in _ALPHA_NAMES]
442 row_entries = [e for e in bank_entries if e["name"] not in _ALPHA_NAMES]
443
444 meta_layers = reader.meta("expert_bank_num_layers")
445 if meta_layers is not None and meta_layers != num_layers:
446 raise RuntimeError(
447 f"{path!r} was converted with {meta_layers} expert-bank layers but the "
448 f"model config says num_moe_layers={num_layers}; the checkpoint does not "
449 "match its config"
450 )
451
452 # Split row entries into the two layouts by name.
453 flat_entries: list[dict] = []
454 per_layer_groups: dict[str, dict[int, dict]] = {}
455 for e in row_entries:
456 m = _LAYER_ENTRY_RE.match(e["name"])
457 if m is None:
458 flat_entries.append(e)
459 continue
460 per_layer_groups.setdefault(m.group("base"), {})[int(m.group("layer"))] = e
461
462 mixed = {e["name"] for e in flat_entries} & per_layer_groups.keys()
463 assert not mixed, f"FTW bank(s) mix flat and per-layer row layouts: {sorted(mixed)}"
464 # experts of several formats: a layer stores only the banks its kernel reads
465 several = reader.meta("layer_quant_formats") is not None
466 assert not (several and flat_entries), "experts of several formats are stored per layer"
467 for base, by_layer in per_layer_groups.items():
468 layers = sorted(by_layer)
469 assert (set(layers) <= set(range(num_layers))) if several else layers == list(range(num_layers)), (
470 f"FTW bank {base!r} has per-layer entries for layers {layers}, "
471 f"expected exactly range({num_layers})"
472 )
473 if several:
474 covered = set().union(*(by_layer.keys() for by_layer in per_layer_groups.values()))
475 assert covered == set(range(num_layers)), f"FTW expert banks have no entries for layers {sorted(set(range(num_layers)) - covered)}"
476 return bank_entries, alpha_entries, flat_entries, per_layer_groups

479def load_ftw_banks(
480 path: str, *, num_layers: int, workers: int = 8, chunk: int = _DEFAULT_CHUNK,
481 layer_residency: list[str] | None = None,
482):
483 """Reconstruct the offload :class:`ExpertBanks` from the FTW's ``experts_bank``
484 entries, on the per-layer host bank contract (one ``[num_experts, ...]``
485 HostBank per layer per bank; see ``moe.offload_cache.set_bank_sources``).
486
487 ``layer_residency`` (default: all pinned) settles each layer's banks per its ``HostResidency`` label as reads complete: PINNED -> cudaHostRegister, LOCKED -> mlock (CPU-executor resident, no pin quota spent).
488 The applied labels are echoed back on ``ExpertBanks.layer_residency``.
489
490 Two on-disk row layouts, distinguished per bank name (a file never mixes them for
491 the same name -- checked below):
492
493 * **Flat region** (pre-existing files, and non-streamable formats): one entry per
494 bank, ONE contiguous ``[num_layers * num_experts, ...]`` region. ``num_layers``
495 isn't part of that region's shape, so the caller passes it
496 (``ModelConfig.num_moe_layers`` -- FTW checkpoints carry the model's config.json);
497 the ``expert_bank_num_layers`` index meta the converter records is used as a
498 cross-check when present. A layer's byte range within the region generally is
499 NOT 4096-aligned (only the whole region's start is guaranteed aligned) -- read it
500 via its ALIGNED enclosing window ``[align_down(off), align_up(off+len))`` into a
501 page-aligned scratch HostBank, and view the real per-layer tensor as a
502 head-offset slice.
503 * **Per-layer** (streamable-format conversion, see :mod:`freetoken.checkpoint.convert`):
504 one entry per ``(bank, layer)``, name ``f"{bank_name}#L{layer_id:05d}"``. Each was
505 written by its own ``add_tensor`` call, so its start is already ALIGN-aligned --
506 no windowing/head-pad needed, read straight into a HostBank shaped like the entry.
507
508 Alphas (``gate_up_alpha``/``down_alpha``) stay flat ``[num_layers*num_experts]``
509 vectors, unaffected by the row split (fixed GPU residency; see
510 ``cache_budget.expert_bytes_per_slot``).
511 """
512 from freetoken.moe.host_banks import (
513 HostBank, HostResidency, PinPipeline, alloc_banks, born_pinned_default,
514 )
515 from freetoken.utils.progress import byte_bar
516
517 residency = layer_residency or [HostResidency.PINNED.value] * num_layers
518 assert len(residency) == num_layers, (len(residency), num_layers)
519
520 # PINNED layers are born-pinned (cudaHostAlloc) where that wins (see born_pinned_default); LOCKED/PAGEABLE layers stay lazy mmaps
521 born = born_pinned_default()
522
523 def _backing(layer_id: int) -> str:
524 if born and residency[layer_id] == HostResidency.PINNED.value:
525 return "cuda"
526 return "mmap"
527
528 reader = FTWReader(path)
529 try:
530 split = _split_bank_entries(reader, path, num_layers)
531 except BaseException:
532 reader.close()
533 raise
534 if split is None:
535 reader.close()
536 return None
537 bank_entries, alpha_entries, flat_entries, per_layer_groups = split
538
539 # Alphas: unchanged, one flat HostBank per entry.
540 alpha_specs = {e["name"]: (tuple(e["shape"]), _dtype_of(e["dtype"])) for e in alpha_entries}
541 alpha_hb = alloc_banks(alpha_specs)
542
543 # Row banks: one padded-window HostBank per (name, layer_id) for the flat layout, plus
544 # how to carve the real [num_experts, *row_shape] tensor out of its head; ``None`` marks
545 # a per-layer entry (direct view, no carving needed).
546 row_hb: dict[str, list] = {}
547 row_view_args: dict[str, list] = {}
548 row_jobs = [] # (name, HostBank, window_off, window_len, layer_bytes) -- flat layout
549 layer_jobs = [] # (name, HostBank, entry) -- per-layer layout, direct aligned read
550
551 for e in flat_entries:
552 name = e["name"]
553 total, *row_shape = e["shape"]
554 assert total % num_layers == 0, (name, total, num_layers)
555 num_experts = total // num_layers
556 dtype = _dtype_of(e["dtype"])
557 row_bytes = (math.prod(row_shape) if row_shape else 1) * _elsize(dtype)
558 layer_bytes = num_experts * row_bytes
559 assert layer_bytes * num_layers == e["nbytes"], (name, layer_bytes, num_layers, e["nbytes"])
560 row_hb[name] = []
561 row_view_args[name] = []
562 for layer_id in range(num_layers):
563 off = e["global_off"] + layer_id * layer_bytes
564 win_off = (off // ALIGN) * ALIGN
565 win_end = _align_up(off + layer_bytes)
566 head_pad = off - win_off
567 bank = HostBank((win_end - win_off,), torch.uint8, backing=_backing(layer_id))
568 row_hb[name].append(bank)
569 row_view_args[name].append((head_pad, layer_bytes, num_experts, tuple(row_shape), dtype))
570 row_jobs.append((name, bank, win_off, win_end - win_off, layer_bytes, layer_id))
571
572 num_experts = next((e["shape"][0] for by_layer in per_layer_groups.values() for e in by_layer.values()), None)
573 for base, by_layer in per_layer_groups.items():
574 row_hb[base] = []
575 row_view_args[base] = []
576 for layer_id in range(num_layers):
577 e = by_layer.get(layer_id)
578 if e is None: # experts of several formats: this layer's kernel reads no such bank
579 row_hb[base].append(None)
580 row_view_args[base].append("placeholder")
581 continue
582 assert e["global_off"] % ALIGN == 0, (base, layer_id, e["global_off"]) # writer invariant
583 bank = HostBank(tuple(e["shape"]), _dtype_of(e["dtype"]), backing=_backing(layer_id))
584 row_hb[base].append(bank)
585 row_view_args[base].append(None)
586 layer_jobs.append((base, bank, e, layer_id))
587
588 total_bytes = sum(e["nbytes"] for e in bank_entries)
589 bar = byte_bar(total_bytes, "Loading expert banks (FTW)")
590
591 # Jobs are per (bank, layer) -- many small reads, so a wider pool; each bank pins
592 # as its read completes, overlapping cudaHostRegister with the remaining reads.
593 n_jobs = len(alpha_entries) + len(row_jobs) + len(layer_jobs)
594 try:
595 with PinPipeline() as pins:
596
597 def _read_alpha(e):
598 bank = alpha_hb[e["name"]]
599 reader.read_into(bank.memoryview(), e, workers=workers, chunk=chunk)
600 pins.submit(bank)
601 bar.update(e["nbytes"])
602
603 def _read_row(job):
604 _name, bank, win_off, win_len, layer_bytes, layer_id = job
605 reader.read_into(bank.memoryview(), {"global_off": win_off, "nbytes": win_len},
606 workers=workers, chunk=chunk)
607 pins.submit(bank, residency[layer_id])
608 bar.update(layer_bytes)
609
610 def _read_layer(job):
611 _name, bank, entry, layer_id = job
612 reader.read_into(bank.memoryview(), entry, workers=workers, chunk=chunk)
613 pins.submit(bank, residency[layer_id])
614 bar.update(entry["nbytes"])
615
616 with ThreadPoolExecutor(min(max(_BANK_CONCURRENCY, 16), max(n_jobs, 1))) as ex:
617 futures = [ex.submit(_read_alpha, e) for e in alpha_entries]
618 futures += [ex.submit(_read_row, job) for job in row_jobs]
619 futures += [ex.submit(_read_layer, job) for job in layer_jobs]
620 for f in futures:
621 f.result()
622 finally:
623 bar.close()
624 reader.close()
625
626 from freetoken.moe.offload_cache import placeholder_bank
627
628 sources: dict[str, list] = {}
629 for name, banks in row_hb.items():
630 views = []
631 for bank, view_args in zip(banks, row_view_args[name]):
632 if view_args == "placeholder":
633 views.append(placeholder_bank(num_experts))
634 continue
635 if view_args is None: # per-layer entry: already shaped [num_experts, ...]
636 views.append(bank.tensor)
637 continue
638 head_pad, layer_bytes, num_experts, row_shape, dtype = view_args
639 raw = bank.tensor[head_pad:head_pad + layer_bytes].view(dtype)
640 views.append(raw.view(num_experts, *row_shape) if row_shape else raw.view(num_experts))
641 sources[name] = views
642
643 from freetoken.moe.legacy_format import canonical_role, kind_kernel_for
644 from freetoken.moe.expert_banks import ExpertBanks
645
646 # the file names the banks the legacy way; the quant_format tag names the (kind, kernel) they were packed for
647 sources = {canonical_role(name): views for name, views in sources.items()}
648 quant_format = reader.meta("quant_format")
649 kind, kernel = kind_kernel_for(quant_format) if quant_format is not None else (None, None)
650
651 # a failed mlock leaves a LOCKED layer pageable; the log and labels report what the banks actually settled at
652 applied = list(residency)
653 for banks in row_hb.values():
654 for layer_id, bank in enumerate(banks):
655 if bank is None:
656 continue
657 if (applied[layer_id] == HostResidency.LOCKED.value
658 and bank.residency is not HostResidency.LOCKED):
659 applied[layer_id] = HostResidency.PAGEABLE.value
660 unpinned = [i for i, r in enumerate(applied) if r != HostResidency.PINNED.value]
661 if unpinned:
662 by_layer = [0] * num_layers
663 for name, banks in row_hb.items():
664 for layer_id, bank in enumerate(banks):
665 by_layer[layer_id] += bank.nbytes if bank is not None else 0
666 locked = [i for i in unpinned if applied[i] == HostResidency.LOCKED.value]
667 pageable = [i for i in unpinned if i not in set(locked)]
668 pinned_b = sum(b for i, b in enumerate(by_layer) if i not in set(unpinned))
669 locked_b = sum(by_layer[i] for i in locked)
670 pageable_part = ""
671 if pageable:
672 pageable_b = sum(by_layer[i] for i in pageable)
673 pageable_part = (
674 f" + {pageable_b / 2**30:.2f} GiB pageable "
675 f"(lock failed, {len(pageable)} CPU layers: {pageable})"
676 )
677 logger.info(
678 f"MoE bank split residency: {pinned_b / 2**30:.2f} GiB pinned "
679 f"({'born-pinned cudaHostAlloc' if born else 'cudaHostRegister'}, "
680 f"{num_layers - len(unpinned)} GPU layers) + "
681 f"{locked_b / 2**30:.2f} GiB OS-locked ({len(locked)} CPU layers: {locked})"
682 f"{pageable_part}"
683 )
684
685 # alphas are the small per-expert scale vectors, distinguished by their reserved names
686 # (not a separate kind); everything else under experts_bank is a weight source.
687 alpha_kw = {n: alpha_hb[n].tensor for n in alpha_hb}
688 layer_formats = reader.meta("layer_quant_formats")
689 return ExpertBanks(
690 quant_format, sources, **alpha_kw,
691 layer_residency=applied, kind=kind, kernel=kernel,
692 layer_tags=layer_formats, layer_formats=layer_formats,
693 )

696def load_ftw_banks_to_device(path: str, *, num_layers: int, device: torch.device, workers: int = 8, chunk: int = _DEFAULT_CHUNK):
697 """:func:`load_ftw_banks` for resident experts: every bank lands on ``device``, one (bank, layer) at a time through a transient host buffer."""
698 from freetoken.moe.expert_banks import ExpertBanks
699 from freetoken.moe.legacy_format import canonical_role, kind_kernel_for
700 from freetoken.utils.progress import byte_bar
701
702 reader = FTWReader(path)
703 try:
704 split = _split_bank_entries(reader, path, num_layers)
705 if split is None:
706 return None
707 if reader.meta("layer_quant_formats") is not None:
708 raise ValueError(f"{path!r} holds experts of several formats, which are served from the offload slot cache only")
709 bank_entries, alpha_entries, flat_entries, per_layer_groups = split
710
711 # (name, layer_id or None for an alpha, read offset, read length, head pad, shape, dtype)
712 jobs = [(e["name"], None, e["global_off"], e["nbytes"], 0, tuple(e["shape"]), _dtype_of(e["dtype"])) for e in alpha_entries]
713 for e in flat_entries:
714 total, *row_shape = e["shape"]
715 assert total % num_layers == 0, (e["name"], total, num_layers)
716 layer_bytes = total // num_layers * math.prod(row_shape) * _elsize(_dtype_of(e["dtype"]))
717 assert layer_bytes * num_layers == e["nbytes"], (e["name"], layer_bytes, num_layers, e["nbytes"])
718 for layer_id in range(num_layers):
719 off = e["global_off"] + layer_id * layer_bytes
720 win_off = (off // ALIGN) * ALIGN
721 jobs.append((e["name"], layer_id, win_off, _align_up(off + layer_bytes) - win_off, off - win_off, (total // num_layers, *row_shape), _dtype_of(e["dtype"])))
722 for base, by_layer in per_layer_groups.items():
723 for layer_id in range(num_layers):
724 e = by_layer[layer_id]
725 jobs.append((base, layer_id, e["global_off"], e["nbytes"], 0, tuple(e["shape"]), _dtype_of(e["dtype"])))
726
727 def _read(job) -> torch.Tensor:
728 _name, _layer, off, nbytes, head_pad, shape, dtype = job
729 buf = _transient_buffer(nbytes)
730 reader.read_into(memoryview(buf), {"global_off": off, "nbytes": nbytes}, workers=workers, chunk=chunk)
731 count = math.prod(shape) * _elsize(dtype)
732 return torch.frombuffer(buf, dtype=torch.uint8, count=head_pad + count)[head_pad:].view(dtype).view(shape)
733
734 sources: dict[str, list] = {}
735 alphas: dict[str, torch.Tensor] = {}
736 bar = byte_bar(sum(e["nbytes"] for e in bank_entries), "Loading expert banks (FTW)")
737 try:
738 # read the next (bank, layer) while the current one copies to the device
739 with ThreadPoolExecutor(1) as ex:
740 ahead = ex.submit(_read, jobs[0])
741 for k, (name, layer_id, *_rest) in enumerate(jobs):
742 host = ahead.result()
743 if k + 1 < len(jobs):
744 ahead = ex.submit(_read, jobs[k + 1])
745 if layer_id is None:
746 alphas[name] = host.to(device)
747 else:
748 sources.setdefault(canonical_role(name), [None] * num_layers)[layer_id] = host.to(device)
749 bar.update(host.numel() * host.element_size())
750 del host
751 finally:
752 bar.close()
753 quant_format = reader.meta("quant_format")
754 finally:
755 reader.close()
756
757 kind, kernel = kind_kernel_for(quant_format) if quant_format is not None else (None, None)
758 return ExpertBanks(quant_format, sources, **alphas, kind=kind, kernel=kernel)
