23def _linear_local_dims(
24 group: LinearGatedDeltaGroupConfig, tp_size: int
25) -> tuple[int, int, int]:
26 """TP-local ``(n_layers, conv_dim, v_heads)`` for the GDN state tensors -- the single
27 source of the sharding math shared by the pool allocation and the byte estimate."""
28 local_k_heads = div_even(group.num_key_heads, tp_size, allow_replicate=True)
29 local_v_heads = div_even(group.num_value_heads, tp_size, allow_replicate=True)
30 local_conv_dim = 2 * local_k_heads * group.key_head_dim + local_v_heads * group.value_head_dim
31 return len(group.layer_ids), local_conv_dim, local_v_heads

34class LinearStatePool:
35 """Per-request recurrent state (conv + SSM) for GatedDeltaNet layers.
36
37 Indexed by ``Req.table_idx`` (0..max_running_req), the same per-request slot the
38 page table uses, so the scheduler's existing admit/free of ``table_idx`` covers the
39 state's lifetime. One fixed slot per running request; no paging, no eviction.
40
41 A model can declare extra per-request tensors on the same slots through
42 ``ModelConfig.slot_states`` (see ``SlotStateSpec``); they advance, snapshot, COW and
43 rebuild with the GDN state and are read back through ``slot_state(name, layer_id)``.
44 Consumers must re-read them each forward: ``rebuild`` replaces the tensors.
45 """
46
47 def __init__(
48 self,
49 group: LinearGatedDeltaGroupConfig,
50 num_slots: int,
51 dtype: torch.dtype,
52 device: torch.device,
53 tp_size: int | None = None,
54 slot_states: tuple[SlotStateSpec, ...] = (),
55 ) -> None:
56 if tp_size is None:
57 tp_size = get_tp_info().size
58
59 self._group = group
60 self._num_slots = num_slots
61 self._device = device
62 self._conv_dtype = dtype
63
64 n_layers, local_conv_dim, local_v_heads = _linear_local_dims(group, tp_size)
65
66 # conv left-context: the last (kernel-1) timesteps of the conv input stream.
67 self.conv_states = torch.zeros(
68 (n_layers, num_slots, local_conv_dim, group.conv_kernel_dim - 1),
69 dtype=dtype,
70 device=device,
71 )
72 # SSM recurrent state. fp32 by default (matches HF mamba_ssm_dtype); the dtype is
73 # overridable via FREETOKEN_MAMBA_SSM_DTYPE (see ssm_state_dtype).
74 self.recurrent_states = torch.zeros(
75 (n_layers, num_slots, local_v_heads, group.key_head_dim, group.value_head_dim),
76 dtype=ssm_state_dtype(),
77 device=device,
78 )
79 self._local_index = {layer_id: i for i, layer_id in enumerate(group.layer_ids)}
80
81 self._slot_specs = tuple(slot_states)
82 names = [spec.name for spec in self._slot_specs]
83 if len(set(names)) != len(names):
84 raise ValueError(f"duplicate slot_state names: {names}")
85 self._state_layer_index = {
86 spec.name: {lid: i for i, lid in enumerate(spec.layer_ids)}
87 for spec in self._slot_specs
88 }
89 self.slot_states: dict[str, torch.Tensor] = self._alloc_slot_states(num_slots)
90
91 # Free-list allocator over slots 1..num_slots-1 (slot 0 reserved as a padding sink,
92 # sglang MambaPool convention). Live working slots, ping-pong track slots, and
93 # radix-tree-donated snapshots are all drawn from this single free-list, so memory
94 # flows between them by demand. Unused by the op harness (which assigns slots by hand).
95 self.padding_slot = 0
96 self._free_slots: list[int] = list(range(1, num_slots))
97
98 def _alloc_slot_states(self, num_slots: int) -> dict[str, torch.Tensor]:
99 return {
100 spec.name: torch.full(
101 (max(1, len(spec.layer_ids)), num_slots, *spec.shape),
102 spec.fill_value,
103 dtype=spec.dtype if spec.dtype is not None else self._conv_dtype,
104 device=self._device,
105 )
106 for spec in self._slot_specs
107 }
108
109 def has_slot_state(self, name: str) -> bool:
110 return name in self.slot_states
111
112 def slot_state(self, name: str, layer_id: int | None = None) -> torch.Tensor:
113 """One declared sibling state, ``[num_slots, *shape]``; ``layer_id`` picks the layer row."""
114 t = self.slot_states[name]
115 if layer_id is None:
116 assert not self._state_layer_index[name], (
117 f"slot_state {name!r} is per-layer, pass layer_id"
118 )
119 return t[0]
120 return t[self._state_layer_index[name][layer_id]]
121
122 @property
123 def num_free_slots(self) -> int:
124 return len(self._free_slots)
125
126 def alloc(self, n: int = 1) -> list[int]:
127 """Pop ``n`` free slot ids (LIFO). Raises if the pool is exhausted."""
128 if n > len(self._free_slots):
129 raise RuntimeError(
130 f"LinearStatePool exhausted: need {n}, have {len(self._free_slots)}"
131 )
132 return [self._free_slots.pop() for _ in range(n)]
133
134 def reclaim_all_slots(self) -> None:
135 """Restore the free-list to all non-padding slots. Idle-only: the caller (e.g. a
136 CacheManager rebuild that discards the tree owning donated snapshots) must guarantee no
137 running request holds a slot, otherwise live state would be handed out twice."""
138 self._free_slots = list(range(1, self._num_slots))
139
140 def rebuild(self, num_slots: int) -> None:
141 """Reallocate the conv + recurrent state tensors for ``num_slots`` slots IN PLACE.
142
143 Geometry (layers, conv dim, head dims) and dtypes are taken from the existing
144 tensors; only the slot count changes. Object identity is preserved so cached
145 references (ctx.linear_state_pool) stay valid. Idle-only and destructive: every
146 live/snapshot state is dropped, so the caller must guarantee no running request
147 holds a slot and the radix tree owning donated snapshots is discarded too.
148 """
149 n_layers, _, local_conv_dim, km1 = self.conv_states.shape
150 _, _, local_v_heads, key_head_dim, value_head_dim = self.recurrent_states.shape
151 conv_dtype, rec_dtype = self.conv_states.dtype, self.recurrent_states.dtype
152 device = self._device
153 self.conv_states = None
154 self.recurrent_states = None
155 self.slot_states = {}
156 if device.type == "cuda":
157 torch.cuda.synchronize(device)
158 torch.cuda.empty_cache()
159 self.conv_states = torch.zeros(
160 (n_layers, num_slots, local_conv_dim, km1), dtype=conv_dtype, device=device
161 )
162 self.recurrent_states = torch.zeros(
163 (n_layers, num_slots, local_v_heads, key_head_dim, value_head_dim),
164 dtype=rec_dtype,
165 device=device,
166 )
167 self.slot_states = self._alloc_slot_states(num_slots)
168 self._num_slots = num_slots
169 self._free_slots = list(range(1, num_slots))
170
171 def free(self, slots) -> None:
172 """Return slot ids to the free-list. Accepts an int, list, or 1-D tensor."""
173 if isinstance(slots, torch.Tensor):
174 slots = slots.flatten().tolist()
175 elif isinstance(slots, int):
176 slots = [slots]
177 self._free_slots.extend(int(s) for s in slots)
178
179 def clear_slots(self, slots) -> None:
180 """Zero conv + recurrent state at ``slots`` across all linear layers (fresh sequence)."""
181 if isinstance(slots, (list, tuple)):
182 slots = torch.as_tensor(slots, dtype=torch.long, device=self._device)
183 self.conv_states[:, slots] = 0
184 self.recurrent_states[:, slots] = 0
185 for spec in self._slot_specs:
186 self.slot_states[spec.name][:, slots] = spec.fill_value
187
188 def copy_from(self, src: int, dst: int) -> None:
189 """Copy a whole-sequence snapshot (conv + recurrent, all layers) from slot ``src`` to
190 ``dst``. Used for COW-on-restore (donated snapshot -> fresh live slot)."""
191 self.conv_states[:, dst].copy_(self.conv_states[:, src])
192 self.recurrent_states[:, dst].copy_(self.recurrent_states[:, src])
193 for t in self.slot_states.values():
194 t[:, dst].copy_(t[:, src])
195
196 def is_linear_layer(self, layer_id: int) -> bool:
197 return layer_id in self._local_index
198
199 def local_index(self, layer_id: int) -> int:
200 return self._local_index[layer_id]
201
202 def conv_state(self, layer_id: int, table_idx: int) -> torch.Tensor:
203 return self.conv_states[self._local_index[layer_id], table_idx]
204
205 def recurrent_state(self, layer_id: int, table_idx: int) -> torch.Tensor:
206 return self.recurrent_states[self._local_index[layer_id], table_idx]
207
208 def reset(self, table_idx: int) -> None:
209 """Zero a slot across all linear layers (new request takes this table_idx)."""
210 self.conv_states[:, table_idx].zero_()
211 self.recurrent_states[:, table_idx].zero_()
212 for spec in self._slot_specs:
213 self.slot_states[spec.name][:, table_idx] = spec.fill_value
214
215 @property
216 def num_linear_layers(self) -> int:
217 return len(self._local_index)
218
219 @property
220 def num_slots(self) -> int:
221 return self._num_slots
222
223 @property
224 def device(self) -> torch.device:
225 return self._device
226
227 def bytes_per_slot(self) -> int:
228 """Total state bytes for one request (all linear layers)."""
229 per = (
230 self.conv_states[:, 0].numel() * self.conv_states.element_size()
231 + self.recurrent_states[:, 0].numel() * self.recurrent_states.element_size()
232 )
233 for t in self.slot_states.values():
234 per += t[:, 0].numel() * t.element_size()
235 return int(per)

238def linear_state_bytes_per_req(
239 group: LinearGatedDeltaGroupConfig,
240 tp_size: int,
241 dtype: torch.dtype,
242 slot_states: tuple[SlotStateSpec, ...] = (),
243) -> int:
244 """Linear-state bytes for one request across all linear layers (TP-local), plus any
245 declared slot_states."""
246 n_layers, local_conv_dim, local_v_heads = _linear_local_dims(group, tp_size)
247
248 conv_elems = local_conv_dim * (group.conv_kernel_dim - 1)
249 rec_elems = local_v_heads * group.key_head_dim * group.value_head_dim
250 conv_bytes = conv_elems * dtype.itemsize # conv state in model dtype
251 rec_bytes = rec_elems * ssm_state_dtype().itemsize # recurrent state (default fp32)
252 total = n_layers * (conv_bytes + rec_bytes)
253
254 for spec in slot_states:
255 item = (spec.dtype if spec.dtype is not None else dtype).itemsize
256 total += max(1, len(spec.layer_ids)) * math.prod(spec.shape) * item
257 return int(total)

263def state_pool_bytes(config, num_slots: int | None = None) -> int:
264 """Total GDN state-pool bytes at ``num_slots`` PHYSICAL slots (default: the startup
265 slot count). The engine adds this to the KV family's fixed cost when budgeting --
266 the state pool is a sibling pool, not a KV tier."""
267 linear_group = config.model_config.linear_attention_group()
268 slot_states = getattr(config.model_config, "slot_states", ())
269 if linear_group is None:
270 if slot_states:
271 raise ValueError("slot_states ride the linear-state slots; model has no linear group")
272 return 0
273 slots = num_slots if num_slots is not None else _linear_pool_num_slots(config)
274 per_req = linear_state_bytes_per_req(
275 linear_group, config.tp_info.size, config.dtype, slot_states
276 )
277 return per_req * slots

280def _linear_pool_num_slots(config) -> int:
281 """LinearStatePool slot count. Hybrid-radix non-evictable peak is 4 slots per running request
282 (1 live + 2 ping-pong + 1 committed snapshot locked through decode), plus a cross-request
283 snapshot cache and a padding sink; naive GDN keeps the old (max_running_req + 1)."""
284 mr = config.max_running_req
285 if config.cache_type != "hybrid_radix":
286 return mr + 1 # live + dummy/padding
287 ratio = config.linear_state_cache_ratio
288 n_cache = max(4, int(ratio * mr))
289 return 4 * mr + n_cache + 1 # live + 2 ping-pong + locked committed snapshot + cache + padding

292def _linear_pool_min_slots(config) -> int:
293 """Floor on LinearStatePool slots that still runs: the non-evictable working set with a
294 zero snapshot cache. Hybrid-radix needs 4 per running request (1 live + 2 ping-pong + 1
295 committed snapshot locked through decode) + the padding sink; naive needs 1 per request +
296 padding. Below this, a full max_running_req batch can't get its slots and admission
297 deadlocks -- so a runtime rebuild rejects a smaller request."""
298 mr = config.max_running_req
299 if config.cache_type != "hybrid_radix":
300 return mr + 1
301 return 4 * mr + 1
