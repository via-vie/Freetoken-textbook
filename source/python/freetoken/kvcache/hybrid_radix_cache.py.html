1"""Hybrid (full-attn KV + GDN linear-state) radix cache.
2
3A SEPARATE class from ``RadixPrefixCache`` (Option C) that REUSES the shared ``RadixTreeNode``
4and walk/split logic, so the production KV radix is untouched (zero risk to non-hybrid models).
5It adds a second "currency": an optional GDN state snapshot (``node.mamba_value`` = a
6LinearStatePool slot id) attached at chunk/page-aligned boundary nodes, with its own LRU
7eviction. Mirrors sglang ``MambaRadixCache`` (donate-not-copy, dual eviction, internal-node
8tombstone, ``full_ref >= mamba_ref``) on FreeToken's tree.
9
10Currency seam: the secondary value + its eviction is the slot a future SWA component plugs
11into. This class is pool-agnostic -- it stores/returns slot ids and KV page indices; the
12caller (CacheManager / scheduler) does the actual LinearStatePool / KV-pool free.
13"""

29@dataclass(frozen=True)
30class HybridCacheHandle(BaseCacheHandle):
31 """Lock handle for a matched hybrid prefix: the matched node (lock target) + the reusable
32 KV page indices. ``cached_len`` is already truncated to the deepest live-snapshot boundary.
33 Plugs into PrefillAdder (reads ``.cached_len`` / ``.get_matched_indices()``) like the plain
34 RadixCacheHandle; the restore slot rides on ``MatchResult.mamba_value``."""
35
36 node: RadixTreeNode
37 kv_indices: torch.Tensor
38
39 def get_matched_indices(self) -> torch.Tensor:
40 return self.kv_indices

43class HybridMatch(NamedTuple):
44 kv_indices: torch.Tensor # reused KV page indices for [0:cached_len)
45 cached_len: int # truncated to the deepest LIVE-snapshot boundary
46 mamba_value: Optional[int] # GDN snapshot slot to restore from (None = cold start)
47 node: RadixTreeNode # the matched node (lock target)

50class EvictResult(NamedTuple):
51 kv_indices: torch.Tensor # KV page indices to free
52 mamba_slots: List[int] # GDN state slots to free

55class HybridRadixCache:
56 def __init__(self, device: torch.device, page_size: int) -> None:
57 from freetoken.kernel.fla.chunk import CHUNK_SIZE
58 # Snapshots land on ×CHUNK_SIZE boundaries; require them to be page-aligned so the KV
59 # node boundary and the GDN-state boundary coincide (page_size in {1,2,4,8,16,32,64}).
60 assert CHUNK_SIZE % page_size == 0, (
61 f"hybrid_radix needs CHUNK_SIZE({CHUNK_SIZE}) % page_size({page_size}) == 0"
62 )
63 self.device = device
64 self.page_size = page_size
65 self.key_fn = _get_key_fn(page_size)
66 self.empty = torch.empty(0, dtype=torch.int32, device=device)
67 self.root = RadixTreeNode(self.key_fn)
68 self.root.set_key_value(self.empty, self.empty)
69 self.root.ref_count = 1 # root is always protected
70 self.full_evictable = 0
71 self.full_protected = 0
72 self.mamba_evictable = 0 # number of live, unlocked snapshots
73 self.mamba_protected = 0
74
75 # ---------------------------------------------------------------- match / insert
76 def match_prefix(self, input_ids: torch.Tensor) -> HybridMatch:
77 """Match the token prefix, then truncate the reusable length to the deepest node on
78 the path that still owns a LIVE snapshot (a continuation can only resume the GDN
79 recurrence from a checkpointed boundary)."""
80 node, _ = self._walk(input_ids)
81 # walk up to the deepest node whose END boundary has a live snapshot
82 cur, end_len = node, self._path_len(node)
83 while not cur.is_root():
84 if cur.mamba_value is not None:
85 return HybridMatch(self._collect_kv(cur), end_len, cur.mamba_value, cur)
86 end_len -= cur.length
87 cur = cur.parent
88 return HybridMatch(self.empty, 0, None, self.root)
89
90 def insert(self, input_ids: torch.Tensor, kv_indices: torch.Tensor,
91 mamba_value: int) -> Tuple[int, bool]:
92 """Insert the committed KV prefix and DONATE ``mamba_value`` at the (page-aligned) end
93 boundary node. Returns (matched_prefix_len, mamba_exist). If the boundary node already
94 owns a live snapshot, returns mamba_exist=True and does not attach (caller frees the
95 donated slot -- dedup)."""
96 insert_len = align_down(len(input_ids), self.page_size)
97 input_ids, kv_indices = input_ids[:insert_len], kv_indices[:insert_len]
98 node, prefix_len = self._walk(input_ids)
99 if prefix_len != insert_len:
100 new_node = RadixTreeNode(self.key_fn)
101 # A view would keep the request's full output-budget token buffer alive.
102 new_node.set_key_value(
103 input_ids[prefix_len:].clone(), kv_indices[prefix_len:].clone()
104 )
105 new_node.set_parent(node)
106 self.full_evictable += new_node.length
107 node = new_node
108 if node.is_root():
109 return prefix_len, True # root can't hold a snapshot; report exist so caller frees it
110 if node.mamba_value is not None:
111 return prefix_len, True # dedup: caller frees its donated slot
112 node.mamba_value = mamba_value # fills a fresh node or a tombstone
113 if node.mamba_ref_count == 0:
114 self.mamba_evictable += 1
115 return prefix_len, False
116
117 # ---------------------------------------------------------------- locking (dual)
118 def inc_lock(self, node: RadixTreeNode) -> None:
119 """Protect a matched node's snapshot (mamba ref on the node) and its KV path
120 (full ref node..root). Enforces full_ref >= mamba_ref: using a snapshot at N pins the
121 whole root..N KV chain."""
122 if node.mamba_value is not None:
123 if node.mamba_ref_count == 0:
124 self.mamba_evictable -= 1
125 self.mamba_protected += 1
126 node.mamba_ref_count += 1
127 cur = node
128 while not cur.is_root():
129 if cur.ref_count == 0:
130 self.full_evictable -= cur.length
131 self.full_protected += cur.length
132 cur.ref_count += 1
133 cur = cur.parent
134
135 def dec_lock(self, node: RadixTreeNode) -> None:
136 if node.mamba_value is not None and node.mamba_ref_count > 0:
137 node.mamba_ref_count -= 1
138 if node.mamba_ref_count == 0:
139 self.mamba_evictable += 1
140 self.mamba_protected -= 1
141 cur = node
142 while not cur.is_root():
143 cur.ref_count -= 1
144 assert cur.ref_count >= 0
145 if cur.ref_count == 0:
146 self.full_evictable += cur.length
147 self.full_protected -= cur.length
148 cur = cur.parent
149
150 # ---------------------------------------------------------------- eviction (dual)
151 def evict_full(self, num_tokens: int) -> EvictResult:
152 """Evict KV tokens by LRU over UNLOCKED LEAF nodes (an internal node's KV is a prefix
153 dependency for all descendants). Frees each evicted node's snapshot too."""
154 leaves = [n for n in self._leaves() if n.ref_count == 0]
155 heapq.heapify(leaves)
156 kv, mamba, freed = [], [], 0
157 while freed < num_tokens and leaves:
158 node = heapq.heappop(leaves)
159 if node.ref_count != 0 or not node.is_leaf() or node.is_root():
160 continue
161 freed += node.length
162 kv.append(node.value)
163 self.full_evictable -= node.length
164 self._free_node_mamba(node, mamba)
165 parent, casc = self._cascade_tombstone_leaves(self._unlink(node), kv)
166 freed += casc
167 if parent.is_leaf() and parent.ref_count == 0 and not parent.is_root():
168 heapq.heappush(leaves, parent)
169 return EvictResult(torch.cat(kv) if kv else self.empty, mamba)
170
171 def evict_mamba(self, num: int) -> EvictResult:
172 """Evict GDN snapshots by LRU over UNLOCKED snapshot-bearing nodes -- internal nodes
173 too. Internal node -> TOMBSTONE (free the slot, keep KV + children). Leaf node -> free
174 both KV and slot and unlink, then cascade-delete any KV-only tombstone leaves it exposes
175 upward (so a leaf always carries a live snapshot -- mirrors sglang)."""
176 cands = [n for n in self._snapshot_nodes() if n.mamba_ref_count == 0]
177 heapq.heapify(cands)
178 kv, mamba, freed = [], [], 0
179 while freed < num and cands:
180 node = heapq.heappop(cands)
181 if node.mamba_value is None or node.mamba_ref_count != 0 or node.is_root():
182 continue
183 if node.is_leaf() and node.ref_count == 0:
184 kv.append(node.value)
185 self.full_evictable -= node.length
186 self._free_node_mamba(node, mamba)
187 freed += 1
188 self._cascade_tombstone_leaves(self._unlink(node), kv)
189 else:
190 self._free_node_mamba(node, mamba) # tombstone internal (or locked-KV) node
191 freed += 1
192 return EvictResult(torch.cat(kv) if kv else self.empty, mamba)

209 def check_integrity(self) -> None:
210 # Structural: every snapshot-bearing node holds a real slot id; ref counts non-negative.
211 # (KV/page conservation is checked by CacheManager.check_integrity.)
212 for n in self._snapshot_nodes():
213 assert n.mamba_value is not None and n.mamba_ref_count >= 0 and n.ref_count >= 0

215 # ---------------------------------------------------------------- helpers
216 def _free_node_mamba(self, node: RadixTreeNode, out: List[int]) -> None:
217 if node.mamba_value is not None:
218 out.append(node.mamba_value)
219 node.mamba_value = None
220 if node.mamba_ref_count == 0:
221 self.mamba_evictable -= 1
222
223 def _unlink(self, node: RadixTreeNode) -> RadixTreeNode:
224 parent = node.parent
225 del parent.children[self.key_fn(node._key)]
226 return parent
227
228 def _cascade_tombstone_leaves(self, parent: RadixTreeNode, kv_out: List[torch.Tensor]):
229 """After a leaf is unlinked, eagerly reclaim the KV-only tombstone leaves it exposes
230 upward (mamba_value None, no children, unlocked): free their KV and unlink, walking up.
231 Keeps the 'a leaf always carries a live snapshot' invariant (sglang
232 _iteratively_delete_tombstone_leaf). Returns (highest surviving ancestor, freed_tokens)."""
233 freed = 0
234 while (parent.mamba_value is None and parent.is_leaf()
235 and parent.ref_count == 0 and not parent.is_root()):
236 kv_out.append(parent.value)
237 self.full_evictable -= parent.length
238 freed += parent.length
239 parent = self._unlink(parent)
240 return parent, freed
241
242 def _path_len(self, node: RadixTreeNode) -> int:
243 n, total = node, 0
244 while not n.is_root():
245 total += n.length
246 n = n.parent
247 return total
248
249 def _collect_kv(self, node: RadixTreeNode) -> torch.Tensor:
250 vals: List[torch.Tensor] = []
251 n = node
252 while not n.is_root():
253 vals.append(n.value)
254 n = n.parent
255 vals.reverse()
256 return torch.cat(vals) if vals else self.empty
257
258 def _leaves(self) -> List[RadixTreeNode]:
259 out, stack = [], [self.root]
260 while stack:
261 n = stack.pop()
262 if n.is_leaf():
263 if not n.is_root():
264 out.append(n)
265 else:
266 stack.extend(n.children.values())
267 return out
268
269 def _snapshot_nodes(self) -> List[RadixTreeNode]:
270 out, stack = [], [self.root]
271 while stack:
272 n = stack.pop()
273 if n.mamba_value is not None and not n.is_root():
274 out.append(n)
275 stack.extend(n.children.values())
276 return out
277
278 def _walk(self, input_ids: torch.Tensor) -> Tuple[RadixTreeNode, int]:
279 prefix_len, total = 0, len(input_ids)
280 node = self.root
281 tic = time.monotonic_ns()
282 while prefix_len < total:
283 child = node.children.get(self.key_fn(input_ids[prefix_len:]))
284 if child is None:
285 return node, prefix_len
286 node = child
287 match_len = align_down(node.get_match_len(input_ids[prefix_len:]), self.page_size)
288 prefix_len += match_len
289 if match_len != node.length:
290 node = node.split_at(match_len)
291 node.timestamp = tic
292 return node, prefix_len
293 node.timestamp = tic
294 return node, prefix_len
