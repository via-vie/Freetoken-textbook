29 # Secondary "currency" for hybrid models (HybridRadixCache): an optional GDN state
30 # snapshot slot attached at this node's END boundary, with its own ref count. None
31 # for plain KV radix. ``split_at`` leaves the new prefix node's ``mamba_value`` None
32 # ("a snapshot cannot be split") -- the snapshot stays on the original (suffix) node,
33 # whose end boundary is unchanged. Forward-compat seam for SWA.
34 self.mamba_value: int | None = None
35 self.mamba_ref_count: int = 0
36
37 # SWA second currency (SWARadixCache). Unlike the GDN snapshot above, SWA stores NO
38 # separate slot: ``value`` (full-pool page indices) is canonical and the swa KV is
39 # reached via the pool's full->swa mapping. ``swa_tombstone`` marks that the swa KV for
40 # these tokens was freed (slid out of window) while the full KV survives; ``swa_ref_count``
41 # is the trailing-window lock; ``swa_uuid`` is the window-boundary handle (lock release).
42 # All default off -> the plain KV radix and the GDN hybrid radix are unaffected.
43 self.swa_tombstone: bool = False
44 self.swa_ref_count: int = 0
45 self.swa_uuid: int | None = None

87 def split_at(self, pos: int) -> RadixTreeNode:
88 assert 0 < pos < self.length
89 parent = self.parent
90
91 new_node = RadixTreeNode(self.key_fn, self.timestamp)
92 new_node.set_key_value(self._key[:pos], self._value[:pos])
93 new_node.set_parent(parent)
94 new_node.ref_count = self.ref_count
95 # SWA: a tombstone covers all the node's tokens, so both halves inherit it; both halves
96 # stay within a locked window, so the swa lock copies; the window-boundary uuid migrates
97 # to the root-side (prefix) node and is cleared on the suffix (matches sglang _split_node).
98 new_node.swa_ref_count = self.swa_ref_count
99 new_node.swa_tombstone = self.swa_tombstone
100 new_node.swa_uuid = self.swa_uuid
101 self.swa_uuid = None
102
103 self.set_key_value(self._key[pos:], self._value[pos:])
104 self.set_parent(new_node)
105
106 return new_node

126class RadixPrefixCache(BasePrefixCache):
127 def __init__(self, device: torch.device, page_size: int | None = None):
128 super().__init__()
129 self.device = device
130 # Explicit page_size beats the global ctx: a manager constructed with page_size!=ctx
131 # (tests, future multi-granularity) must not silently split the two (the tree would
132 # align by one granularity while the manager frees by the other -> orphaned pages).
133 self.page_size = get_global_ctx().page_size if page_size is None else int(page_size)
134 self.key_fn = _get_key_fn(self.page_size)
135 self.empty_tensor = torch.empty(0, dtype=torch.int32, device=device)
136 self.evictable_size = 0
137 self.protected_size = 0
138 self.root_node = RadixTreeNode(self.key_fn)
139 self.root_node.ref_count = 1 # root is always protected
140
141 def lock_handle(self, handle: BaseCacheHandle, unlock: bool = False) -> None:
142 assert isinstance(handle, RadixCacheHandle)
143 node = handle.node
144 if unlock:
145 while not node.is_root():
146 node.ref_count -= 1
147 assert node.ref_count >= 0
148 if node.ref_count == 0:
149 self.evictable_size += node.length
150 self.protected_size -= node.length
151 node = node.parent
152 else:
153 while not node.is_root():
154 if node.ref_count == 0:
155 self.evictable_size -= node.length
156 self.protected_size += node.length
157 node.ref_count += 1
158 node = node.parent
159
160 def match_prefix(self, input_ids: torch.Tensor) -> MatchResult:
161 node, prefix_len = self._tree_walk(input_ids)
162 return MatchResult(RadixCacheHandle(prefix_len, node))
163
164 def insert_prefix(self, input_ids: torch.Tensor, indices: torch.Tensor) -> InsertResult:
165 insert_len = align_down(len(input_ids), self.page_size)
166 input_ids, indices = input_ids[:insert_len], indices[:insert_len]
167 node, prefix_len = self._tree_walk(input_ids)
168 if prefix_len != insert_len: # NOTE: prefix_len < insert_len
169 new_node = RadixTreeNode(self.key_fn)
170 # A view would keep the request's full output-budget token buffer alive.
171 new_node.set_key_value(
172 input_ids[prefix_len:].clone(), indices[prefix_len:].clone()
173 )
174 new_node.set_parent(node)
175 self.evictable_size += new_node.length
176 node = new_node
177 return InsertResult(prefix_len, RadixCacheHandle(insert_len, node))

179 def evict(self, size: int) -> torch.Tensor:
180 if size == 0:
181 return self.empty_tensor
182 assert (
183 size <= self.evictable_size
184 ), f"Cannot evict {size}, only {self.evictable_size} is evictable"
185
186 leave_nodes = self._collect_leave_nodes_for_evict()
187 heapq.heapify(leave_nodes)
188 evicted_indices: List[torch.Tensor] = []
189 evicted_size = 0
190
191 while evicted_size < size:
192 assert (
193 leave_nodes
194 ), f"Cannot evict enough cache, need {size}, only {evicted_size} evicted"
195 node = heapq.heappop(leave_nodes)
196 assert node.ref_count == 0 and node.is_leaf() and not node.is_root()
197 evicted_size += node.length
198 evicted_indices.append(node.value)
199 self.evictable_size -= node.length
200 parent = node.parent
201 del parent.children[self.key_fn(node._key)]
202 # NOTE: root is always protected, so won't be evicted
203 if parent.is_leaf() and parent.ref_count == 0:
204 heapq.heappush(leave_nodes, parent)
205
206 return torch.cat(evicted_indices)

221 def _collect_leave_nodes_for_evict(self) -> List[RadixTreeNode]:
222 nodes: List[RadixTreeNode] = [self.root_node]
223 leave_nodes: List[RadixTreeNode] = []
224
225 while len(nodes) > 0:
226 node = nodes.pop()
227 if node.is_leaf():
228 if node.ref_count == 0:
229 leave_nodes.append(node)
230 else:
231 for child in node.children.values():
232 nodes.append(child)
233
234 return leave_nodes

236 def _tree_walk(self, input_ids: torch.Tensor) -> Tuple[RadixTreeNode, int]:
237 prefix_len = 0
238 indice_len = len(input_ids)
239 node = self.root_node
240 tic = time.monotonic_ns()
241
242 while prefix_len < indice_len:
243 child_node = node.children.get(self.key_fn(input_ids[prefix_len:]))
244 if child_node is None:
245 return node, prefix_len
246 node = child_node # walk to child node
247
248 # NOTE: at least 1 page is matched, so match_len >= page_size
249 match_len = node.get_match_len(input_ids[prefix_len:])
250 match_len = align_down(match_len, self.page_size)
251 prefix_len += match_len
252
253 # need to split the node if not fully matched
254 if match_len != node.length:
255 node = node.split_at(match_len)
256 node.timestamp = tic
257 return node, prefix_len
258
259 # update timestamp for accessed node
260 node.timestamp = tic
261
262 return node, prefix_len
