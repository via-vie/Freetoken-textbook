30class LLM(Scheduler):
31 def __init__(self, model_path: str, dtype: torch.dtype = torch.bfloat16, **kwargs):
32 config = SchedulerConfig(
33 model_path=model_path,
34 tp_info=DistributedInfo(0, 1),
35 dtype=dtype,
36 offline_mode=True,
37 **kwargs,
38 )
39 super().__init__(config)
40 self.pending_requests: List[Tuple[List[int] | str, SamplingParams, List[bytes] | None]] = []
41 self.status_map: Dict[int, RequestStatus] = {}
42 self.counter = 0

47 def _tokenize_one(self, prompt: List[int] | str) -> torch.Tensor:
48 if isinstance(prompt, str):
49 return self.tokenizer.encode(prompt, return_tensors="pt").view(-1).to(torch.int32)
50 else:
51 return torch.tensor(prompt, dtype=torch.int32, device="cpu")

53 def offline_receive_msg(self, blocking: bool = False) -> List[BaseBackendMsg]:
54 if blocking and len(self.pending_requests) == 0:
55 raise RequestAllFinished()
56 results: List[BaseBackendMsg] = []
57 added, sum_input_len = 0, 0
58 for tokens_or_prompt, sampling_params, images in self.pending_requests:
59 if sum_input_len >= self.prefill_budget:
60 break
61 input_ids = self._tokenize_one(tokens_or_prompt)
62 msg = UserMsg(uid=0, input_ids=input_ids, sampling_params=sampling_params)
63 if images:
64 if self._mm_processor is None:
65 raise ValueError("image input is not supported for this model")
66 r = self._mm_processor.apply(input_ids, images)
67 input_ids = r.input_ids
68 msg = UserMsg(
69 uid=0,
70 input_ids=input_ids,
71 sampling_params=sampling_params,
72 mm_items=r.mm_items,
73 mrope_positions=r.mrope_positions,
74 mrope_delta=r.mrope_delta,
75 )
76 sum_input_len += len(input_ids)
77 uid, added = self.counter + added, added + 1
78 msg.uid = uid
79 results.append(msg)
80 self.status_map[uid] = RequestStatus(
81 uid=uid,
82 input_ids=input_ids.tolist(),
83 output_ids=[],
84 )
85 self.counter += added
86 self.pending_requests = self.pending_requests[added:]
87 return results

89 def offline_send_result(self, reply: List[BaseTokenizerMsg]) -> None:
90 for msg in reply:
91 if isinstance(msg, PromptAdmittedMsg):
92 # PromptAdmittedMsg feeds the online server's global accounting. Offline
93 # generation already owns its inputs and has no FrontendManager stats sink.
94 continue
95 assert isinstance(msg, DetokenizeMsg)
96 status = self.status_map[msg.uid]
97 if not (msg.finished and msg.next_token in self.eos_token_ids):
98 status.output_ids.append(msg.next_token)

100 def generate(
101 self,
102 prompts: List[str] | List[List[int]],
103 sampling_params: List[SamplingParams] | SamplingParams,
104 images: List[List[bytes] | None] | None = None,
105 ) -> List[Dict[str, str | List[int]]]:
106 """Offline generation; images is aligned with prompts: the raw image files of each prompt in placeholder order, or None."""
107 self.pending_requests = []
108 self.status_map = {}
109 self.counter = 0
110 if isinstance(sampling_params, SamplingParams):
111 sampling_params = [sampling_params] * len(prompts)
112 if images is None:
113 images = [None] * len(prompts)
114 for prompt, sp, imgs in zip(prompts, sampling_params, images, strict=True):
115 self.pending_requests.append((prompt, sp, imgs))
116 try:
117 self.run_forever()
118 except RequestAllFinished:
119 pass
120 results: List[Dict[str, str | List[int]]] = []
121 for i in range(len(prompts)):
122 status = self.status_map[i]
123 output_text = self.tokenizer.decode(status.output_ids)
124 results.append({"text": output_text, "token_ids": status.output_ids})
125 return results
