20@dataclass
21class PromptReplacement:
22 """The token sequence that replaces one placeholder; is_embed marks the positions that take image embeddings (None = all of them)."""
23
24 full: list[int]
25 is_embed: list[bool] | None = None
26
27 @classmethod
28 def select_token_id(cls, full: list[int], embed_token_id: int) -> PromptReplacement:
29 return cls(full, [t == embed_token_id for t in full])

31 def embed_spans(self) -> list[list[int]]:
32 """Half-open [lo, hi) runs of embedding positions, relative to the start of full."""
33 mask = self.is_embed if self.is_embed is not None else [True] * len(self.full)
34 spans: list[list[int]] = []
35 lo: int | None = None
36 for i, flag in enumerate([*mask, False]):
37 if flag and lo is None:
38 lo = i
39 elif not flag and lo is not None:
40 spans.append([lo, i])
41 lo = None
42 return spans

53def _find_all(ids: list[int], target: list[int]) -> list[int]:
54 """Start index of every non-overlapping occurrence of target."""
55 n = len(target)
56 found: list[int] = []
57 i = 0
58 while i <= len(ids) - n:
59 if ids[i : i + n] == target:
60 found.append(i)
61 i += n
62 else:
63 i += 1
64 return found

67class MMProcessor(ABC):
68 """How a checkpoint's images become items, tokens and positions; apply is the family-agnostic driver, subclasses fill the hooks.
69
70 Built from the checkpoint config alone and import-light: tokenizer workers call apply without the model layer, the engine calls dummy_items.
71 Image preprocessing must stay on the CPU with one fixed implementation: radix keys and the embedding cache are content hashes of pixel_values, and resize results differ across backends/devices.
72 """
73
74 # the token sequence the chat template renders once per image
75 placeholder: list[int]

77 def __init__(self, model_path: str, mm: MultimodalConfig) -> None:

79 self.mm = mm
80 self._image_processor_instance: Any = None

84 """The checkpoint's image processor, loaded on first use (tokenizer workers share the instance across threads)."""

87 from transformers import AutoImageProcessor

112 def apply(self, input_ids: torch.Tensor, images: list[bytes]) -> MMResult:
113 """Swap each image placeholder for its replacement sequence and precompute rope positions."""
114 from PIL import Image
115
116 ids = input_ids.tolist()
117 target = self.placeholder
118 slots = _find_all(ids, target)
119 if len(slots) != len(images):
120 raise ValueError(
121 f"prompt renders {len(slots)} image placeholders but the request "
122 f"carries {len(images)} images"
123 )
124
125 pils = [Image.open(io.BytesIO(raw)).convert("RGB") for raw in images]
126 items = self.process(pils)
127 out: list[int] = []
128 cursor = 0
129 for slot, item in zip(slots, items):
130 repl = self.prompt_replacement(item)
131 out.extend(ids[cursor:slot])
132 base = len(out)
133 full = list(repl.full)
134 spans = repl.embed_spans()
135 # embedding slots carry the content pad id so radix keys and the model's scatter mask see the image
136 for lo, hi in spans:
137 full[lo:hi] = [item.pad_value] * (hi - lo)
138 out.extend(full)
139 item.offsets = [[base + lo, base + hi] for lo, hi in spans]
140 item.validate()
141 cursor = slot + len(target)
142 out.extend(ids[cursor:])
143 new_ids = torch.tensor(out, dtype=input_ids.dtype)
144
145 positions, delta = self.positions(len(out), items) or (None, 0)
146 return MMResult(new_ids, items, positions, delta)

149def image_positions(
150 length: int, blocks: list[tuple[int, int, int, int]]
151) -> tuple[torch.Tensor, int]:
152 """t/h/w positions: text advances all three rows, an image freezes t and spreads h/w over its grid, then the running position jumps by max(h, w).
153
154 blocks are (offset, n_tokens, grid_h, grid_w) in prompt order; delta shifts decode positions (sequence index + delta)."""
155 pos = torch.empty((3, length), dtype=torch.int32)
156 st = 0
157 cursor = 0
158 for offset, n_tokens, h_m, w_m in blocks:
159 run = offset - cursor
160 if run:
161 text = torch.arange(st, st + run, dtype=torch.int32)
162 pos[:, cursor:offset] = text
163 st += run
164 cursor = offset
165 idx = torch.arange(n_tokens, dtype=torch.int32)
166 pos[0, cursor : cursor + n_tokens] = st
167 pos[1, cursor : cursor + n_tokens] = st + idx // w_m
168 pos[2, cursor : cursor + n_tokens] = st + idx % w_m
169 st += max(h_m, w_m)
170 cursor += n_tokens
171 run = length - cursor
172 if run:
173 pos[:, cursor:] = torch.arange(st, st + run, dtype=torch.int32)
174 st += run
175 delta = int(pos.max().item()) + 1 - length
176 return pos, delta

179def content_hash(feature: torch.Tensor, extra: bytes = b"") -> int:
180 """62-bit hash of the wire bytes plus family extras (e.g. the grid, so transposed images differ)."""
181 buf = feature.contiguous()
182 if buf.dtype == torch.bfloat16:
183 buf = buf.view(torch.uint16)
184 hasher = hashlib.sha256(memoryview(buf.numpy()))
185 hasher.update(extra)
186 return int.from_bytes(hasher.digest()[:8], "little") % (1 << 62)

189def get_mm_processor(model_path: str, mm: MultimodalConfig | None = None) -> MMProcessor | None:
190 """A processor for the checkpoint under the mm knobs, or None when no encoder is left to serve: the family registers none, the checkpoint ships none, or mm.disabled_encoders names them all; callers keep the instance, it holds the loaded image processor."""
191 from freetoken.models.register import get_model_spec
192 from freetoken.utils import cached_load_hf_config
193
194 try:
195 config = cached_load_hf_config(model_path)
196 spec = get_model_spec((config.architectures or [None])[0])
197 except Exception: # noqa: BLE001 -- missing/foreign config or unknown architecture: no multimodal input
198 return None
199 mm = mm or MultimodalConfig()
200 served = [e for e in spec.encoders if getattr(config, e.config_key, None) is not None and e.kind not in mm.disabled_encoders]
201 if spec.mm_processor is None or not served:
202 return None
203 # families with a flat checkpoint config (deepseek_v4) carry vocab_size at the top level
204 check_mm_pad_shift(getattr(config, "text_config", config).vocab_size)
205 module, _, cls = spec.mm_processor.partition(":")
206 return getattr(importlib.import_module(module), cls)(config, model_path, mm)
