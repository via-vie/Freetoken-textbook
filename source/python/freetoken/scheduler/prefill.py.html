43@dataclass
44class PrefillAdder:
45 token_budget: int
46 reserved_size: int
47 cache_manager: CacheManager
48 table_manager: TableManager
49 encoder_cache: EncoderCache | None = None
50 # end a chunk before an image it would cut; only models whose image spans attend in both directions need it
51 keep_images_whole: bool = False
52 # the whole budget of this pass; token_budget shrinks as requests are admitted
53 pass_budget: int = 0
54 # SWA-pool tokens charged to reqs admitted so far this pass. Mirrors reserved_size: swa is
55 # allocated only in allocate_paged (after the pass), so swa_available_size does not decrement
56 # across the admission loop -- without this, successive admits all see the full pool.
57 reserved_swa: int = 0

63 def _kv_reservation_size(self, total_len: int, cached_len: int) -> int:
64 """Return the token-equivalent cost of the additional KV pages for a request."""
65 page_size = self.cache_manager.page_size
66 return (
67 div_ceil(total_len, page_size) - div_ceil(cached_len, page_size)
68 ) * page_size

70 def _try_allocate_one(self, req: PendingReq):
71 if self.table_manager.available_size == 0:
72 return None
73
74 # TODO: consider host cache match case (it needs the same replay cap)
75 replay = self.cache_manager.prefix_replay_tokens
76 mr = self.cache_manager.match_req(req, max_len=max(0, req.input_len - replay) if replay else None)
77 handle = mr.cuda_handle
78 cached_len = handle.cached_len
79 # TODO: better estimate policy
80 extend_len = req.input_len - cached_len
81 estimated_size = self._kv_reservation_size(
82 req.input_len + req.output_len, cached_len
83 )
84
85 if estimated_size + self.reserved_size > self.cache_manager.available_size:
86 return None
87 self.cache_manager.lock(handle)
88 if estimated_size + self.reserved_size > self.cache_manager.available_size:
89 return self.cache_manager.unlock(handle)
90
91 # Second currency (hybrid GDN): reserve 1 live + 2 ping-pong state slots; evict tree
92 # snapshots if the pool is short, fail admission if still short (mirrors the KV gate).
93 if self.cache_manager.is_hybrid:
94 pool = self.cache_manager.linear_state_pool
95 if pool.num_free_slots < 3:
96 self.cache_manager.ensure_mamba_slots(3)
97 if pool.num_free_slots < 3:
98 return self.cache_manager.unlock(handle)
99
100 # Third currency (SWA): refuse admission unless the swa pool can seat this request's first
101 # chunk / one window (the per-chunk charge is in _add_one_req; the reclaim -- radix
102 # evict_swa -- happens in allocate_paged, so no ensure here; swa_available_size already
103 # folds the evictable tree). For naive (no tree) this can only refuse, which is correct.
104 if self.cache_manager.swa_paged:
105 ps = self.cache_manager.page_size
106 # swa is charged per WHOLE page (allocate_paged -> alloc_swa), so the seat check is
107 # in page units too; identical at page_size==1.
108 need_swa = div_ceil(
109 min(max(extend_len, 1), self.cache_manager.sliding_window_size) + 1, ps
110 ) * ps
111 if self.cache_manager.swa_available_size - self.reserved_swa < need_swa:
112 return self.cache_manager.unlock(handle)
113
114 table_idx = self.table_manager.allocate()
115 if cached_len > 0: # NOTE: set the cached part
116 device_ids = self.table_manager.token_pool[table_idx][:cached_len]
117 device_ids.copy_(_maybe_pinned(req.input_ids[:cached_len]), non_blocking=True)
118 # Write the matched indices into the TAIL of the page_entry: a cache may return
119 # fewer matched indices than cached_len, in which case only the trailing n slots are
120 # known-live. Today both the generic radix and the SWA radix match a prefix whose
121 # full-loc row is entirely live (n == cached_len), so the tail IS the whole prefix.
122 # (DSV4 reads this table too: its pool's full_loc_map is attached to it.)
123 matched = handle.get_matched_indices()
124 n = int(matched.numel())
125 self.table_manager.page_table[table_idx][cached_len - n : cached_len].copy_(matched)
126
127 linear_slot_idx = ping_pong = None
128 if self.cache_manager.is_hybrid:
129 pool = self.cache_manager.linear_state_pool
130 linear_slot_idx = pool.alloc(1)[0]
131 ping_pong = tuple(pool.alloc(2))
132
133 return handle, table_idx, linear_slot_idx, ping_pong, mr.mamba_value

135 def _add_one_req(
136 self,
137 pending_req: PendingReq,
138 cache_handle: BaseCacheHandle,
139 table_idx: int,
140 cached_len: int,
141 linear_slot_idx: int | None = None,
142 ping_pong: tuple | None = None,
143 next_track_idx: int = 0,
144 restore_src: int | None = None,
145 swa_evicted_seqlen: int = 0,
146 ) -> Req | None:
147 remain_len = pending_req.input_len - cached_len
148 chunk_size = min(self.token_budget, remain_len)
149 if self.cache_manager.swa_paged:
150 # Cap this chunk by the swa the pool can back this pass. swa is allocated per token in
151 # allocate_paged, and token_budget (max_extend_tokens, default 8192) won't chunk a
152 # shorter prompt -- so this cap is what forces a prompt whose swa footprint exceeds the
153 # pool to chunk. Credit the slots THIS request's own extend-free (in _prepare_batch,
154 # which runs AFTER this sizing) will release this batch, else a continuation sees a
155 # drained pool and stalls at chunk_size 0.
156 cm = self.cache_manager
157 window, ps = cm.sliding_window_size, cm.page_size
158 floor = cache_handle.cached_len
159 new_evicted = align_down(cached_len - window - ps, ps)
160 self_reclaim = max(0, new_evicted - max(swa_evicted_seqlen, floor))
161 swa_budget = cm.swa_available_size + self_reclaim - self.reserved_swa
162 # swa is charged per WHOLE page: cap the chunk so its PAGE-SPAN cost fits the budget
163 # (the extend [cached_len, cached_len+chunk) pulls div_ceil(end,ps)-div_ceil(start,ps)
164 # fresh pages -- the partial head page was charged by the previous chunk), and reserve
165 # that cost, not the raw token count. Degenerates to the token math at page_size==1.
166 max_end = (div_ceil(cached_len, ps) + max(swa_budget, 0) // ps) * ps
167 chunk_size = min(chunk_size, max(max_end - cached_len, 0))
168 # A continuation resumes the compressor carry at its boundary, which must be
169 # page-aligned; the token_budget leftover (unlike max_end) is not. Align the end
170 # down when the chunk mints a continuation; no whole page -> retry next pass.
171 # 0 <: a chunk the swa cap collapsed to 0 must NOT bail (undersized pool --
172 # bailing would livelock; the floor tests pin the loud failure).
173 if 0 < chunk_size < remain_len:
174 aligned = align_down(cached_len + chunk_size, ps) - cached_len
175 if aligned <= 0:
176 return None
177 chunk_size = aligned
178 align = self.cache_manager.prefill_chunk_align
179 if align > 1 and 0 < chunk_size < remain_len:
180 # An unaligned chunk end is correct, it just loses this prompt's snapshot boundaries --
181 # so keep it when the leftover budget cannot fill one whole unit instead of stalling
182 # the request until it gets a bigger turn.
183 aligned = align_down(cached_len + chunk_size, align) - cached_len
184 chunk_size = aligned if aligned > 0 else chunk_size
185 if self.keep_images_whole and pending_req.mm_items and chunk_size < remain_len:
186 # a cut image would attend within only the part already in the cache: end the chunk before it, decided last because the caps above only move the end earlier and would undo it
187 unit = math.lcm(self.cache_manager.page_size if self.cache_manager.swa_paged else 1, align if align > 1 else 1)
188 end = mm_chunk_end(pending_req.mm_items, cached_len, cached_len + chunk_size, unit)
189 cut = next((hi for item in pending_req.mm_items for lo, hi in item.offsets if lo < end < hi), None)
190 if cut is not None and self.token_budget < self.pass_budget and cut - cached_len <= self.pass_budget:
191 # other requests took part of this pass; a pass of its own holds the image whole
192 return None
193 chunk_size = end - cached_len
194 if self.cache_manager.swa_paged:
195 ps = self.cache_manager.page_size
196 self.reserved_swa += (div_ceil(cached_len + chunk_size, ps) - div_ceil(cached_len, ps)) * ps
197 is_chunked = chunk_size < remain_len
198 CLS = ChunkedReq if is_chunked else Req
199 self.token_budget -= chunk_size
200 # CacheManager allocates each request independently in whole pages. Reserve the same
201 # page span here; charging raw tokens can admit several short requests against one page
202 # even though allocate_paged() needs a separate page for each request.
203 self.reserved_size += self._kv_reservation_size(
204 pending_req.input_len + pending_req.output_len, cached_len
205 )
206 # NOTE: update the tokens ids only; new pages will be allocated in the scheduler
207 _slice = slice(cached_len, cached_len + chunk_size)
208 device_ids = self.table_manager.token_pool[table_idx, _slice]
209 device_ids.copy_(_maybe_pinned(pending_req.input_ids[_slice]), non_blocking=True)
210 req = CLS(
211 input_ids=pending_req.input_ids[: cached_len + chunk_size],
212 table_idx=table_idx,
213 cached_len=cached_len,
214 output_len=pending_req.output_len,
215 uid=pending_req.uid,
216 cache_handle=cache_handle,
217 sampling_params=pending_req.sampling_params,
218 prompt_len=pending_req.input_len,
219 )
220 req.mm_items = pending_req.mm_items
221 req.mrope_positions_full = pending_req.mrope_positions_full
222 req.mrope_delta = pending_req.mrope_delta
223 # Hybrid GDN per-request state slots (None for non-hybrid). On a fresh admit these are
224 # freshly allocated; on a chunked continuation they are inherited from the prior chunk.
225 req.linear_slot_idx = linear_slot_idx
226 req.mamba_ping_pong = ping_pong
227 req.mamba_next_track_idx = next_track_idx
228 req.mamba_restore_src = restore_src
229 req.swa_evicted_seqlen = swa_evicted_seqlen # carry the extend-free watermark across chunks
230 return req

232 def try_add_one(self, pending_req: PendingReq) -> Req | None:
233 if self.token_budget <= 0:
234 return None
235
236 if chunked_req := pending_req.chunked_req:
237 return self._add_one_req(
238 pending_req=pending_req,
239 cache_handle=chunked_req.cache_handle,
240 table_idx=chunked_req.table_idx,
241 cached_len=chunked_req.cached_len,
242 linear_slot_idx=chunked_req.linear_slot_idx,
243 ping_pong=chunked_req.mamba_ping_pong,
244 next_track_idx=chunked_req.mamba_next_track_idx,
245 restore_src=None, # continuation chunk already has live state
246 swa_evicted_seqlen=chunked_req.swa_evicted_seqlen, # extend-free watermark so far
247 )
248
249 if resource := self._try_allocate_one(pending_req):
250 cache_handle, table_idx, linear_slot_idx, ping_pong, restore_src = resource
251 req = self._add_one_req(
252 pending_req=pending_req,
253 cache_handle=cache_handle,
254 table_idx=table_idx,
255 cached_len=cache_handle.cached_len,
256 linear_slot_idx=linear_slot_idx,
257 ping_pong=ping_pong,
258 next_track_idx=0,
259 restore_src=restore_src,
260 )
261 if req is None:
262 # no aligned chunk this pass: undo the admission (a continuation keeps its
263 # resources -- they belong to the prior chunk's Req)
264 self.cache_manager.unlock(cache_handle)
265 self.table_manager.free(table_idx)
266 if linear_slot_idx is not None:
267 self.cache_manager.linear_state_pool.free([linear_slot_idx, *ping_pong])
268 return req
269
270 return None

294 def schedule_next_batch(self, prefill_budget: int) -> Batch | None:
295 if len(self.pending_list) == 0:
296 return None
297
298 # estimated offset due to in-flight decode
299 adder = PrefillAdder(
300 token_budget=prefill_budget,
301 reserved_size=self.decode_manager.inflight_tokens,
302 cache_manager=self.cache_manager,
303 table_manager=self.table_manager,
304 encoder_cache=self.encoder_cache,
305 keep_images_whole=self.keep_images_whole,
306 )
307 reqs: List[Req] = []
308 chunked_list: List[PendingReq] = []
309 prompt_admissions: List[Tuple[int, int, int]] = []
310 # Snapshot here, before the forward's complete_one() advances cached_len: the tokens
311 # forwarded this batch (extend_len) and the prefix-cache hit. SGLang counts the hit
312 # once at admission, so continuation chunks (already-chunked reqs) contribute 0.
313 log_new_tokens = 0
314 log_cached_tokens = 0
315 for pending_req in self.pending_list:
316 is_continuation = pending_req.chunked_req is not None
317 if req := adder.try_add_one(pending_req):
318 pending_req.chunked_req = None
319 if isinstance(req, ChunkedReq):
320 pending_req.chunked_req = req
321 chunked_list.append(pending_req)
322 reqs.append(req)
323 if not is_continuation:
324 # Record the COMPLETE prompt length and the prefix-cache hit on the
325 # first chunk. The scheduler publishes them only after _prepare_batch
326 # succeeds; continuation chunks must never publish them again.
327 prompt_admissions.append(
328 (req.uid, pending_req.input_len, req.cache_handle.cached_len)
329 )
330 if pending_req.mm_items and self.encoder_cache is not None:
331 # claim the rows every chunk of this request will gather; the entry outlives the chunks
332 for item in pending_req.mm_items:
333 self.encoder_cache.register(
334 item.hash, req.uid, mm_rows_after(item, req.cache_handle.cached_len)
335 )
336 log_new_tokens += req.extend_len
337 if not is_continuation:
338 log_cached_tokens += req.cache_handle.cached_len
339 else:
340 break # We cannot add more requests
341 if len(reqs) == 0:
342 return None
343 self.pending_list = chunked_list + self.pending_list[len(reqs) :]
344 batch = Batch(reqs=reqs, phase="prefill")
345 batch.log_new_tokens = log_new_tokens
346 batch.log_cached_tokens = log_cached_tokens
347 batch.prompt_admissions = prompt_admissions
348 return batch

350 def abort_req(self, uid: int) -> Req | None:
351 for i, req in enumerate(self.pending_list):
352 if req.uid == uid:
353 self.pending_list.pop(i)
354 return req.chunked_req
355 return None
