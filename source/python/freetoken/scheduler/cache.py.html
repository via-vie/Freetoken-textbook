26# Finish-time retention keeps [P - window - gap, P) swa-live for the next turn's cut near the
27# prompt end. The gap covers templates whose generation prompt injects tokens that vanish when
28# the client drops reasoning (Qwen's "<think>\n": the re-render diverges 2 tokens BEFORE P).

33 def __init__(self, num_pages: int, page_size: int, page_table: torch.Tensor, type: str,
34 linear_state_pool=None, swa_pool=None, sliding_window_size=None):
35 # The `_free_slots` follows a page-aligned manner. For example, if page_size = 2,
36 # the `_free_slots` may look like [0, 2, 4, 6, ...], and each slot represents a page.
37 device = page_table.device
38 self.free_slots = torch.arange(num_pages, dtype=torch.int32, device=device) * page_size
39 # Hybrid GDN models drive a second currency (GDN state snapshots in LinearStatePool)
40 # through a HybridRadixCache; SWA models drive a second currency (swa-pool KV slots in
41 # the HybridSWAKVCache global-paged mode) through a SWARadixCache; non-hybrid models keep
42 # the plain naive/radix path.
43 self.linear_state_pool = linear_state_pool
44 self.swa_pool = swa_pool
45 self.sliding_window_size = sliding_window_size
46 self.is_hybrid = type == "hybrid_radix"
47 self.is_swa = type == "swa_radix"
48 # swa_paged: this SWA model drives the global-paged swa pool -- true for BOTH the naive
49 # (NaivePrefixCache, no reuse) and radix (SWARadixCache) paths. Gates the swa slot
50 # lifecycle (alloc_swa / out-of-window free / free-on-finish). is_swa gates only the extra
51 # SWARadixCache reuse machinery (tree match/insert/evict_swa/swa_uuid lock).
52 self.swa_paged = swa_pool is not None and getattr(swa_pool, "swa_paged", False)
53 # Owned-pool capability pickup: a plugged-in swa pool may cap the prefill chunk (DSV4:
54 # ~half the window working set). Instance attrs shadow the class defaults; absent
55 # attributes leave the defaults untouched (Gemma4).
56 if swa_pool is not None:
57 self.prefill_chunk_budget = getattr(swa_pool, "prefill_chunk_budget", None)
58 self.prefix_cache = self._make_prefix_cache(device, page_size, type)
59 self.device = device
60 self.num_pages = num_pages
61 self.page_table = page_table
62 self.page_size = page_size
63 self.cache_type = type

131 def ensure_swa_slots(self, n: int) -> None:
132 """Free swa-pool slots until >= ``n`` are available by tombstoning LRU tree swa nodes
133 (evict_swa, internal -> tombstone in place / leaf -> free both pools), returning their swa
134 slots to the pool and any deleted-leaf full KV to free_slots."""
135 while self.swa_pool.swa_available_size() < n:
136 ev = self.prefix_cache.evict_swa(n - self.swa_pool.swa_available_size())
137 if ev.swa_indices.numel() == 0:
138 break
139 self.swa_pool.free_swa(ev.swa_indices)
140 if ev.kv_indices.numel():
141 self._free(ev.kv_indices)

143 def ensure_mamba_slots(self, n: int) -> None:
144 """Free GDN state slots until >= ``n`` are available by tombstoning LRU tree snapshots
145 (evict_mamba), returning their slots + any freed KV to the pools."""
146 while self.linear_state_pool.num_free_slots < n:
147 er = self.prefix_cache.evict_mamba(n - self.linear_state_pool.num_free_slots)
148 if not er.mamba_slots:
149 break
150 self.linear_state_pool.free(er.mamba_slots)
151 self._free(er.kv_indices)

153 def snapshot_toolcall_anchor(self, reqs: List[Req]) -> None:
154 """Freeze each decoding request's GDN state at its tool-call anchor, into the ping-pong
155 slot that is idle during decode (the kernel-side ×CHUNK track only runs on prefill
156 extends). Must run on the engine stream before the current step's kernels: cached_len
157 equals the anchor exactly when every enqueued step up to the anchor-consuming one has
158 been issued and the next (current) one has not, so the copy lands between them in
159 stream order. Reuses ``mamba_last_track_seqlen`` as the pending-donate mark -- the
160 prefill track's own pending freeze was consumed by the prefill-commit ``cache_req``
161 before any decode drain could set an anchor."""
162 if not self.is_hybrid:
163 return
164 pool = self.linear_state_pool
165 for r in reqs:
166 a = r.toolcall_anchor_len
167 if (
168 a is None
169 or r.mamba_ping_pong is None
170 or r.mamba_last_track_seqlen is not None
171 or r.cached_len != a
172 or align_down(a, self.page_size) != a
173 ):
174 continue
175 dst = r.mamba_ping_pong[r.mamba_next_track_idx]
176 pool.copy_from(r.linear_slot_idx, dst)
177 r.mamba_last_track_seqlen = a
178 r.mamba_next_track_idx = 1 - r.mamba_next_track_idx

180 def maybe_free_swa_out_of_window(self, reqs: List[Req], *, forward_iter: int) -> None:
181 """Proactively free each decoding request's now-out-of-window SWA slots, bounding its swa
182 footprint to ~one window so a smaller-than-full swa pool (swa_full_tokens_ratio<1) stays
183 viable. Mirrors sglang ``ScheduleBatch.maybe_evict_swa`` / ``free_swa_out_of_window_slots``:
184 evict every ``interval`` forwards; skip a request's first decode step (its extend forward
185 may still be in-flight under overlap); floor the frontier at the request's protected
186 (reused) prefix so it only frees its OWN slots, never the tree-shared prefix's swa; and keep
187 a ``window + page_size`` margin so the freed slots are out-of-window for every in-flight
188 forward."""
189 if not self.swa_paged or forward_iter % _SWA_EVICTION_INTERVAL != 0:
190 return
191 window = self.sliding_window_size
192 for req in reqs:
193 if req.decode_batch_idx < 1:
194 continue # overlap guard: extend forward may still be running
195 floor = req.cache_handle.cached_len # reused prefix -> its swa is tree-owned, not ours
196 threshold = (req.device_len - 1) - window - self.page_size
197 if req.toolcall_anchor_len is not None:
198 # Keep the window ending at the anchor resumable: a client-side rewrite of the
199 # echoed tool call forks after the anchor, and a resume there needs
200 # [anchor - window, anchor) live. The finish-insert then adopts (rather than
201 # tombstones) these never-evicted slots; they stay unlocked, so real pool
202 # pressure can still reclaim them (same soft retention as the prompt-end pin).
203 cap = req.toolcall_anchor_len - window - _SWA_RETAIN_GAP
204 if threshold - cap > window + _SWA_RETAIN_GAP:
205 # The decode ran on far past the anchor (a tool call is normally within
206 # tens of tokens of the end). Holding the cap would grow this request's
207 # live swa without bound ("SWA pool exhausted" is unhandled) -- drop the
208 # anchor and let normal eviction resume. This bound is what the
209 # anchor-retention term in _swa_per_req_swa_floor sizes the pool for.
210 req.toolcall_anchor_len = None
211 else:
212 threshold = min(threshold, cap)
213 new_evicted = align_down(threshold, self.page_size)
214 start = max(req.swa_evicted_seqlen, floor)
215 if new_evicted > start:
216 self._free_swa(self.page_table[req.table_idx, start:new_evicted])
217 req.swa_evicted_seqlen = new_evicted

219 def free_swa_out_of_window_extend(self, reqs: List[Req]) -> None:
220 """Prefill sibling of ``maybe_free_swa_out_of_window``: before allocating a chunk, return
221 each request's now-out-of-window SWA slots so a chunked prompt's live swa stays ~one window
222 regardless of prompt length (else a prompt longer than the swa pool exhausts alloc_swa).
223 Runs on EVERY prefill batch -- no eviction-interval cadence, since a long prompt would
224 overflow the pool before a cadence fires. The frontier is based on ``cached_len`` (the
225 pre-chunk, already-forwarded length; the chunk ``[cached_len, device_len)`` is allocated by
226 the following ``allocate_paged``, not here), so only positions prior chunks consumed are
227 freed, floored at the tree-owned reused prefix. Overlap-safe by the same scheduler stream
228 gate + ``window + page_size`` margin the decode driver relies on; ``free_swa`` is idempotent
229 over the sentinel, so re-freeing an earlier chunk's range is a no-op. The pool is always
230 sized > one window (see the swa-pool floor), so a chunk can always make forward progress."""
231 if not self.swa_paged:
232 return
233 window = self.sliding_window_size
234 for req in reqs:
235 floor = req.cache_handle.cached_len # reused prefix -> its swa is tree-owned, not ours
236 new_evicted = align_down(req.cached_len - window - self.page_size, self.page_size)
237 start = max(req.swa_evicted_seqlen, floor)
238 if new_evicted > start:
239 self._free_swa(self.page_table[req.table_idx, start:new_evicted])
240 req.swa_evicted_seqlen = new_evicted

242 def lock(self, handle: BaseCacheHandle) -> None:
243 if self.is_swa:
244 # records the window boundary on the (frozen) handle for unlock/dec_lock.

265 def allocate_paged(self, reqs: List[Req]) -> None:
266 needed_pages = 0
267 allocation_info: List[Tuple[int, int, int]] = []
268 for req in reqs:
269 first_page = div_ceil(req.cached_len, self.page_size)
270 last_page = div_ceil(req.device_len, self.page_size)
271 if last_page > first_page:
272 needed_pages += last_page - first_page
273 allocation_info.append((req.table_idx, first_page, last_page))
274 if needed_pages > 0:
275 allocated = self._page_to_token(self._allocate(needed_pages))
276 if self.swa_paged:
277 # Each newly-allocated full token needs a swa-pool slot (where its SWA-layer KV is
278 # written; read back via the full->swa mapping). radix reuses the existing prefix's
279 # (live, mapped) swa slots and evicts tree swa if the pool is short; naive has no
280 # tree (the pool is sized concurrency x window so it always fits).
281 if self.is_swa:
282 self.ensure_swa_slots(len(allocated))
283 self.swa_pool.alloc_swa(allocated)
284 _write_page_table(self.page_table, allocated, allocation_info, self.page_size)

286 def cache_req(self, req: Req, *, finished: bool) -> None:
287 if self.is_swa:
288 return self._cache_req_swa(req, finished=finished)
289 if self.is_hybrid:
290 return self._cache_req_hybrid(req, finished=finished)
291 # ==================================== valid cache region ====================================
292 # [0, req.cached_len) This part is valid for attention kernel read/write.
293 # [0, old_handle.cached_len) This part is in the prefix cache before prefill.
294 # [old_handle.cached_len, req.cached_len) This part is allocated by cache manager for this request.
295 # ================================== allocated cache region ==================================
296 # [old_handle.cached_len, cached_len) This part was not in the prefix cache when prefill,
297 # but later cached by other requests.
298 # We must free them to avoid memory leak.
299 # [cached_len, new_handle.cached_len) This part is newly inserted into the prefix cache.
300 # [new_handle.cached_len, req.cached_len) This part is tailing part that can not inserted into the prefix cache.
301 # We should free it if the request has finished.
302 page_indices = self.page_table[req.table_idx, : req.cached_len]
303 old_handle = req.cache_handle
304 insert_ids = req.input_ids[: req.cached_len]
305 cached_len, new_handle = self.prefix_cache.insert_prefix(insert_ids, page_indices)
306 # unlock until all operations on handle is done
307 self.unlock(old_handle)
308 # this part is already in the prefix cache, free it. A naive-SWA request (swa_paged, no
309 # reuse) also returns the swa slots backing every full slot it frees; the out-of-window
310 # ones were already freed by the decode driver (free_swa is idempotent over the sentinel).
311 if self.swa_paged:
312 self._free_swa(page_indices[old_handle.cached_len : cached_len])
313 self._free(page_indices[old_handle.cached_len : cached_len])
314 if finished: # this tail part should be freed
315 tail = self._padded_tail(req, new_handle.cached_len)
316 if self.swa_paged:
317 self._free_swa(tail)
318 self._free(tail)
319 else: # keep the tail part, update the handle
320 # Re-point the deduped span at the tree's canonical pages: the request's own pages
321 # for [old_handle.cached_len, cached_len) went back on the free list above, but the
322 # attention backends read this row every step and the next allocation hands those
323 # pages to someone else. [0, old_handle.cached_len) needs no rewrite -- it has been
324 # locked since admission, so the row already equals canonical there.
325 if cached_len > old_handle.cached_len:
326 canonical = new_handle.get_matched_indices()
327 self.page_table[req.table_idx, old_handle.cached_len : cached_len].copy_(
328 canonical[old_handle.cached_len : cached_len])
329 req.cache_handle = new_handle
330 self.lock(new_handle)

332 def _cache_req_hybrid(self, req: Req, *, finished: bool) -> None:
333 """Hybrid (GDN) cache_req: commit KV like radix AND manage the GDN state snapshot.
334 Prefill chunk commit: DONATE the frozen ping-pong slot (the snapshot the forward wrote
335 at the tracked ×64 boundary mamba_last_track_seqlen) into the tree; replace it with a
336 fresh slot if the tree took it (dedup keeps it for reuse). Finish: donate the live slot
337 (final full-sequence state, zero-copy since the req is done) and free the req's slots."""
338 from freetoken.kvcache.hybrid_radix_cache import HybridCacheHandle
339
340 pool = self.linear_state_pool
341 old_handle = req.cache_handle
342 page_indices = self.page_table[req.table_idx, : req.cached_len]
343
344 if finished:
345 # A pending freeze (the tool-call anchor, or a prefill ×64 track the request
346 # finished too early to chunk-commit) is a strictly shorter prefix than the live
347 # donate below: insert it first and advance the dedup-free floor to its boundary
348 # -- [prefix_len, L) is now tree-owned by the donated node, so only [old, prefix_len)
349 # is this request's dup to free. The frozen slot is consumed either way (taken by
350 # the tree or freed here) and both ping-pong refs are dropped before
351 # _free_req_slots so nothing double-frees.
352 free_upto = old_handle.cached_len
353 L = req.mamba_last_track_seqlen
354 if (
355 L is not None
356 and 0 < L <= req.cached_len
357 and align_down(L, self.page_size) == L
358 and req.mamba_ping_pong is not None
359 ):
360 frozen_idx = 1 - req.mamba_next_track_idx
361 frozen = req.mamba_ping_pong[frozen_idx]
362 prefix_len, mamba_exist = self.prefix_cache.insert(
363 req.input_ids[:L], page_indices[:L], frozen)
364 pool.free([s for s in req.mamba_ping_pong if mamba_exist or s != frozen])
365 req.mamba_ping_pong = None
366 self._free(page_indices[free_upto : max(free_upto, prefix_len)])
367 free_upto = max(free_upto, L)
368 # Donate the live slot (final full-sequence state). The live state is at cached_len;
369 # only attach it when cached_len is itself the page-aligned node boundary (always for
370 # page_size==1). For page_size>1 a non-aligned cached_len would attach an over-advanced
371 # state to a shorter prefix node -> skip the finish-donate (the ×64 prefill snapshots
372 # remain as reuse points).
373 insert_len = align_down(req.cached_len, self.page_size)
374 keep_live = False
375 if insert_len == req.cached_len and insert_len > 0:
376 prefix_len, mamba_exist = self.prefix_cache.insert(
377 req.input_ids[:insert_len], page_indices[:insert_len], req.linear_slot_idx)
378 self.unlock(old_handle)
379 self._free(page_indices[free_upto : max(free_upto, prefix_len)])
380 keep_live = not mamba_exist # tree now owns linear_slot_idx
381 else:
382 self.unlock(old_handle)
383 self._free(page_indices[free_upto :])
384 self._free_req_slots(req, keep_live=keep_live)
385 return
386
387 # Prefill chunk commit: donate the frozen snapshot at the tracked ×64 boundary.
388 L = req.mamba_last_track_seqlen
389 if L is None:
390 return # no ×64 boundary crossed this chunk; req keeps its pages (committed later)
391 if align_down(L, self.page_size) != L:
392 # page_size>1 only: insert would align the key down, attaching a state that encodes
393 # L tokens to a SHORTER node -- a future hit would COW-restore an over-advanced
394 # state. Skip; the next aligned boundary (or the finish-donate) commits instead.
395 req.mamba_last_track_seqlen = None
396 return
397 frozen_idx = 1 - req.mamba_next_track_idx # the slot the forward just wrote
398 frozen = req.mamba_ping_pong[frozen_idx]
399 prefix_len, mamba_exist = self.prefix_cache.insert(
400 req.input_ids[:L], page_indices[:L], frozen)
401 self.unlock(old_handle)
402 self._free(page_indices[old_handle.cached_len : prefix_len])
403 # Lock the committed snapshot node FIRST: the replacement-slot alloc below can trigger
404 # evict_mamba (via ensure_mamba_slots), which would otherwise reclaim this still-unlocked
405 # just-donated node -- freeing its KV pages under the still-decoding request.
406 m = self.prefix_cache.match_prefix(req.input_ids[:L])
407 # Same re-point as the generic path: the dedup free above returned this request's own
408 # pages for [old_handle.cached_len, prefix_len) while its row still named them.
409 if prefix_len > old_handle.cached_len:
410 self.page_table[req.table_idx, old_handle.cached_len : prefix_len].copy_(
411 m.kv_indices[old_handle.cached_len : prefix_len])
412 req.cache_handle = HybridCacheHandle(m.cached_len, m.node, m.kv_indices)
413 self.lock(req.cache_handle)
414 if not mamba_exist: # tree took `frozen`; replace it
415 self.ensure_mamba_slots(1)
416 pp = list(req.mamba_ping_pong)
417 pp[frozen_idx] = pool.alloc(1)[0]
418 req.mamba_ping_pong = tuple(pp)
419 req.mamba_last_track_seqlen = None

421 def _cache_req_swa(self, req: Req, *, finished: bool) -> None:
422 """SWA cache_req: commit the request's full KV prefix into the SWARadixCache (node.value =
423 the canonical full-pool page indices; the swa KV rides along via the full->swa mapping).
424 Tokens < req.swa_evicted_seqlen are marked tombstone on insert. No donate/COW (the swa KV
425 is in the pool already). On any dup/tail free, free both pools (full slot + its swa slot)."""
426 from freetoken.kvcache.swa_radix_cache import SWACacheHandle
427
428 old_handle = req.cache_handle
429 page_indices = self.page_table[req.table_idx, : req.cached_len]
430
431 insert_len = align_down(req.cached_len, self.page_size)
432 freed = page_indices[:0]
433 if insert_len > 0:
434 # insert reconciles tombstones (revives the in-window ones by ADOPTING the request's
435 # live-swa slots into node.value) and returns every full slot to reclaim: the displaced
436 # old tree slots + the request's non-adopted dups. swa_evicted_seqlen is the request's
437 # own extend/decode free frontier: insert must tombstone [.., swa_evicted_seqlen) rather
438 # than adopt those (now-sentinel) swa slots. This holds for BOTH finished and unfinished
439 # commits -- the extend driver frees out-of-window swa during chunked prefill too, so an
440 # unfinished chunk's frontier is already > 0 and must be honored (else insert adopts
441 # sentinel slots -> the request's later SWA gathers read slot 0 -> corruption).
442 _, freed = self.prefix_cache.insert(
443 req.input_ids[:insert_len], page_indices[:insert_len],
444 swa_evicted_seqlen=req.swa_evicted_seqlen,
445 update_kv_after_len=old_handle.cached_len)
446 self.unlock(old_handle)
447 self._free_swa(freed) # idempotent: revived/out-of-window slots are already sentinel -> no-op
448 self._free(freed)
449 if finished:
450 # Page-unaligned tail (page_size>1) not inserted. The padded slice reaches to the
451 # page-ceil bound: allocate_paged charged a swa slot for EVERY token of the last
452 # partial page (whole-page alloc_swa), so the padding slots must return with it or
453 # they leak (-cached_len mod page_size slots per request, permanently).
454 tail = self._padded_tail(req, insert_len)
455 self._free_swa(tail)
456 self._free(tail)
457 # Soft-pin the prompt-end window: decode never re-stamps the prompt path, so after
458 # the unlock above it is the stalest LRU entry and the first evict_swa victim. A
459 # follow-up turn diverges at the prompt end when the client drops reasoning; a cut
460 # there only needs the trailing window live, so eagerly reclaim the head's swa
461 # (full KV stays for the full-attn layers) and re-stamp the retained tail -- still
462 # unlocked, so it remains reclaimable under real pressure.
463 prompt_len = align_down(req.max_device_len - req.output_len, self.page_size)
464 if prompt_len > 0:
465 keep_from = align_down(
466 max(prompt_len - self.sliding_window_size - _SWA_RETAIN_GAP, 0),
467 self.page_size,
468 )
469 if keep_from > 0:
470 self._free_swa(
471 self.prefix_cache.trim_head_swa(req.input_ids[:prompt_len], keep_from))
472 self.prefix_cache.match_prefix(req.input_ids[:prompt_len])
473 else:
474 # inc_lock is node-granular, and the suffix insert just made this chunk's whole
475 # extend one node: locking it would pin the entire chunk's swa for all of decode,
476 # though the request reads only its trailing window from here on. Force a node
477 # boundary a window back (match_prefix splits) so the lock lands on that window
478 # alone. The head stays live and unlocked -- still reusable while the pool is
479 # roomy, evictable the moment it is not.
480 keep_from = align_down(
481 max(insert_len - self.sliding_window_size - _SWA_RETAIN_GAP, 0), self.page_size)
482 if keep_from > 0:
483 self.prefix_cache.match_prefix(req.input_ids[:keep_from])
484 m = self.prefix_cache.match_prefix(req.input_ids[:insert_len])
485 # Re-point the page table to the tree's live slots for the committed region. Any dup
486 # slots insert reclaimed had their full->swa mapping reset to the 0 sentinel; unlike the
487 # full pool (KV survives in place until realloc), a stale swa mapping would make the
488 # request's subsequent SWA gathers read the sentinel -> corruption. The reconcile revived
489 # the in-window tombstones, so the re-matched slots are live.
490 if m.cached_len > 0:
491 self.page_table[req.table_idx, : m.cached_len].copy_(m.kv_indices)
492 req.cache_handle = SWACacheHandle(m.cached_len, m.node, m.kv_indices)
493 self.lock(req.cache_handle)
494
495 def _padded_tail(self, req: Req, start: int) -> torch.Tensor:
496 """The request's OWN slice [start, page_ceil(cached_len)) of the page table. A finish
497 frees through the page-CEIL bound, not cached_len: allocate_paged allocates (and, when
498 swa_paged, charges swa for) whole pages, so the padding [cached_len, page_ceil) belongs
499 to the finishing request. ``start`` is page-aligned (a match/insert boundary), so the
500 full-pool page bases derived via ``[::page_size]`` are identical to the unpadded slice."""
501 end = div_ceil(req.cached_len, self.page_size) * self.page_size
502 return self.page_table[req.table_idx, start:end]

504 def _free_req_slots(self, req: Req, keep_live: bool = False) -> None:
505 """Return a finished request's GDN pool slots: both ping-pong slots, plus the live slot
506 unless it was donated to the tree. Idempotent -- clears the refs so a re-entry frees
507 nothing (defense-in-depth against the abort/finish double-free, see _free_req_resources)."""
508 slots = list(req.mamba_ping_pong) if req.mamba_ping_pong is not None else []
509 if not keep_live and req.linear_slot_idx is not None:
510 slots.append(req.linear_slot_idx)
511 if slots:
512 self.linear_state_pool.free(slots)
513 req.mamba_ping_pong = None
514 req.linear_slot_idx = None

516 def check_integrity(self) -> None:
517 if self.is_hybrid:
518 pc = self.prefix_cache
519 pc.check_integrity() # structural: every snapshot node owns a slot, refs >= 0
520 cache_pages = (pc.full_evictable + pc.full_protected) // self.page_size
521 # GDN-slot conservation upper bound: free slots + tree-held snapshots can never
522 # exceed the (non-padding) pool capacity; the remainder is held by running requests.
523 pool = self.linear_state_pool
524 tree_slots = pc.mamba_evictable_size + pc.mamba_protected
525 assert pool.num_free_slots + tree_slots <= pool.num_slots - 1, (
526 f"GDN-slot leak: free({pool.num_free_slots}) + tree({tree_slots}) > "
527 f"capacity({pool.num_slots - 1})"
528 )
529 elif self.is_swa:
530 pc = self.prefix_cache
531 pc.check_integrity() # full>=swa refs, tombstone => no swa lock
532 cache_pages = (pc.full_evictable + pc.full_protected) // self.page_size
533 # swa-slot conservation upper bound: free swa slots + tree-held live swa tokens can
534 # never exceed the (non-sentinel) swa-pool capacity; the rest is held by running reqs.
535 tree_swa = pc.swa_evictable + pc.swa_protected
536 cap = self.swa_pool.swa_num_tokens - 1 # slot 0 is the reserved sentinel
537 # check_integrity is idle-only (like the exact full-pool check below), so no request
538 # holds a swa slot: free + tree must equal cap exactly. `==` (not `<=`) so a LEAK
539 # (free + tree < cap) is caught, not just a double-free (> cap).
540 assert self.swa_pool.swa_available_size() + tree_swa == cap, (
541 f"SWA-slot leak/double-free: free({self.swa_pool.swa_available_size()}) + "
542 f"tree({tree_swa}) != capacity({cap})"
543 )
544 else:
545 self.prefix_cache.check_integrity()
546 cache_pages = self.prefix_cache.size_info.total_size // self.page_size
547 if len(self.free_slots) + cache_pages != self.num_pages:
548 raise RuntimeError(
549 "CacheManager integrity check failed:"
550 f" free_pages({len(self.free_slots)}) +"
551 f" cache_pages({cache_pages}) != num_pages({self.num_pages})"
552 )
553 if self.page_size > 1:
554 assert torch.all(self.free_slots % self.page_size == 0)
555
556 def rebuild(self, num_pages: int, page_table: torch.Tensor) -> None:
557 """Re-point the page table and reset page accounting + prefix cache IN PLACE.
558
559 Idle-only: assumes no request holds a live handle. Builds a brand-new prefix
560 cache (RadixPrefixCache.reset() is an unimplemented stub) rather than mutating
561 the old one.
562 """
563 device = page_table.device
564 self.device = device
565 self.num_pages = num_pages
566 self.page_table = page_table
567 self.free_slots = torch.arange(num_pages, dtype=torch.int32, device=device) * self.page_size
568 self.prefix_cache = self._make_prefix_cache(device, self.page_size, self.cache_type)
569 # The discarded hybrid tree owned donated GDN-snapshot slots; rebuild is idle-only, so
570 # reclaim the whole LinearStatePool free-list (else those slots leak -> admission hangs).
571 if self.is_hybrid:
572 self.linear_state_pool.reclaim_all_slots()

574 @contextmanager
575 def lazy_free_region(self):
576 def lazy_free(indices: torch.Tensor) -> None:
577 # clone: callers pass page-table VIEWS, and the deferred concat below only reads them
578 # when the region exits. A commit that re-points the row in between (the dedup
579 # re-point, the SWA one) would otherwise rewrite the pending free list underneath us
580 # and return the tree's canonical pages instead of the request's duplicates.
581 lazy_free_list.append(indices[:: self.page_size].clone())
582
583 lazy_free_list: List[torch.Tensor] = []
584 try:
585 self._free = lazy_free
586 yield
587 finally:
588 del self._free
589 self.free_slots = torch.cat([self.free_slots] + lazy_free_list)

591 def _allocate(self, needed_pages: int) -> torch.Tensor:
592 if needed_pages > (free_pages := len(self.free_slots)):
593 need = (needed_pages - free_pages) * self.page_size
594 if self.is_swa:
595 # Evicting KV leaf nodes drops their swa slots too -> return both pools.
596 ev = self.prefix_cache.evict_full(need)
597 evicted = ev.kv_indices
598 self._free_swa(ev.swa_indices)
599 elif self.is_hybrid:
600 # Evicting KV leaf nodes drops their GDN snapshots too -> return both pools.
601 er = self.prefix_cache.evict_full(need)
602 evicted = er.kv_indices
603 if er.mamba_slots:
604 self.linear_state_pool.free(er.mamba_slots)
605 else:
606 evicted = self.prefix_cache.evict(need)
607 self.free_slots = torch.cat([self.free_slots, evicted[:: self.page_size]])
608 assert len(self.free_slots) >= needed_pages, "Eviction did not free enough space."
609 allocated = self.free_slots[:needed_pages]
610 self.free_slots = self.free_slots[needed_pages:]
611 return allocated
612
613 def _free(self, indices: torch.Tensor) -> None:
614 if len(indices) > 0:
615 self.free_slots = torch.cat([self.free_slots, indices[:: self.page_size]])

617 def _page_to_token(self, pages: torch.Tensor) -> torch.Tensor:
618 if self.page_size == 1:
619 return pages
620 # [X * page_size] -> [X * page_size, ..., X * page_size + page_size - 1]
621 offsets = torch.arange(self.page_size, device=self.device, dtype=torch.int32)
622 return (pages.unsqueeze(1) + offsets).flatten()

625def _write_page_table(
626 page_table: torch.Tensor,
627 allocated: torch.Tensor,
628 allocation_info: List[Tuple[int, int, int]],
629 page_size: int,
630) -> None:
631 needed_tokens = len(allocated)
632 # Pinned only when there is a device to copy to asynchronously; CPU-only runs (unit tests,
633 # a CPU CI runner) would otherwise raise instead of just doing a plain host allocation.
634 pin = torch.cuda.is_available()
635 table_idx_host = torch.empty(needed_tokens, dtype=torch.int64, pin_memory=pin)
636 positions_host = torch.empty(needed_tokens, dtype=torch.int64, pin_memory=pin)
637 offset = 0
638 for table_idx, first_page, last_page in allocation_info:
639 first_pos, last_pos = first_page * page_size, last_page * page_size
640 length = last_pos - first_pos
641 table_idx_host[offset : offset + length].fill_(table_idx)
642 torch.arange(first_pos, last_pos, out=positions_host[offset : offset + length])
643 offset += length
644 assert offset == needed_tokens, "Mismatch in allocated tokens and filled tokens."
645 table_idxs = table_idx_host.to(page_table.device, non_blocking=True)
646 offsets = positions_host.to(page_table.device, non_blocking=True)
647 assert allocated.dtype == page_table.dtype, (
648 f"allocated dtype {allocated.dtype} != page_table dtype {page_table.dtype}"
649 )
650 page_table[table_idxs, offsets] = allocated
