159 @torch.inference_mode()
160 def rebuild_cache(
161 self,
162 *,
163 moe_cache_size: int | None = None,
164 num_pages: int | None = None,
165 num_mamba_slots: int | None = None,
166 num_swa_pages: int | None = None,
167 ) -> None:
168 """Idle-only runtime cache rebuild: resize the MoE slot cache, KV pages, GDN (mamba) state
169 pool, and/or the window pool (num_swa_pages), re-capture CUDA graphs, and re-thread the
170 page managers (clearing the prefix cache on a KV/mamba/window resize). The caller MUST
171 guarantee the scheduler is idle — no pending prefill, no running decode, no in-flight
172 finished requests. All TP ranks must call this with identical arguments.
173 """
174 assert not self.prefill_manager.runnable, "rebuild requires no pending prefill"
175 assert not self.decode_manager.runnable, "rebuild requires no running decode"
176 torch.cuda.synchronize(self.device)
177 if self.config.tp_info.size > 1:
178 self.sync_all_ranks()
179 self.engine.rebuild_runtime_cache(
180 moe_cache_size=moe_cache_size, num_pages=num_pages, num_mamba_slots=num_mamba_slots,
181 num_swa_pages=num_swa_pages,
182 )
183 if num_pages is not None or num_mamba_slots is not None or num_swa_pages is not None:
184 # Any of these resizes invalidates the prefix cache: a KV resize leaves stale page
185 # indices, a mamba resize leaves stale GDN-snapshot slot ids, and a window-pool resize
186 # (num_swa_pages) reallocates the SWA/window token pool, leaving stale slot ids in the
187 # radix tree. Rebuild the prefix cache + reclaim the resized free-lists.
188 self.cache_manager.rebuild(self.engine.num_pages, self.engine.page_table)
189 if num_pages is not None:
190 # token_pool is sized to the page table; only a KV-page resize reallocates it.
191 # A mamba-only rebuild leaves the page table untouched, so skip this (else it
192 # needlessly reallocates + zeros the whole GPU token_pool every mamba resize).
193 self.table_manager.rebuild(self.engine.page_table)
194 self.token_pool = self.table_manager.token_pool
195 self.cache_manager.check_integrity()
196 # The prefill chunk cap tracks the CURRENT window-pool size (DSV4); a rebuild that
197 # shrank the pool must shrink the cap too, or the next long prompt is chunked against
198 # the stale budget and crashes _alloc_window.
199 _chunk_cap = self.cache_manager.prefill_chunk_budget
200 self.prefill_budget = (
201 min(self.config.max_extend_tokens, _chunk_cap)
202 if _chunk_cap else self.config.max_extend_tokens
203 )
204 if self.config.tp_info.size > 1:
205 self.sync_all_ranks()

207 def overlap_loop(self, last_data: ForwardData | None) -> ForwardData | None:
208 """
209 The main loop of overlapping scheduling and execution.
210
211 It will overlap the execution of current batch and processing of last batch's results,
212 which can effectively hide CPU latency and improve GPU utilization.
213 """
214 # Expose the un-drained batch to _process_one_msg (abort in-flight check). Assigning
215 # before the message loop is what makes the check airtight: the batch launched later
216 # this iteration can only be probed by messages of the NEXT iteration, which sees it here.
217 self._last_data = last_data
218 blocking = not (
219 last_data is not None # don't block if we have a batch to be processed
220 or self.prefill_manager.runnable
221 or self.decode_manager.runnable
222 or self._pending_rebuild is not None # a queued rebuild to drain toward + execute
223 )
224 for msg in self.receive_msg(blocking=blocking):
225 self._process_one_msg(msg)
226
227 # Execute a queued cache rebuild once the scheduler is fully idle (the safe point):
228 # no last batch to process, no pending prefill, no running decode. finished_reqs is
229 # NOT a gate — those requests are already freed (no live GPU/page resources).
230 if self._pending_rebuild is not None and last_data is None and not (
231 self.prefill_manager.runnable or self.decode_manager.runnable
232 ):
233 self._execute_pending_rebuild()
234
235 # Order this iteration's host->device token_pool copies (issued on ``self.stream``
236 # during scheduling) after the previous batch's sampled-token writes (issued on the
237 # engine stream in ``_forward``). Without this, a request that reuses a just-freed
238 # table_idx can have its freshly copied prompt clobbered by the prior occupant's
239 # still-pending output write -- corrupting tokens (e.g. dropping an image
240 # placeholder, which the multimodal merge then rejects).
241 self.stream.wait_stream(self.engine.stream)
242 forward_input = self._schedule_next_batch()
243 ongoing_data = None
244 if forward_input is not None:
245 with self.engine_stream_ctx: # run the batch in the engine's stream
246 self.engine.stream.wait_stream(self.stream)
247 # COW-restore GDN snapshots for prefix hits ON THE ENGINE STREAM, after the
248 # cross-stream wait and before the forward reads the live slot (program order
249 # vs the prior batch's snapshot writes). Doing this on self.stream would race.
250 self._restore_linear_states(forward_input.batch)
251 ongoing_data = (forward_input, self._forward(forward_input))
252
253 # The drain issues GPU-visible writes to state the batch just launched still reads: the
254 # page-table re-point and, for the paged-SWA pools, the full->swa (DSV4: full->window)
255 # sentinel scatter. DSV4 stages the page table at replay time and translates
256 # full_to_window INSIDE the captured graph, so an unordered drain can redirect an
257 # in-flight forward. copy_done only covers batch N; order against N+1 explicitly.
258 self.stream.wait_stream(self.engine.stream)
259 self._process_last_data(last_data)
260 self._flush_abort_acks()
261 return ongoing_data

263 def normal_loop(self) -> None:
264 blocking = not (
265 self.prefill_manager.runnable
266 or self.decode_manager.runnable
267 or self._pending_rebuild is not None # a queued rebuild to execute at idle
268 )
269 for msg in self.receive_msg(blocking=blocking):
270 self._process_one_msg(msg)
271
272 # Non-overlap mode has no last_data to drain; execute a queued rebuild as soon as
273 # the scheduler is idle (no pending prefill / running decode). Without this, a
274 # rebuild in DISABLE_OVERLAP_SCHEDULING mode stays pending until the HTTP timeout.
275 if self._pending_rebuild is not None and not (
276 self.prefill_manager.runnable or self.decode_manager.runnable
277 ):
278 self._execute_pending_rebuild()
279
280 forward_input = self._schedule_next_batch()
281 ongoing_data = None
282 if forward_input is not None:
283 # already inside engine_stream_ctx (run_forever); restore on the engine stream
284 self._restore_linear_states(forward_input.batch)
285 ongoing_data = (forward_input, self._forward(forward_input))
286
287 self._process_last_data(ongoing_data)
288 self._flush_abort_acks()

290 @torch.inference_mode()
291 def run_forever(self) -> NoReturn:
292 # DSV4 (owned-KV) decode reads its per-token window/cmp/idx slot maps off the attention
293 # backend's per-batch SNAPSHOT (staged in prepare_for_replay right before the replay, on
294 # the same stream, like the generic out_loc copy_from), not the live slot maps -- so the
295 # next batch's allocate_paged cannot corrupt the in-flight graph replay. DSV4 overlaps.
296 if ENV.DISABLE_OVERLAP_SCHEDULING:
297 with self.engine_stream_ctx:
298 self.engine.stream.wait_stream(self.stream)
299 while True:
300 self.normal_loop()
301 else:
302 assert torch.cuda.current_stream() == self.stream
303 data = None
304 while True:
305 data = self.overlap_loop(data)

312 def _process_last_data(self, last_data: ForwardData | None) -> None:
313 if last_data is None:
314 return
315
316 batch, (_, next_tokens_cpu, copy_done) = last_data[0].batch, last_data[1]
317 copy_done.synchronize()
318 reply: List[DetokenizeMsg] = []
319 new_finished_reqs: Set[Req] = set()
320 with self.cache_manager.lazy_free_region():
321 for i, req in enumerate(batch.reqs):
322 if isinstance(req, ChunkedReq):
323 # Don't cache intermediate chunks; the full prompt is cached once when the
324 # final chunk is processed. Caching here snapshots a handle the next chunk
325 # already copied (overlap), so cache_req double-frees the prior chunk.
326 if req.aborted:
327 # Aborted mid-chunked-prefill while this chunk was in flight: the abort
328 # popped the pending continuation (no next chunk launches), and this
329 # drain point frees the chunk's pages/slots exactly once.
330 self._free_req_resources(req)
331 continue
332 if req.aborted:
333 # Aborted while this final-chunk prefill / decode step was in flight: free
334 # here (the forward is drained) and finish the request. No DetokenizeMsg --
335 # the abort ack flushed after this method stays the uid's terminal reply.
336 self.decode_manager.remove_req(req)
337 self._free_req_resources(req)
338 new_finished_reqs.add(req)
339 continue
340 if req in self.finished_reqs:
341 # Overlap scheduling launched one more decode step for a request that
342 # already terminated (filter_reqs keeps it while output budget remains,
343 # and the next batch is scheduled before this drain runs). Its resources
344 # are freed below/already; shipping this token would append past the
345 # client's terminal reply.
346 continue
347 next_token = next_tokens_cpu[i]
348 req.append_host(next_token.unsqueeze(0))
349 next_token = int(next_token.item())
350 # EOS / stop-string -> "stop", output budget exhausted -> "length";
351 # EOS and stop strings win over length.
352 # Overlap can advance device_len one token ahead of this host reply.
353 hit_length = req.input_ids.numel() >= req.max_device_len
354 hit_eos = (
355 not req.sampling_params.ignore_eos and next_token in self.eos_token_ids
356 )
357 matched_stop = (
358 self._match_stop_str(req)
359 if not hit_eos and req.sampling_params.stop_strs
360 else None
361 )
362 finished = hit_length or hit_eos or matched_stop is not None
363 finish_reason = (
364 ("stop" if (hit_eos or matched_stop is not None) else "length")
365 if finished
366 else None
367 )
368 if (
369 next_token == self.toolcall_anchor_id
370 and req.toolcall_anchor_len is None
371 and not finished
372 ):
373 req.toolcall_anchor_len = req.input_ids.numel()
374 reply.append(
375 DetokenizeMsg(
376 uid=req.uid,
377 next_token=next_token,
378 finished=finished,
379 finish_reason=finish_reason,
380 matched_stop=matched_stop,
381 stop_strs=req.sampling_params.stop_strs or None,
382 )
383 )
384
385 # NOTE: overlap scheduling may make the request freed twice, skip second free
386 if finished and req not in self.finished_reqs:
387 self.decode_manager.remove_req(req)
388 self._free_req_resources(req)
389 new_finished_reqs.add(req)
390 elif batch.is_prefill and req.table_idx != -1:
391 # for prefill, non-chunk req, cache the prefix.
392 # Polymorphic: the DSV4 naive manager keeps the request's slots (no-op);
393 # the generic manager inserts the prefix into its radix/naive cache.
394 # table_idx == -1 is defense-in-depth: aborts mark in-flight requests
395 # instead of freeing them (handled above), so a freed request should
396 # never reach this commit -- but if a future path frees one early, skip
397 # rather than re-read the freed page-table row (and on hybrid, deref the
398 # None'd GDN ping-pong slots).
399 self.cache_manager.cache_req(req, finished=False)
400
401 self.finished_reqs = new_finished_reqs
402 # Stamp each reply with the post-batch KV page occupancy so the frontend (shell
403 # status bar) can show live KV usage without a separate query.
404 used, total = self._kv_usage_pages()
405 mamba_slots = self._mamba_slot_usage()
406 swa_tokens = self._swa_token_usage()
407 if reply:
408 mem = self._gpu_mem_bytes()
409 mamba_used, mamba_total = mamba_slots or (0, 0)
410 swa_used, swa_total = swa_tokens or (0, 0)
411 for m in reply:
412 m.kv_used_pages = used
413 m.kv_total_pages = total
414 m.mamba_used_slots = mamba_used
415 m.mamba_total_slots = mamba_total
416 m.swa_used_tokens = swa_used
417 m.swa_total_tokens = swa_total
418 m.gpu_mem_bytes = mem
419 self.status_reporter.report_batch(
420 batch,
421 running_reqs=len(self.decode_manager.running_reqs),
422 queue_reqs=len(self.prefill_manager.pending_list),
423 kv_used_pages=used,
424 kv_total_pages=total,
425 page_size=self.config.page_size,
426 mamba_slots=mamba_slots,
427 swa_tokens=swa_tokens,
428 )
429 self.send_result(reply)

431 def _match_stop_str(self, req: Req) -> str | None:
432 """First stop string present in this request's generated tail, else None. Decodes
433 only a short suffix (bounded by the longest stop string's char length, so a stop of
434 N chars spans at most N tokens) to keep the per-step cost small."""
435 stop_strs = req.sampling_params.stop_strs
436 prompt_len = req.max_device_len - req.output_len
437 if len(req.input_ids) <= prompt_len:
438 return None
439 max_chars = max(len(s) for s in stop_strs)
440 tail_start = max(prompt_len, len(req.input_ids) - (max_chars + 1))
441 tail = self.tokenizer.decode(req.input_ids[tail_start:].tolist())
442 for s in stop_strs:
443 if s in tail:
444 return s
445 return None

447 def _kv_usage_pages(self) -> Tuple[int, int]:
448 """(used_pages, total_pages) of the KV page pool.
449
450 ``used`` follows SGLang's logging semantics: allocated pages that are not
451 evictable (active requests + protected prefix cache). Evictable prefix-cache
452 pages are available to future requests, so they are excluded from usage.
453 Always the manager's own primary pool (for DSV4 the FULL cmp/idx tier); the
454 window (swa) tier is reported separately by ``_swa_token_usage``.
455 """
456 return self.cache_manager.page_usage()
457
458 def _mamba_slot_usage(self) -> Tuple[int, int] | None:
459 """(used_slots, total_slots) of the GDN-state (mamba) pool for hybrid models, else None.
460
461 Mirrors SGLang's mamba-pool semantics: ``total`` excludes the reserved padding
462 sink (slot 0); ``used`` excludes free slots and evictable tree snapshots.
463 """
464 if not self.cache_manager.is_hybrid:
465 return None
466 total = self.cache_manager.linear_state_pool.num_slots - 1
467 return total - self.cache_manager.mamba_available_size, total
468
469 def _swa_token_usage(self) -> Tuple[int, int] | None:
470 """(used_tokens, total_tokens) of the window (swa) pool for SWA models, else None.
471
472 Mirrors the mamba accounting: ``total`` excludes the pool's reserved sentinel
473 unit; ``used`` excludes free slots and evictable (unlocked) tree tokens.
474 """
475 cm = self.cache_manager
476 if not cm.swa_paged:
477 return None
478 total = cm.swa_pool.swa_num_tokens - 1
479 return total - cm.swa_available_size, total

481 def _gpu_mem_bytes(self) -> int:
482 """Bytes this engine process holds on the GPU (torch's reserved caching-allocator
483 pool: weights + KV + MoE cache + graphs). 0 on CPU. Cheap, no device sync."""
484 if self.device.type != "cuda":
485 return 0
486 return torch.cuda.memory_reserved(self.device)

488 def _process_one_msg(self, msg: BaseBackendMsg) -> None:
489 if isinstance(msg, BatchBackendMsg):
490 for msg in msg.data:
491 self._process_one_msg(msg)
492 elif isinstance(msg, ExitMsg):
493 raise KeyboardInterrupt
494 elif isinstance(msg, UserMsg):
495 logger.debug_rank0("Received user msg: %s", msg)
496 tombstones = getattr(self, "_abort_tombstones", None)
497 if tombstones is not None and msg.uid in tombstones:
498 tombstones.pop(msg.uid, None)
499 logger.debug_rank0(
500 "Dropping request %d because its abort arrived before admission", msg.uid
501 )
502 return
503 if msg.mm_items and self.engine.encoder_cache is None:
504 # no encoder runtime: fail loudly instead of decoding unexpanded placeholders
505 self.send_result(
506 [
507 ErrorReplyMsg(
508 uid=msg.uid,
509 error="image input is not supported by this server",
510 )
511 ]
512 )
513 return
514 input_len, max_seq_len = len(msg.input_ids), self.engine.max_seq_len
515 max_output_len = max_seq_len - input_len
516 if max_output_len <= 0:
517 logger.warning_rank0(
518 f"Input sequence length {input_len} exceeds {max_seq_len}, "
519 f"request {msg.uid} is dropped."
520 )
521 # Tell the client instead of dropping silently — otherwise its wait_for_ack
522 # never sees a `finished` reply and hangs until the request times out.
523 self.send_result(
524 [
525 ErrorReplyMsg(
526 uid=msg.uid,
527 # "prompt is too long: N tokens > M" is the phrasing Claude Code and
528 # OpenClaw match on; the Anthropic wire has no error code to read.
529 error=(
530 f"prompt is too long: {input_len} tokens > {max_seq_len} maximum "
531 f"(prompt + generation); shorten the prompt or increase the KV "
532 f"cache budget"
533 ),
534 # OpenAI's standard class for this, for clients that read a code.
535 code="context_length_exceeded",
536 )
537 ]
538 )
539 return
540 if msg.sampling_params.max_tokens > max_output_len:
541 msg.sampling_params.max_tokens = max_output_len
542 logger.warning_rank0(
543 f"Adjust max_tokens to {max_output_len} for request {msg.uid}."
544 )
545 self.prefill_manager.add_one_req(msg)
546 elif isinstance(msg, AbortBackendMsg):
547 logger.debug_rank0("Aborting request %d", msg.uid)
548 tombstones = getattr(self, "_abort_tombstones", None)
549 if tombstones is None:
550 tombstones = self._abort_tombstones = {}
551 tombstones[msg.uid] = None
552 # Unknown aborts normally consume their tombstone when the cross-worker UserMsg
553 # catches up. Bound hostile/no-followup abort traffic without affecting realistic
554 # in-flight concurrency.
555 while len(tombstones) > 65_536:
556 tombstones.pop(next(iter(tombstones)))
557 req_to_free = self.prefill_manager.abort_req(msg.uid)
558 req_to_free = req_to_free or self.decode_manager.abort_req(msg.uid)
559 if (
560 req_to_free is not None
561 and req_to_free.mm_items
562 and self.engine.encoder_cache is not None
563 ):
564 # drop the aborted request's claims; entries it held alone die here
565 self.engine.encoder_cache.release(
566 msg.uid, [item.hash for item in req_to_free.mm_items]
567 )
568 if req_to_free is not None:
569 # SGLang-style abort: never free resources under an in-flight forward. If the
570 # request is in the launched-but-not-drained batch (overlap), only mark it;
571 # _process_last_data frees it this same iteration, after copy_done.synchronize()
572 # -- so its KV pages / GDN slots are never recycled mid-write, and the
573 # finished=False prefix-commit can't run on a freed request. A request with no
574 # forward in flight (e.g. a decode req starved behind a long chunked prefill)
575 # is freed immediately -- deferring would leak until its next batch, which
576 # strict prefill-priority puts arbitrarily far away.
577 inflight = (
578 self._last_data is not None
579 and req_to_free in self._last_data[0].batch.reqs
580 )
581 if inflight:
582 req_to_free.aborted = True
583 else:
584 self._free_req_resources(req_to_free)
585 # Always acknowledge the abort, even when the request already left the manager,
586 # but NOT yet: overlap_loop still has to publish the prior forward's sampled reply.
587 # _flush_abort_acks runs after _process_last_data, making this a true terminal
588 # accounting barrier for FrontendManager/prepare-stop.
589 self._pending_abort_acks.add(msg.uid)
590 elif isinstance(msg, CacheRebuildBackendMsg):
591 # v1 scope: only if_idle, single-rank, non-owned-KV. drain mode and TP rebuild
592 # need the drain-gate / all-rank failure-agreement machinery (deferred), so we
593 # reject them cleanly rather than ship hang-prone half-wired paths.
594 if not self.cache_manager.supports_runtime_rebuild:
595 self._reply_rebuild(
596 msg.request_id, "unsupported", "this model's cache does not support runtime rebuild"
597 )
598 elif msg.mode != "if_idle":
599 self._reply_rebuild(
600 msg.request_id, "unsupported", f"mode {msg.mode!r} unsupported (use if_idle)"
601 )
602 elif self.config.tp_info.size > 1:
603 self._reply_rebuild(
604 msg.request_id, "unsupported", "runtime rebuild unsupported under TP > 1"
605 )
606 elif self.prefill_manager.runnable or self.decode_manager.runnable:
607 # if_idle: refuse rather than wait. (finished_reqs hold no resources — they
608 # are already freed — so they do not block a rebuild.)
609 self._reply_rebuild(msg.request_id, "busy")
610 else:
611 self._pending_rebuild = msg
612 else:
613 logger.error(f"Unknown message type: {type(msg)}")
614 raise NotImplementedError

616 def _restore_linear_states(self, batch) -> None:
617 """COW-restore a hybrid prefix hit's GDN snapshot into its freshly-allocated live slot
618 (first chunk only). MUST run on the ENGINE stream so it is program-ordered after the
619 prior batch's snapshot writes and before this forward reads the live slot."""
620 pool = self.engine.linear_state_pool
621 if pool is None or not batch.is_prefill:
622 return
623 for req in batch.reqs:
624 if req.mamba_restore_src is not None:
625 pool.copy_from(req.mamba_restore_src, req.linear_slot_idx)
626 req.mamba_restore_src = None # consumed: restore exactly once

628 def _free_req_resources(self, req: Req) -> None:
629 # Idempotent: an EOS-finished request can stay in running_reqs (output budget left), so an
630 # abort in the same overlap iteration races _process_last_data and would free it twice --
631 # double-freeing its table_idx and (hybrid) GDN slots onto the free-list, handing the same
632 # slots to two later requests. table_idx == -1 marks an already-freed request.
633 if req.table_idx == -1:
634 return
635 # Polymorphic free: the DSV4 manager returns the request's window pages + cmp/idx blocks
636 # to their tier free-lists; the generic manager frees its KV pages (it reads
637 # page_table[req.table_idx], so free the table entry after).
638 self.cache_manager.cache_req(req, finished=True)
639 self.table_manager.free(req.table_idx)
640 req.table_idx = -1

642 def _reply_rebuild(self, request_id: str, status: str, error: str | None = None) -> None:
643 # Single source of truth with the rollback snapshot (_current_cache_geometry): mamba is
644 # usable slots (padding sink excluded, matching the status-bar gauge), and num_swa_pages
645 # reports 0 unless the model actually has a window pool.
646 geo = self._current_cache_geometry()
647 self.send_result(
648 [
649 CacheRebuildResultMsg(
650 request_id=request_id,
651 status=status,
652 moe_cache_size=geo["moe_cache_size"] or 0,
653 num_pages=geo["num_pages"],
654 mamba_slots=geo["num_mamba_slots"] or 0,
655 num_swa_pages=geo["num_swa_pages"] or 0,
656 error=error,
657 )
658 ]
659 )

661 def _execute_pending_rebuild(self) -> None:
662 from freetoken.engine.engine import CacheRebuildRejected
663
664 msg = self._pending_rebuild
665 assert msg is not None
666 self._pending_rebuild = None
667 requested = {
668 "moe_cache_size": msg.moe_cache_size,
669 "num_pages": msg.num_pages,
670 "num_mamba_slots": msg.num_mamba_slots,
671 "num_swa_pages": msg.num_swa_pages,
672 }
673 # Rollback target: the CURRENT (serving) sizes of ONLY the pools this request touches.
674 # Passing the untouched pools too would trip rebuild_cache's KV/mamba/SWA gate and wipe
675 # the prefix cache that a successful resize of just the requested pool preserves.
676 snapshot = self._current_cache_geometry()
677 prior = {k: snapshot[k] for k, v in requested.items() if v is not None}
678 # Cleared here, set by engine.rebuild_runtime_cache at its point of no return — lets the
679 # except below tell a pre-teardown failure (engine untouched) from a mid-teardown one.
680 self.engine.rebuild_teardown_started = False
681 try:
682 self.rebuild_cache(**requested)
683 except CacheRebuildRejected as e:
684 # Rejected before any destructive free — old cache intact, keep serving.
685 logger.warning(f"cache rebuild rejected: {e}")
686 self._reply_rebuild(msg.request_id, "rejected", error=str(e))
687 return
688 except Exception as e: # noqa: BLE001
689 if not getattr(self.engine, "rebuild_teardown_started", True):
690 # Failed before the destructive phase began: graphs and pools are untouched and
691 # the engine is still serving. A destructive rollback would only add risk.
692 logger.error(f"cache rebuild failed before teardown: {e!r} — old cache intact")
693 self._reply_rebuild(msg.request_id, "rejected", error=repr(e))
694 return
695 if self.config.tp_info.size > 1:
696 # A lone-rank failure cannot be rolled back symmetrically: rebuild_cache runs TP
697 # barriers, and ranks that succeeded will not re-enter them — a solo rollback
698 # would desync the group. Keep the latch-failed behavior for tp>1.
699 logger.error(f"cache rebuild failed: {e!r} — tp>1, latching failed")
700 self._reply_rebuild(msg.request_id, "failed", error=repr(e))
701 return
702 # The destructive phase failed — typically a CUDA OOM while reallocating a pool or
703 # recapturing graphs. The graphs/pools are already torn down, so the engine cannot
704 # serve as-is. Rather than latch "failed" (which forces a full process restart),
705 # rebuild the touched pools back to the sizes that were serving a moment ago: they
706 # fit before, so shrinking back frees the just-attempted allocation and restores
707 # service. Only if the rollback ALSO fails is the engine genuinely wedged. (Post-OOM
708 # CUDA state is not guaranteed sane — a rollback that succeeds here may still surface
709 # a deferred fault on a later request; that residual risk is accepted over always
710 # forcing a restart.)
711 logger.error(f"cache rebuild failed: {e!r} — rolling back to the previous geometry")
712 try:
713 self.rebuild_cache(**prior)
714 except Exception as e2: # noqa: BLE001 — rollback failed too; genuinely unrecoverable
715 logger.error(f"cache rebuild rollback failed: {e2!r} — server latched failed")
716 self._reply_rebuild(
717 msg.request_id,
718 "failed",
719 error=f"{e!r}; rollback to the prior geometry also failed: {e2!r}",
720 )
721 return
722 logger.warning("cache rebuild rolled back to the previous geometry — still serving")
723 self._log_cache_geometry("Cache rolled back")
724 self._reply_rebuild(
725 msg.request_id, "rejected", error=f"rebuild failed and was rolled back: {e!r}"
726 )
727 return
728 # Outside the try: an ack/send failure after a fully-applied rebuild must not be
729 # mistaken for a rebuild failure and roll back the geometry the engine now serves.
730 self._log_cache_geometry("Cache rebuilt")
731 self._reply_rebuild(msg.request_id, "ok")

733 def _current_cache_geometry(self) -> dict:
734 """The pools' current (serving) sizes as rebuild_cache kwargs — the rollback snapshot and
735 the single source for _reply_rebuild's readout. None for a pool this model lacks
736 (rebuild_cache skips those; the reply maps them to the wire format's 0). num_swa_pages is
737 the CONCRETE current window (usable pages) so a rollback restores it byte-for-byte,
738 whether it was pinned or ratio-derived."""
739 eng = self.engine
740 from freetoken.kvcache.cache_status import window_pool_spec
741
742 spec = window_pool_spec(self.config)
743 num_swa_pages = eng.kv_cache.window_pages if spec is not None else None
744 return dict(
745 num_pages=eng.num_pages,
746 moe_cache_size=eng.moe_offload_cache.cache_size if eng.moe_offload_cache is not None else None,
747 num_mamba_slots=(eng.linear_state_pool.num_slots - 1) if eng.linear_state_pool is not None else None,
748 num_swa_pages=num_swa_pages,
749 )

751 def _log_cache_geometry(self, event: str) -> None:
752 """One-line readout of every pool's new size + VRAM after a rebuild changed them:
753 full KV always; swa/mamba/MoE only for models with the pool. Byte figures are
754 best-effort (0 when a unit cost cannot be measured) and must never block the reply."""
755 from freetoken.kvcache.cache_status import compute_cache_pools, compute_cache_unit_bytes
756
757 try:
758 pools = compute_cache_pools(self.engine)
759 unit = compute_cache_unit_bytes(self.engine)
760 kv_tokens = pools["num_pages"] * pools["page_size"]
761 parts = [
762 f"KV {pools['num_pages']} pages"
763 f" ({kv_tokens} tokens, {_gib(kv_tokens * unit['kv_bytes_per_token'])})"
764 ]
765 if pools["num_swa_pages"]:
766 swa_tokens = pools["num_swa_pages"] * pools["swa_page_size"]
767 parts.append(
768 f"swa {pools['num_swa_pages']} pages"
769 f" ({swa_tokens} tokens, {_gib(swa_tokens * unit['swa_bytes_per_token'])})"
770 )
771 if pools["num_mamba_slots"]:
772 parts.append(
773 f"mamba {pools['num_mamba_slots']} slots"
774 f" ({_gib(pools['num_mamba_slots'] * unit['mamba_bytes_per_slot'])})"
775 )
776 moe = self.engine.moe_offload_cache
777 if moe is not None:
778 parts.append(
779 f"MoE cache {moe.cache_size}/{moe.num_layers * moe.num_experts}"
780 f" ({_gib(moe.cache_size * unit['moe_bytes_per_expert'])})"
781 )
782 logger.info_rank0(f"{event}: " + ", ".join(parts))
783 except Exception as e: # noqa: BLE001
784 logger.warning(f"could not log cache geometry: {e!r}")

786 def _prepare_batch(self, batch: Batch) -> ForwardInput:
787 self.engine.graph_runner.pad_batch(batch)
788 self._forward_iter += 1
789 if batch.is_decode:
790 # Free each decoding request's now-out-of-window SWA slots BEFORE the alloc below,
791 # so they can back the new token -- this is what bounds the per-request swa
792 # footprint during decode. (no-op unless the model is SWA / paged swa pool.)
793 self.cache_manager.maybe_free_swa_out_of_window(
794 batch.reqs, forward_iter=self._forward_iter)
795 for req in batch.reqs:
796 req.decode_batch_idx += 1
797 else:
798 # Prefill sibling of the decode driver: free out-of-window swa BEFORE allocating
799 # this chunk, so a chunked prompt longer than the swa pool never accumulates its
800 # whole swa footprint (which would exhaust alloc_swa). No-op unless SWA/paged.
801 self.cache_manager.free_swa_out_of_window_extend(batch.reqs)
802 # Polymorphic page allocation: DSV4 allocates window pages + cmp/idx blocks into its
803 # slot maps; the generic manager allocates KV pages into the page table.
804 self.cache_manager.allocate_paged(batch.reqs)
805 if batch.is_prefill:
806 self._gather_multimodal(batch)
807 batch.positions = _make_positions(batch, self.device)
808 if self._model_is_mrope:
809 batch.mrope_positions = _make_mrope_positions(batch, self.device)
810 input_mapping = _make_input_tuple(batch, self.device)
811 write_mapping = _make_write_tuple(batch, self.device)
812 batch.out_loc = self.engine.page_table[input_mapping]
813 if self.engine.linear_state_pool is not None:
814 if batch.is_decode:
815 # GPU GDN-state slot (one per padded request) for the decode gather/scatter;
816 # lands in the CUDA-graph input buffer via copy_from. Gate on the cache mode,
817 # NOT on whether any padded req has a linear_slot_idx -- the persistent dummy
818 # req always carries one (= padding_slot), so that test is True even for naive
819 # and would collapse all real naive reqs onto the padding slot. Hybrid: build
820 # per padded req from Req.linear_slot_idx (dummy -> padding_slot). Naive: keep
821 # the old keying = input_mapping's table_idx column (already staged, no H2D).
822 if self.cache_manager.is_hybrid:
823 pool = self.engine.linear_state_pool
824 slots = [r.linear_slot_idx if r.linear_slot_idx is not None
825 else pool.padding_slot for r in batch.padded_reqs]
826 batch.linear_table_idx = torch.tensor(
827 slots, dtype=torch.int32, device="cpu", pin_memory=True
828 ).to(self.device, non_blocking=True)
829 else:
830 batch.linear_table_idx = input_mapping[0].to(torch.int32)
831 # Per-forward GDN metadata (cu_seqlens / cache_indices / continuation flags),
832 # built once here instead of rebuilt in each of the 30 GDN layers. For decode
833 # under CUDA graph the persistent cu_seqlens buffer is supplied by set_batch.
834 batch.fla_metadata = build_fla_metadata(batch, self.device)
835 if batch.is_decode:
836 # This batch's padded per-row page-table rows. Backends that snapshot the table for
837 # a captured replay (DSV4) read them in prepare_metadata / prepare_for_replay.
838 batch.active_table_idx = input_mapping[0].view(-1)
839 self.engine.attn_backend.prepare_metadata(batch)
840 return ForwardInput(
841 batch=batch,
842 sample_args=self.engine.sampler.prepare(batch),
843 input_tuple=input_mapping,
844 write_tuple=write_mapping,
845 )

847 def _gather_multimodal(self, batch: Batch) -> None:
848 """Plan the chunk's encoder jobs, gather rows and scatter rows over the batch; the engine runs them before the LM forward."""
849 jobs, plan, rows, block_ends = plan_mm_batch(batch.padded_reqs, self.engine.encoder_cache)
850 if plan:
851 batch.mm_encoder_jobs = jobs
852 batch.mm_gather_plan = plan
853 batch.mm_rows = torch.tensor(rows, dtype=torch.int64, pin_memory=True).to(self.device, non_blocking=True)
854 batch.mm_block_ends = torch.tensor(block_ends, dtype=torch.int32, pin_memory=True).to(self.device, non_blocking=True)
855 if self._bidirectional_mm and not self._warned_cut_image and (cut := cut_image_spans(batch.padded_reqs)):
856 # only a bidirectional image span loses context when cut, and only an image longer than the chunk still gets cut
857 lo, hi = cut[0]
858 self._warned_cut_image = True
859 logger.warning_rank0(
860 f"an image of {hi - lo} tokens does not fit one prefill chunk (--max-extend-tokens {self.prefill_budget}, or the sliding-window pool's share of it): its earlier rows attend within the first part only"
861 )

890 def _flush_abort_acks(self) -> None:
891 pending = getattr(self, "_pending_abort_acks", None)
892 if not pending:

898 def _forward(self, forward_input: ForwardInput) -> ForwardOutput:
899 batch, sample_args, input_mapping, output_mapping = forward_input
900 batch.input_ids = self.token_pool[input_mapping]
901 if self.toolcall_anchor_id is not None and not batch.is_prefill:

906 return forward_output

909def _make_mrope_positions(batch: Batch, device: torch.device) -> torch.Tensor:
910 """[3, N] rope rows: an image request's prompt tokens use their precomputed columns, everything else is sequence index + per-request delta."""
911 needed = sum(r.extend_len for r in batch.padded_reqs)
912 host = torch.empty((3, needed), dtype=torch.int32, pin_memory=True)
913 offset = 0
914 for req in batch.padded_reqs:
915 length = req.extend_len
916 out = host[:, offset : offset + length]
917 full = req.mrope_positions_full
918 if full is not None and req.device_len <= full.shape[1]:
919 out.copy_(full[:, req.cached_len : req.device_len])
920 else:
921 row = torch.arange(
922 req.cached_len + req.mrope_delta,
923 req.device_len + req.mrope_delta,
924 dtype=torch.int32,
925 )
926 out.copy_(row.unsqueeze(0).expand(3, -1))
927 offset += length
928 return host.to(device, non_blocking=True)

931def _make_positions(batch: Batch, device: torch.device) -> torch.Tensor:
932 needed_size = sum(r.extend_len for r in batch.padded_reqs)
933 indices_host = torch.empty(needed_size, dtype=torch.int32, pin_memory=True)
934 offset = 0
935 for req in batch.padded_reqs:
936 length = req.extend_len
937 torch.arange(
938 req.cached_len,
939 req.device_len,
940 dtype=torch.int32,
941 out=indices_host[offset : offset + length],
942 )
943 offset += length
944 return indices_host.to(device, non_blocking=True)

947def _make_input_tuple(batch: Batch, device: torch.device) -> Indice2D:
948 mapping_host = torch.empty(len(batch.positions), dtype=torch.int64, pin_memory=True)
949 offset = 0

957def _make_write_tuple(batch: Batch, device: torch.device) -> Indice2D:
958 mapping_list = [req.table_idx for req in batch.reqs]
