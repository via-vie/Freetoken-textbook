18def _report_startup_error(ack_queue: mp.Queue, exc: BaseException) -> None:
19 """Tell the parent WHY this worker is dying — push an ("error", reason) ack before it exits,
20 so the supervisor reports the real cause (e.g. a config ValueError) instead of the generic
21 "backend worker … exited during load". Best-effort and flushed (close + join_thread), since
22 the process is about to terminate; a failure to report must never mask the original error."""
23 try:
24 ack_queue.put(("error", f"{type(exc).__name__}: {exc}"))
25 ack_queue.close()
26 ack_queue.join_thread()
27 except Exception: # noqa: BLE001 -- reporting is a nicety; never shadow the real exception
28 pass

31def _detach_process_group() -> None:
32 """Shell mode only: move this worker out of the terminal's foreground process group.
33
34 The shell binds ^C to "cancel this turn", but a terminal delivers SIGINT to the whole
35 foreground group — which, with the engine running in this same process, includes the
36 workers. They would take the same ^C and exit (``_run_scheduler`` below stops gracefully on
37 KeyboardInterrupt), leaving the shell chatting with a dead engine. Nothing depends on the
38 signal reaching them: the parent tears the workers down explicitly on every stop path
39 (uvicorn's lifespan, plus the SIGTERM/SIGHUP handler and the reap backstop in api_server).
40
41 ``ft serve`` keeps the default — uvicorn owns ^C there, and the group-wide delivery is part
42 of how it stops."""
43 try:
44 os.setpgrp()
45 except OSError: # no job control (already a group leader / unusual environment)
46 pass

58def _run_scheduler(args: ServerArgs, ack_queue: mp.Queue[str]) -> None:
59 if args.shell_mode:
60 _detach_process_group()
61
62 # published (not bound) here: the engine binds it after the allocator setup
63 from freetoken.gpu_select import set_assigned_gpu
64
65 # resolved UUIDs when we have them, the raw --gpu entries when NVML could not resolve them, else one CUDA ordinal per rank
66 targets = args.gpu_assigned or args.gpu or tuple(str(r) for r in range(args.tp_info.size))
67 set_assigned_gpu(targets[args.tp_info.rank])
68
69 import torch
70 from freetoken.scheduler import Scheduler
71
72 if args.tp_info.is_primary():
73 from freetoken.utils.progress import set_progress_sink
74
75 set_progress_sink(
76 lambda desc, done, total: ack_queue.put(("progress", desc, done, total))
77 )
78
79 with torch.inference_mode():
80 try:
81 scheduler = Scheduler(args)
82 scheduler.sync_all_ranks()
83 except Exception as exc: # noqa: BLE001 -- surface the reason, then let it propagate
84 # A startup failure (bad config, OOM, corrupt weights) would otherwise reach the
85 # parent only as a dead process -> a generic "exited during load". Push the real
86 # reason first so the supervisor (and the desktop failure modal) can surface it;
87 # the traceback still prints and the process still exits non-zero.
88 _report_startup_error(ack_queue, exc)
89 raise
90
91 if args.tp_info.is_primary():
92 # Report the real per-unit cache VRAM costs (KV/expert/mamba), the device-wide free
93 # VRAM, and the per-pool rebuild floors before the ready ack, so the supervisor has
94 # them (and the desktop's slider bounds) by the time the gate flips. Optional +
95 # best-effort: a failure here must never keep the model from serving, and older
96 # consumers ignore ("meta", …).
97 try:
98 from freetoken.kvcache.cache_status import compute_cache_status_meta
99
100 meta = compute_cache_status_meta(scheduler.engine)
101 # the parent must not touch CUDA to learn this
102 meta["gpus"] = scheduler.gpus
103 ack_queue.put(("meta", meta))
104 except Exception: # noqa: BLE001 -- metadata is a nicety; readiness is not
105 pass
106 ack_queue.put("Scheduler is ready")
107 # The supervisor stops draining ack_queue once ready, so uninstall the sink:
108 # runtime cache rebuilds re-run the graph capture (which emits progress) and
109 # would otherwise push onto a queue nobody reads for the server's lifetime.
110 set_progress_sink(None)
111
112 if args.silent_output:
113 logging.disable(logging.INFO)
114
115 try:
116 scheduler.run_forever()
117 except KeyboardInterrupt:
118 logger = init_logger(__name__)
119 if args.tp_info.is_primary():
120 print() # for a clean newline after ^C
121 logger.info("Scheduler exiting gracefully...")
122 scheduler.shutdown()

125def launch_server(
126 run_shell: bool = False,
127 argv: list[str] | None = None,
128 prog: str | None = None,

153 def start_subprocess() -> "BackendHandle":
154 import multiprocessing as mp
155
156 from .supervisor import BackendHandle
157
158 mp.set_start_method("spawn", force=True)
159 detach = server_args.shell_mode # see _detach_process_group
160
161 world_size = server_args.tp_info.size
162 ack_queue: mp.Queue = mp.Queue()
163 processes: list[mp.Process] = []
164
165 for i in range(world_size):
166 new_args = replace(server_args, tp_info=DistributedInfo(i, world_size))
167 p = mp.Process(
168 target=_run_scheduler,
169 args=(new_args, ack_queue),
170 daemon=False,
171 name=f"freetoken-TP{i}-scheduler",
172 )
173 p.start()
174 processes.append(p)
175
176 num_tokenizers = server_args.num_tokenizer
177 p = mp.Process(
178 target=_run_tokenize_worker,
179 kwargs={
180 "detach": detach,
181 "tokenizer_path": server_args.model_path,
182 "addr": server_args.zmq_detokenizer_addr,
183 "backend_addr": server_args.zmq_backend_addr,
184 "frontend_addr": server_args.zmq_frontend_addr,
185 "local_bs": 1,
186 "mm": server_args.mm,
187 "create": server_args.tokenizer_create_addr,
188 "tokenizer_id": num_tokenizers,
189 "ack_queue": ack_queue,
190 },
191 daemon=False,
192 name="freetoken-detokenizer-0",
193 )
194 p.start()
195 processes.append(p)
196 for i in range(num_tokenizers):
197 p = mp.Process(
198 target=_run_tokenize_worker,
199 kwargs={
200 "detach": detach,
201 "tokenizer_path": server_args.model_path,
202 "addr": server_args.zmq_tokenizer_addr,
203 "backend_addr": server_args.zmq_backend_addr,
204 "frontend_addr": server_args.zmq_frontend_addr,
205 "local_bs": 1,
206 "mm": server_args.mm,
207 "create": server_args.tokenizer_create_addr,
208 "tokenizer_id": i,
209 "ack_queue": ack_queue,
210 },
211 daemon=False,
212 name=f"freetoken-tokenizer-{i}",
213 )
214 p.start()
215 processes.append(p)
216
217 # Expected ready acks: 1 primary scheduler + num_tokenizers + 1 detokenizer.
218 return BackendHandle(
219 ack_queue=ack_queue,
220 processes=processes,
221 expected_acks=num_tokenizers + 2,
222 )
223
224 run_api_server(server_args, start_subprocess, run_shell=run_shell)
