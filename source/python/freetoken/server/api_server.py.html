71def _terminate_backend_workers(processes: List[Any]) -> None:
72 """Best-effort, non-blocking teardown of the backend worker processes on an orderly stop.
73
74 Called from the shutdown path AFTER ``_SHUTTING_DOWN`` is set, so the supervisor's liveness
75 watch is guaranteed to observe the flag before it sees these deaths — the exits are then
76 attributed to the stop, not misreported as a crash (the flag/death race, otherwise decided
77 by OS signal-delivery order, is settled in our favor). Also ensures the non-daemon workers
78 are actually torn down when an external stop signals only the main process.
79
80 Never blocks (no join) and never raises: a worker already gone / an unqueryable handle is
81 fine — this only nudges live ones toward exit."""
82 for p in processes or []:
83 try:
84 if p.is_alive():
85 p.terminate()
86 except Exception: # noqa: BLE001 -- already-gone / unqueryable handle: nothing to do
87 continue

90def _exit_after_backend_death(grace_s: float) -> threading.Timer:
91 def _stop() -> None:
92 if _SHUTTING_DOWN.is_set():
93 return # an external stop got here first
94 logger.error("Backend worker is gone and cannot be restarted; stopping the API server")
95 os.kill(os.getpid(), signal.SIGTERM)
96
97 timer = threading.Timer(grace_s, _stop)
98 timer.daemon = True
99 timer.start()
100 return timer

103def _reap_backend_workers(processes: List[Any], timeout: float = 5.0) -> None:
104 """Wait out a preceding ``_terminate_backend_workers`` and SIGKILL whatever is still
105 standing. Only the shell path needs this: it owns the process lifetime end to end (no
106 outer signal takes the process down for it), and a worker that ignored SIGTERM would keep
107 the GPU and the IPC sockets after the shell has already returned to the user's terminal."""
108 for p in processes or []:
109 try:
110 p.join(timeout=timeout)
111 if p.is_alive():
112 p.kill()
113 except Exception: # noqa: BLE001 -- already-gone / unqueryable handle: nothing to do
114 continue

249 async def listen(self):
250 while True:
251 msg = await self.recv_tokenizer.get()
252 if isinstance(msg, CacheRebuildReply):
253 self._resolve_rebuild(msg)
254 continue
255 for msg in _unwrap_msg(msg):
256 # Global accounting follows actual admitted/sampled work even after the HTTP
257 # client disconnects and abort_user removes its ack queue. Delivery to a live
258 # request remains gated below, but observation must happen first.
259 self.stats.observe(msg)
260 if msg.uid not in self.ack_map:
261 continue
262 self.ack_map[msg.uid].append(msg)
263 self.event_map[msg.uid].set()

265 def _resolve_rebuild(self, msg: CacheRebuildReply) -> None:
266 """Terminal transition for a rebuild: rebuilding -> serving | failed. This is the ONLY
267 path that reopens the gate dispatch_rebuild latches to "rebuilding", so it must always
268 land on a definite state — including for a reply that arrives after the HTTP wait timed
269 out (its future is already gone) — so a rebuild can never wedge the server in
270 "rebuilding" forever.
271
272 Ordering matters. Wake any waiter and record the result first, then decide the gate:
273 - A fatal worker death latched "failed" (the watchdog) OUTRANKS this reply. A reply
274 that raced a crash (e.g. a buffered "ok") must not resurrect a dead engine to
275 "serving" — a crashed backend cannot serve. Leave it latched.
276 - Otherwise only a genuine destructive "failed" latches maintenance; "ok"/"busy"/
277 "rejected"/"unsupported" all leave the prior cache intact, so the engine keeps
278 serving."""
279 self.last_rebuild = {
280 "request_id": msg.request_id,
281 "status": msg.status,
282 "moe_cache_size": msg.moe_cache_size,
283 "num_pages": msg.num_pages,
284 "mamba_slots": msg.mamba_slots,
285 "num_swa_pages": msg.num_swa_pages,
286 "error": msg.error,
287 }
288 fut = self.rebuild_futures.pop(msg.request_id, None)
289 if fut is not None and not fut.done():
290 fut.set_result(self.last_rebuild)
291 if self.fatal_error is not None:
292 # A dead backend stays failed regardless of any (possibly stale/buffered) reply.
293 self.maintenance_state = "failed"
294 return
295 self.maintenance_state = "failed" if msg.status == "failed" else "serving"

297 def fail_pending_rebuilds(self, message: str) -> None:
298 """Resolve every in-flight rebuild waiter as failed. Called from the supervisor thread
299 when a worker death latches a fatal error: no CacheRebuildReply will ever arrive, so a
300 caller blocked in dispatch_rebuild's ``await`` would otherwise hang until its full
301 timeout (up to 300s). Runs the resolution ON the event loop via call_soon_threadsafe —
302 asyncio Futures are not thread-safe, and _resolve_rebuild only ever calls set_result
303 from the loop thread (the listen() task), so we mirror that here from off-thread."""
304 loop = self._loop
305 if loop is None:
306 return # listener never started -> no futures could be pending
307 result = {"status": "failed", "error": message}

309 def _resolve_all() -> None:
310 for request_id in list(self.rebuild_futures):
311 fut = self.rebuild_futures.pop(request_id, None)
312 if fut is not None and not fut.done():
313 fut.set_result(dict(result))
314
315 try:
316 loop.call_soon_threadsafe(_resolve_all)
317 except RuntimeError:
318 # Loop already closed (shutdown racing the crash): nothing left to wake.
319 pass

331 async def wait_for_ack(self, uid: int):
332 event = self.event_map[uid]
333 # finally, not a trailing statement: every consumer breaks out of its `async for` at
334 # the terminal ack, leaving this generator suspended at the yield. Cleanup written
335 # after the loop would then only run on paths nobody takes, leaking both maps once
336 # per completed request. GeneratorExit runs the finally.
337 try:
338 while True:
339 await event.wait()
340 event.clear()
341
342 pending = self.ack_map[uid]
343 self.ack_map[uid] = []
344 ack = None
345 for ack in pending:
346 yield ack
347 if ack and ack.finished:
348 break
349 finally:
350 self.ack_map.pop(uid, None)
351 self.event_map.pop(uid, None)

353 async def stream_generate(self, uid: int):
354 async for ack in self.wait_for_ack(uid):
355 if ack.incremental_output:
356 # SSE: JSON-encode (so a newline inside a token can't break the frame)
357 # and terminate every event with a blank line.
358 yield f"data: {json.dumps({'text': ack.incremental_output}, ensure_ascii=False)}\n\n".encode()
359 if ack.finished:
360 break
361 yield b"data: [DONE]\n\n"
362 logger.debug("Finished streaming response for user %s", uid)

519class CacheRebuildRequest(BaseModel):
520 moe_cache_size: int | None = None
521 num_pages: int | None = None
522 # Usable GDN (mamba) state-pool slots (matches the status-bar total); the engine adds the
523 # reserved padding sink internally.

525 # Window pool sizing (DSV4 or a sliding-window model in --cache-type radix). Two forms, at
526 # most one: num_swa_pages pins the window to an absolute page count (usable, in the pool's own
527 # page unit -- P for DSV4, 1 token for radix-SWA); swa_full_tokens_ratio in (0, 1] is a
528 # convenience the server converts to num_swa_pages = ceil(ratio x full window) at the current
529 # (or requested) anchor. Internally only num_swa_pages flows to the engine.
530 num_swa_pages: int | None = None
531 swa_full_tokens_ratio: float | None = None
532 # Only "if_idle" (reject unless the scheduler is idle) is supported today. "drain" mode
533 # is deferred (needs the drain-gate machinery); constraining the Literal makes an
534 # unsupported value fail fast with a 422 at the API layer instead of a generic 503.
535 mode: Literal["if_idle"] = "if_idle"
536 timeout: float = 300.0

539async def dispatch_rebuild(
540 state: FrontendManager,
541 *,
542 moe_cache_size: int | None,
543 num_pages: int | None,
544 num_mamba_slots: int | None = None,
545 num_swa_pages: int | None = None,
546 mode: str = "if_idle",
547 timeout: float = 300.0,
548) -> Dict[str, Any]:
549 """Send a cache-rebuild request to the scheduler and await its result, managing the
550 maintenance gate. Returns the scheduler's result dict, or a synthesized
551 ``{"status": "failed"|"timeout"}`` on dispatch error / timeout. Every caller reaches it
552 through ``POST /v1/cache/rebuild`` (``ft ctl cache``, the desktop panel, the shell's
553 ``/cache``), which does the pre-flight maintenance_state checks (409/503 short-circuits)."""
554 request_id = str(uuid.uuid4())
555 fut = asyncio.get_running_loop().create_future()
556 state.rebuild_futures[request_id] = fut
557 state.maintenance_state = "rebuilding"
558 try:
559 await state.send_one(
560 CacheRebuildMsg(
561 request_id=request_id,
562 moe_cache_size=moe_cache_size,
563 num_pages=num_pages,
564 num_mamba_slots=num_mamba_slots,
565 num_swa_pages=num_swa_pages,
566 mode=mode,
567 )
568 )
569 except Exception as e: # noqa: BLE001
570 # The enqueue failed, so the scheduler never received the request and the engine is
571 # untouched. Roll the gate back to serving (else a transient ZMQ error would latch
572 # maintenance forever with no reply ever arriving to clear it) and surface the error.
573 state.rebuild_futures.pop(request_id, None)
574 state.maintenance_state = "serving"
575 return {"status": "failed", "error": f"failed to dispatch rebuild: {e!r}"}
576 try:
577 return await asyncio.wait_for(fut, timeout=timeout)
578 except asyncio.TimeoutError:
579 # Do NOT reopen the maintenance gate here: the scheduler may still be mid-rebuild
580 # (e.g. a slow CUDA-graph recapture) and the backend request was not cancelled.
581 # Leave maintenance_state == "rebuilding" so new generation and new rebuilds stay
582 # blocked; the eventual CacheRebuildReply flips it to serving/failed via
583 # _resolve_rebuild. Drop the now-cancelled future so it does not linger.
584 state.rebuild_futures.pop(request_id, None)
585 return {"status": "timeout", "request_id": request_id}

589 """External accepts num_swa_pages OR swa_full_tokens_ratio; internally only num_swa_pages
590 flows. Convert a ratio to an absolute window at the requested (or current) full anchor, in the
591 window pool's own page unit (P for DSV4, 1 token for radix-SWA)."""
592 if req.swa_full_tokens_ratio is None:
593 return req.num_swa_pages
594 config = state.config
595 pools = getattr(state, "cache_pools", None) or {}
596 last = getattr(state, "last_rebuild", None) or {}

665def _cache_limits(geo: dict, unit_bytes: dict, pool_budget: int, floors: dict) -> dict:
666 """Per-pool adjustable {min, max} bounds for the desktop cache sliders, so their ranges stay
667 sensible instead of spanning the whole theoretical space:
668
669 min -- the engine's rebuild floor for that pool (from the ("meta", …) floors; falls back to
670 page_size for KV and num_experts for MoE, both known here, and 0 for mamba).
671 max -- the ideal-case unit count if the whole cache budget went to this one pool:
672 max = pool_budget // unit_cost,
673 where pool_budget is ``cache_budget_bytes`` -- the SAME ceiling the rebuild
674 fit-check enforces (memory_ratio x pre-model baseline − weights), so a slider can
675 never offer a size the rebuild would reject. It excludes the (1-memory_ratio)
676 CUDA-graph/activation headroom, is constant for the engine's lifetime, and is
677 measured before any pool existed, so each pool's ideal max is simply budget/cost
678 -- no current-occupancy correction -- and the bounds are stable across rebuilds.
679
680 Still optimistic in one way: it is the ``fixed=0`` whole-cache ceiling, while a
681 real rebuild also subtracts ``fixed_cache_size`` (GDN state pool, the radix-SWA
682 single-request floor / pinned window). Pools sized near their max on a model that
683 carries those may still be rejected; the desktop's whole-config budget line is
684 what catches that.
685
686 MoE is additionally capped at the model's own routed-expert count: past that the
687 budget could buy slots the model has nothing to put in.
688
689 A pool whose unit cost or the budget signal is unknown (0) reports max 0 -- the desktop
690 reads that as "unknown" and keeps its own bounds. Non-MoE / non-hybrid models therefore
691 report 0 max (and a 0 floor) for the pools they lack. Defensive like the unit_bytes/num_experts
692 blocks: any bad read degrades to all-zero bounds and never raises."""
693 try:
694 page_size = int(geo["page_size"])
695 kv_per_token = int(unit_bytes["kv_per_token"])
696 moe_per_expert = int(unit_bytes["moe_per_expert"])
697 mamba_per_slot = int(unit_bytes["mamba_per_slot"])
698 budget = int(pool_budget) if pool_budget and int(pool_budget) > 0 else 0
699
700 def ideal(unit_cost: int) -> int:
701 if unit_cost <= 0 or budget <= 0:
702 return 0 # unit cost or budget signal unknown -> "unknown", desktop falls back
703 return budget // unit_cost
704
705 swa_per_token = int(unit_bytes.get("swa_per_token", 0) or 0)
706 kv_min = int(floors.get("kv_tokens", page_size) or 0)
707 moe_min = int(floors.get("moe_experts", geo["num_experts"]) or 0)
708 mamba_min = int(floors.get("mamba_slots", 0) or 0)
709 # SWA pool floor (single-request working set); the desktop uses it to lower-bound the
710 # SWA-token estimate at a given reuse ratio. max is the ideal per-token ceiling.
711 swa_min = int(floors.get("swa_tokens", 0) or 0)
712 # The expert cache is the one pool with a ceiling of its own: caching more slots than
713 # the model HAS routed experts buys nothing (every expert is already resident), so on a
714 # small MoE model with a big card budget//cost overshoots the whole model. The other
715 # pools have no such bound -- KV, window and GDN capacity all keep paying off with more
716 # concurrent requests and longer prefix reuse.
717 total_experts = int(geo["num_experts"]) * int(geo["num_moe_layers"])
718 moe_max = ideal(moe_per_expert)
719 if total_experts > 0:
720 moe_max = min(moe_max, total_experts)
721 return {
722 "kv_tokens": {"min": kv_min, "max": ideal(kv_per_token)},
723 "moe_experts": {"min": moe_min, "max": moe_max},
724 "mamba_slots": {"min": mamba_min, "max": ideal(mamba_per_slot)},
725 "swa_tokens": {"min": swa_min, "max": ideal(swa_per_token)},
726 }
727 except Exception: # noqa: BLE001 -- limits are a nicety; a bad read must not 500 the poll
728 return {
729 "kv_tokens": {"min": 0, "max": 0},
730 "moe_experts": {"min": 0, "max": 0},
731 "mamba_slots": {"min": 0, "max": 0},
732 "swa_tokens": {"min": 0, "max": 0},
733 }

757def cache_geometry(state: Any) -> dict:
758 """Current cache geometry for the desktop cache panel. Each pool size resolves
759 most-recent-truth first: the last rebuild's result, else the running UserReply snapshot
760 (updated per generation), else the load-time allocation from the ("meta", …) ack — so the
761 panel shows the real allocation from the moment the server is ready, not zeros until the
762 first chat. moe_cache_size falls back further to the configured size (rate resolved to a
763 slot count, matching the shell status bar). num_experts / num_moe_layers are the per-layer
764 expert count and MoE layer count from model_config, so callers can derive the moe pool
765 bounds; both are 0 when config.model_config is absent/raises (dummy configs) or non-MoE."""
766 from .model_meta import moe_cache_size as configured_moe_cache_size
767
768 tr = state.stats
769 config = state.config
770 last = getattr(state, "last_rebuild", None) or {}
771 pools = getattr(state, "cache_pools", None) or {}
772 num_pages = int(last.get("num_pages") or tr.kv_total_pages or pools.get("num_pages", 0) or 0)
773 num_mamba_slots = int(
774 last.get("mamba_slots") or tr.mamba_total_slots or pools.get("num_mamba_slots", 0) or 0
775 )
776 page_size = int(pools.get("page_size", 0) or getattr(config, "page_size", 1) or 1)
777 moe_cache_size = last.get("moe_cache_size")
778 if moe_cache_size is None:
779 moe_cache_size = int(pools.get("moe_cache_size", 0) or 0) or configured_moe_cache_size(
780 config
781 )
782 # Effective window/full ratio: derived from the last rebuild's pinned window (num_swa_pages,
783 # the internal currency), else the load-time meta value. 0.0 for models without a window pool.
784 swa_full_tokens_ratio = float(getattr(state, "swa_full_tokens_ratio", 0.0) or 0.0)
785 last_swa_pages = last.get("num_swa_pages")
786 swa_page_size = int(pools.get("swa_page_size", 0) or 0)

788 full = num_pages * page_size
789 if full > 0:
790 swa_full_tokens_ratio = min(1.0, last_swa_pages * swa_page_size / full)
791 try:
792 model_config = config.model_config
793 num_experts = int(getattr(model_config, "num_experts", 0) or 0)
794 num_moe_layers = int(getattr(model_config, "num_moe_layers", 0) or 0)
795 except Exception:
796 num_experts = 0
797 num_moe_layers = 0
798 # Per-unit VRAM costs from the backend's ("meta", …) ack (compute_cache_unit_bytes). All 0
799 # when the meta never arrived (older engine build / still loading) or a dummy config is in
800 # play; same defensive guard as num_experts so a missing/odd unit_bytes can't 500 the poll.
801 try:
802 ub = getattr(state, "unit_bytes", None) or {}
803 unit_bytes = {
804 "kv_per_token": int(ub.get("kv_bytes_per_token", 0) or 0),
805 "moe_per_expert": int(ub.get("moe_bytes_per_expert", 0) or 0),
806 "mamba_per_slot": int(ub.get("mamba_bytes_per_slot", 0) or 0),
807 "swa_per_token": int(ub.get("swa_bytes_per_token", 0) or 0),
808 }
809 except Exception:
810 unit_bytes = {"kv_per_token": 0, "moe_per_expert": 0, "mamba_per_slot": 0, "swa_per_token": 0}
811 # Thinking control: the gears a client can offer and the chat_template_kwargs
812 # each selects, derived from the checkpoint's own probed template (no
813 # per-family registry). None until the frontend tokenizer is warm — the
814 # first status poll kicks the warmup thread and later polls see the gears,
815 # so the picker self-populates without ever blocking this route.
816 try:
817 reasoning = _reasoning_geometry(state)
818 except Exception:
819 reasoning = None
820 geo = {
821 "num_pages": num_pages,
822 "page_size": page_size,
823 "moe_cache_size": moe_cache_size,
824 "num_mamba_slots": num_mamba_slots,
825 "num_experts": num_experts,
826 "num_moe_layers": num_moe_layers,
827 # Eviction policy of the MoE slot cache ("lru"). Reported so a client can label the
828 # pool without having to know how the server was started.
829 "moe_cache_policy": getattr(config, "moe_cache_policy", None),
830 "unit_bytes": unit_bytes,
831 "swa_full_tokens_ratio": swa_full_tokens_ratio,
832 # The pool's window page unit; 0 for models without a resizable window pool.
833 "swa_page_size": swa_page_size,
834 # Concrete current window size in that unit (usable pages): the last rebuild's value if it
835 # pinned/derived one, else the load-time pool size. 0 for models without a window pool.
836 "num_swa_pages": int(last.get("num_swa_pages") or pools.get("num_swa_pages", 0) or 0),
837 # Engine's exact total cache VRAM budget (all pools), from the ("meta", …) ack. 0 when
838 # unknown (pre-budget engine) — the desktop then reverse-derives it from the limits.
839 "cache_budget_bytes": int(getattr(state, "cache_budget_bytes", 0) or 0),
840 "reasoning": reasoning,
841 }
842 # Per-pool slider bounds, sized against the cache budget the rebuild fit-check actually
843 # enforces — NOT the raw post-weights free VRAM, which is larger by the (1-memory_ratio)
844 # graph/activation headroom and would offer slider room every rebuild rejects. Both come
845 # from the ("meta", …) ack and are constant for the engine's lifetime; the free-VRAM value
846 # stays the fallback for an ack without a budget. 0 -> limits report 0 max and the desktop
847 # keeps its own bounds.
848 pool_budget = int(getattr(state, "cache_budget_bytes", 0) or 0) or int(
849 getattr(state, "free_vram_bytes", 0) or 0
850 )
851 floors = getattr(state, "cache_floors", None) or {}
852 geo["limits"] = _cache_limits(geo, unit_bytes, pool_budget, floors)

894def _install_shell_stop_handlers() -> None:
895 """Shell mode runs uvicorn off the main thread, where its own signal capture is a no-op, so
896 an external stop would kill this process and orphan the non-daemon backend workers. Flag the
897 orderly stop FIRST -- so the supervisor attributes the ensuing worker exits to the stop
898 instead of reporting a crash -- terminate them, then chain to whatever handler was there
899 before (SIG_DFL restored and re-raised when it isn't callable, so the signal still stops us).
900
901 SIGHUP is handled for the same reason as SIGTERM, and it matters more here: the workers run
902 outside our process group in shell mode (see launch.py:_detach_process_group), so a closed
903 terminal no longer reaches them on its own and this handler is what tears them down.
904
905 SIGINT is deliberately left alone: the shell binds it, per turn, to "cancel this turn"."""
906 previous = {sig: signal.getsignal(sig) for sig in (signal.SIGTERM, signal.SIGHUP)}
907
908 def _flag_shutdown(signum, frame) -> None:
909 _SHUTTING_DOWN.set()
910 _terminate_backend_workers(_GLOBAL_STATE.backend_processes)
911 prev = previous.get(signum)
912 if callable(prev):
913 prev(signum, frame)
914 else:
915 signal.signal(signum, prev if prev is not None else signal.SIG_DFL)
916 os.kill(os.getpid(), signum)
917
918 for sig in previous:
919 signal.signal(sig, _flag_shutdown)

922def _serve_and_run_shell(host: str, port: int, api_key: str | None) -> None:
923 """Shell mode: serve the API here, and attach the terminal client to it over the loopback.
924
925 The shell is an ordinary API client (see ``freetoken.shell``), so shell mode is just
926 ``ft serve`` plus that client -- one generation path for every caller, none of it
927 shell-private. uvicorn owns the HTTP surface on a worker thread and, through its lifespan,
928 the orderly shutdown that flags ``_SHUTTING_DOWN`` before the workers exit; the main thread
929 belongs to the TUI. Access logging is off because those lines would land in the middle of
930 the chat as it streams -- ``/v1/requests`` still records every request for ``ft ctl``.
931
932 Signals: uvicorn's capture is a no-op off the main thread, so ^C stays with the shell, which
933 binds it to "cancel this turn". That only works because the engine workers leave our process
934 group in shell mode (launch.py:_detach_process_group) -- otherwise the terminal would deliver
935 that same ^C to them and the shell would go on chatting with a dead engine. In exchange, the
936 stop signals they no longer receive are relayed by _install_shell_stop_handlers below."""
937 from freetoken.launch import resolve_server_url
938 from freetoken.shell.tui import run_shell
939
940 # Resolved before anything is started, so a bad address fails while there is still nothing
941 # to tear down. 0.0.0.0/:: are bind addresses, not destinations; resolve_server_url maps
942 # them to loopback. An IPv6 bind host has to be bracketed before it can go into a URL.
943 netloc = f"[{host}]:{port}" if ":" in host else f"{host}:{port}"
944 origin = resolve_server_url(f"http://{netloc}").origin
945
946 server = uvicorn.Server(uvicorn.Config(app, host=host, port=port, access_log=False))
947 thread = threading.Thread(target=server.run, name="freetoken-uvicorn", daemon=True)
948 thread.start()
949 _install_shell_stop_handlers()
950 try:
951 # The engine is still loading here (uvicorn binds first, by design): the client waits on
952 # /health and echoes the same load progress the desktop app polls for. A ^C during that
953 # wait is a stop, not a crash -- exit through the teardown below, not a traceback.
954 with contextlib.suppress(KeyboardInterrupt):
955 asyncio.run(run_shell(origin, connect_grace=30.0, api_key=api_key))
956 finally:
957 server.should_exit = True
958 thread.join(timeout=15)
959 # Belt and braces: if uvicorn's lifespan shutdown did not run (thread wedged), flag the
960 # stop and tear the workers down here so nothing outlives the shell.
961 _SHUTTING_DOWN.set()
962 _terminate_backend_workers(_GLOBAL_STATE.backend_processes)
963 _reap_backend_workers(_GLOBAL_STATE.backend_processes)

1053 # sizes allocated at load, delivered once on the ack path; surfaced by cache_geometry
