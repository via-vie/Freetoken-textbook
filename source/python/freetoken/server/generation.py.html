1"""Protocol-neutral generation core.
2
3The narrow interface every wire protocol (OpenAI chat, Anthropic messages, OpenAI
4Responses) sits on. A protocol adapter converts its request into a ``GenSpec``
5(messages + sampling + tools), submits it, and formats the resulting semantic
6events (``GenEvent``) / ``GenResult`` into its own wire shape. This mirrors vLLM's
7split of per-request ``to_sampling_params`` + shared preprocess + a neutral
8``SamplingParams`` — no wire request type reaches the engine path.
9
10This module is imported by ``openai_api`` / ``anthropic_api`` / ``responses_api``;
11it depends on none of them.

47class GenerationError(ValueError):
48 """A request failed before producing output (surfaced via ``UserReply.error`` — e.g. a
49 chat template the tokenizer cannot render, or a prompt that exceeds the KV budget). Each
50 adapter turns this into its own wire-level error instead of hanging on a reply that never
51 arrives. ``code`` carries the stable class (``UserReply.error_code``) when the failure has
52 one, so an adapter can emit it as OpenAI's error ``code`` and clients need not parse prose.
53
54 Subclasses ValueError so every adapter's invalid-request handler maps it to a 400."""
55
56 def __init__(self, message: str, code: str | None = None):
57 super().__init__(message)
58 self.code = code

162def resolve_sampling(
163 *,
164 temperature: float | None,
165 top_k: int | None,
166 top_p: float | None,
167 max_tokens: int | None,
168 ignore_eos: bool,
169 model_sampling: dict[str, Any],
170 stop: str | list[str] | None = None,
171 default_max_tokens: int = DEFAULT_MAX_OUTPUT_TOKENS,
172 presence_penalty: float = 0.0,
173 frequency_penalty: float = 0.0,
174) -> SamplingParams:
175 """Map a protocol's sampling fields onto the engine's neutral SamplingParams,
176 filling unspecified fields from the checkpoint's recommended defaults."""
177
178 def pick(value, key, framework):
179 return value if value is not None else model_sampling.get(key, framework)
180
181 stop_list = [stop] if isinstance(stop, str) else list(stop or [])
182 # `is not None`, not truthiness: an explicit max_tokens=0 must not read as "unset" and
183 # silently become the 32k default. The engine cannot serve a zero-token budget either
184 # (the request would never become decodable, so the client would wait forever), so a
185 # non-positive value is a client error.
186 if max_tokens is not None and max_tokens < 1:
187 raise ValueError(f"max_tokens must be at least 1, got {max_tokens}")
188 return SamplingParams(
189 ignore_eos=ignore_eos,
190 max_tokens=default_max_tokens if max_tokens is None else max_tokens,
191 temperature=pick(temperature, "temperature", 0.0),
192 top_k=pick(top_k, "top_k", -1),
193 top_p=pick(top_p, "top_p", 1.0),
194 stop_strs=[s for s in stop_list if s], # drop empty strings (would match everything)
195 presence_penalty=presence_penalty,
196 frequency_penalty=frequency_penalty,
197 )

200def render_messages(messages: list[dict[str, Any]]) -> list[dict[str, Any]]:
201 """Normalize OpenAI-shaped message dicts for the chat template: flatten text
202 content parts to a string and decode tool-call arguments from JSON. Raises
203 ValueError on a non-text content part (text-only server). Shared by all adapters."""
204 return [_render_message(m) for m in messages]

207def _render_message(message: dict[str, Any]) -> dict[str, Any]:
208 m = dict(message)
209 content = m.get("content")
210 if isinstance(content, list):
211 m["content"] = _flatten_text_parts(content)
212 # Templates read different reasoning keys (reasoning_content: most; reasoning:
213 # gemma4; thinking: gpt-oss) — accept any, emit both.
214 reasoning = m.get("reasoning_content") or m.get("reasoning") or m.get("thinking")
215 if reasoning:
216 m.setdefault("reasoning_content", reasoning)
217 # gpt-oss's template raises when a tool-call assistant turn carries BOTH content and
218 # thinking -- it renders one or the other. Visible text wins: dropping it would lose
219 # what the user saw, and every other family reads `content` too.
220 if not (m.get("tool_calls") and m.get("content")):
221 m.setdefault("thinking", reasoning)
222 if m.get("role") == "assistant" and m.get("content") is None:
223 # gpt-oss templates concatenate message.content unconditionally.
224 m["content"] = ""
225 tool_calls = m.get("tool_calls")
226 if tool_calls:
227 rendered = []
228 for tc in tool_calls:
229 tc = dict(tc)
230 fn = dict(tc.get("function") or {})
231 arguments = fn.get("arguments")
232 if isinstance(arguments, str):
233 try:
234 fn["arguments"] = json.loads(arguments)
235 except json.JSONDecodeError:
236 pass
237 tc["function"] = fn
238 rendered.append(tc)
239 m["tool_calls"] = rendered
240 return m

243def _flatten_text_parts(parts: list[Any]) -> str | list[dict[str, Any]]:
244 """Plain string for text-only parts; a template-ready part list with {"type": "image"} entries (source kept under freetoken_ref) when images are present."""
245 out: list[dict[str, Any]] = []
246 has_image = False
247 for part in parts:
248 ptype = part.get("type") if isinstance(part, dict) else None
249 if ptype == "text":
250 out.append({"type": "text", "text": part.get("text") or ""})
251 elif ptype in ("image_url", "input_image"):
252 url = part.get("image_url")
253 if isinstance(url, dict):
254 url = url.get("url")
255 url = url or part.get("url")
256 if not url:
257 raise ValueError("image content part carries no url")
258 out.append({"type": "image", "freetoken_ref": {"kind": "url", "data": url}})
259 has_image = True
260 elif ptype == "image" and isinstance(part.get("freetoken_ref"), dict):
261 out.append(part)
262 has_image = True
263 else:
264 raise ValueError(f"Unsupported content part type: {ptype}")
265 if not has_image:
266 return "".join(p["text"] for p in out)
267 return out

297async def submit_generation(spec: GenSpec, state: Any) -> int:
298 """Enqueue one generation from a GenSpec; return its uid. Every protocol adapter
299 calls this — it takes the neutral spec, not a wire request type."""
300 refs = collect_image_refs(spec.messages)
301 images = await _resolve_images(refs, state) if refs else None
302 uid = state.new_user()
303 await state.send_one(
304 TokenizeMsg(
305 uid=uid,
306 text=spec.messages,
307 sampling_params=spec.sampling_params,
308 chat_template_kwargs=spec.chat_template_kwargs,
309 tools=spec.template_tools,
310 images=images,
311 inline_system_policy=spec.inline_system_policy,
312 )
313 )
314 return uid

317async def count_prompt_tokens(
318 messages: list[dict[str, Any]],
319 tools: list[dict[str, Any]] | None,
320 chat_template_kwargs: dict[str, Any],
321 state: Any,
322 *,
323 inline_system_policy: str | None = None,
324) -> int:
325 """Token count of an already-converted (messages, tools, chat_template_kwargs) prompt,
326 using the frontend's own tokenizer (``state.frontend_tokenizer()``) so the count equals the
327 ``usage.input_tokens`` a real generation of the same prompt would report. The neutral
328 counterpart to ``submit_generation`` — any protocol's count endpoint converts to this triple
329 and calls it. The caller validates the prompt first (non-empty, has tokenizable content).
330
331 Failure classification mirrors ``/v1/messages``: a chat template that rejects the specific
332 conversation (bad role ordering, an unmatched tool_result, an explicit raise_exception) is
333 re-raised as ``GenerationError`` — an input-driven, client-classifiable failure, the same
334 class the generation path maps to a 400. A tokenizer *initialization* failure (missing
335 template, load error) propagates as its original exception, a server fault. Load + tokenize
336 run in a worker thread so the event loop is never blocked."""
337 refs = collect_image_refs(messages)
338 images = await _resolve_images(refs, state) if refs else None
339 msg = TokenizeMsg(
340 uid=0,
341 text=messages,
342 sampling_params=SamplingParams(),
343 chat_template_kwargs=chat_template_kwargs,
344 tools=tools,
345 images=images,
346 inline_system_policy=inline_system_policy,
347 )
348 manager = await asyncio.to_thread(state.frontend_tokenizer) # init failure -> server fault
349 try:
350 (user_msg,) = await asyncio.to_thread(manager.tokenize, [msg])
351 except InlineSystemError as exc:
352 raise GenerationError(str(exc)) from exc
353 except _TemplateError as exc:
354 raise GenerationError(str(exc)) from exc
355 return int(user_msg.input_ids.numel())

358async def prerender_error(spec: GenSpec, state: Any) -> GenerationError | None:
359 """Render ``spec``'s prompt frontend-side, returning the failure a streaming
360 adapter should surface as an HTTP 400 *before* committing an SSE stream —
361 once headers go out, a template rejection can only ride in-stream, where
362 some agents show nothing but "empty response". Render only; the worker
363 still renders and encodes authoritatively. Best-effort: a state without a
364 frontend tokenizer, or one that fails to *initialize*, skips validation
365 rather than blocking the generation path.
366 """
367 build = getattr(state, "frontend_tokenizer", None)
368 if build is None:
369 return None
370 msg = TokenizeMsg(
371 uid=0,
372 text=spec.messages,
373 sampling_params=SamplingParams(),
374 chat_template_kwargs=spec.chat_template_kwargs,
375 tools=spec.template_tools,
376 inline_system_policy=spec.inline_system_policy,
377 )
378 try:
379 manager = await asyncio.to_thread(build)
380 except Exception: # noqa: BLE001 -- server fault, not this request's problem
381 return None
382 try:
383 await asyncio.to_thread(manager.render_prompt, msg)
384 except Exception as exc: # noqa: BLE001 -- mirror the worker's classification
385 return GenerationError(f"could not encode request: {exc}")
386 return None

389def _make_reasoning_parser(spec: GenSpec, state: Any) -> ReasoningParser | None:
390 """Build a reasoning parser for this generation, or None if the server has no
391 reasoning parser configured. ``force_reasoning`` matches the encode-side
392 thinking mode so chat-mode content is never mislabeled as reasoning."""
393 parser_name = getattr(state.config, "reasoning_parser", None)
394 if parser_name == "qwen3":
395 # The qwen3 chat template opens an implicit <think> (thinking on) unless
396 # enable_thinking is explicitly false, so the model emits only the closing
397 # </think>. Mirror that default here, else the chain-of-thought leaks into content.
398 force_reasoning = (spec.chat_template_kwargs or {}).get("enable_thinking") is not False
399 elif parser_name == "glm":
400 # GLM's template honors enable_thinking (default on) even with tools; the
401 # generic fallback would force thinking and mislabel disabled output as reasoning.
402 force_reasoning = (spec.chat_template_kwargs or {}).get("enable_thinking") is not False
403 elif parser_name == "gemma4":
404 # Gemma4 defaults thinking off even when tools are present: its template injects an
405 # empty thought channel before generation. Do not let Codex tool definitions make all
406 # visible text look like hidden reasoning.
407 ctk = spec.chat_template_kwargs or {}
408 force_reasoning = (
409 ctk.get("thinking_mode") == "thinking"
410 or bool(ctk.get("enable_thinking"))
411 or bool(ctk.get("thinking"))
412 )
413 elif parser_name == "minimax_m3":
414 # M3's template pre-opens <mm:think> only in thinking_mode "enabled" (the
415 # model then emits just the closing tag); "adaptive" (default) leaves the
416 # model to open the tag itself and "disabled" pre-closes it.
417 force_reasoning = (spec.chat_template_kwargs or {}).get("thinking_mode") == "enabled"
418 else:
419 force_reasoning = (
420 resolve_thinking_mode(spec.chat_template_kwargs, spec.template_tools) == "thinking"
421 )
422 return build_reasoning_parser(state.config, force_reasoning)

481
482async def with_keepalive(events: AsyncIterator[GenEvent], interval: float):
483 """Yield events from ``events``, interspersing the KEEPALIVE sentinel whenever
484 ``interval`` seconds pass without one (covers queue/prefill silence before the
485 first event too). Exceptions propagate unchanged; the pending read is cancelled
486 when the consumer closes."""
487 aiter = events.__aiter__()
488 task = None
489 try:
490 while True:
491 if task is None:
492 task = asyncio.ensure_future(aiter.__anext__())
493 try:
494 ev = await asyncio.wait_for(asyncio.shield(task), interval)
495 except asyncio.TimeoutError:
496 yield KEEPALIVE
497 continue
498 except StopAsyncIteration:
499 return
500 task = None
501 yield ev
502 finally:
503 if task is not None:
504 task.cancel()

508 *,

542async def generate_events(
543 uid: int, spec: GenSpec, state: Any, *, source: str | None = None
544) -> AsyncIterator[GenEvent]:
545 """Wraps `_generate_events_impl` to log the request with its totals, read off the terminal
546 `GenDone`. The `finally` still records the row on a mid-stream disconnect — but with 0 tokens
547 if the drop lands before `GenDone`, the only event carrying the totals."""
548 start = time.monotonic()
549 prompt_tokens = 0
550 completion_tokens = 0
551 first_token_at: float | None = None
552 error: str | None = None
553 try:
554 async for ev in _generate_events_impl(uid, spec, state):
555 if isinstance(ev, GenDone):
556 prompt_tokens = ev.prompt_tokens
557 completion_tokens = ev.completion_tokens
558 elif first_token_at is None:
559 first_token_at = time.monotonic()
560 yield ev
561 except GenerationError as exc:
562 error = str(exc)
563 raise
564 finally:
565 _record_generation(
566 source=source, stream=True, start=start,
567 prompt_tokens=prompt_tokens, completion_tokens=completion_tokens, error=error,
568 first_token_at=first_token_at,
569 )

595async def _generate_events_impl(uid: int, spec: GenSpec, state: Any) -> AsyncIterator[GenEvent]:
596 """Protocol-neutral streaming generation. Yields semantic events (reasoning /
597 content / tool-call deltas) terminated by exactly one GenDone. Produces no wire
598 format — the OpenAI/Anthropic/Responses streamers format these into their own.
599
600 With tools configured, content is parsed incrementally: plain text streams live
601 (the detector holds back only a suffix that could still grow into a tool-call
602 tag) and each tool call is emitted as one complete ToolCallsDelta as soon as it
603 closes — mid-generation, not after the whole response. Emitted ToolCallItems
604 carry the output ordinal (0, 1, ...) in tool_index. Formats whose detector
605 cannot parse incrementally (supports_streaming=False) keep the previous
606 buffer-everything-then-parse-at-the-end behavior."""
607 prompt_tokens = 0
608 completion_tokens = 0
609 cached_tokens = 0
610 pending = ""
611 parse_tools = spec.parse_tools
612 reasoning_parser = _make_reasoning_parser(spec, state)
613 specials = _leaked_special_tokens(state)
614
615 tool_parser: FunctionCallParser | None = None
616 if parse_tools:
617 try:
618 candidate = _make_tool_parser(spec, state)
619 except ValueError:
620 candidate = None # unsupported parser name: keep the buffered path's behavior
621 if candidate is not None and candidate.supports_streaming():
622 tool_parser = candidate
623 frag_stable = tool_parser.args_fragments_prefix_stable() if tool_parser else True
624
625 # Streaming tool-call assembly: detectors emit fragments (name first, then
626 # argument diffs); they accumulate here and the call is emitted complete when
627 # the next call starts, trailing text arrives, or the stream ends.
628 open_call: dict[str, Any] | None = None
629 calls_emitted = 0
630 # Swallow whitespace-only text only between a call's close and the next real
631 # text (markup separators) — NOT inside post-call prose, where a lone " "
632 # chunk is a legitimate word gap.
633 suppress_ws = False
634
635 def _close_open_call() -> ToolCallsDelta | None:
636 nonlocal open_call, calls_emitted
637 if open_call is None:
638 return None
639 # Streamed fragments concatenate to the exact final arguments (detectors
640 # emit prefix-stable fragments and close the JSON before the call ends);
641 # fall back to the detector's parse state when the stream was cut short.
642 params = open_call["params"]
643 if not _valid_json(params):
644 params = tool_parser.unstreamed_arguments(open_call["detector_index"]) or params
645 call = ToolCallItem(
646 tool_index=open_call["ordinal"], name=open_call["name"], parameters=params or "{}"
647 )
648 open_call = None
649 calls_emitted += 1
650 nonlocal suppress_ws
651 suppress_ws = True
652 return ToolCallsDelta([call])
653
654 def _route_tool_text(piece: str) -> list[GenEvent]:
655 nonlocal open_call, suppress_ws
656 out: list[GenEvent] = []
657 if not piece:
658 return out
659 for kind, payload in tool_parser.parse_stream_events(piece):
660 if kind == "text":
661 # Text arriving while a call is open means the call finished:
662 # close it first so wire order matches generation order.
663 done = _close_open_call()
664 if done is not None:
665 out.append(done)
666 stripped = strip_special_tokens(payload, specials)
667 if stripped and not (stripped.strip() == "" and suppress_ws):
668 out.append(ContentDelta(stripped))
669 if stripped.strip():
670 suppress_ws = False
671 continue
672 for frag in payload:
673 starts_new = frag.name is not None and (
674 open_call is None or frag.tool_index != open_call["detector_index"]
675 )
676 if starts_new:
677 done = _close_open_call()
678 if done is not None:
679 out.append(done)
680 if open_call is None or starts_new:
681 open_call = {
682 "detector_index": frag.tool_index,
683 "name": frag.name,
684 "params": "",
685 "ordinal": calls_emitted,
686 }
687 out.append(ToolCallStart(
688 tool_index=calls_emitted, name=frag.name, args_prefix_stable=frag_stable,
689 ))
690 if frag.parameters:
691 open_call["params"] += frag.parameters
692 out.append(
693 ToolCallArgsDelta(tool_index=open_call["ordinal"], fragment=frag.parameters)
694 )
695 return out
696
697 engine_finish_reason: str | None = None
698 engine_matched_stop: str | None = None
699 async for ack in state.wait_for_ack(uid):
700 if getattr(ack, "error", None):
701 raise GenerationError(ack.error, getattr(ack, "error_code", None))
702 prompt_tokens += ack.prompt_tokens_delta
703 completion_tokens += ack.completion_tokens_delta
704 cached_tokens += ack.cached_tokens
705 content_delta = ack.incremental_output
706 if reasoning_parser is not None and content_delta:
707 reasoning_delta, content_delta = reasoning_parser.parse_stream_chunk(content_delta)
708 if reasoning_delta:
709 stripped_reasoning = strip_special_tokens(reasoning_delta, specials)
710 if stripped_reasoning: # a bare special token must not open a thinking block
711 yield ReasoningDelta(stripped_reasoning)
712 if content_delta:
713 if tool_parser is not None:
714 for ev in _route_tool_text(content_delta):
715 yield ev
716 elif parse_tools:
717 pending += content_delta
718 else:
719 yield ContentDelta(strip_special_tokens(content_delta, specials))
720 if ack.finished:
721 engine_finish_reason = getattr(ack, "finish_reason", None)
722 engine_matched_stop = getattr(ack, "matched_stop", None)
723 break
724
725 # Drain residue held in the reasoning parser (a deferred tool block, or a
726 # trailing partial token) so it is not silently dropped.
727 if reasoning_parser is not None:
728 flush_reasoning, flush_content = reasoning_parser.flush()
729 if flush_reasoning:
730 stripped_reasoning = strip_special_tokens(flush_reasoning, specials)
731 if stripped_reasoning:
732 yield ReasoningDelta(stripped_reasoning)
733 if flush_content:
734 if tool_parser is not None:
735 for ev in _route_tool_text(flush_content):
736 yield ev
737 elif parse_tools:
738 pending += flush_content
739 else:
740 yield ContentDelta(strip_special_tokens(flush_content, specials))
741
742 # Engine reason ("stop"/"length"); a tool call overrides it, but a truncation (length) wins.
743 finish_reason = engine_finish_reason or "stop"
744
745 if tool_parser is not None:
746 # End-of-stream drain: let the detector finalize a call cut off mid-arguments
747 # (closing fragments keep the client's concatenated JSON valid), close a call
748 # whose end marker never arrived (truncated generation), best-effort recover a
749 # call cut off inside an unterminated tag block, then release text still held
750 # back for tag disambiguation.
751 for frag in tool_parser.finalize_stream():
752 if open_call is not None and frag.parameters:
753 open_call["params"] += frag.parameters
754 yield ToolCallArgsDelta(tool_index=open_call["ordinal"], fragment=frag.parameters)
755 done = _close_open_call()
756 if done is not None:
757 yield done
758 for item in tool_parser.recover_truncated_call():
759 yield ToolCallStart(tool_index=calls_emitted, name=item.name)
760 yield ToolCallsDelta(
761 [ToolCallItem(tool_index=calls_emitted, name=item.name, parameters=item.parameters)]
762 )
763 calls_emitted += 1
764 residual = tool_parser.finish_stream()
765 if residual:
766 stripped = strip_special_tokens(residual, specials)
767 if stripped and not (stripped.strip() == "" and suppress_ws):
768 yield ContentDelta(stripped)
769 if calls_emitted and finish_reason != "length":
770 finish_reason = "tool_calls"
771 else:
772 parsed = _parse_tool_response(pending, spec, state) if parse_tools else None
773 if parsed is not None:
774 normal_text, tool_calls = parsed
775 normal_text = strip_special_tokens(normal_text, specials)
776 if normal_text:
777 yield ContentDelta(normal_text)
778 yield ToolCallsDelta(tool_calls)
779 if finish_reason != "length":
780 finish_reason = "tool_calls"
781 elif parse_tools and pending:
782 yield ContentDelta(strip_special_tokens(pending, specials))
783
784 yield GenDone(
785 finish_reason, prompt_tokens, completion_tokens,
786 matched_stop=engine_matched_stop, cached_tokens=cached_tokens,
787 )
