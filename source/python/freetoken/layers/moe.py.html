29class MoELayer(BaseOP):
30 """Resident routed experts.
31
32 The expert format comes from ``quant_method`` (declared by ``create_weights``, run by
33 ``apply``); without a ``quant_config`` the experts are plain bf16. The gated activation is
34 ``act(clamp(g, limit) * alpha) * (clamp(u) + beta)`` with ``interleaved`` gate|up rows
35 for gpt-oss."""

107 def routed_forward(
108 self,
109 hidden_states: torch.Tensor,
110 topk_weights: torch.Tensor,
111 topk_ids: torch.Tensor,
112 ) -> torch.Tensor:
113 """Expert compute for an externally computed routing decision (``TopK``).
114
115 Same name and shape as ``OffloadMoELayer.routed_forward`` so a model with
116 its own router calls ``experts.routed_forward(...)`` without knowing whether
117 the experts are resident or offloaded. The shared contract is the offload
118 one: ``topk_ids`` must be safe to mutate in place (the offload decode
119 rewrites expert ids into cache slot ids); pass a fresh tensor or a clone.
120 The resident path does not mutate it today, but callers must not rely on
121 that.
122
123 ``hidden_states`` may also be overwritten by the expert kernel. Compute
124 shared branches that need the original input before calling this method.
125 """
126 out = self._resident_gemm(hidden_states, topk_weights, topk_ids)
127 return self._maybe_all_reduce(out)

203 def routed_forward(
204 self,
205 hidden_states: torch.Tensor,
206 topk_weights: torch.Tensor,
207 topk_ids: torch.Tensor,
208 ) -> torch.Tensor:
209 """Expert compute for an externally computed routing decision (``TopK``).
210
211 The entry point for models whose router does not fit ``fused_topk`` (sigmoid
212 scores, selection bias, group-limited top-k, ...); identical to ``forward``
213 past the router. ``topk_ids`` must be safe to mutate in place (decode
214 rewrites expert ids into cache slot ids); pass a fresh tensor or a clone.

217 shared branches that need the original input before calling this method.
218 """
219 ctx = get_global_ctx()
220 if ctx.batch.is_prefill:
221 out = self._prefill_routed(hidden_states, topk_weights, topk_ids)
222 else:
223 out = self._decode_routed(hidden_states, topk_weights, topk_ids)
224 return self._maybe_all_reduce(out)

252 # ------------------------------------------------------------------
253 # Data movement -- one decision tree for every quant format (the banks
254 # registry makes the cache machinery bank-count agnostic). Decode loads
255 # on demand; prefill streams whole layers, double-buffered when overlap
256 # is enabled. The kernels only ever see bank views plus row indices;
257 # which kernel runs is decided afterwards, in ``_expert_gemm``.
258 # ------------------------------------------------------------------
259
260 def _decode_routed(
261 self,
262 hidden_states: torch.Tensor,
263 topk_weights: torch.Tensor,
264 topk_ids: torch.Tensor,
265 ) -> torch.Tensor:
266 """On-demand load: ``ensure_experts`` rewrites ``topk_ids`` into cache slot
267 ids in place (loading missing experts), then the GEMM reads the full slot
268 cache. All device-side with fixed shapes, so the decode call is CUDA-graph
269 capturable.
270
271 For ``decode_target == "cpu"`` the experts are instead computed on the CPU
272 (high RAM bandwidth) straight from the host banks: ship hidden/routing to
273 pinned host memory, run the GEMV on the worker pool via host nodes, ship the
274 result back. The GPU slot cache is untouched (topk_ids keep their raw expert
275 ids), so no ``ensure_experts``/``copy_missing`` here."""
276 cache = self.offload_cache
277 assert cache is not None
278 if cache.is_cpu_layer(self.layer_id):
279 executor = cache.cpu_executor
280 assert executor is not None, "CPU MoE executor was not initialized"
281 return executor.decode(self.layer_id, hidden_states, topk_weights, topk_ids)
282 if cache.decode_target == "hybrid":
283 return self._decode_hybrid(cache, hidden_states, topk_weights, topk_ids)
284 cache.ensure_experts(self.layer_id, topk_ids)
285 cache.copy_missing()
286 return self._expert_gemm(
287 cache,
288 hidden_states,
289 topk_weights,
290 topk_ids,
291 views=cache.pool_bank_views(self.layer_id),
292 n=None,
293 alphas=cache.alphas_for_slots(self.layer_id),
294 is_prefill=False,
295 )

297 def _decode_hybrid(
298 self,
299 cache: OffloadMoeCache,
300 hidden_states: torch.Tensor,
301 topk_weights: torch.Tensor,
302 topk_ids: torch.Tensor,
303 ) -> torch.Tensor:
304 """Hybrid decode: GPU computes cache hits + <=K freshly-fetched experts, the CPU
305 computes the overflow misses, overlapped, then the partials merge.
306
307 The CPU pool is kicked off (``decode_submit``) before the GPU PCIe fetch + GEMM so
308 the CPU overflow GEMV runs concurrently with the GPU work. Capture-safe: the
309 routing split runs inside the hybrid ensure kernel and the CPU submit/sync are host nodes.
310 Each route is computed exactly once -- CPU-assigned routes reach the GPU as slot -1 with
311 weight 0 (a kernel with ``supports_inactive_slots`` skips them without a read; any other reads
312 slot 0, whose storage is a payload or the bank's zero fill) and the CPU ids are -1 for
313 GPU-assigned routes (the C++ kernel skips id<0).
314 """
315 executor = cache.cpu_executor
316 assert executor is not None, "CPU MoE executor was not initialized"
317 # the kernel reads every route's slot: give the inactive ones slot 0, which holds a
318 # complete payload or the bank's zero fill (never uninitialized storage), times weight 0
319 # with several quants this holds only because the cache zeroes its staging area after prefill
320 clamp = not self.skips_inactive_slots
321 gpu_slots, gpu_w, cpu_ids = cache.ensure_experts_hybrid(self.layer_id, topk_ids, topk_weights, clamp=clamp)
322 pending = executor.decode_submit(self.layer_id, hidden_states, topk_weights, cpu_ids)
323
324 # Measurement knob: FREETOKEN_HYBRID_OVERLAP=0 syncs the CPU pool *before* the
325 # PCIe fetch + GPU GEMM, serializing the two so an A/B isolates the overlap win.
326 cpu_routed_early = (
327 executor.decode_sync(pending) if not _HYBRID_OVERLAP else None
328 )
329
330 cache.copy_missing()
331 gpu_routed = self._expert_gemm(
332 cache,
333 hidden_states,
334 gpu_w,
335 gpu_slots,
336 views=cache.pool_bank_views(self.layer_id),
337 n=None,
338 alphas=cache.alphas_for_slots(self.layer_id),
339 is_prefill=False,
340 )
341 if not _HYBRID_OVERLAP:
342 return gpu_routed + cpu_routed_early
343 return executor.decode_sync(pending, add_to=gpu_routed)

345 def _prefill_routed(
346 self,
347 hidden_states: torch.Tensor,
348 topk_weights: torch.Tensor,
349 topk_ids: torch.Tensor,
350 ) -> torch.Tensor:
351 """Prefill movement: stream whole layers -- double-buffered behind the
352 previous layer's GEMMs when ``prefill_overlap`` is on, else a synchronous
353 ``materialize_layer``. In both, position == expert id, so the routing ids
354 pass through unmapped."""
355 cache = self.offload_cache
356 assert cache is not None
357 if cache.prefill_overlap:
358 views = self._wait_prefill_overlap(cache)
359 out = self._expert_gemm(
360 cache,
361 hidden_states,
362 topk_weights,
363 topk_ids,
364 views=views,
365 n=self.num_experts,
366 alphas=cache.alphas_for_layer(self.layer_id),
367 is_prefill=True,
368 )
369 cache.release_prefill_layer(self.layer_id)
370 return out
371 cache.materialize_layer(self.layer_id)
372 cache.copy_missing()
373 out = self._expert_gemm(
374 cache,
375 hidden_states,
376 topk_weights,
377 topk_ids,
378 views=cache.stage_views(self.layer_id, 0),
379 n=self.num_experts,
380 alphas=cache.alphas_for_layer(self.layer_id),
381 is_prefill=True,
382 )
383 cache.release_prefill_layer(self.layer_id)
384 return out

386 def _wait_prefill_overlap(self, cache: OffloadMoeCache) -> tuple[torch.Tensor, ...]:
387 """Double-buffer choreography for this layer's overlap prefill: kick off the
388 next layer's full-layer H2D copy, then return this layer's bank views (in
389 bank registration order; buffer position == expert id, so routing ids pass
390 through unmapped). The caller runs ``release_prefill_layer`` after its GEMMs.
391 """
392 if self.layer_id == 0:
393 cache.begin_prefill()
394 cache.prefetch_prefill_layer(self.layer_id)
395 cache.prefetch_prefill_layer(self.layer_id + 1)
396 return cache.wait_prefill_layer(self.layer_id)

398 # ------------------------------------------------------------------
399 # Kernel dispatch: ``views`` are the bank tensors the movement step produced (in bank registration order) and ``topk_ids`` already index their rows.
400 # ------------------------------------------------------------------
401
402 def _expert_gemm(
403 self,
404 cache: OffloadMoeCache,
405 hidden_states: torch.Tensor,
406 topk_weights: torch.Tensor,
407 topk_ids: torch.Tensor,
408 *,
409 views: tuple[torch.Tensor, ...],
410 n: int | None,
411 alphas: tuple[torch.Tensor, torch.Tensor] | None,
412 is_prefill: bool,
413 ) -> torch.Tensor:
414 from freetoken.moe.legacy_format import canonical_role # legacy_format imports this package
415
416 view = ExpertView(
417 {canonical_role(name): t for name, t in zip(cache.bank_schema, views)},
418 slots=None if n is not None else topk_ids, n=n, alphas=alphas,
419 )
420 return self.quant_method.apply(
421 hidden_states, topk_weights, topk_ids, view, layer=self, is_prefill=is_prefill
422 )

425def iter_moe_layers(model) -> Iterator[MoELayer]:
426 if isinstance(model, MoELayer):
427 yield model
428 if not isinstance(model, BaseOP):
429 return
430 for value in model.__dict__.values():
431 if isinstance(value, BaseOP):
432 yield from iter_moe_layers(value)
433 elif isinstance(value, (list, tuple)):
434 for item in value:
435 yield from iter_moe_layers(item)

438def make_moe_layer(
439 config: "ModelConfig",
440 *,
441 layer_id: int | None = None,
442 activation: str = "silu",
443 renormalize: bool | None = None,
444 apply_router_weight_on_input: bool = False,
445 num_experts: int | None = None,
446 top_k: int | None = None,
447 hidden_size: int | None = None,
448 intermediate_size: int | None = None,
449 resident_cls: type[MoELayer] | None = None,
450 offload_cls: "type[OffloadMoELayer] | None" = None,
451 alpha: float = 1.0,
452 beta: float = 0.0,
453 limit: float | None = None,
454 interleaved: bool = False,
455 has_bias: bool = False,
456 quant_config: QuantConfig | None = None,
457 prefix: str = "",
458) -> MoELayer:
459 """Build the experts layer for ``config.moe_strategy`` -- the one construction
460 seam between a model and the MoE strategy.
461
462 Picks ``OffloadMoELayer`` for the offload family (offload/cpu/hybrid) and
463 ``MoELayer`` otherwise. Geometry defaults come from ``config``; pass overrides
464 for models whose fields deviate. ``resident_cls``/``offload_cls`` keep model-specific
465 subclasses constructible through the same seam.
466 """
467 offload = is_offload_moe_strategy(config.moe_strategy)
468 layer_cls = (offload_cls or OffloadMoELayer) if offload else (resident_cls or MoELayer)
469 kwargs = dict(
470 num_experts=num_experts if num_experts is not None else config.num_experts,
471 top_k=top_k if top_k is not None else config.num_experts_per_tok,
472 hidden_size=hidden_size if hidden_size is not None else config.hidden_size,
473 intermediate_size=(
474 intermediate_size if intermediate_size is not None else config.moe_intermediate_size
475 ),
476 renormalize=renormalize if renormalize is not None else config.norm_topk_prob,
477 activation=activation,
478 apply_router_weight_on_input=apply_router_weight_on_input,
479 alpha=alpha,
480 beta=beta,
481 limit=limit,
482 interleaved=interleaved,
483 has_bias=has_bias,
484 quant_config=quant_config,
485 prefix=prefix,
486 layer_id=layer_id,
487 )
488 if offload:
489 assert layer_id is not None, "offload MoE backends need the layer_id"
490 kwargs["strategy"] = config.moe_strategy
491 kwargs["decode_target"] = config.decode_target
492 return layer_cls(**kwargs)
