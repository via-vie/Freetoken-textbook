1"""NVFP4 experts: served resident or from the offload cache through the Triton inline-dequant kernels, vLLM's Marlin or flashinfer's b12x.
2
3FreeToken stores MoE experts as ModelOpt NVFP4 (packed e2m1 codes + fp8-e4m3 per-16
4block scales + per-tensor global scale) in pinned host banks and gathers the routed
5experts into a GPU slot cache. Rules every kernel here follows:
6
7* A kernel owns ``{load-time pack, forward}`` as a unit. Prefill and decode consume
8 the *same* bank layout -- there is never a second copy of the experts.
9* Every pack keeps the expert dimension outermost with per-expert blocks contiguous
10 and byte-identical in size to the native layout, so the banks are repacked *in
11 place* and the offload cache's slot gather (``copy_missing``) works unchanged.
12* The per-tensor global scales never enter the offload banks: they are tiny
13 ``[L*E]`` vectors kept resident on the GPU and gathered per forward call
14 (device-side, CUDA-graph safe).
15"""
16

174# Marlin pack: mirrors vLLM's prepare_moe_fp4_layer_for_marlin so the host prep always matches the AOT op.
175def _marlin_pack_proj(
176 packed: torch.Tensor,
177 scale: torch.Tensor,
178 row_global: torch.Tensor,
179 *,
180 size_k: int,
181 size_n: int,
182) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
183 """Pack one expert projection (on GPU) into the Marlin layout.
184
185 ``packed`` ``[N, K//2]`` uint8 e2m1 pairs, ``scale`` ``[N, K//16]`` e4m3,
186 ``row_global`` ``[N]`` fp16 per-row global (rows of a merged gate_up may carry
187 w1's and w3's different globals). Returns ``(qweight [K//16, 2N] int32,
188 scales [K//16, N] e4m3-coded, global scalar bf16)``.
189
190 Marlin takes one global scalar per projection, so when the row globals differ
191 the ratio is folded into the fp16 block scales before they are re-encoded
192 (exact when all rows share one global, <=1 scale ulp otherwise).
193 """
194 from vllm import _custom_ops as ops
195 from vllm.model_executor.layers.quantization.utils.marlin_utils import (
196 marlin_permute_scales,
197 )
198 from vllm.model_executor.layers.quantization.utils.marlin_utils_fp4 import (
199 nvfp4_marlin_process_global_scale,
200 nvfp4_marlin_process_scales,
201 )
202
203 assert size_n % 64 == 0, f"Marlin requires N % 64 == 0, got {size_n}"
204 qweight = ops.gptq_marlin_repack(
205 b_q_weight=packed.view(torch.int32).T.contiguous(),
206 perm=torch.empty(0, dtype=torch.int, device=packed.device),
207 size_k=size_k,
208 size_n=size_n,
209 num_bits=4,
210 )
211
212 g = row_global.to(torch.float32)
213 g_max = g.max()
214 s = scale.to(torch.bfloat16)
215 if not torch.all(g == g_max):
216 s = (s.float() * (g / g_max).unsqueeze(1)).to(torch.bfloat16)
217 s = marlin_permute_scales(s.T, size_k=size_k, size_n=size_n, group_size=16)
218 s = nvfp4_marlin_process_scales(s)
219 if isinstance(s, tuple): # newer vLLM returns (scales, scale_factor)
220 s, factor = s
221 g_max = g_max / factor
222 g_out = nvfp4_marlin_process_global_scale(g_max.to(torch.bfloat16).reshape(1))
223 return qweight, s, g_out

274class MarlinNvfp4MoEKernel(MoEKernel):
275 """vLLM's fused Marlin MoE over pre-tiled banks; the globals fold into GPU-resident alphas."""
276
277 name = "marlin"
278 max_slots = MARLIN_MAX_SLOTS
279
280 def unusable_reason(self, cfg: MoEConfig) -> str | None:
281 if backend.is_rocm():
282 return "Marlin is CUDA-only and unavailable on ROCm; use triton"
283 if not backend.is_vllm_installed():
284 return "vLLM is not installed"
285 reason = self._common_reject(cfg, tp_ok=False, cpu_ok=False, plain_silu_only=True)
286 if reason:
287 return reason
288 if not _marlin_symbols_ok():
289 return "vLLM Marlin donor symbols are unusable"
290 return None
291
292 def worth_it(self, cfg: MoEConfig) -> bool:
293 return (8, 0) <= backend.device_capability() < (10, 0)
294
295 def layout(self, cfg: MoEConfig) -> dict[str, BankSpec]:
296 i, h = cfg.intermediate, cfg.hidden
297 return {
298 "gate_up": BankSpec((h // GROUP, 4 * i), torch.int32),
299 "gate_up_scale": BankSpec((h // GROUP, 2 * i), FP8),
300 "down": BankSpec((i // GROUP, 2 * h), torch.int32),
301 "down_scale": BankSpec((i // GROUP, h), FP8),
302 "gate_up_alpha": BankSpec((), torch.bfloat16, resident=True),
303 "down_alpha": BankSpec((), torch.bfloat16, resident=True),
304 }
305
306 def pack(self, pieces, cfg: MoEConfig, out):
307 i, h = cfg.intermediate, cfg.hidden
308 device = torch.device("cuda")
309 gu, gus, gug = fused_piece(pieces, "gate_up"), fused_piece(pieces, "gate_up_scale"), fused_global(pieces, i)
310 dn, dns, dng = pieces["down"], pieces["down_scale"], global_rows(pieces["down_global"], h)
311 e = gu.shape[0]
312 gate_up_alpha = torch.empty(e, dtype=torch.bfloat16, device=device)
313 down_alpha = torch.empty(e, dtype=torch.bfloat16, device=device)
314 for k in range(e):
315 qw, sc, al = _marlin_pack_proj(gu[k].to(device), gus[k].to(device), gug[k].to(device), size_k=h, size_n=2 * i)
316 out["gate_up"][k].copy_(qw)
317 out["gate_up_scale"][k].copy_(sc.view(FP8))
318 gate_up_alpha[k] = al[0]
319 qw, sc, al = _marlin_pack_proj(dn[k].to(device), dns[k].to(device), dng[k].to(device), size_k=i, size_n=h)
320 out["down"][k].copy_(qw)
321 out["down_scale"][k].copy_(sc.view(FP8))
322 down_alpha[k] = al[0]
323 torch.cuda.synchronize(device)
324 return {"gate_up_alpha": gate_up_alpha, "down_alpha": down_alpha}
325
326 def apply(self, layer, x, topk_weights, topk_ids, view: ExpertView, *, is_prefill: bool):
327 t = view.tensors
328 assert view.alphas is not None
329 return marlin_fused_experts(x, t["gate_up"], t["gate_up_scale"], view.alphas[0], t["down"], t["down_scale"], view.alphas[1], topk_weights, topk_ids, layer.activation, layer.apply_router_weight_on_input)

332# ---------------------------------------------------------------------------
333# b12x: flashinfer's SM12x CuTe-DSL fused MoE (W4A16) over b12x-packed banks.
334# The pack (``prepare_w4a16_packed_weights``) runs on any CUDA build, but the fused-MoE
335# kernel needs sm_120/121 AND a CUDA>=13 *driver* (it JIT-compiles PTX at runtime). The
336# kernel object gates on that, so the forward is only reached on hardware that can run it.
337#
338# Layout contract (verified against flashinfer 0.6.12 prepare_w4a16_packed_weights):
339# * prepare() takes the block scales already *swizzled* (it calls unswizzle_expert_
340# scales internally), so the native row-major fp8 scales are run through
341# nvfp4_block_scale_interleave first.
342# * prepare() reorders the w13 rows with reorder_w13_to_gate_up == cat([second, first]),
343# i.e. it expects the merged proj as [up, gate] and emits [gate, up]. FreeToken stores
344# [gate, up], so the halves are swapped before prepare() to come out [gate, up] again
345# (the kernel computes silu(first) * second == silu(gate) * up).
346# * modelopt globals are passed straight through (no 1/g inversion).
347# The fused forward reconstructs the W4A16PackedWeights from the banks and uses the
348# prepared-weights launch (_launch_sm120_w4a16_moe) directly, so weights are prepared
349# exactly once at load time (b12x_fused_moe would re-prepare + ptr-cache every call).
350# ---------------------------------------------------------------------------
351

395 """None if the flashinfer b12x decode/prefill kernel can actually run on this device,
396 else a human-readable reason. The b12x *pack* works anywhere, but the SM12x CuTe-DSL
397 kernel itself requires sm_120+ and a CUDA>=13 *driver* (it JIT-compiles PTX through the

401

403 return f"b12x requires sm_120+, got sm_{cc[0]}{cc[1]}"

429def _b12x_swizzle_block_scales(scale: torch.Tensor) -> torch.Tensor:
430 """Row-major per-16 fp8 block scales ``[n, N, K//16]`` -> flashinfer's expert-leading
431 *swizzled* storage, which is what :func:`prepare_w4a16_packed_weights` expects on input
432 (it unswizzles internally). The 128x4 interleave is per-expert; ``N`` is a multiple of
433 128 and ``K//16`` a multiple of 4 for the supported models, so the shape is preserved."""
434 from flashinfer import nvfp4_block_scale_interleave
435
436 pieces = [
437 nvfp4_block_scale_interleave(scale[e].view(torch.uint8).contiguous())
438 for e in range(scale.shape[0])
439 ]
440 return torch.stack(pieces, dim=0).view(torch.float8_e4m3fn)

461def b12x_fused_experts(
462 hidden_states: torch.Tensor,
463 gate_up_q: torch.Tensor,
464 gate_up_s: torch.Tensor,
465 gate_up_alpha: torch.Tensor,
466 down_q: torch.Tensor,
467 down_s: torch.Tensor,
468 down_alpha: torch.Tensor,
469 topk_weights: torch.Tensor,
470 topk_ids: torch.Tensor,
471 activation: str,
472 apply_router_weight_on_input: bool,
473) -> torch.Tensor:
474 """SM12x W4A16 fused MoE over cache slots (same calling convention as
475 :func:`marlin_fused_experts`). Requires sm_120/121 + a CUDA>=13 driver; reached only
476 when the b12x kernel object has confirmed the runtime supports it.
477
478 The banks already hold flashinfer's prepared (tiled) layout from
479 the b12x kernel object's pack, so this wraps them back into a
480 ``W4A16PackedWeights`` and calls the prepared-weights launch directly -- the public
481 ``b12x_fused_moe`` would re-prepare the raw modelopt weights (and ptr-cache them) on
482 every call, which both costs a prepare per step and breaks for the offload cache,
483 whose slot contents move between calls.
484
485 CUDA-graph note: the launch resolves a (shape-keyed, module-cached) scratch workspace
486 on first use and flashinfer raises if that happens *during* capture, so the decode
487 path must be warmed once eagerly before graph capture (FreeToken already does)."""
488 from flashinfer.fused_moe.cute_dsl.blackwell_sm12x.moe_dispatch import (
489 _launch_sm120_w4a16_moe,
490 )
491 from flashinfer.fused_moe.cute_dsl.blackwell_sm12x.moe_w4a16_prepare import (
492 W4A16PackedWeights,
493 )
494
495 assert activation == "silu", "b12x backend supports gated silu only"
496 assert not apply_router_weight_on_input
497 num_experts = gate_up_q.size(0)
498 hidden_size = hidden_states.size(-1)
499 # down bank is the prepared w2 == [E, K_tiles, ...] with K_tiles == intermediate//16.
500 intermediate_size = down_q.size(1) * 16
501 prepared = W4A16PackedWeights(
502 w13=gate_up_q,
503 w13_scale=gate_up_s,
504 w13_global_scale=gate_up_alpha,
505 w2=down_q,
506 w2_scale=down_s,
507 w2_global_scale=down_alpha,
508 workspace=_b12x_small_workspace(hidden_states.device),
509 hidden_size=hidden_size,
510 intermediate_size=intermediate_size,
511 num_experts=num_experts,

540class B12xNvfp4MoEKernel(MoEKernel):
541 """flashinfer's SM12x CuTe-DSL W4A16 MoE; the packed blocks reuse the native bank bytes."""

543 name = "b12x"
544
545 def unusable_reason(self, cfg: MoEConfig) -> str | None:
546 if backend.is_rocm():
547 return "b12x is CUDA-only and unavailable on ROCm; use triton"
548 cc = backend.device_capability()
549 if cc < (12, 0):
550 return f"b12x requires sm_120+, got sm_{cc[0]}{cc[1]}"
551 if not backend.is_flashinfer_installed():
552 return "flashinfer is not installed"
553 reason = self._common_reject(cfg, tp_ok=False, cpu_ok=False, plain_silu_only=True)
554 if reason:
555 return reason
556 return _b12x_unusable_reason(cc)
557
558 def worth_it(self, cfg: MoEConfig) -> bool:
559 # NOTE: never auto-selected. flashinfer's cute launcher indexes each bank with int32 element offsets, so the GPU slot cache is capped at (2^31 - 1) / elements-per-slot (about 1000 slots for GLM-5.3-Flash's 12960 experts); until flashinfer lifts that, b12x stays behind triton in the table; the selection does not weigh the cap, cache-auto clamps to slot_limit() and OffloadMoeCache refuses a larger cache.
560 return cfg.intermediate >= B12X_MIN_INTERMEDIATE
561
562 def slot_limit(self, cfg: MoEConfig) -> int | None:
563 # the cute launcher indexes each bank with int32 element offsets
564 per_slot = max(math.prod(spec.shape) for spec in self.layout(cfg).values() if not spec.resident)
565 return (2**31 - 1) // per_slot
566
567 def layout(self, cfg: MoEConfig) -> dict[str, BankSpec]:
568 # flashinfer's prepared tiles: the Marlin shapes, byte-identical to the native rows
569 i, h = cfg.intermediate, cfg.hidden
570 return {
571 "gate_up": BankSpec((h // GROUP, 4 * i), torch.int32),
572 "gate_up_scale": BankSpec((h // GROUP, 2 * i), FP8),
573 "down": BankSpec((i // GROUP, 2 * h), torch.int32),
574 "down_scale": BankSpec((i // GROUP, h), FP8),
575 "gate_up_alpha": BankSpec((), torch.float32, resident=True),
576 "down_alpha": BankSpec((), torch.float32, resident=True),
577 }

579 def pack(self, pieces, cfg: MoEConfig, out):
580 from flashinfer.fused_moe.cute_dsl.blackwell_sm12x.moe_w4a16_prepare import prepare_w4a16_packed_weights
581
582 i, h = cfg.intermediate, cfg.hidden
583 device = torch.device("cuda")
584 gu = fused_piece(pieces, "gate_up").to(device)
585 gug = fused_global(pieces, i).to(device).float()
586 g_max = gug.max(dim=1, keepdim=True).values
587 gus = fused_piece(pieces, "gate_up_scale").to(device).to(torch.float16)
588 ratio = gug / g_max
589 if not torch.all(ratio == 1.0):
590 gus = (gus.float() * ratio.unsqueeze(-1)).to(torch.float16)
591 dng = global_rows(pieces["down_global"], h).to(device)
592 if not torch.all(dng == dng[:, :1]):
593 raise ValueError("b12x pack requires a row-constant per-expert down_proj global scale")
594 # b12x wants up|gate row order
595 gu = torch.cat([gu[:, i:], gu[:, :i]], dim=1).contiguous()
596 gus = torch.cat([gus[:, i:], gus[:, :i]], dim=1).contiguous()
597 prepared = prepare_w4a16_packed_weights(
598 gu,
599 _b12x_swizzle_block_scales(gus.to(FP8)),
600 g_max.squeeze(1),
601 pieces["down"].to(device),
602 _b12x_swizzle_block_scales(pieces["down_scale"].to(device)),
603 dng[:, 0].float(),
604 activation="silu",
605 params_dtype=torch.bfloat16,
606 source_format="modelopt",
607 )
608 e = gu.shape[0]
609 for role, t in (("gate_up", prepared.w13), ("gate_up_scale", prepared.w13_scale), ("down", prepared.w2), ("down_scale", prepared.w2_scale)):
610 bank = out[role]
611 assert t.shape[1:] == bank.shape[1:] and t.dtype == bank.dtype, (role, t.shape, t.dtype, bank.shape, bank.dtype)
612 bank.copy_(t[:e])
613 return {"gate_up_alpha": prepared.w13_global_scale[:e].float(), "down_alpha": prepared.w2_global_scale[:e].float()}

616 t = view.tensors
617 assert view.alphas is not None
618 return b12x_fused_experts(x, t["gate_up"], t["gate_up_scale"], view.alphas[0], t["down"], t["down_scale"], view.alphas[1], topk_weights, topk_ids, layer.activation, layer.apply_router_weight_on_input)
