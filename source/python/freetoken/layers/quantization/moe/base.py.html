90def fused_piece(pieces: dict[str, torch.Tensor], role: str) -> torch.Tensor:
91 """A gate_up-family piece, concatenating separate gate / up pieces along the row axis."""
92 if role in pieces:
93 return pieces[role]
94 suffix = role[len("gate_up"):]
95 return torch.cat([pieces["gate" + suffix], pieces["up" + suffix]], dim=1)
96
97
98def global_rows(piece: torch.Tensor, rows: int) -> torch.Tensor:
99 """Per-expert global scale as one fp16 value per output row (a scalar per expert is broadcast)."""
100 flat = piece.reshape(piece.shape[0], -1).to(torch.float16)
101 if flat.shape[1] == rows:
102 return flat
103 assert flat.shape[1] == 1, (flat.shape, rows)
104 return flat.expand(-1, rows)
105
106
107def fused_global(pieces: dict[str, torch.Tensor], half_rows: int) -> torch.Tensor:
108 if "gate_up_global" in pieces:
109 return global_rows(pieces["gate_up_global"], 2 * half_rows)
110 return torch.cat([global_rows(pieces["gate_global"], half_rows), global_rows(pieces["up_global"], half_rows)], dim=1)

118def gated_epilogue_reason(cfg: "MoEConfig") -> str | None:
119 """Why ``gated_act_and_mul`` cannot run ``cfg``'s activation quadruple, or None; the plain kinds have no alpha / limit and nothing takes beta or interleaved rows."""
120 from freetoken.layers import GATED_ACTIVATIONS
121
122 if cfg.activation not in GATED_ACTIVATIONS:
123 return f"no {cfg.activation!r} epilogue"
124 if cfg.interleaved:
125 return "the epilogue reads uninterleaved [gate; up] halves"
126 if cfg.beta != 0.0:
127 return f"the epilogue has no beta (got {cfg.beta})"
128 if cfg.activation in ("silu", "gelu", "gelu_tanh") and (cfg.alpha != 1.0 or cfg.limit is not None):
129 return f"{cfg.activation} takes no alpha / limit (got alpha={cfg.alpha}, limit={cfg.limit})"
130 return None

137class MoEKernel(ABC):
138 """One backend for one expert kind: bank layout, packing from checkpoint pieces, and the fused forward."""
139
140 name: ClassVar[str]
141 cpu_format: ClassVar[str | None] = None
142 max_slots: ClassVar[int | None] = None
143 # Whether ``apply`` treats a slot of -1 as an inactive route (zero output, no read of any slot).
144 # Hybrid decode hands the GPU a route split with such holes; kernels without this get the holes
145 # pointed at slot 0 with weight 0 (slot storage is zero-filled at allocation, so that is exact).
146 supports_inactive_slots: ClassVar[bool] = False
147
148 def unusable_reason(self, cfg: MoEConfig) -> str | None:
149 return None
150
151 def worth_it(self, cfg: MoEConfig) -> bool:
152 return True
153
154 def cpu_options(self, cfg: MoEConfig) -> dict[str, int]:
155 """Integer options of ``cpu_format`` for these experts."""
156 return {}
157
158 def slot_limit(self, cfg: MoEConfig) -> int | None:
159 """Most GPU cache slots the kernel can address for ``cfg``; None for no limit."""
160 return self.max_slots
161
162 def _common_reject(self, cfg: MoEConfig, *, tp_ok: bool, cpu_ok: bool, plain_silu_only: bool) -> str | None:
163 if not tp_ok and cfg.tp_size > 1:
164 return "TP > 1 is not supported for this expert format"
165 if not cpu_ok and cfg.decode_target != "gpu":
166 return "has no CPU executor format; decode must run on the GPU"
167 if cfg.has_bias:
168 return "kernel has no bias epilogue"
169 if plain_silu_only and not cfg.plain_silu:
170 return f"kernel is plain-silu only, experts use ({cfg.activation}, alpha={cfg.alpha}, beta={cfg.beta}, limit={cfg.limit})"
171 return None
