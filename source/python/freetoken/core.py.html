35@dataclass(eq=False)
36class Req:
37 input_ids: torch.Tensor # cpu tensor
38 table_idx: int
39 cached_len: int
40 output_len: int
41 uid: int
42 sampling_params: SamplingParams
43 cache_handle: BaseCacheHandle
44 # per-item processor outputs and the tokenizer's precomputed mrope rows and delta
45 mm_items: list | None = None
46 mrope_positions_full: torch.Tensor | None = None # [3, prompt_len] int32, CPU
47 mrope_delta: int = 0
48 # The whole prompt's length; 0 = this request's input_ids (a ChunkedReq's end at its chunk).
49 prompt_len: int = 0
50
51 # --- hybrid-radix (GDN linear-state) per-request slots; None for non-hybrid models or
52 # until allocated from LinearStatePool. Set by the scheduler (P2). ---
53 linear_slot_idx: int | None = None # live GDN state slot (sglang mamba_pool_idx)
54 mamba_ping_pong: tuple[int, int] | None = None # 2 donatable track slots under overlap
55 mamba_next_track_idx: int = 0 # which ping-pong slot is the next snapshot dst (0/1)
56 mamba_last_track_seqlen: int | None = None # chunk-aligned committed len of the last snapshot
57 mamba_restore_src: int | None = None # on a prefix hit: tree snapshot slot to COW into the live slot (first chunk only)
58 swa_evicted_seqlen: int = 0 # SWA radix: positions < this had their swa KV freed (slid out of window) during decode
59 decode_batch_idx: int = 0 # SWA radix: # of decode forwards done; the proactive free_swa skips the first (overlap guard)
60 # Set once, at the first sampled tool-call opener token (scheduler detection): the state
61 # length just after that token (its index + 1). A client-side rewrite of the echoed tool
62 # call diverges strictly after this point, so it is the deepest reuse boundary that
63 # survives such a rewrite. GDN: the state is frozen into a ping-pong slot when cached_len
64 # reaches it (snapshot_toolcall_anchor) and donated at finish. SWA: caps the proactive
65 # out-of-window eviction so the window ending here stays resumable.
66 toolcall_anchor_len: int | None = None
67 # Abort arrived while this request's forward was in flight (overlap scheduling). The abort
68 # handler must not free resources under an in-flight forward; it sets this flag and
69 # _process_last_data frees the request when the batch drains (after copy_done.synchronize).
70 aborted: bool = False
71 output_token_counts: torch.Tensor | None = field(default=None, init=False, repr=False)

73 def __post_init__(self) -> None:
74 assert self.input_ids.is_cpu
75 self.device_len = len(self.input_ids)
76 self.max_device_len = len(self.input_ids) + self.output_len
77 self.prompt_len = self.prompt_len or self.device_len
78 assert 0 <= self.cached_len < self.device_len <= self.max_device_len
79 self._alloc_ids_buf()
80
81 def _alloc_ids_buf(self) -> None:
82 self._ids_buf = torch.empty(self.max_device_len, dtype=self.input_ids.dtype)
83 self._ids_buf[: self.device_len] = self.input_ids
84 self.input_ids = self._ids_buf[: self.device_len]
85
86 @property
87 def remain_len(self) -> int:
88 return self.max_device_len - self.device_len
89
90 @property
91 def extend_len(self) -> int:
92 return self.device_len - self.cached_len
93
94 def complete_one(self) -> None:
95 self.cached_len = self.device_len
96 self.device_len += 1
97
98 def append_host(self, next_token: torch.Tensor) -> None:
99 n = self.input_ids.numel()
100 m = n + next_token.numel()
101 assert m <= self.max_device_len
102 self._ids_buf[n:m] = next_token
103 self.input_ids = self._ids_buf[:m]
104
105 @property
106 def can_decode(self) -> bool:
107 return self.remain_len > 0
108
109 def __repr__(self) -> str:
110 return (
111 f"{type(self)}(table_idx={self.table_idx}, "
112 f"cached_len={self.cached_len}, device_len={self.device_len}, "
113 f"max_device_len={self.max_device_len})"

118@dataclass
119class Batch:
120 reqs: List[Req]
121 phase: Literal["prefill", "decode"]
122 # these fields should be set by scheduler
123 input_ids: torch.Tensor = field(init=False)
124 positions: torch.Tensor = field(init=False)
125 # [3, n] t/h/w rope positions on mrope models; positions keeps its sequence-index meaning for token_pool / page_table
126 mrope_positions: torch.Tensor | None = field(default=None, init=False)
127 out_loc: torch.Tensor | None = field(init=False)
128 # Per-(padded-)request table_idx as a GPU int64 tensor, used by GatedDeltaNet
129 # decode to gather/scatter recurrent+conv state without host-side loops (so the
130 # decode step is CUDA-graph capturable). Set by the scheduler / graph buffer.
131 linear_table_idx: torch.Tensor | None = field(default=None, init=False)
132 # Per-forward GatedDeltaNet metadata (cu_seqlens / cache_indices / continuation
133 # flags), built once and shared by all GDN layers. Lazily built by the GDN op if
134 # the scheduler/graph didn't set it.
135 fla_metadata: "FLAMetadata | None" = field(default=None, init=False)
136 padded_reqs: List[Req] = field(init=False)
137 # DSV4 paged-KV out-locations for this batch (None for non-DSV4 models). Set by the scheduler.
138 # This decode batch's padded per-row page-table rows. Attention backends that must read
139 # positions anywhere in a request's history snapshot those rows before a captured replay
140 # (DSV4), since the next batch's allocate_paged mutates the live table.
141 active_table_idx: "torch.Tensor | None" = None
142 # this field should be set by attention backend
143 attn_metadata: BaseAttnMetadata = field(init=False)
144 # concatenated multimodal soft-token embeddings for a prefill batch (or None) and the batch rows they land on
145 mm_embeds: torch.Tensor | None = field(default=None, init=False)
146 mm_rows: torch.Tensor | None = field(default=None, init=False)
147 # per batch token, the end (exclusive, in its request) of the image span holding it, 0 for text: the block a bidirectional layer attends within
148 mm_block_ends: torch.Tensor | None = field(default=None, init=False)
149 # this chunk's cache-miss items to encode and the gather plan [(uid, hash, row_lo, row_hi, n, pos), ...] in scatter order
150 mm_encoder_jobs: list | None = field(default=None, init=False)
151 mm_gather_plan: list | None = field(default=None, init=False)
152 # Prefill log stats snapshotted at schedule time (before forward's complete_one()
153 # advances cached_len), so the prefill log reports the tokens actually forwarded and
154 # the prefix-cache hit -- matching SGLang's #new-token / #cached-token. Set by the
155 # PrefillManager; 0 on decode batches.
156 log_new_tokens: int = field(default=0, init=False)
157 log_cached_tokens: int = field(default=0, init=False)
158 # (uid, complete prompt length, prefix-cache hit) for requests entering their first
159 # prepared prefill batch. The scheduler turns these into PromptAdmittedMsg only AFTER
160 # _prepare_batch succeeds. Continuation chunks leave this empty, so accounting is
161 # exactly-once.
162 prompt_admissions: List[Tuple[int, int, int]] = field(default_factory=list, init=False)
