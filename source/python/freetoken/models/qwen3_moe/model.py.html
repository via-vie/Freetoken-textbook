19class Qwen3DecoderLayer(BaseOP):
20 def __init__(self, config: ModelConfig, layer_id: int, *, prefix: str = ""):
21 self.self_attn = Qwen3Attn(config, layer_id, has_qk_norm=True, prefix=f"{prefix}.self_attn")
22 self.mlp = Qwen3MLP(config, layer_id, prefix=f"{prefix}.mlp")
23 self.input_layernorm = RMSNormFused(
24 size=config.hidden_size,
25 eps=config.rms_norm_eps,
26 )
27 self.post_attention_layernorm = RMSNormFused(
28 size=config.hidden_size,
29 eps=config.rms_norm_eps,
30 )
31
32 self._layer_id = layer_id
33
34 @nvtx_annotate("Layer_{}", layer_id_field="_layer_id")
35 def forward(
36 self, x: torch.Tensor, residual: torch.Tensor | None = None
37 ) -> Tuple[torch.Tensor, torch.Tensor]:
38 x, residual = self.input_layernorm.forward(x, residual)
39 x = self.self_attn.forward(x)
40 x, residual = self.post_attention_layernorm.forward(x, residual)
41 x = self.mlp.forward(x)
42 return x, residual

45class Qwen3Model(BaseOP):
46 def __init__(self, config: ModelConfig, *, prefix: str = "model"):
47 self.embed_tokens = VocabParallelEmbedding(
48 num_embeddings=config.vocab_size,
49 embedding_dim=config.hidden_size,
50 quant_config=config.quant,
51 prefix=f"{prefix}.embed_tokens",
52 )
53 self.layers = OPList(
54 [
55 Qwen3DecoderLayer(config, layer_id, prefix=f"{prefix}.layers.{layer_id}")
56 for layer_id in range(config.num_layers)
57 ]
58 )
59 self.norm = RMSNormFused(
60 size=config.hidden_size,
61 eps=config.rms_norm_eps,
62 )
63
64 def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
65 x = self.embed_tokens.forward(input_ids)
66 residual: torch.Tensor | None = None
67 for layer in self.layers.op_list:
68 x, residual = layer.forward(x, residual)
69 return self.norm.forward(x, residual)[0]

72class Qwen3MoeForCausalLM(BaseLLMModel):
