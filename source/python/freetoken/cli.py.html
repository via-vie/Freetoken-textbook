76def _run_bench(argv: list[str]) -> int:
77 if not argv:
78 _print_bench_help(sys.stderr)
79 return 2
80 sub = argv[0]
81 if sub in {"-h", "--help"}:
82 _print_bench_help(sys.stdout)
83 return 0
84 if sub == "bw":
85 from freetoken.moe.benchbw import main
86
87 return main(argv[1:], prog="ft bench bw")
88 print(f"unknown ft bench subcommand: {sub}", file=sys.stderr)
89 _print_bench_help(sys.stderr)

104def main(argv: Sequence[str] | None = None) -> int:
105 args = list(sys.argv[1:] if argv is None else argv)
106 if not args:
107 _print_help(sys.stderr)
108 return 2
109 if args[0] in {"-h", "--help"}:
110 _print_help(sys.stdout)
111 return 0
112 if args[0] in {"-V", "--version"}:
113 from freetoken.version import __version__
114
115 print(f"freetoken version {__version__}")
116 return 0
117
118 command = args[0]
119 runner_name = COMMANDS.get(command)
120 if runner_name is None:
121 print(f"unknown ft command: {command}", file=sys.stderr)
122 _print_help(sys.stderr)
123 return 2
124
125 runner = globals()[runner_name]
126 return runner(args[1:])
