1"""Reusable pinned host-bank primitives shared by the fast expert-load paths.
2
3Two ideas the parallel read of the original checkpoint and FTW (read a repacked
4contiguous cache) paths both rely on:
5
6* **pin-after-fill** -- allocate the bank as a *lazy* anonymous ``mmap`` (no pages
7 resident, instant), fill it with real data, and only THEN ``cudaHostRegister`` it.
8 Registering already-resident pages just page-locks them; registering a lazy mmap first
9 faults+zero-fills every page (~137 GiB -> ~47 s for DSV4) and that zero-fill is then
10 immediately overwritten by the read. So pin-after-fill removes a whole redundant pass.
11* **chunked multi-threaded O_DIRECT** -- DMA straight from disk into the (page-aligned)
12 bank, bypassing the page cache, with many concurrent ``preadv`` on one fd (scales to the
13 device's queue-depth ceiling even for a single file).
14
15The mmaps are held for the process lifetime (the banks live as long as the offload cache).
16"""
17

39class PinFailed(RuntimeError):
40 """cudaHostRegister refused a bank: the host is out of pinnable RAM or over its pin quota."""

43class HostResidency(str, Enum):
44 """Residency class of a host bank layer.
45
46 Only PINNED (cudaHostRegister'd) memory can feed the GPU movement paths; LOCKED (mlock'd, no device address) and PAGEABLE layers must decode on the CPU executor.
47 The non-pinned classes exist for hosts that cap CUDA pin quota (WSL/WDDM: ~half of RAM).
48 """
49
50 PINNED = "pinned"
51 LOCKED = "locked"
52 PAGEABLE = "pageable"
53
54
55_DEFAULT_CHUNK = 8 << 20
56
57# Hold the mmaps for the process lifetime; the offload cache reads from these banks forever.
58_LIVE_BUFFERS: list[mmap.mmap] = []

60def _env_born_pinned() -> bool | None:
61 """``FREETOKEN_BANK_CUDA_ALLOC`` tri-state: unset -> ``None`` (default applies), else the parsed boolean."""
62 v = os.environ.get("FREETOKEN_BANK_CUDA_ALLOC", "").strip().lower()
63 if not v:
64 return None
65 return v in ("1", "true", "yes", "on")

68def born_pinned_default() -> bool:
69 """Whether PINNED serving banks use cudaHostAlloc instead of mmap + register-after-fill.
70
71 Off by default: registered mmaps already read at the PCIe roofline and lazy mmaps commit pages only on fill. ``FREETOKEN_BANK_CUDA_ALLOC`` overrides."""
72 env = _env_born_pinned()
73 if env is not None:
74 return env
75 return False

78class HostBank:
79 """A page-aligned host buffer + its torch view, page-locked on demand: allocate -> fill -> ``pin()``/``lock()``.
80
81 * ``"mmap"`` (default) -- lazy anonymous mmap; pages materialize on fill, then ``pin()`` registers or ``lock()`` OS-locks it.
82 * ``"cuda"`` -- cudaHostAlloc, born pinned+mapped; ``pin()``/``lock()``/``release()`` are no-ops and it never takes LOCKED. See :func:`born_pinned_default`.
83
84 The buffer is rounded up to the O_DIRECT block; ``tensor`` views exactly ``nbytes``. ``backing=None`` follows ``FREETOKEN_BANK_CUDA_ALLOC``."""
85
86 __slots__ = ("tensor", "addr", "nbytes", "_buf", "_pinned", "_locked")
87
88 def __init__(self, shape: tuple[int, ...], dtype: torch.dtype,
89 *, backing: str | None = None):
90 if backing is None:
91 plan = _requested_residency
92 # a plan with non-pinned labels vetoes born-pinned: cudaHostAlloc spends the pin quota the plan exists to save
93 born = _env_born_pinned() and (plan is None or not plan.has_unpinned)
94 backing = "cuda" if born else "mmap"
95 assert backing in ("mmap", "cuda"), backing
96 elsize = torch.empty((), dtype=dtype).element_size()
97 self.nbytes = math.prod(shape) * elsize
98 asize = ((self.nbytes + _BLK - 1) // _BLK) * _BLK
99 if backing == "cuda":
100 from freetoken.kernel.pinned import alloc_pinned_tensor
101
102 # direct-IO readers need page alignment, but cudaHostAlloc only guarantees ~512 in practice
103 # over-allocate one block and carve the aligned window; the numpy slice keeps the pinned storage alive via .base
104 raw = alloc_pinned_tensor(asize + _BLK, dtype=torch.uint8) # cudaMallocHost
105 raw.zero_() # keep the anonymous-mmap guarantee: unwritten regions stay zero
106 off = (-raw.data_ptr()) % _BLK
107 self._buf = raw.numpy()[off:off + asize]
108 self.addr = raw.data_ptr() + off
109 assert self.addr % _BLK == 0
110 self._pinned = True # born pinned+mapped; pin() is a no-op
111 else:
112 self._buf = mmap.mmap(-1, asize) # lazy: address space only, no resident pages yet
113 _LIVE_BUFFERS.append(self._buf)
114 self.addr = ctypes.addressof(ctypes.c_char.from_buffer(self._buf))
115 self._pinned = False
116 self.tensor = torch.frombuffer(self._buf, dtype=dtype, count=self.nbytes // elsize).view(*shape)
117 self._locked = False
118
119 @property
120 def residency(self) -> HostResidency:
121 if self._pinned:
122 return HostResidency.PINNED
123 if self._locked:
124 return HostResidency.LOCKED
125 return HostResidency.PAGEABLE
126
127 def memoryview(self) -> memoryview:
128 return memoryview(self._buf)
129
130 def pin(self) -> None:
131 """cudaHostRegister the (now-filled) buffer -- pin-after-fill.
132
133 ``FREETOKEN_SKIP_BANK_PIN=1`` makes this a no-op for CPU-only tooling (the FTW converter); never set it when serving, the GPU paths need registered banks."""
134 if self._pinned:
135 return
136 if os.environ.get("FREETOKEN_SKIP_BANK_PIN", "").strip().lower() in ("1", "true", "yes", "on"):
137 return
138 from freetoken.kernel.pinned import host_register
139
140 try:
141 host_register(self.addr, len(self._buf))
142 except RuntimeError as exc:
143 raise PinFailed(f"cudaHostRegister failed for {len(self._buf) / 2**30:.1f} GiB") from exc
144 self._pinned = True
145
146 def release(self) -> None:
147 """Drop the resident pages; the address space stays valid, the contents become undefined.
148
149 For buffers that are done being read (the converter). No-op for born-pinned banks: registered pages cannot be dropped."""
150 if self._pinned:
151 return
152 self._buf.madvise(mmap.MADV_DONTNEED)
153
154 def lock(self) -> None:
155 """mlock the (now-filled) buffer: resident without CUDA pin quota, but no device address -- only the CPU executor can serve a locked layer.
156
157 Lock after fill, or the lazy mmap faults+zero-fills every page. A failed lock (RLIMIT_MEMLOCK) warns once and leaves the bank PAGEABLE, which every consumer treats the same."""
158 if self._locked or self._pinned: # cudaHostRegister already page-locks
159 return
160 global _os_lock_failed
161 if _os_lock_failed:
162 return # the quota is exhausted for good; skip the syscall spam
163 try:
164 _os_lock(self.addr, len(self._buf))
165 except (OSError, ImportError) as exc:
166 _os_lock_failed = True
167 logger.warning(f"bank lock failed; leaving this and later banks pageable: {exc}")
168 return
169 self._locked = True
170
171
172_os_locked_total = 0 # bytes locked so far; the OS lock ceiling is a per-process quota
173_os_lock_failed = False # sticky: once over quota, later (bigger-total) locks fail too
174
175
176def _os_lock(addr: int, nbytes: int) -> None:
177 global _os_locked_total
178 import resource
179
180 # grow the soft RLIMIT_MEMLOCK (defaults to a few MiB); the hard limit needs privilege, past it mlock fails below
181 want = _os_locked_total + nbytes + (256 << 20)
182 soft, hard = resource.getrlimit(resource.RLIMIT_MEMLOCK)
183 if soft != resource.RLIM_INFINITY and soft < want:
184 new_soft = want if hard == resource.RLIM_INFINITY else min(want, hard)
185 if new_soft > soft:
186 try:
187 resource.setrlimit(resource.RLIMIT_MEMLOCK, (new_soft, hard))
188 except (OSError, ValueError):
189 pass # keep the old limit; mlock below reports the real ceiling
190 libc = ctypes.CDLL(None, use_errno=True)
191 if libc.mlock(ctypes.c_void_p(addr), ctypes.c_size_t(nbytes)):
192 err = ctypes.get_errno()
193 raise OSError(
194 err,
195 f"mlock({nbytes / 2**30:.1f} GiB): {os.strerror(err)} "
196 f"(RLIMIT_MEMLOCK / `ulimit -l` caps OS-locked bytes; raise it or "
197 f"shrink --moe-cpu-layers)",
198 )
199 _os_locked_total += nbytes

207class _ResidencyPlan:
208 """Per-layer ``HostResidency`` labels, ambiently visible to the bank settle points.
209
210 Installed by ``load_expert_banks`` around the provider dispatch so every loader honors --moe-cpu-layers without a new parameter in each signature. ``applied`` flips once a settle point consults the plan."""
211
212 __slots__ = ("labels", "applied", "has_unpinned", "actual")
213
214 def __init__(self, labels: list[str]):
215 self.labels = list(labels)
216 self.applied = False
217 self.has_unpinned = any(r != HostResidency.PINNED.value for r in labels)
218 self.actual: dict[int, str] = {}
219
220 def residency_for(self, layer_id: int) -> str:
221 self.applied = True
222 return self.labels[layer_id]
223
224 def record(self, layer_id: int, achieved: str) -> None:
225 """One pageable bank downgrades the whole layer (a failed lock settles PAGEABLE)."""
226 if self.actual.get(layer_id) != HostResidency.PAGEABLE.value:
227 self.actual[layer_id] = achieved
228
229
230_requested_residency: _ResidencyPlan | None = None
231
232
233@contextlib.contextmanager
234def requested_residency(labels: list[str] | None):
235 """Install the ambient per-layer residency plan for the enclosed bank load (``None`` = no plan, everything pins)."""
236 global _requested_residency
237 if labels is None:
238 yield None
239 return
240 plan = _ResidencyPlan(labels)
241 prev, _requested_residency = _requested_residency, plan
242 try:
243 yield plan
244 finally:
245 _requested_residency = prev

248def _settle(bank: HostBank, residency: str) -> None:
249 """Route a filled bank to its residency class (PAGEABLE = leave the plain mmap)."""
250 if residency == HostResidency.PINNED.value:
251 bank.pin()
252 elif residency == HostResidency.LOCKED.value:
253 bank.lock()

256def pin_banks(banks: dict[str, HostBank | list[HostBank] | dict[int, HostBank]]) -> None:
257 """Settle every bank after it has been filled -- pin-after-fill by default.
258 List-valued entries (indexed by layer id) and dict-valued ones (layer id -> bank) are per-layer and honor the ambient :func:`requested_residency` plan; scalar banks always pin."""
259 plan = _requested_residency
260 for bank in banks.values():
261 if isinstance(bank, (list, dict)):
262 for layer_id, layer_bank in bank.items() if isinstance(bank, dict) else enumerate(bank):
263 residency = (
264 HostResidency.PINNED.value if plan is None
265 else plan.residency_for(layer_id)
266 )
267 _settle(layer_bank, residency)
268 if plan is not None and residency == HostResidency.LOCKED.value:
269 plan.record(layer_id, layer_bank.residency.value)
270 else:
271 bank.pin()

274class PinPipeline:
275 """Settle (pin or lock) filled banks while other banks are still being read.
276
277 cudaHostRegister is driver-serialized, so one background thread drains a queue and submitters never block: load time ~= max(read, settle).
278 LOCKED banks mlock on the same thread (the quota bookkeeping in ``_os_lock`` is not thread-safe).
279 A clean context-manager exit drains the queue and re-raises the first settle failure.
280 """
281
282 def __init__(self) -> None:
283 self._q: queue.SimpleQueue = queue.SimpleQueue()
284 self._exc: BaseException | None = None
285 # the current device is thread-local: a fresh thread sits on device 0 and cudaHostRegister would build its context there -- carry the creator's (bound) device into the worker
286 self._device = torch.cuda.current_device() if torch.cuda.is_available() else None
287 self._thread = threading.Thread(target=self._run, daemon=True)
288 self._thread.start()
289
290 def _run(self) -> None:
291 if self._device is not None:
292 torch.cuda.set_device(self._device)
293 while True:
294 item = self._q.get()
295 if item is None:
296 return
297 if self._exc is not None:
298 continue # drain without settling after a failure
299 bank, residency, plan, layer_id = item
300 try:
301 _settle(bank, residency)
302 if plan is not None and residency == HostResidency.LOCKED.value:
303 plan.record(layer_id, bank.residency.value)
304 except BaseException as exc: # surfaced by wait()/__exit__
305 self._exc = exc
306
307 def submit(self, bank: HostBank, residency: str = HostResidency.PINNED.value,
308 plan=None, layer_id: int | None = None) -> None:
309 self._q.put((bank, residency, plan, layer_id))
310
311 def __call__(self, layer_id: int, banks: dict[str, HostBank]) -> None:
312 """Layer-completion sink: queue every bank of the completed layer at its ambient :func:`requested_residency` label."""
313 plan = _requested_residency
314 residency = (
315 HostResidency.PINNED.value if plan is None else plan.residency_for(layer_id)
316 )
317 for bank in banks.values():
318 self.submit(bank, residency, plan, layer_id)
319
320 def _join(self) -> None:
321 self._q.put(None)
322 self._thread.join()
323
324 def wait(self) -> None:
325 self._join()
326 if self._exc is not None:
327 raise self._exc
328
329 def __enter__(self) -> "PinPipeline":
330 return self
331
332 def __exit__(self, exc_type, exc, tb) -> None:
333 if exc_type is not None:
334 self._join() # no thread leak; the in-flight exception wins
335 return
336 self.wait()

339class LayerCompletionTracker:
340 """Fire a sink once per layer, when all of that layer's writes have landed.
341
342 ``note(layer_id)`` is called after each write; at ``expected_per_layer``
343 notes the layer's banks are handed to ``on_layer(layer_id, {name: bank})``
344 exactly once. Thread-safe (shard-driven loaders write layers from many
345 threads in arbitrary order).
346 """
347
348 def __init__(
349 self,
350 expected_per_layer: int,
351 banks: dict[str, list],
352 on_layer,
353 ) -> None:
354 assert expected_per_layer > 0
355 self._expected = expected_per_layer
356 self._banks = banks
357 self._on_layer = on_layer
358 self._counts: dict[int, int] = {}
359 self._lock = threading.Lock()
360
361 def note(self, layer_id: int) -> None:
362 with self._lock:
363 n = self._counts.get(layer_id, 0) + 1
364 self._counts[layer_id] = n
365 fire = n == self._expected
366 if fire:
367 self._on_layer(layer_id, {name: per[layer_id] for name, per in self._banks.items()})

370def read_file_into(buf: memoryview | mmap.mmap, path: str, *, workers: int = 8,
371 chunk: int = _DEFAULT_CHUNK, drop_cache: bool = True) -> int:
372 """Chunked multi-threaded O_DIRECT read of the whole file ``path`` into ``buf``
373 (page-aligned). Returns the file size. The buffer must be >= the rounded-up file size."""
374 size = os.path.getsize(path)
375 if drop_cache:
376 try:
377 fd0 = os.open(path, os.O_RDONLY)
378 os.posix_fadvise(fd0, 0, 0, os.POSIX_FADV_DONTNEED)
379 os.close(fd0)
380 except OSError:
381 pass
382 mv = buf if isinstance(buf, memoryview) else memoryview(buf)
383 fd = os.open(path, os.O_RDONLY | os.O_DIRECT)
384 offs = list(range(0, size, chunk))
385
386 def rd(o):
387 want = min(chunk, len(mv) - o)
388 want = min(want, ((size - o + _BLK - 1) // _BLK) * _BLK)
389 os.preadv(fd, [mv[o:o + want]], o)
390
391 try:
392 if len(offs) <= 1:
393 for o in offs:
394 rd(o)
395 else:
396 with ThreadPoolExecutor(workers) as ex:
397 list(ex.map(rd, offs))
398 finally:
399 os.close(fd)
400 return size

403def _preadv_all(fd: int, dst: memoryview, offset: int, need: int) -> None:
404 """preadv into ``dst`` until ``need`` bytes have landed; O_DIRECT may return a short count."""
405 done = 0
406 while done < need:
407 if done % _BLK: # a continuation read has to stay block-aligned on both sides
408 raise OSError(f"unaligned short O_DIRECT read: {done} of {need} bytes at {offset}")
409 got = os.preadv(fd, [dst[done:]], offset + done)
410 if got <= 0:
411 raise OSError(f"short O_DIRECT read: {done} of {need} bytes at {offset}")
412 done += got

415def read_range_into(buf: memoryview | mmap.mmap, path: str, *, file_offset: int, nbytes: int,
416 dest_offset: int = 0, workers: int = 8, chunk: int = _DEFAULT_CHUNK,
417 drop_cache: bool = True) -> int:
418 """Chunked multi-threaded O_DIRECT read of ``path[file_offset : file_offset + nbytes]`` into ``buf`` at ``dest_offset``. Returns ``nbytes``.
419
420 Byte-range counterpart of :func:`read_file_into`, for one tensor inside a shard. O_DIRECT needs the file offset AND the destination address block-aligned at the same time, which only holds when the two share their offset mod 4096 -- a safetensors data offset practically never lines up with the tensor's slot in the bank. Chunks that do line up DMA straight into ``buf``; the rest DMA into a page-aligned bounce (source window rounded out to whole blocks) and are copied into place, which also covers the unaligned head and tail.
421 """
422 mv = (buf if isinstance(buf, memoryview) else memoryview(buf)).cast("B")
423 if dest_offset + nbytes > len(mv):
424 raise ValueError(f"destination holds {len(mv)} bytes, need {dest_offset + nbytes}")
425 base = ctypes.addressof(ctypes.c_char.from_buffer(mv))
426 if drop_cache:
427 try:
428 fd0 = os.open(path, os.O_RDONLY)
429 os.posix_fadvise(fd0, file_offset, nbytes, os.POSIX_FADV_DONTNEED)
430 os.close(fd0)
431 except OSError:
432 pass
433 fd = os.open(path, os.O_RDONLY | os.O_DIRECT)
434 scratch = threading.local()
435
436 def rd(i: int) -> None:
437 n = min(chunk, nbytes - i)
438 src, dst = file_offset + i, dest_offset + i
439 if src % _BLK == 0 and (base + dst) % _BLK == 0 and n % _BLK == 0:
440 _preadv_all(fd, mv[dst:dst + n], src, n)
441 return
442 head = src % _BLK
443 span = ((head + n + _BLK - 1) // _BLK) * _BLK
444 bounce = getattr(scratch, "buf", None)
445 if bounce is None or len(bounce) < span:
446 bounce = scratch.buf = mmap.mmap(-1, span) # anonymous mmaps are page-aligned
447 bmv = memoryview(bounce)
448 _preadv_all(fd, bmv[:span], src - head, head + n)
449 mv[dst:dst + n] = bmv[head:head + n]
450
451 try:
452 offs = list(range(0, nbytes, chunk))
453 if len(offs) <= 1:
454 for o in offs:
455 rd(o)
456 else:
457 with ThreadPoolExecutor(workers) as ex:
458 list(ex.map(rd, offs))
459 finally:
460 os.close(fd)
461 return nbytes
