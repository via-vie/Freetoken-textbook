1"""Torch-free reader for the ``ft bench bw`` hardware profile (``benchbw/<gpu-uuid>.json``).
2
3The profile holds one row per MoE layer kind (``layer_kind_key``): the CPU MoE and PCIe gather
4bandwidths benched for that kind on this GPU and host. The engine reads it to set each hybrid
5layer's fetch split, without importing the (torch-heavy) benchmark. ``benchbw.py`` writes it.
6"""
7
8from __future__ import annotations
9
10import json
11import os
12
13from freetoken.utils import init_logger
14
15logger = init_logger(__name__)
16
17# Bump when the row key or the measurements change: rows of another version are not comparable.
18PROFILE_VERSION = 6

21def _cache_dir() -> str:
22 cache = os.environ.get("XDG_CACHE_HOME") or os.path.expanduser("~/.cache")
23 return os.path.join(cache, "freetoken")
24

26def default_profile_path(gpu_uuid: str | None = None) -> str:
27 """``$XDG_CACHE_HOME/freetoken/benchbw/<gpu-uuid>.json``, or the legacy ``benchbw.json`` without a uuid.
28
29 One file per GPU: bandwidth differs between slots.
30 """
31 if gpu_uuid:
32 return os.path.join(_cache_dir(), "benchbw", f"{gpu_uuid}.json")
33 return os.path.join(_cache_dir(), "benchbw.json")
34
35
36def latest_profile_path() -> str | None:
37 """Newest ``benchbw/*.json``, else the legacy ``benchbw.json``, else None."""
38 per_gpu = os.path.join(_cache_dir(), "benchbw")
39 newest: tuple[float, str] | None = None
40 try:
41 for name in os.listdir(per_gpu):
42 if not name.endswith(".json"):
43 continue
44 path = os.path.join(per_gpu, name)
45 try:
46 mtime = os.path.getmtime(path)
47 except OSError:
48 continue
49 if newest is None or mtime > newest[0]:
50 newest = (mtime, path)
51 except OSError:
52 pass
53 if newest is not None:
54 return newest[1]
55 legacy = default_profile_path()
56 return legacy if os.path.isfile(legacy) else None
57
58
59def _load(path: str) -> dict | None:
60 try:
61 with open(path) as f:
62 return json.load(f)
63 except (OSError, ValueError):
64 return None
65
66
67def _usable_profile(
68 gpu_name: str | None, path: str | None, gpu_uuid: str | None = None
69) -> dict | None:
70 """The cached profile, or ``None`` when there is no file / it was benched on another GPU
71 (bandwidths are hardware-specific, so a mismatch is ignored rather than trusted).
72
73 Lookup: explicit ``path`` (else ``FREETOKEN_BENCHBW_PATH``) -> ``benchbw/<gpu_uuid>.json`` -> legacy ``benchbw.json``.
74 """
75 explicit = path or os.environ.get("FREETOKEN_BENCHBW_PATH")
76 if explicit:
77 candidates = [explicit]
78 else:
79 candidates = [default_profile_path(gpu_uuid)] if gpu_uuid else []
80 candidates.append(default_profile_path())
81 prof = None
82 for src in candidates:
83 prof = _load(src)
84 if isinstance(prof, dict):
85 break
86 if os.path.exists(src):
87 # unreadable profile for this card: stay on the safe default, do not borrow the legacy file
88 return None
89 if not isinstance(prof, dict):
90 return None
91 if prof.get("version") != PROFILE_VERSION:
92 logger.warning(
93 f"benchbw profile {src} has version {prof.get('version')}, this engine reads "
94 f"{PROFILE_VERSION}; ignoring it, rerun `ft bench bw`"
95 )
96 return None
97 prof_gpu = (prof.get("gpu") or {}).get("name")
98 if gpu_name and prof_gpu and prof_gpu != gpu_name:
99 logger.warning(
100 f"benchbw profile {src} was measured on {prof_gpu!r}, not this GPU "
101 f"({gpu_name!r}); ignoring it"
102 )
103 return None
104 return prof

107def layer_kind_key(fmt: str, options: dict[str, int], hidden: int, inter: int, top_k: int, cpu_threads: int) -> str:
108 """Row key of one MoE layer kind, e.g. ``gguf[down_type=8,gate_up_type=8]:2816x704:top8:t12``.
109
110 ``cpu_threads`` is the CPU executor's worker count: the per-step pool overhead makes the CPU bandwidth depend on it and on ``top_k``."""
111 opts = ",".join(f"{k}={v}" for k, v in sorted(options.items()))
112 fmt = f"{fmt}[{opts}]" if opts else fmt
113 return f"{fmt}:{hidden}x{inter}:top{top_k}:t{cpu_threads}"
114

116def fetch_fraction(row: dict | None) -> float | None:
117 """The hybrid fetch split of a row: of a decode step's expert misses, the fraction to fetch over PCIe so the fetch and the CPU compute finish together.
118
119 From the overlapped pair (both sides measured while running concurrently): pcie_ov / (pcie_ov + cpu_ov); without it, pcie / cpu. Clamped to 1."""
120 if not isinstance(row, dict):
121 return None
122 cpu_ov, pcie_ov = row.get("cpu_moe_overlap_gbs"), row.get("pcie_gather_overlap_gbs")
123 if cpu_ov and pcie_ov:
124 return min(1.0, pcie_ov / (pcie_ov + cpu_ov))
125 cpu, pcie = row.get("cpu_moe_gbs"), row.get("pcie_gather_gbs")
126 if cpu and pcie:
127 return min(1.0, pcie / cpu)
128 return None

131def load_layer_kinds(gpu_name: str | None = None, path: str | None = None, gpu_uuid: str | None = None) -> dict | None:
132 """The rows of the usable profile for this GPU (key -> row), or None without one."""
133 prof = _usable_profile(gpu_name, path, gpu_uuid)
134 if prof is None:
135 return None
136 rows = prof.get("layer_kinds")
137 return rows if isinstance(rows, dict) else {}
