1"""Expert banks for the offload MoE cache: load, pack and pin the routed experts.
2
3The expert kernel (``QuantMethod.kernel``) owns the bank layout and the pack step; the
4checkpoint side delivers pieces (``moe.expert_pieces``) and this module fills the pinned host
5banks from them (``build_expert_banks``). GGUF experts keep their packed ggml rows
6(``models.gguf.experts``).
7"""

30@dataclass(frozen=True)
31class ExpertBanks:
32 """Loaded expert banks, normalized for ``OffloadMoeCache`` wiring."""
33
34 quant_format: str # _BANK_SCHEMAS key
35 # Pinned host banks, keyed by the format's schema: one [num_experts, ...]
36 # tensor per layer (independent allocations -> per-layer host attributes).
37 sources: dict[str, list[torch.Tensor]]
38 # marlin/b12x per-expert global scales ([L*E]); None for formats without them
39 gate_up_alpha: torch.Tensor | None = field(default=None)
40 down_alpha: torch.Tensor | None = field(default=None)
41 # per-layer HostResidency values actually applied by the loader; None -> all pinned (also the degrade signal when a request was not honored)
42 layer_residency: list[str] | None = field(default=None)
43 # True iff the ``layer_sink`` passed to the loader was actually engaged (each layer
44 # streamed straight to its sink instead of staying materialized here) -- set by
45 # convert.py's per-format streaming gate; ``sources`` may hold released tensors.
46 streamed: bool = False
47 # the expert (kind, kernel) the banks were packed for; None for the legacy providers
48 kind: QuantKind | None = None
49 kernel: str | None = None
50 layout: dict | None = None
51 # per-layer quant tag (e.g. the gate_up and down GGUF types), for layers whose banks match in dtype and row shape but not in quant; None = one quant
52 layer_tags: list | None = None
53 # per bank, per layer: bytes past an expert matrix's last row that the kernel reads and that must read as zeros (ggml MMQ); None = no such reads
54 row_tail_bytes: dict[str, list[int]] | None = None
55 # per-layer quant_format of experts in several formats (``quant_format``/``kind``/``kernel`` then name the most common one); None = every layer is ``quant_format``
56 layer_formats: list[str] | None = None

59def _dummy_fill(role: str, tensor: torch.Tensor) -> None:
60 """Random but finite bank contents for --use-dummy-weight."""
61 if role.endswith("_scale"):
62 if tensor.dtype is torch.uint8:
63 tensor.fill_(127) # e8m0 exponent code for 1.0
64 else:
65 tensor.fill_(1.0)
66 elif role.endswith("_global"):
67 tensor.fill_(0.01)
68 elif tensor.dtype in (torch.uint8, torch.int32):
69 tensor.view(torch.uint8).random_(0, 256)
70 elif tensor.dtype is torch.float8_e4m3fn:
71 tensor.view(torch.uint8).random_(0, 16) # small codes, no NaN / inf
72 else:
73 tensor.normal_()

76def build_expert_banks(
77 method,
78 num_layers: int,
79 pieces,
80 *,
81 device: torch.device,
82 layer_sink=None,
83 dummy: bool = False,
84 resident: bool = False,
85 layers: list[int] | None = None,
86) -> ExpertBanks:
87 """Fill host banks in the kernel's layout from a stream of expert pieces.
88
89 ``pieces`` yields ``(layer_id, e0, e1, {role: tensor[e1 - e0, ...]})`` in any order;
90 each batch is packed in place into rows ``e0:e1`` of that layer's banks. A layer is
91 complete once its ``num_experts`` rows have arrived: with ``layer_sink=None`` its banks
92 are pinned in the background, otherwise the sink receives them (converter). ``dummy``
93 skips the pieces and fills the banks with finite random contents.
94 ``resident`` packs into banks on ``device`` instead of host banks (the resident 'fused' experts).
95 ``layers`` fills only those bank layers (the ones ``method`` serves, for experts of several
96 formats): pieces of other layers are skipped and the other layers' sources are None.
97 """
98 from freetoken.moe.host_banks import LayerCompletionTracker, PinPipeline, pin_banks
99 from freetoken.moe.legacy_format import legacy_format_for
100
101 kernel = method.kernel
102 layout = method.layout()
103 E = method.cfg.num_experts
104 ids = list(range(num_layers)) if layers is None else sorted(layers)
105 filled = set(ids)
106 specs = {role: ((E, *spec.shape), spec.dtype) for role, spec in layout.items() if not spec.resident}
107 if resident:
108 assert layer_sink is None
109 hb = None
110 banks = {role: {l: torch.empty(shape, dtype=dtype, device=device) for l in ids} for role, (shape, dtype) in specs.items()}
111 else:
112 hb = {role: {l: HostBank(shape, dtype) for l in ids} for role, (shape, dtype) in specs.items()}
113 banks = {role: {l: b.tensor for l, b in per_layer.items()} for role, per_layer in hb.items()}
114 alphas = {
115 role: torch.empty(num_layers * E, dtype=spec.dtype, device=device)
116 for role, spec in layout.items() if spec.resident
117 }
118
119 def _sources() -> dict[str, list]:
120 return {role: [per_layer.get(l) for l in range(num_layers)] for role, per_layer in banks.items()}
121
122 if dummy:
123 for role, per_layer in banks.items():
124 for tensor in per_layer.values():
125 _dummy_fill(role, tensor)
126 for alpha in alphas.values():
127 alpha.fill_(1.0)
128 if hb is not None and torch.cuda.is_available():
129 pin_banks(hb)
130 return ExpertBanks(
131 legacy_format_for(method.kind, kernel.name), _sources(),
132 gate_up_alpha=alphas.get("gate_up_alpha"), down_alpha=alphas.get("down_alpha"),
133 kind=method.kind, kernel=kernel.name, layout=layout,
134 )
135
136 def _fill(sink) -> None:
137 tracker = LayerCompletionTracker(E, hb, sink) if sink is not None else None
138 # a reader that skips a layer or mislabels a piece must fail here, not serve uninitialized rows
139 written = torch.zeros(num_layers, E, dtype=torch.int32)
140 for layer_id, e0, e1, piece in pieces:
141 if not (0 <= layer_id < num_layers and 0 <= e0 < e1 <= E):
142 raise ValueError(f"expert piece out of range: layer {layer_id}, experts {e0}:{e1} of {num_layers} x {E}")
143 if layer_id not in filled:
144 continue
145 # refuse before writing: a duplicate row would also complete the layer early and hand the sink a half-filled bank
146 if written[layer_id, e0:e1].any():
147 raise ValueError(f"expert rows written more than once: layer {layer_id}, experts {e0}:{e1}")
148 written[layer_id, e0:e1] = 1
149 out = {role: banks[role][layer_id][e0:e1] for role in specs}
150 got = method.pack(piece, out)
151 for role, values in got.items():
152 alphas[role][layer_id * E + e0 : layer_id * E + e1] = values.to(alphas[role].dtype)
153 if tracker is not None:
154 for _ in range(e1 - e0):
155 tracker.note(layer_id)
156 missing = (written[ids] == 0).nonzero().tolist()
157 if missing:
158 missing = [[ids[l], e] for l, e in missing]
159 raise ValueError(f"expert banks were not filled: {len(missing)} (layer, expert) rows missing (first {missing[:4]})")
160
161 if layer_sink is not None:
162 _fill(layer_sink)
163 elif torch.cuda.is_available() and not resident:
164 with PinPipeline() as pins:
165 _fill(pins)
166 else:
167 _fill(None)
168
169 return ExpertBanks(
170 legacy_format_for(method.kind, kernel.name), _sources(),
171 gate_up_alpha=alphas.get("gate_up_alpha"), down_alpha=alphas.get("down_alpha"),
172 streamed=layer_sink is not None, kind=method.kind, kernel=kernel.name, layout=layout,
173 )

176def method_groups(layer_methods) -> list[tuple]:
177 """``[(method, [bank layers])]`` of the layers' expert methods, one entry per (kind, kernel), most layers first."""
178 groups: dict = {}
179 for layer, method in enumerate(layer_methods):
180 groups.setdefault((method.kind, method.kernel.name), (method, []))[1].append(layer)
181 return sorted(groups.values(), key=lambda g: (-len(g[1]), g[1][0]))

188def merge_expert_banks(parts: list[ExpertBanks], layer_methods, num_experts: int) -> ExpertBanks:
189 """One ``ExpertBanks`` over the per-format ``parts`` (each filled only on its layers): the banks are the
190 union of the roles, a layer without a role gets a zero-byte placeholder (see ``OffloadMoeCache.layer_layouts``)."""
191 from freetoken.moe.legacy_format import legacy_format_for
192
193 num_layers = len(layer_methods)
194 sources: dict[str, list] = {}
195 for part in parts:
196 if part.gate_up_alpha is not None or part.down_alpha is not None:
197 raise ValueError(f"experts of several formats cannot use the {part.kernel} kernel, which keeps GPU-resident alphas")
198 for role, per_layer in part.sources.items():
199 merged = sources.setdefault(role, [None] * num_layers)
200 for layer, tensor in enumerate(per_layer):
201 if tensor is not None:
202 assert merged[layer] is None, (role, layer)
203 merged[layer] = tensor
204 for per_layer in sources.values():
205 for layer, tensor in enumerate(per_layer):
206 if tensor is None:
207 per_layer[layer] = placeholder_bank(num_experts)
208 formats = [legacy_format_for(m.kind, m.kernel.name) for m in layer_methods]
209 primary = parts[0]
210 return ExpertBanks(
211 primary.quant_format, sources, streamed=all(p.streamed for p in parts),
212 kind=primary.kind, kernel=primary.kernel, layer_tags=formats, layer_formats=formats,
213 )

216def _mixed_expert_banks(model_path, model_config, layer_methods, device, dummy, parallel, workers, chunk, layer_sink=None) -> ExpertBanks:
217 from freetoken.moe.expert_pieces import iter_expert_pieces
218
219 num_layers = model_config.num_moe_layers
220 assert len(layer_methods) == num_layers, (len(layer_methods), num_layers)
221 groups = method_groups(layer_methods)
222 # resolve every format's reader before any bank exists, so a missing parallel reader falls back to serial with nothing allocated
223 readers = [
224 None if dummy else iter_expert_pieces(
225 model_path, model_config, method.kind, layers=layers, parallel=parallel, workers=workers, chunk=chunk
226 )
227 for method, layers in groups
228 ]
229 parts = []
230 for (method, layers), pieces in zip(groups, readers):
231 parts.append(build_expert_banks(
232 method, num_layers, pieces, device=device, layer_sink=layer_sink, dummy=dummy, layers=layers,
233 ))
234 logger.info_rank0(f"expert banks: {method.kind} / {method.kernel.name} on {len(layers)} of {num_layers} MoE layers")
235 return merge_expert_banks(parts, layer_methods, model_config.num_experts)

253def _host_ram_fits_parallel(model_path: str) -> bool:
254 """Best-effort: can free host RAM hold the expert banks plus the parallel reader's one
255 extra (non-reclaimable) whole-shard buffer? Unknown (non-local path / no /proc) -> True,
256 i.e. keep the fast path. Banks ~= checkpoint size (experts dominate); transient ~= the
257 largest shard. Uses MemAvailable (counts reclaimable cache) -- the OOM-relevant figure."""
258 avail = None
259 try:
260 with open("/proc/meminfo") as f:
261 for line in f:
262 if line.startswith("MemAvailable:"):
263 avail = int(line.split()[1]) * 1024
264 break
265 except OSError:
266 pass
267 if avail is None:
268 return True
269 try: # resolve a hub id to its local cache dir (no-op for a local path) so glob sees the shards
270 from freetoken.utils.hf import download_hf_weight
271
272 model_path = download_hf_weight(model_path)
273 except Exception:
274 return True
275 sizes = [os.path.getsize(p) for p in glob.glob(os.path.join(model_path, "*.safetensors"))]
276 if not sizes:
277 return True
278 return avail > sum(sizes) + max(sizes)

281def ftw_bank_bytes(model_path: str) -> int | None:
282 """Total expert-bank bytes of an FTW checkpoint, from its metadata (no bank IO).
283 ``None`` when the checkpoint is not FTW -- callers that size things pre-load (auto split residency) then leave the load unchanged."""
284 import json
285
286 meta = os.path.join(model_path, "freetoken_weight.json")
287 if not os.path.isfile(meta):
288 return None
289 with open(meta, encoding="utf-8") as f:
290 tensors = json.load(f).get("tensors", [])
291 return sum(t["nbytes"] for t in tensors if t.get("kind") == "experts_bank")

301def bank_bytes_estimate(model_config, method=None, layer_methods=None) -> int | None:
302 """Estimated total expert-bank bytes of a raw checkpoint before loading it.
303
304 With a bound expert ``method`` the kernel's layout gives the exact host bytes (``layer_methods``,
305 one per MoE layer, for experts of several formats); otherwise the format-tag table sizes the
306 format. ``None`` for unknown formats or missing dims (callers then skip the pre-load sizing)."""
307 layers = getattr(model_config, "num_moe_layers", None)
308 if layer_methods is not None:
309 return sum(_method_expert_bytes(m) for m in layer_methods)
310 if method is not None and layers:
311 return layers * _method_expert_bytes(method)
312 expert_quant = getattr(model_config, "expert_quant", "none")
313 fmt = expert_quant if expert_quant != "none" else (
314 getattr(model_config, "moe_weight_format", None) or "bf16"
315 )
316 per_expert = _BANK_BYTES_PER_EXPERT.get(fmt)
317 layers = getattr(model_config, "num_moe_layers", None)
318 experts = getattr(model_config, "num_experts", None)
319 hidden = getattr(model_config, "hidden_size", None)
320 inter = getattr(model_config, "moe_intermediate_size", None)
321 if per_expert is None or not all((layers, experts, hidden, inter)):
322 return None
323 return layers * experts * per_expert(hidden, inter)

326def load_expert_banks(
327 model_path: str,
328 model_config,
329 *,
330 method=None,
331 device: torch.device,
332 dummy: bool = False,
333 parallel: bool | None = None,
334 workers: int = 8,
335 chunk: int = _PARALLEL_CHUNK,
336 layer_sink=None,
337 layer_residency: list[str] | None = None,
338 resident: bool = False,
339 layer_methods: list | None = None,
340) -> ExpertBanks:
341 """Load (or fabricate, with ``dummy=True``) the expert banks. Two paths, both returning
342 the same normalized ``ExpertBanks`` and both pinning after fill:
343
344 * **Fast path (FTW)**: if ``model_path`` is a converted FTW checkpoint, read its
345 repacked banks directly (contiguous chunked O_DIRECT). No auto-conversion.
346 * **Slow path** (the original checkpoint): auto-pick **parallel** (the common parallel chunked
347 O_DIRECT reader) when experts are stored as many small tensors -- the serial read is
348 slow there -- else the **serial baseline** (packed experts: serial already saturates,
349 parallel only adds read amplification). parallel unavailable for a quant falls back to serial.
350
351 ``parallel`` overrides the slow-path auto-pick: ``None`` = auto (production), ``True`` /
352 ``False`` = force parallel / serial (used by the loader benchmark and the converter).
353
354 ``layer_sink`` (the converter only): forwarded to whichever provider is picked; a
355 provider only engages it (and reports ``ExpertBanks.streamed=True``) for its own
356 streamable formats, so callers must check ``streamed`` rather than assume it fired.
357
358 ``method`` (the bound expert quant method of the model's offload layers) selects
359 the generic path: the family's pieces packed by the method's kernel. Without it only an
360 FTW checkpoint loads.
361
362 ``layer_residency``: per-layer ``HostResidency`` labels applied at settle time -- explicitly on the FTW fast path, ambiently (``requested_residency``) in the slow-path providers.
363 Applied labels are echoed on ``ExpertBanks.layer_residency``; a loader that settles some other way leaves it ``None`` (CPU-layer decode still works on pinned banks, it just saves no pin quota).
364
365 ``resident`` (with ``method``): the banks land on ``device`` for the resident 'fused' experts; no host bank outlives the load.
366
367 ``layer_methods`` (one bound method per MoE layer) serves experts of several formats: each
368 format fills its own layers and the banks merge over the union of roles (``merge_expert_banks``).
369 """
370 from freetoken.checkpoint.ftw import is_ftw_checkpoint, load_ftw_banks, load_ftw_banks_to_device
371
372 mixed = is_mixed(layer_methods)
373 if mixed and (resident or layer_residency is not None):
374 raise ValueError("experts of several formats are served from the offload slot cache only (no resident experts, no --moe-cpu-layers)")
375 if resident:
376 assert method is not None and layer_sink is None and layer_residency is None
377 if model_path and is_ftw_checkpoint(model_path) and not dummy:
378 if resident:
379 banks = load_ftw_banks_to_device(model_path, num_layers=model_config.num_moe_layers, device=device, workers=workers, chunk=chunk)
380 else:
381 banks = load_ftw_banks(
382 model_path, num_layers=model_config.num_moe_layers, workers=workers, chunk=chunk,
383 layer_residency=layer_residency,
384 )
385 if banks is not None:
386 logger.info_rank0(f"expert banks: FTW fast path (FTW checkpoint {model_path})")
387 return banks
388
389 if parallel and not _PARALLEL_READER_SUPPORTED:
390 logger.warning_rank0(
391 "expert banks: parallel O_DIRECT reader unsupported on this platform "
392 "(no os.O_DIRECT/preadv) -> serial build"
393 )
394 parallel = False
395
396 auto = parallel is None
397 if auto:
398 from freetoken.models.weight import experts_scattered
399
400 parallel = _PARALLEL_READER_SUPPORTED and not dummy and experts_scattered(model_path)
401 # Low-RAM fallback: the parallel reader holds whole-shard ANONYMOUS buffers
402 # (non-reclaimable) on top of the ~bank-sized resident set, so on a memory-tight box
403 # it OOMs where the serial path (reclaimable file mmap) survives. Drop to serial when
404 # free RAM can't cover the banks + one shard's transient. (--expert-load serial/parallel
405 # bypass this by forcing ``parallel`` explicitly.)
406 if parallel and not _host_ram_fits_parallel(model_path):
407 logger.warning_rank0(
408 "expert banks: low free RAM -> serial build (avoids parallel-reader OOM; "
409 "override with --expert-load parallel)"
410 )
411 parallel = False
412 logger.info_rank0(f"expert banks: slow path ({'parallel' if parallel else 'serial'} build)")
413 # parallel's reader resolves hub ids + handles single-file/no-index checkpoints, so it won't
414 # OSError on those (which would leak the banks it pre-allocated, since host banks live for
415 # the process). Only NotImplementedError (quant has no parallel reader; raised before any
416 # allocation) falls back to serial.
417 from freetoken.moe.host_banks import requested_residency
418
419 def _build(par: bool) -> ExpertBanks:
420 if mixed:
421 return _mixed_expert_banks(model_path, model_config, layer_methods, device, dummy, par, workers, chunk, layer_sink)
422 if method is not None and method.kind is QuantKind.GGUF:
423 from freetoken.models.gguf.experts import load_gguf_expert_banks
424
425 # each layer keeps its own ggml types, so the banks come from the file, not from one method's layout
426 return load_gguf_expert_banks(model_path, model_config.num_moe_layers, first_layer=model_config.first_k_dense_replace, dummy=dummy, device=device if resident else None)
427 if method is None:
428 raise ValueError("the experts of a raw checkpoint are packed by the offload layers' MoE quant method; pass method=")
429 return _method_expert_banks(model_path, model_config, method, device, dummy, par, workers, chunk, layer_sink, resident)
430
431 with requested_residency(layer_residency) as residency_plan:
432 try:
433 banks = _build(parallel)
434 except NotImplementedError as exc:
435 if not parallel:
436 raise
437 logger.warning_rank0(f"parallel reader unavailable ({exc}); falling back to serial build")
438 banks = _build(False)
439 return _echo_residency(banks, layer_residency, residency_plan)

442def _echo_residency(banks: ExpertBanks, requested, plan) -> ExpertBanks:
443 """Stamp an honored residency request onto the ExpertBanks; keep None (and warn) when no settle point consulted the plan."""
444 if requested is None or banks.layer_residency is not None:
445 return banks
446 if plan is not None and plan.applied:
447 import dataclasses
448
449 labels = [plan.actual.get(i, r) for i, r in enumerate(requested)]
450 downgraded = [i for i, r in enumerate(requested) if labels[i] != r]
451 if downgraded:
452 logger.warning_rank0(
453 f"--moe-cpu-layers: layers {downgraded} settled pageable instead of "
454 f"OS-locked (lock failed); they still decode on the CPU executor but "
455 f"may swap under memory pressure"
456 )
457 return dataclasses.replace(banks, layer_residency=labels)
458 from freetoken.moe.host_banks import HostResidency
459
460 if any(r != HostResidency.PINNED.value for r in requested):
461 logger.warning_rank0(
462 "--moe-cpu-layers: this checkpoint's bank loader settles banks without "
463 "per-layer residency (pre-pins everything); CPU-layer decode still works "
464 "but saves no pinned quota"
465 )
466 return banks

469def attach_resident_banks(layers, banks: ExpertBanks) -> None:
470 """Give each resident MoE layer its rows of the device ``banks``; bank layer ``i`` goes to the layer whose ``layer_id`` is ``i``."""
471 ids = sorted(layer.layer_id for layer in layers if layer.layer_id is not None)
472 num_layers = len(next(iter(banks.sources.values())))
473 if ids != list(range(num_layers)) or len(ids) != len(layers):
474 raise ValueError(f"resident MoE layer ids {ids} do not cover the {num_layers} expert bank layers")
475 for layer in layers:
476 i, e = layer.layer_id, layer.num_experts
477 layout = layer.quant_method.layout()
478 rows = {role: (e, *spec.shape) for role, spec in layout.items() if not spec.resident}
479 if set(banks.sources) != set(rows):
480 raise ValueError(f"expert banks hold {sorted(banks.sources)}, the {layer.quant_method.kernel.name} kernel reads {sorted(rows)}")
481 for role, shape in rows.items():
482 bank = banks.sources[role][i]
483 if tuple(bank.shape) != shape or bank.dtype != layout[role].dtype:
484 raise ValueError(
485 f"bank {role!r} of layer {i} is {tuple(bank.shape)} {bank.dtype} but the {layer.quant_method.kernel.name} kernel "
486 f"wants {shape} {layout[role].dtype}; the banks were packed for another kernel or model"
487 )
488 alphas = []
489 # apply() reads view.alphas as (gate_up, down)
490 for role in (role for role in ("gate_up_alpha", "down_alpha") if role in layout):
491 a = getattr(banks, role)
492 if a is None or a.numel() != num_layers * e or a.dtype != layout[role].dtype:
493 raise ValueError(f"the {layer.quant_method.kernel.name} kernel needs {num_layers * e} {layout[role].dtype} {role} values, the expert banks carry {None if a is None else (a.numel(), a.dtype)}")
494 alphas.append(a)
495 layer._expert_banks = {role: per_layer[i] for role, per_layer in banks.sources.items()}
496 layer._expert_alphas = tuple(a[i * e : (i + 1) * e] for a in alphas) or None
