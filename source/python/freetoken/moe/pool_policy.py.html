1"""Slot pool policy for the offload MoE cache: which layers share a pool, and how many slots each pool gets.
2
3Pure functions. The engine supplies the constraints and hands the result to ``OffloadMoeCache``:
4
5 groups = group_layers(scope, layouts, slot_bytes) # --moe-cache-scope, once per load
6 pools = make_pools(groups, num_experts, min_num_slots, slot_limit) # the minimum follows the decode batch
7 sizes = split_slots(total, pools, counts) # --moe-cache-split, counts from resolve_profile
8 OffloadMoeCache(pool_of_layer=pool_of_layer(pools, num_layers), pool_sizes=sizes, ...)
9"""
10

23SCOPES = ("unified", "per-layer")

26@dataclass(frozen=True)
27class PoolSpec:
28 layers: tuple[int, ...]
29 min_num_slots: int
30 max_num_slots: int
31

33def group_layers(scope: str, layouts: list, slot_bytes: list[int]) -> list[list[int]]:
34 """The layers of each pool, in placement order.
35
36 ``unified``: one pool per layout key (see ``bank_layouts``); ``per-layer``: one pool per layer.
37 Pools with wider slots come first: the prefill staging buffers sit at the start of every bank, so they then cover fewer slots.
38 """
39 if scope == "unified":
40 groups: dict = {}
41 for layer, layout in enumerate(layouts):
42 groups.setdefault(layout, []).append(layer)
43 pools = list(groups.values())
44 elif scope == "per-layer":
45 pools = [[layer] for layer in range(len(layouts))]
46 else:
47 raise ValueError(f"unknown --moe-cache-scope {scope!r}; want one of {SCOPES}")
48 return sorted(pools, key=lambda layers: (-slot_bytes[layers[0]], layers[0]))
49

51def make_pools(groups: list[list[int]], num_experts: int, min_num_slots: int, slot_limit: int | None) -> list[PoolSpec]:
52 """Each group's slot range: at least ``min_num_slots``, at most one slot per expert of its layers and the kernel's ``slot_limit``."""
53 pools = []
54 for layers in groups:
55 max_num_slots = len(layers) * num_experts
56 if slot_limit is not None:
57 max_num_slots = min(max_num_slots, slot_limit)
58 pools.append(PoolSpec(tuple(layers), min(min_num_slots, max_num_slots), max_num_slots))
59 return pools

62def pool_of_layer(pools: list[PoolSpec], num_layers: int) -> list[int]:
63 """Each layer's pool index, the form ``OffloadMoeCache`` takes."""
64 owner = [-1] * num_layers
65 for p, pool in enumerate(pools):
66 for layer in pool.layers:
67 owner[layer] = p
68 assert min(owner) >= 0, owner
69 return owner

72def split_slots(total: int, pools: list[PoolSpec], counts: torch.Tensor | None = None) -> list[int]:
73 """Slots per pool, each in [min_num_slots, max_num_slots], summing to ``total``.
74
75 With ``counts`` (per-layer expert use from ``resolve_profile``) the slots follow expert use, else the pools' layer counts.
76 """
77 if len(pools) == 1:
78 return [total]
79 _check_budget(total, pools)
80 return _split_equal(total, pools) if counts is None else _split_hot(total, pools, counts)
81
82
83def _check_budget(total: int, pools: list[PoolSpec]) -> None:
84 need = sum(p.min_num_slots for p in pools)
85 room = sum(p.max_num_slots for p in pools)
86 if total < need:
87 raise ValueError(f"MoE cache of {total} slots is below what the pools need ({need} slots)")
88 if total > room:
89 raise ValueError(f"MoE cache of {total} slots exceeds what the pools can hold ({room} slots)")
90
91
92def _split_equal(total: int, pools: list[PoolSpec]) -> list[int]:
93 sizes = [p.min_num_slots for p in pools]
94 weights = [len(p.layers) for p in pools]
95 left = total - sum(sizes)
96 while left > 0:
97 open_ = [i for i, p in enumerate(pools) if sizes[i] < p.max_num_slots]
98 # give the next slot to the pool with the fewest slots per layer
99 i = min(open_, key=lambda j: (sizes[j] / weights[j], j))
100 sizes[i] += 1
101 left -= 1
102 return sizes

105def _split_hot(total: int, pools: list[PoolSpec], counts: torch.Tensor) -> list[int]:
106 sizes = [p.min_num_slots for p in pools]
107 # a pool's minimum covers its hottest experts; the rest, up to its maximum, competes across pools
108 cand = []
109 for p, pool in enumerate(pools):
110 hot = counts[list(pool.layers)].flatten().sort(descending=True).values
111 cand.extend((float(c), p) for c in hot[pool.min_num_slots : pool.max_num_slots])
112 cand.sort(key=lambda c: -c[0])
113 for _, p in cand[: total - sum(sizes)]:
114 sizes[p] += 1
115 return sizes

118def resolve_profile(spec: str, model_path: str, num_layers: int, num_experts: int) -> torch.Tensor | None:
119 """Expert use for ``--moe-cache-split``: None for ``equal``, else the profile at ``profile:<path>``."""
120 if spec == "equal":
121 return None
122 path = Path(spec.removeprefix("profile:"))
123 logger.info_rank0(f"MoE pool profile: {path}")
124 return load_profile(path, num_layers, num_experts, model_path)

126
127def load_profile(path: str | Path, num_layers: int, num_experts: int, model_path: str = "") -> torch.Tensor:
128 """Per-layer expert use ``[num_layers, num_experts]`` from a profile, each row scaled to sum to 1."""
129 data = json.loads(Path(path).read_text())
130 model = str(data.get("model", ""))
131 if model and model_path and not any(n.lower().startswith(model.lower()) for n in _model_names(model_path)):
132 logger.warning_rank0(f"MoE pool profile {path} was taken on {model!r}, not on {model_path}")
133 counts = torch.tensor(data["expert_counts"], dtype=torch.float64)
134 if tuple(counts.shape) != (num_layers, num_experts):
135 raise ValueError(
136 f"profile {path} has {tuple(counts.shape)} expert counts; this model needs ({num_layers}, {num_experts})"
137 )
138 # every layer decodes the same tokens, so equal row sums keep the layers comparable
139 return counts / counts.sum(dim=1, keepdim=True).clamp(min=1e-12)

142def _model_names(model_path: str) -> list[str]:
143 # every path component, so an HF cache snapshot (models--Org--Name/snapshots/<hash>) still finds its name
144 names = []
145 for part in reversed(Path(model_path).parts):
146 name = part.split("--")[-1].removesuffix(".gguf")
147 if name:
148 names.append(name)
149 return names
