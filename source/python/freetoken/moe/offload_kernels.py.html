11# Hybrid backend: which of a step's missing experts to fetch (when capped below the miss
12# count). "recency" (default) fetches the experts most-recently active before this step
13# (LRU on the expert -> prioritizes recurring misses, lowering the steady miss rate);
14# "lowest_id" fetches the smallest expert ids (the original, routing-blind heuristic).
15_HYBRID_FETCH_BY_RECENCY = (
16 os.getenv("FREETOKEN_HYBRID_FETCH", "recency").strip().lower() != "lowest_id"

20def ensure_experts(cache, layer_id: int, expert_ids: torch.Tensor) -> None:
21 """Make this layer's routed experts resident; rewrite ``expert_ids`` to slot ids.
22
23 Delegates to flashlib's slot cache. ``id_base`` maps this layer's expert ids into the
24 flat ``layer * num_experts + expert`` space the cache indexes by, and maps
25 ``src_indices`` back, so ``copy_missing`` still resolves against this layer's own host
26 tensor. ``out_indices`` aliases the input, preserving the in-place rewrite every
27 downstream GEMM depends on.
28 """
29 id_of_slot, usage = cache.pool_state(layer_id)
30 lru_ensure(
31 expert_ids,
32 cache.slot_for_id.view(-1),
33 id_of_slot,
34 usage,
35 cache.step,
36 expert_ids,
37 cache.src_indices,
38 cache.evict_slots,
39 cache.num_indices,
40 stats=cache.lru_stats[layer_id] if cache.collect_stats else None,
41 id_base=layer_id * cache.num_experts,
42 )

53def ensure_experts_hybrid(
54 cache,
55 layer_id: int,
56 expert_ids: torch.Tensor,
57 weights: torch.Tensor,
58 max_fetch: int,
59 fetch_fraction: float = 0.0,
60 clamp: bool = False,
61) -> HybridRoutes:
62 """Capped-fetch variant of ``ensure_experts`` (hybrid backend).
63
64 Identical LRU bookkeeping, but only the first ``max_fetch`` of this step's missing
65 experts are given a slot and scheduled for copy; the overflow misses stay
66 non-resident and their ``expert_ids`` positions are rewritten to ``-1`` (compute on
67 the CPU). ``fetch_fraction`` > 0 replaces the fixed cap with the bandwidth-matched
68 split (fraction = pcie_bw / cpu_bw): fetch ~fraction of the step's misses, rounded to
69 the integer that makes the PCIe fetch and the CPU overflow compute finish closest to
70 together. ``num_indices`` = capped fetch count (copy_missing); ``num_missing_full`` =
71 pre-cap miss count (stats).
72
73 The same launch splits the routes: ``expert_ids`` becomes the GPU slots in place, and
74 ``clamp`` gives CPU routes slot 0 instead of -1 for GEMM kernels that read every slot."""
75 assert expert_ids.is_cuda, "the hybrid ensure kernel runs on the GPU"
76 # Q16 fixed point: the kernel caps in integers.
77 frac_q16 = min(1 << 16, max(0, round(fetch_fraction * (1 << 16))))
78 weights = weights.contiguous()
79 routes = HybridRoutes(expert_ids, torch.empty_like(weights), torch.empty_like(expert_ids))
80 _ensure_experts_hybrid_gpu(cache, layer_id, expert_ids, weights, routes, clamp, max_fetch, frac_q16)
81 return routes

84def prefill_hit_compact(cache, layer_id: int) -> None:
85 """Compact this layer's cache-resident experts into gather indices, device-side.
86
87 hit = slot_for_id[layer_id][e] >= the pool's staging threshold (the double buffer owns the
88 slots below, so those bytes are volatile within a prefill chunk and classify
89 as miss). Writes fixed-shape ``_prefill_hit_dst``/``_prefill_hit_src`` (buffer
90 row / pool-local slot) and the count into ``_prefill_hit_num``; one launch on the
91 current stream, no host sync. Safe against the concurrent buffer invalidation
92 on the copy stream: that only rewrites entries already below the threshold."""
93 num_experts = cache.num_experts
94 _prefill_hit_compact_kernel[(1,)](
95 cache.slot_for_id[layer_id],
96 cache._prefill_hit_dst,
97 cache._prefill_hit_src,
98 cache._prefill_hit_num,
99 cache._pool_stable_from[cache.pool_of_layer[layer_id]],
100 num_experts,
101 BLOCK=triton.next_power_of_2(num_experts),
102 )

230@triton.jit
231def _materialize_layer_kernel(
232 slot_for_id_ptr,
233 id_of_slot_ptr,
234 usage_ptr,
235 step_ptr,
236 evict_slots_ptr,
237 src_indices_ptr,
238 num_indices_ptr,
239 layer_id: tl.constexpr,
240 num_experts: tl.constexpr,
241 cache_size: tl.constexpr,
242 BLOCK: tl.constexpr,
243):
244 off = tl.arange(0, BLOCK)
245 expert_mask = off < num_experts
246 slot_mask = off < cache_size
247 slot = off
248
249 base = layer_id * num_experts
250 old_id = tl.load(id_of_slot_ptr + slot, mask=slot_mask, other=-1)
251 # Flat ids make "belongs to this layer" a range check instead of a field compare.
252 same_layer = slot_mask & (old_id >= base) & (old_id < base + num_experts)
253 tl.store(id_of_slot_ptr + slot, -1, mask=same_layer)
254 tl.store(usage_ptr + slot, 0, mask=same_layer)
255
256 old_valid = expert_mask & (old_id >= 0) & (~same_layer)
257 tl.store(slot_for_id_ptr + old_id, -1, mask=old_valid)
258
259 step = tl.load(step_ptr) + 1
260 tl.store(step_ptr, step)
261 tl.store(id_of_slot_ptr + slot, base + off, mask=expert_mask)
262 tl.store(slot_for_id_ptr + base + off, slot, mask=expert_mask)
263 tl.store(usage_ptr + slot, step, mask=expert_mask)
264 tl.store(evict_slots_ptr + off, slot, mask=expert_mask)
265 tl.store(src_indices_ptr + off, off, mask=expert_mask) # layer-local row
266 tl.store(num_indices_ptr, num_experts)

271@triton.jit(do_not_specialize=["layer_id", "num_active", "max_fetch", "fetch_frac_q16", "cache_size", "id_of_slot_ptr", "usage_ptr"])
272def _ensure_experts_hybrid_kernel(
273 expert_ids_ptr,
274 slot_for_id_ptr,
275 id_of_slot_ptr,
276 usage_ptr,
277 step_ptr,
278 active_mask_ptr,
279 evict_slots_ptr,
280 src_indices_ptr,
281 num_indices_ptr,
282 num_missing_full_ptr,
283 expert_recency_ptr,
284 weights_ptr,
285 cpu_ids_ptr,
286 gpu_w_ptr,
287 stats_ptr,
288 stats_layer_ptr,
289 layer_id,
290 num_active,
291 max_fetch,
292 fetch_frac_q16,
293 num_experts: tl.constexpr,
294 cache_size,
295 BLOCK_E: tl.constexpr,
296 BLOCK_C: tl.constexpr,
297 BY_RECENCY: tl.constexpr,
298 CLAMP: tl.constexpr,
299 STATS: tl.constexpr,
300):
301 """Capped-fetch timestamp-LRU (hybrid backend).
302
303 Same as ``_ensure_experts_lru_v2_kernel`` but only ``min(num_missing, max_fetch)``
304 missing experts are evicted-into / scheduled for copy; the overflow misses stay
305 non-resident, so Phase 3 rewrites their positions to -1 (the layer computes those on
306 the CPU). ``fetch_frac_q16`` > 0 (Q16 fixed point) replaces the fixed cap with the
307 bandwidth-matched split ``~frac * num_missing`` (see the Phase-1 comment), computed
308 in-kernel because ``num_missing`` only exists device-side (CUDA graph). ``num_indices``
309 = the capped fetch count (copy_missing), ``num_missing_full`` = the pre-cap miss count
310 (stats).
311
312 Which misses to fetch is the cap policy. ``BY_RECENCY`` (default) fetches the experts
313 most-recently active before this step (LRU on the expert, via ``expert_recency``),
314 breaking ties toward the lower expert id -- this prioritizes *recurring* misses for
315 caching, lowering the steady miss rate. Otherwise the lowest expert ids are fetched
316 (``missing_rank``), the original routing-blind heuristic."""
317 step = tl.load(step_ptr) + 1
318 tl.store(step_ptr, step)
319 base = layer_id * num_experts
320
321 # ---- Phase 1: active + missing over experts ----
322 off_e = tl.arange(0, BLOCK_E)
323 e_mask = off_e < num_experts
324 is_active = tl.zeros((BLOCK_E,), dtype=tl.int1)
325 for i in tl.range(num_active):
326 e = tl.load(expert_ids_ptr + i)
327 is_active = is_active | (off_e == e)
328 tl.store(active_mask_ptr + off_e, is_active.to(tl.int32), mask=e_mask)
329 slot = tl.load(slot_for_id_ptr + base + off_e, mask=e_mask, other=-1)
330 is_missing = is_active & (slot == -1) & e_mask
331 num_missing = tl.sum(is_missing.to(tl.int32))
332 # Cap the fetches; the overflow misses are computed on the CPU (left non-resident).
333 if fetch_frac_q16 > 0:
334 # Bandwidth-matched split (fetch_frac = pcie_bw / cpu_bw): fetch time scales with
335 # F * (1 - frac), CPU time with (M - F) * frac; they balance at F = frac * M. Pick
336 # the integer neighbor that minimizes the slower (max) side of the overlap.
337 lo = (num_missing * fetch_frac_q16) >> 16
338 cost_lo = tl.maximum(lo * ((1 << 16) - fetch_frac_q16), (num_missing - lo) * fetch_frac_q16)
339 cost_hi = tl.maximum(
340 (lo + 1) * ((1 << 16) - fetch_frac_q16), (num_missing - lo - 1) * fetch_frac_q16
341 )
342 max_fetch = tl.where(cost_lo <= cost_hi, lo, lo + 1)
343 is_hit = is_active & (slot >= 0)
344 # a pool smaller than num_experts may hold fewer free slots than misses
345 num_fetch = tl.minimum(tl.minimum(num_missing, max_fetch), cache_size - tl.sum(is_hit.to(tl.int32)))
346 tl.store(num_missing_full_ptr, num_missing.to(tl.int64))
347 tl.store(num_indices_ptr, num_fetch.to(tl.int64))
348 tl.store(usage_ptr + slot, step, mask=is_hit)
349
350 # Fetch-selection priority: encode (recency desc, id asc) into one strictly-ordered
351 # score so argmax has no ties (rec deltas are multiples of num_experts; the id term
352 # spans only [0, num_experts), so it can only break exact-recency ties).
353 if BY_RECENCY:
354 rec = tl.load(expert_recency_ptr + base + off_e, mask=e_mask, other=-1).to(tl.int64)
355 score = tl.where(
356 is_missing, rec * num_experts + (num_experts - 1 - off_e), -1152921504606846976
357 ).to(tl.int64)
358 else:
359 missing_rank = tl.cumsum(is_missing.to(tl.int32)) - 1
360
361 # ---- Phase 2: evict victims by argmin(usage), only for the capped fetches ----
362 if num_fetch > 0:
363 off_c = tl.arange(0, BLOCK_C)
364 c_mask = off_c < cache_size
365 oid = tl.load(id_of_slot_ptr + off_c, mask=c_mask, other=-1)
366 u = tl.load(usage_ptr + off_c, mask=c_mask, other=9223372036854775807).to(tl.int64)
367 owner_active = c_mask & False
368 for i in tl.range(num_active):
369 ei = tl.load(expert_ids_ptr + i)
370 owner_active = owner_active | (oid == base + ei)
371 u = tl.where(owner_active | (~c_mask), 9223372036854775807, u)
372 for i in tl.range(num_fetch):
373 victim = tl.argmin(u, axis=0).to(tl.int32)
374 old_id = tl.sum(tl.where(off_c == victim, oid, 0))
375 if old_id >= 0:
376 tl.store(slot_for_id_ptr + old_id, -1)
377 if BY_RECENCY:
378 e = tl.argmax(score, axis=0).to(tl.int32)
379 score = tl.where(off_e == e, -1152921504606846976, score)
380 else:
381 e = tl.sum(tl.where((missing_rank == i) & is_missing, off_e, 0))
382 tl.store(id_of_slot_ptr + victim, base + e)
383 tl.store(slot_for_id_ptr + base + e, victim)
384 tl.store(usage_ptr + victim, step)
385 tl.store(evict_slots_ptr + i, victim)
386 tl.store(src_indices_ptr + i, e) # layer-local row
387 u = tl.where(off_c == victim, 9223372036854775807, u)
388
389 # ---- Phase 3: rewrite expert_ids -> slot id (hit/fetched) or -1 (overflow -> CPU) ----
390 for i in tl.range(num_active):
391 e = tl.load(expert_ids_ptr + i)
392 s = tl.load(slot_for_id_ptr + base + e)
393 on_gpu = s >= 0
394 tl.store(cpu_ids_ptr + i, tl.where(on_gpu, -1, e))
395 tl.store(gpu_w_ptr + i, tl.where(on_gpu, tl.load(weights_ptr + i), 0.0))
396 if CLAMP:
397 s = tl.maximum(s, 0)
398 tl.store(expert_ids_ptr + i, s)
399
400 # stats_ptr: (missing, fetched, active, calls); stats_layer_ptr: the same per layer, [num_layers, 4]
401 if STATS:
402 n_active = tl.sum((is_active & e_mask).to(tl.int64))
403 tl.store(stats_ptr + 0, tl.load(stats_ptr + 0) + num_missing)
404 tl.store(stats_ptr + 1, tl.load(stats_ptr + 1) + num_fetch)
405 tl.store(stats_ptr + 2, tl.load(stats_ptr + 2) + n_active)
406 tl.store(stats_ptr + 3, tl.load(stats_ptr + 3) + 1)
407 row = stats_layer_ptr + layer_id * 4
408 tl.store(row + 0, tl.load(row + 0) + num_missing)
409 tl.store(row + 1, tl.load(row + 1) + num_fetch)
410 tl.store(row + 2, tl.load(row + 2) + n_active)
411 tl.store(row + 3, tl.load(row + 3) + 1)
412
413 # Bump every active expert's recency to this step (LRU on the expert): an overflow miss
414 # computed on the CPU now ranks high if it recurs, so it gets fetched next time.
415 if BY_RECENCY:
416 step_vec = tl.zeros((BLOCK_E,), dtype=tl.int64) + step
417 tl.store(expert_recency_ptr + base + off_e, step_vec, mask=is_active & e_mask)
418

420@triton.jit(do_not_specialize=["threshold"])
421def _prefill_hit_compact_kernel(
422 slot_ptr, # [num_experts] int32: this layer's slot_for_id row
423 dst_ptr, # [num_experts] int32 out: buffer rows (expert ids), compacted
424 src_ptr, # [num_experts] int32 out: pool-local slots, compacted
425 num_ptr, # [1] int64 out: hit count
426 threshold, # first pool-local slot outside the staging buffers
427 num_experts,
428 BLOCK: tl.constexpr,
429):
430 offs = tl.arange(0, BLOCK)
431 lane = offs < num_experts
432 slots = tl.load(slot_ptr + offs, mask=lane, other=-1)
433 is_hit = lane & (slots >= threshold)
434 pos = tl.cumsum(is_hit.to(tl.int32)) - 1
435 tl.store(dst_ptr + pos, offs.to(tl.int32), mask=is_hit)
436 tl.store(src_ptr + pos, slots, mask=is_hit)
437 tl.store(num_ptr, tl.sum(is_hit.to(tl.int64)))
