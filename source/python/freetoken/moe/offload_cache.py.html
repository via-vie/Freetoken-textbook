18# large entries with sub-~256KB entries on registered host memory (H100 + CUDA 13.0,

142 # GPU-tiled banks, all decode on GPU (stream misses over PCIe into the slot

144 # decode computes experts on the CPU (the slot cache only backs the prefill
145 # double buffer). "hybrid": native banks + a CPU executor + a full slot cache;
146 # each layer fetches a capped subset of its misses over PCIe (``hybrid_max_fetch``
147 # / ``hybrid_fetch_fraction`` below; the GPU computes those plus the hits) and the
148 # CPU absorbs the overflow misses, then the partials merge. The CPU executor is
149 # attached (set_cpu_executor) for cpu/hybrid, set whenever >=1 layer decodes on the CPU.

152 # of that step's misses are computed on the CPU. 0 -> never fetch (CPU does every
153 # miss, the GPU cache stays cold); large -> behaves like pure offload.
154 hybrid_max_fetch: int = 1
155 # hybrid only, one per MoE layer: when > 0, replaces the fixed cap with a per-step fraction -- fetch
156 # ~fraction * misses experts over PCIe (rounded to whichever integer balances the
157 # overlap best), the CPU computes the rest. The engine sets it to the benched
158 # pcie_bw / cpu_bw ratio so the PCIe fetch and the CPU overflow GEMV take equal
159 # time (perfect overlap): fetched : cpu = pcie : cpu - pcie.

348 def _bank_rows(self, k: int, layout: int, offset: int, nrows: int) -> torch.Tensor:
349 """``nrows`` rows of bank ``k`` in ``layout``, starting ``offset`` bytes into the bank cache."""
350 cache = self.banks[k][1]
351 row = self._row_bytes[layout][k]
352 if not self._raw_banks:
353 first = offset // row
354 return cache if first == 0 and nrows == cache.shape[0] else cache[first : first + nrows]
355 dtype, shape = self.layouts[layout][k]
356 return cache[offset : offset + nrows * row].view(dtype).view(nrows, *shape)

358 def _bank_bytes(self, k: int, pool_sizes: list[int]) -> int:
359 """Bytes of raw bank ``k``: each pool's slots and pad, plus the largest pad, so two staging buffers fit wherever two layers of slots do."""
360 layouts = [self._pool_layout(p) for p in range(len(pool_sizes))]
361 slots = sum(s * self._row_bytes[g][k] + self._pad_bytes[g][k] for g, s in zip(layouts, pool_sizes))
362 return slots + max(pads[k] for pads in self._pad_bytes)

367 def _slice_pools(self) -> None:
368 E = self.num_experts
369 bases = [0]
370 for size in self.pool_sizes[:-1]:
371 bases.append(bases[-1] + size)
372 self.pool_bases = bases

375 def cut(t: torch.Tensor, b: int, s: int) -> torch.Tensor:
376 return t if s == t.shape[0] else t[b : b + s]

378 self._pool_id_of_slot = [cut(self.id_of_slot, b, s) for b, s in spans]
379 self._pool_usage = [cut(self.usage, b, s) for b, s in spans]
380 # rows[k][p]: bytes of one slot of pool p in bank k; before the banks exist, one unit per slot with a stage of E
381 if self.banks:
382 rows = [[self._row_bytes[self._pool_layout(p)][k] for p in range(len(spans))] for k in range(len(self.banks))]
383 pads = [[self._pad_bytes[self._pool_layout(p)][k] for p in range(len(spans))] for k in range(len(self.banks))]
384 stage = self._stage_bytes
385 else:
386 rows, pads, stage = [[1] * len(spans)], [[0] * len(spans)], [E]
387 offsets = []
388 for row, pad in zip(rows, pads):
389 offsets.append([0])
390 for s, r, d in zip(self.pool_sizes[:-1], row, pad):
391 offsets[-1].append(offsets[-1][-1] + s * r + d)
392 self._pool_offsets = offsets
393 # local slots below this index overlap the prefill staging area in some bank; a pool's zero-byte rows (a role its kernel does not read) overlap nothing
394 self._pool_stable_from = [
395 max((min(max(-(-(2 * stage[k] - offsets[k][p]) // rows[k][p]), 0), s) for k in range(len(rows)) if rows[k][p]), default=0)
396 for p, s in enumerate(self.pool_sizes)
397 ]
398 self._inval = []
399 for b in range(2):
400 mask = torch.zeros(self.cache_size, dtype=torch.bool)
401 for k in range(len(rows)):
402 lo, hi = b * stage[k], (b + 1) * stage[k]
403 for p, (base, s) in enumerate(spans):
404 if not rows[k][p]:
405 continue
406 first = max((lo - offsets[k][p]) // rows[k][p], 0)
407 end = min(-(-(hi - offsets[k][p]) // rows[k][p]), s)
408 mask[base + first : base + max(first, end)] = True
409 idx = mask.nonzero().flatten()
410 if idx.numel() == 0 or int(idx[-1] - idx[0]) + 1 == idx.numel():
411 self._inval.append(slice(int(idx[0]), int(idx[-1]) + 1) if idx.numel() else slice(0, 0))
412 else:
413 self._inval.append(idx.to(self.device))
414 self._stage_dirty = [False, False]
415 if not self.banks:
416 self._pool_bank_views, self._stage_views, self._stage_tails = [], [], []
417 return
418 self._pool_bank_views = [
419 tuple(self._bank_rows(k, self._pool_layout(p), offsets[k][p], s) for k in range(len(self.banks)))
420 for p, s in enumerate(self.pool_sizes)
421 ]
422 self._stage_views = [
423 [
424 tuple(self._bank_rows(k, g, b * stage[k], E) for k in range(len(self.banks))) if self._stage_fits(b + 1) else None
425 for b in range(2)
426 ]
427 for g in range(len(self.layouts))
428 ]
429 self._stage_tails = [
430 [[self.banks[k][1][b * stage[k] + E * row[k] :][: pad[k]] for k in range(len(self.banks)) if pad[k] and self._stage_fits(b + 1)] for b in range(2)]
431 for row, pad in zip(self._row_bytes, self._pad_bytes)

453 def set_bank_sources(
454 self,
455 sources: dict[str, list[torch.Tensor]],
456 layer_residency: list[str] | None = None,
457 layer_tags: list | None = None,
458 row_tail_bytes: dict[str, list[int]] | None = None,
459 ) -> None:
460 """Attach the host (CPU pinned) expert source banks and allocate a GPU slot
461 cache per bank, following the format's bank schema.
462
463 Every bank is a list of ``num_layers`` tensors, one ``[num_experts, ...]``
464 per layer (independent allocations, so each layer can carry its own host
465 attributes); each slot cache mirrors the bank's row shape and dtype as one
466 unified GPU pool. The row layouts are produced by the weight loaders /
467 repackers (see ``_BANK_SCHEMAS`` and :mod:`freetoken.layers.quantization.moe.nvfp4`)
468 -- the cache machinery is layout-agnostic and just moves rows.
469
470 ``layer_residency`` labels each layer with a ``HostResidency`` value (default: all pinned).
471 Non-pinned (LOCKED/PAGEABLE) layers have no device address: they must already be routed to the CPU executor (``cpu_layer_ids``, set BEFORE this call), the copy plan skips their rows, and their only movement is ``copy_missing``'s whole-layer pageable prefill branch -- which is why prefill overlap is incompatible with them.
472
473 ``layer_tags`` names each layer's quant (see :func:`bank_layouts`); layers of different tags never share a pool.
474
475 ``row_tail_bytes`` gives, per bank and layer, how many bytes past the last row the kernel reads; every pool and staging buffer is then followed by that many zero bytes.
476 """
477 from freetoken.moe.legacy_format import canonical_role
478 from freetoken.moe.host_banks import HostResidency
479
480 # loaders and FTW files may still name the banks the old way (gate_up_packed, ...)
481 by_role = {canonical_role(name): per_layer for name, per_layer in sources.items()}
482 if set(by_role) != {canonical_role(n) for n in self.bank_schema}:
483 raise AssertionError(
484 f"banks {sorted(sources)} do not match the {self.quant_format!r} schema {self.bank_schema}"
485 )
486 sources = {name: by_role[canonical_role(name)] for name in self.bank_schema}
487 residency = layer_residency or [HostResidency.PINNED.value] * self.num_layers
488 assert len(residency) == self.num_layers, (len(residency), self.num_layers)
489 unpinned = frozenset(
490 i for i, r in enumerate(residency) if r != HostResidency.PINNED.value
491 )
492 if unpinned:
493 if not unpinned <= self.cpu_layer_ids:
494 raise ValueError(
495 f"non-pinned layers {sorted(unpinned - self.cpu_layer_ids)} are not in "
496 f"cpu_layer_ids: a layer without a device address can only decode on "
497 f"the CPU executor (set cache.cpu_layer_ids before set_bank_sources)"
498 )
499 if self.prefill_overlap:
500 raise ValueError(
501 "prefill overlap DMAs from registered banks; it must be disabled "
502 "when any layer is LOCKED/PAGEABLE (the engine does this)"
503 )
504 self._unpinned_layers = unpinned
505 self.layer_residency = list(residency)
506 for name in self.bank_schema:
507 per_layer = sources[name]
508 assert len(per_layer) == self.num_layers, (name, len(per_layer))
509 for layer_id, source in enumerate(per_layer):
510 assert source.is_contiguous(), f"bank {name!r} layer {layer_id} must be contiguous"
511 assert source.size(0) == self.num_experts, (name, layer_id, source.shape)
512 spec = self.layout[name] if self.layout is not None else None
513 if self.layer_layouts is not None:
514 spec = self.layer_layouts[layer_id].get(name)
515 if spec is None and source.numel() != 0:
516 raise ValueError(f"layer {layer_id}'s expert kernel reads no {name!r} bank, so it takes a zero-byte placeholder")
517 if spec is not None:
518 if tuple(source.shape[1:]) != tuple(spec.shape) or source.dtype != spec.dtype:
519 raise ValueError(
520 f"bank {name!r} rows are {tuple(source.shape[1:])} {source.dtype} but the expert kernel's layout "
521 f"wants {tuple(spec.shape)} {spec.dtype}; the banks were packed for another kernel"
522 )
523 self.bank_sources[name] = list(per_layer)
524 keys = bank_layouts(sources, layer_tags)
525 unique = list(dict.fromkeys(keys))
526 self.layouts = [banks for _, banks in unique]
527 self.layer_layout = [unique.index(key) for key in keys]
528 for p in range(len(self.pool_sizes)):
529 mixed = {self.layer_layout[l] for l, q in enumerate(self.pool_of_layer) if q == p}
530 assert len(mixed) == 1, f"slot pool {p} holds layers of {len(mixed)} layouts; a pool needs one"
531 self._pad_bytes = [[0] * len(self.bank_schema) for _ in self.layouts]
532 if row_tail_bytes is not None:
533 tails = {canonical_role(name): per_layer for name, per_layer in row_tail_bytes.items()}
534 for k, name in enumerate(self.bank_schema):
535 for layer_id, tail in enumerate(tails[canonical_role(name)]):
536 g = self.layer_layout[layer_id]
537 self._pad_bytes[g][k] = max(self._pad_bytes[g][k], -(-tail // 16) * 16)
538 self._raw_banks = len(set(self.layouts)) > 1 or any(map(any, self._pad_bytes))
539 self._row_bytes = [[math.prod(shape) * dtype.itemsize for dtype, shape in layout] for layout in self.layouts]
540 if self._raw_banks:
541 # every pool view and staging buffer starts on a row boundary of its own layout
542 assert all(r % 16 == 0 for rows in self._row_bytes for r in rows), self._row_bytes
543 self._stage_bytes = [
544 self.num_experts * max(rows[k] for rows in self._row_bytes) + max(pads[k] for pads in self._pad_bytes)
545 for k in range(len(self.bank_schema))
546 ]
547 self._alloc_banks()
548 if not self._stage_fits(1):
549 raise ValueError(f"MoE cache of {self.cache_size} slots cannot stage one whole layer of experts")
550 if self.prefill_overlap and not self._stage_fits(2):
551 logger.warning("Disabling MoE prefill overlap: the slot cache cannot hold two whole layers of experts")
552 self.prefill_overlap = False
553 self._slice_pools()
554 self._build_copy_plan()
555 if self.prefill_overlap:
556 self._init_prefill_overlap_buffers()

565 self.banks = [(self.bank_sources[n], self.bank_caches[n]) for n in self.bank_schema]

579 def _build_fused_copy_plan(self) -> None:
580 """Precompute the fused multi-bank copy descriptor (base addrs + per-row bytes).
581
582 Built once here (and on :meth:`rebuild`, which reallocates the slot caches);
583 the addresses are fixed for the cache's lifetime so the descriptor tensors are
584 CUDA-graph safe. Disabled (-> per-bank fallback) if any bank's row bytes or base
585 address is not 16-byte aligned, or via FREETOKEN_FUSED_COPY=0.
586 """
587 self._copy_fused_ok = False
588 self._copy_dst_ptrs = None
589 self._copy_dst_ptrs_pool = None
590 self._copy_src_ptrs = None
591 self._copy_feat_bytes = None
592 self._copy_dst_ptrs_host: list[int] = []
593 self._copy_src_ptrs_host: list[list[int]] = []
594 self._gather_bank_ids: list[list[int]] = []
595 self._gather_feat_bytes: list[torch.Tensor | None] = []
596 self._gather_stage_ptrs: list[list[torch.Tensor] | None] = []
597 self._gather_pool_ptrs: list[torch.Tensor | None] = []
598 if not _FUSED_COPY or self.device.type != "cuda" or not self.banks:
599 return
600 from freetoken.kernel.pinned import device_ptr
601
602 aligned = [self._stage_bytes] + self._row_bytes + self._pool_offsets
603 if any(n % 16 for ns in aligned for n in ns) or any(c.data_ptr() % 16 for _, c in self.banks):
604 return # leave fused disabled; copy_missing uses the per-bank path
605 dst_ptrs = [cache.data_ptr() for _, cache in self.banks]
606 layer_src_ptrs = [[] for _ in range(self.num_layers)]
607 for per_layer, _ in self.banks:
608 for layer_id, source in enumerate(per_layer):
609 if layer_id in self._unpinned_layers:
610 # unregistered layer: no device alias exists, and the row is never consumed (CPU decode; pageable prefill)
611 # a 0 placeholder keeps the descriptor shape
612 layer_src_ptrs[layer_id].append(0)
613 continue
614 if source.numel() == 0:
615 layer_src_ptrs[layer_id].append(0) # placeholder bank: zero bytes per row, never dereferenced
616 continue
617 # The kernel dereferences these on the GPU, so store each host bank's
618 # device alias (== data_ptr() under UVA identity; differs on
619 # Windows/WDDM).
620 src_dev = device_ptr(source)
621 if src_dev % 16 != 0:
622 return
623 layer_src_ptrs[layer_id].append(src_dev)

625 def ptrs(values: list[int]) -> torch.Tensor:
626 return torch.tensor(values, dtype=torch.int64, device=self.device)
627
628 self._copy_dst_ptrs = ptrs(dst_ptrs)
629 self._copy_dst_ptrs_pool = [
630 self._copy_dst_ptrs if all(o[p] == 0 for o in self._pool_offsets)
631 else ptrs([d + o[p] for d, o in zip(dst_ptrs, self._pool_offsets)])
632 for p in range(len(self.pool_sizes))
633 ]
634 self._copy_src_ptrs = [ptrs(values) for values in layer_src_ptrs]
635 self._copy_feat_bytes = [ptrs(rows) for rows in self._row_bytes]
636 self._copy_dst_ptrs_host = dst_ptrs
637 self._copy_src_ptrs_host = layer_src_ptrs
638 # hit-D2D gather serves only the big banks; small banks are whole-layer
639 # H2D entries (see _SMALL_BANK_FEAT_BYTES), so their rows never need D2D.
640 # The gather copies cache rows of the layer's pool into its staging buffer, row == expert id.
641 self._gather_bank_ids = [[k for k, r in enumerate(rows) if r >= _SMALL_BANK_FEAT_BYTES] for rows in self._row_bytes]
642 for g, ids in enumerate(self._gather_bank_ids):
643 self._gather_feat_bytes.append(ptrs([self._row_bytes[g][k] for k in ids]) if ids else None)
644 self._gather_stage_ptrs.append(
645 [ptrs([dst_ptrs[k] + b * self._stage_bytes[k] for k in ids]) for b in range(2)] if ids else None
646 )
647 for p in range(len(self.pool_sizes)):
648 ids = self._gather_bank_ids[self._pool_layout(p)]
649 self._gather_pool_ptrs.append(ptrs([dst_ptrs[k] + self._pool_offsets[k][p] for k in ids]) if ids else None)
650 self._copy_blocks_per_bank = default_blocks_per_bank(self.device)
651 self._copy_fused_ok = True

653 def validate_rebuild(self, cache_size: int, pool_sizes: list[int] | None = None) -> None:
654 """Pure geometry validation of a rebuild target (no GPU side effects).
655
656 Raises ``ValueError`` if ``cache_size`` is below the ``num_experts`` floor or
657 a pool is above the expert kernel's slot cap, or the bank byte segments cannot stage one layer. Called by :meth:`rebuild` and by the engine's
658 pre-teardown check, so an invalid target rejects with the old cache intact
659 (no destructive free first).
660 """
661 if cache_size < self.num_experts:
662 raise ValueError(f"cache_size {cache_size} < num_experts {self.num_experts}")
663 # decode kernels see one pool's slots, so the cap applies per pool
664 largest = cache_size if pool_sizes is None else max(pool_sizes)
665 what = f"moe_cache_size={cache_size}" if largest == cache_size else f"a MoE slot pool of {largest} slots"
666 if self.max_slots is not None and largest > self.max_slots:
667 raise ValueError(
668 f"{what} exceeds the expert kernel's slot limit of {self.max_slots}; "
669 f"pass --moe-cache-size {self.max_slots} or less, or let the default kernel serve the experts"
670 )
671 if self.layout is None and self.quant_format == "nvfp4_marlin" and largest > MARLIN_MAX_CACHE_SIZE:
672 raise ValueError(
673 f"{what} exceeds the marlin backend's slot limit of "
674 f"{MARLIN_MAX_CACHE_SIZE} (vLLM moe_align_block_size caps padded experts at "
675 "1024); reduce moe_cache_size or force --quant-backend moe.nvfp4=triton"
676 )
677 if self._raw_banks and pool_sizes is not None:
678 for k, stage in enumerate(self._stage_bytes):
679 if self._bank_bytes(k, pool_sizes) < stage:
680 raise ValueError(f"moe_cache_size={cache_size} cannot stage one whole layer of experts")

682 def rebuild(self, cache_size: int, pool_sizes: list[int] | None = None) -> None:
683 """Resize the GPU slot cache + bookkeeping to ``cache_size`` IN PLACE.
684
685 Keeps the CPU/pinned ``bank_sources`` and the GPU-resident alphas; never
686 reloads banks. Tears down prefill-overlap buffers first (their views alias
687 the old ``bank_caches``), frees the old GPU tensors, then reallocates. Slots
688 cold-start after rebuild. Object identity is preserved so attached layers and
689 ``ctx.moe_offload_cache`` stay valid.
690 """
691 assert self.bank_sources, "set_bank_sources must run before rebuild"
692 sizes = [cache_size] if pool_sizes is None else list(pool_sizes)
693 assert len(sizes) == len(self.pool_sizes) and sum(sizes) == cache_size and min(sizes) > 0, (sizes, cache_size)
694 self.validate_rebuild(cache_size, sizes)
695 # 1. Tear down prefill-overlap (its buffer views alias the old bank_caches).
696 self.prefill_copy_stream = None
697 self.prefill_begin_event = None
698 self.prefill_ready_events = []
699 self.prefill_release_events = []
700 self._prefill_buffer_layer = [None, None]
701 self._prefill_buffer_released = [True, True]
702 self._prefill_buffer_has_release_event = [False, False]
703 # 2. Drop old GPU tensors (free-before-alloc).
704 self.banks = []
705 self.bank_caches = {}
706 self._pool_bank_views = []
707 self._stage_views = []
708 self._stage_tails = []
709 self.cache_size = cache_size
710 self.pool_sizes = sizes
711 if self.device.type == "cuda":
712 torch.cuda.synchronize(self.device)
713 torch.cuda.empty_cache()
714 # 3. Reallocate the slot cache from the retained host sources.
715 self._alloc_banks()
716 # 4. Reallocate cache_size-shaped bookkeeping; reset the slot map (cold start).
717 self.slot_for_id.fill_(-1)
718 self.id_of_slot = torch.full((cache_size,), -1, dtype=torch.int32, device=self.device)
719 self.usage = torch.zeros((cache_size,), dtype=torch.int64, device=self.device)
720 self._slice_pools()
721 self._build_copy_plan() # slot caches were reallocated -> refresh fused-copy addrs
722 plan_slots = max(self.num_experts, cache_size)
723 self.evict_slots = torch.empty((plan_slots,), dtype=torch.int32, device=self.device)
724 self.src_indices = torch.empty((plan_slots,), dtype=torch.int32, device=self.device)
725 self.step.zero_()
726 self.active_mask.zero_()
727 self.num_indices.zero_()
728 self.num_missing_full.zero_()
729 self.expert_recency.fill_(-1)
730 self.stat_missing.zero_()
731 self.stat_active.zero_()
732 self.stat_calls.zero_()
733 self.stat_fetched.zero_()
734 self.stat_missing_layer.zero_()
735 # a rebuild is a cold start for the cache; carrying pre-rebuild hit/miss counts over would skew every post-rebuild stats report
736 self.lru_stats.zero_()
737 self.stat_active_layer.zero_()
738 self.stat_fetched_layer.zero_()
739 self.stat_steps_layer.zero_()
740 self.decode_freq.zero_()
741 self.prefill_hit_rows = 0
742 self.prefill_total_rows = 0
743 self._hit_d2d_fallback_logged = False # geometry changed; re-log if still unusable
744 # 5. Re-evaluate prefill overlap against the new size.
745 if self.prefill_overlap and not self._stage_fits(2):
746 logger.warning(
747 f"Disabling MoE prefill overlap on rebuild: cache_size {cache_size} "
748 f"cannot hold two whole layers of experts (2*num_experts {2 * self.num_experts} slots of one layout)."
749 )
750 self.prefill_overlap = False
751 if self.prefill_overlap:
752 self._init_prefill_overlap_buffers()

841 def _invalidate_prefill_buffer(self, buffer_id: int, layer_id: int) -> None:
842 """Take buffer ``buffer_id``'s slots for staging ``layer_id``."""

855 def begin_prefill(self) -> None:
856 if not self.prefill_overlap:
857 return
858 self._prefill_buffer_layer = [None, None]
859 self._prefill_buffer_released = [True, True]
860 if self.prefill_copy_stream is not None:
861 # Fence this prefill's copy-stream work behind everything already enqueued
862 # on the compute stream. The release/ready events only order against the
863 # *previous prefill*; under overlap scheduling a new prefill can be enqueued
864 # while the preceding decode batch is still running, and that decode may
865 # have loaded experts into the slots the buffers borrow -- without this
866 # fence the first prefetch would stomp bytes a running GEMM is reading.
867 self.prefill_begin_event.record(torch.cuda.current_stream(self.device))
868 self.prefill_copy_stream.wait_event(self.prefill_begin_event)
869 self._prefill_hit_d2d_active = self.prefill_hit_d2d and self._hit_d2d_usable()
870 if self._prefill_hit_d2d_active:
871 # The copy stream is fenced behind the previous decode, so the snapshot
872 # observes its final slot map; one host sync per chunk, then per-layer
873 # classification is pure host math.
874 with torch.cuda.stream(self.prefill_copy_stream):
875 self._prefill_slot_snapshot.copy_(self.slot_for_id, non_blocking=True)
876 self.prefill_copy_stream.synchronize()

878 def prefetch_prefill_layer(self, layer_id: int) -> None:
879 if not self.prefill_overlap or layer_id >= self.num_layers:
880 return
881 if layer_id < 0:
882 raise ValueError(f"Invalid prefill layer id: {layer_id}")
883
884 assert self.banks and self._stage_views
885
886 buffer_id = layer_id % 2
887 if self._prefill_buffer_layer[buffer_id] == layer_id:
888 return
889 if self._prefill_buffer_layer[buffer_id] is not None:
890 assert self._prefill_buffer_released[buffer_id], (
891 "Prefill overlap buffer is being reused before release"
892 )

894 def copy() -> None:

896 for (per_layer, _), view in zip(self.banks, self.stage_views(layer_id, buffer_id)):
897 view.copy_(per_layer[layer_id], non_blocking=True)
898
899 if self._prefill_hit_d2d_active:

901 elif self.prefill_copy_stream is None:

905 if self._prefill_buffer_has_release_event[buffer_id]:
906 self.prefill_copy_stream.wait_event(self.prefill_release_events[buffer_id])
907 copy()
908 self.prefill_ready_events[buffer_id].record(self.prefill_copy_stream)
909
910 self._prefill_buffer_layer[buffer_id] = layer_id
911 self._prefill_buffer_released[buffer_id] = False

918 configuration that silently runs the legacy path is visible.

956 def _prefetch_split(self, layer_id: int, buffer_id: int) -> None:
957 """Hit/miss-split prefetch of one expert layer into the double buffer.
958
959 Resident experts are gathered cache -> buffer on the CURRENT stream, fully
960 device-side: a one-launch compaction reads the LIVE slot_for_id row into
961 fixed-shape gather indices (no host round trip), then fast_index_copy_multi
962 moves the rows. Serializing the gather before this layer's GEMMs costs its
963 plain duration instead of nondeterministic SM contention. Misses cross
964 PCIe as ONE cudaMemcpyBatchAsync of coalesced expert-id runs on the copy
965 stream, under the existing release/ready event discipline; its host-built
966 run list comes from the begin-of-chunk snapshot because the batch API
967 takes HOST pointer arrays. Live-vs-snapshot cannot disagree: the only
968 chunk-internal writer (buffer invalidation) rewrites slots already below
969 the pool's staging threshold, and those slots (and -1) are misses on both sides
970 -- the buffers own those slots, so their bytes are volatile within the
971 chunk. Hit and miss row sets are disjoint, so the streams need no
972 ordering against each other.
973 """
974 import numpy as np
975
976 from freetoken.kernel.fast_index_copy import fast_index_copy_multi_jit
977 from freetoken.moe.offload_kernels import prefill_hit_compact
978
979 E = self.num_experts
980 layout = self.layer_layout[layer_id]
981 pool = self.pool_of_layer[layer_id]
982 snap = self._prefill_snapshot_np[layer_id]
983 hit_mask = snap >= self._pool_stable_from[pool]
984 self.prefill_hit_rows += int(hit_mask.sum())
985 self.prefill_total_rows += E
986 if self._gather_feat_bytes[layout] is not None:
987 prefill_hit_compact(self, layer_id)
988 # blocks_per_bank=64 vs the PCIe-tuned default of 8: HBM D2D needs the
989 # wider grid (~22 GB/s per 1024-thread block on H100).
990 fast_index_copy_multi_jit(
991 self._gather_stage_ptrs[layout][buffer_id],
992 self._gather_pool_ptrs[pool],
993 self._gather_feat_bytes[layout],
994 self._prefill_hit_dst,
995 self._prefill_hit_src,
996 self._prefill_hit_num,
997 blocks_per_bank=64,
998 )
999 miss = np.nonzero(~hit_mask)[0]
1000 with torch.cuda.stream(self.prefill_copy_stream):
1001 if self._prefill_buffer_has_release_event[buffer_id]:
1002 self.prefill_copy_stream.wait_event(self.prefill_release_events[buffer_id])
1003 self._invalidate_prefill_buffer(buffer_id, layer_id)
1004 if miss.size:
1005 run_starts = np.concatenate(([0], np.nonzero(np.diff(miss) != 1)[0] + 1))
1006 starts = miss[run_starts]
1007 lengths = np.diff(np.concatenate((run_starts, [miss.size])))
1008 dst, src, nbytes = [], [], []
1009 for b, feat in enumerate(self._row_bytes[layout]):
1010 stage = self._copy_dst_ptrs_host[b] + buffer_id * self._stage_bytes[b]
1011 if feat == 0:
1012 continue
1013 if feat < _SMALL_BANK_FEAT_BYTES:
1014 # Whole layer as one entry, EVEN with zero misses: it keeps every
1015 # batch entry above the driver's async floor and covers the hit
1016 # rows the gather skips for these banks.
1017 dst.append(stage)
1018 src.append(self._copy_src_ptrs_host[layer_id][b])
1019 nbytes.append(E * feat)
1020 elif miss.size:
1021 dst.extend(stage + starts * feat)
1022 src.extend(self._copy_src_ptrs_host[layer_id][b] + starts * feat)
1023 nbytes.extend(lengths * feat)
1024 if dst:
1025 self._batch_memcpy(
1026 torch.tensor(dst, dtype=torch.int64),
1027 torch.tensor(src, dtype=torch.int64),
1028 torch.tensor(nbytes, dtype=torch.int64),
1029 torch.cuda.current_stream(self.device).cuda_stream,
1030 )
1031 self.prefill_ready_events[buffer_id].record(self.prefill_copy_stream)

1189 def decode_routing_stats(self) -> dict:
1190 """Per-layer decode routing concentration, for cache-skew analysis.
1191
1192 Uses the histogram from ``collect_decode_freq``. The ``oracle_hit`` is the best a
1193 per-layer LRU holding ``cache_size/num_layers`` slots could achieve on the observed
1194 (stationary) routing distribution -- i.e. an upper bound on hit rate that depends
1195 purely on how skewed routing is, independent of any LRU/LFU dynamics.
1196 """
1197 freq = self.decode_freq.float()
1198 total = freq.sum(dim=1)
1199 valid = total > 0
1200 if int(valid.sum()) == 0:
1201 return {}
1202 slots_per_layer = self.cache_size / self.num_layers
1203 C = max(1, int(round(slots_per_layer)))
1204 sorted_f, _ = torch.sort(freq, dim=1, descending=True)
1205 oracle_hit = (sorted_f[:, :C].sum(dim=1)[valid] / total[valid]).mean().item()
1206 ws = (freq > 0).sum(dim=1).float()
1207 cdf = torch.cumsum(sorted_f, dim=1) / total.clamp(min=1).unsqueeze(1)
1208 cover90 = ((cdf < 0.9).sum(dim=1).float() + 1)[valid]
1209 p = freq / total.clamp(min=1).unsqueeze(1)
1210 ent = -(p * p.clamp(min=1e-12).log()).sum(dim=1)[valid]
1211 norm_ent = (ent / torch.log(torch.tensor(float(self.num_experts)))).mean().item()
1212 return {
1213 "slots_per_layer": slots_per_layer,
1214 "working_set_mean": ws[valid].mean().item(),
1215 "working_set_max": int(ws[valid].max().item()),
1216 "experts_for_90pct": cover90.mean().item(),
1217 "oracle_hit_at_slots": oracle_hit,
1218 "norm_entropy": norm_ent,
1219 }
