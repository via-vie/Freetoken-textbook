1"""Python wrapper around the ``_cpu_moe`` C++ executor (--moe-strategy cpu).
2
3Owns the persistent CPU worker pool and one pinned IO block of ``max_tokens`` rows that
4every layer and batch size shares (one task is in flight at a time). ``decode`` issues
5the whole CUDA-graph-capturable sequence on the current stream:
6
7 D2H (hidden, topk_ids, topk_weights -> pinned IO rows [:bs])
8 -> submit (doorbell naming (layer, bs), or a cudaLaunchHostFunc host node)
9 -> sync (wait for the CPU pool to finish)
10 -> H2D (pinned expert output -> GPU)
11
12With the doorbell on CUDA, a publish kernel does the D2H and the doorbell, and skips the step when no route goes to the CPU; a combine kernel does the H2D (csrc/cpu_moe/gpu_sync_io.h).
13"""

29# Flag-based GPU<->CPU handshake for hybrid/cpu decode. The default host-func path
30# (cudaLaunchHostFunc submit+sync per layer) pays ~30-50us of callback dispatch latency
31# per call with the GPU stream idle -- 2 calls per MoE layer per decode step (~6 ms/step
32# on a 75-layer model). Instead the GPU raises a mapped-pinned "ready" flag at submit; a
33# persistent CPU coordinator (in _cpu_moe) polls it, runs the layer, and sets a "done"
34# flag the GPU waits on at sync -- no host-func round-trip. Both GPU-side operations are
35# STREAM MEMORY OPERATIONS (cuStreamWriteValue64 / cuStreamWaitValue64 on CUDA, see
36# gpu_sync.h): they execute on the GPU front end with no SM-resident kernel, so
37# GPU utilization stays truthful during the CPU compute window. (The first cut used a
38# spin-wait kernel; that pinned reported utilization at 99% and laptop CPU/GPU dynamic
39# power schedulers responded by clamping the CPU frequency -- a net decode regression on
40# power-coupled edge devices.) The doorbell word names the (layer, decode batch size)
41# task, so one ready/done pair serves every layer and every captured graph (gpu_sync.h).
42# Where memops are unavailable (some driver models, vGPU, old drivers -- functionally probed at
43# startup), decode keeps the host-func path (functional, slower). ROCm 7.14 cannot replay
44# that path from a graph, so capture fails loudly there instead. A Python watchdog thread
45# turns a wedged coordinator into a loud RuntimeError (via raise_if_unhealthy) instead of
46# an indefinite stream stall.
47# Caveat: the coordinator busy-polls one core while decode traffic flows (idle backoff
48# otherwise); FREETOKEN_CPU_MOE_FLAG_SYNC=0 opts out entirely.
49_FLAG_SYNC = os.getenv("FREETOKEN_CPU_MOE_FLAG_SYNC", "1") != "0"
50# Pool threads spin up to 50 ms before they park. FREETOKEN_CPU_MOE_SPIN=0 opts out.
51_SPIN_WAIT = os.getenv("FREETOKEN_CPU_MOE_SPIN", "1") != "0"
52# ds_fp4 runs its input FP8 round-trip on the GPU when it can; tests turn this off to compare against the CPU round-trip.
53_DSFP4_GPU_PREQUANT = True

70def compiled_extension_supports(activation: str) -> bool:
71 """Whether the compiled ``_cpu_moe`` extension can serve ``activation``
72 through its generic epilogue. A stale prebuilt .so accepts newer act ids
73 while silently computing the wrong math; the executor hard-errors on that,
74 but the engine's auto offload->hybrid upgrade consults this first so a
75 default boot degrades to offload instead of crashing after weight load."""
76 if activation not in _ACT_IDS:
77 return False
78 if _ACT_IDS[activation] < 3:
79 return True
80 try:
81 from freetoken.kernel import _cpu_moe
82 except ImportError:
83 return False
84 return _ACT_IDS[activation] <= getattr(_cpu_moe, "max_generic_act_id", lambda: 2)()

87def physical_core_cpus() -> list[int]:
88 """One logical CPU per physical core, restricted to this process's affinity.
89
90 MoE decode is memory-bandwidth-bound, so SMT siblings only contend for the
91 same core's load ports without adding bandwidth. Picking one logical CPU per
92 physical core (and pinning to it) gives the best, most stable bandwidth.
93 Falls back to the full affinity set when sysfs topology is unavailable.
94 """
95 try:
96 allowed = sorted(os.sched_getaffinity(0))
97 except AttributeError:
98 allowed = list(range(os.cpu_count() or 1))
99 reps: list[int] = []
100 seen: set[str] = set()
101 for cpu in allowed:
102 try:
103 with open(f"/sys/devices/system/cpu/cpu{cpu}/topology/thread_siblings_list") as f:
104 key = f.read().strip()
105 except OSError:
106 reps.append(cpu)
107 continue
108 if key not in seen:
109 seen.add(key)
110 reps.append(cpu)
111 return reps or allowed or [0]

114def _spin_wait_ok(core_ids: list[int], coord_core: int) -> bool:
115 """Spin only if each pool thread owns a distinct CPU and two CPUs stay free for the main and CUDA callback threads."""
116 if not _SPIN_WAIT or not hasattr(os, "sched_getaffinity"):
117 return False
118 pool = list(core_ids) + ([coord_core] if coord_core >= 0 else [])
119 if len(set(pool)) != len(pool):
120 return False
121 spare = set(os.sched_getaffinity(0)) - set(pool)
122 return len(spare) >= 2

125def resolve_threads_and_affinity(requested: int) -> tuple[int, list[int]]:
126 """Return (num_threads, core_ids) for the worker pool.
127
128 ``requested == 0`` -> one thread per physical core, pinned to it (best for the
129 bandwidth-bound GEMV: SMT siblings only contend for a core's load ports and the
130 spin-barrier degrades badly when oversubscribed). An explicit count is honored,
131 spreading first across physical cores, then across the remaining logical CPUs
132 (so distinct hardware threads are used before any core is doubled up).
133 """
134 reps = physical_core_cpus()
135 if requested and requested > 0:
136 n = int(requested)
137 try:
138 allowed = sorted(os.sched_getaffinity(0))
139 except AttributeError:
140 allowed = list(range(os.cpu_count() or 1))
141 # physical-core reps first, then the rest of the logical CPUs.
142 order = reps + [c for c in allowed if c not in set(reps)]
143 if not order:
144 order = [0]
145 core_ids = [order[i % len(order)] for i in range(n)]
146 return n, core_ids
147 return len(reps), list(reps)

208 # ABI probe: a stale prebuilt _cpu_moe.so accepts newer act ids without
209 # error and silently computes the wrong activation in the generic
210 # epilogue -- fail loudly with the rebuild instruction instead. (mxfp4
211 # handles its act inside the kernel and predates the marker.)
212 if _ACT_IDS[activation] >= 3 and names != ["mxfp4_triton"]:
213 supported = getattr(_cpu_moe, "max_generic_act_id", lambda: 2)()
214 if _ACT_IDS[activation] > supported:
215 raise RuntimeError(
216 f"the compiled _cpu_moe extension predates activation "
217 f"{activation!r} (max generic act id {supported}); rebuild it "
218 "with `python setup.py build_ext --inplace` (or reinstall the "
219 "wheel) before serving this model on the cpu/hybrid backend."
220 )

222 self.num_layers = int(cache.num_layers)
223 self.num_experts = int(cache.num_experts)
224 self.top_k = int(top_k)
225 self.quant_format = ",".join(names)
226 self.device = device
227 self.max_tokens = int(max_tokens)
228 self.apply_router_weight_on_input = bool(apply_router_weight_on_input)
229 # The per-layer tensors and their pointer tables must outlive the executor
230 # (C++ holds raw addresses into both).
231 self._banks: list[torch.Tensor] = []
232 by_role = {canonical_role(name): per_layer for name, per_layer in cache.bank_sources.items()}
233 # every format's tables address every layer, but a format reads only its own layers' rows
234 formats, shapes = [], set()
235 for name, opts in specs:
236 banks, options, shape = self._resolve_banks(by_role, name, keys.index((name, opts)), dict(opts))
237 formats.append((name, banks, {**dict(opts), **options}))
238 shapes.add(shape)
239 if len(shapes) != 1:
240 raise NotImplementedError(f"CPU MoE formats disagree on the expert shape (hidden, intermediate): {sorted(shapes)}")
241 (self.H, self.I), = shapes
242 layer_counts = {spec: keys.count(spec) for spec in specs}

244 # ds_fp4: the reference FP8 activation round-trip is a scalar per-element chain
245 # that the C++ side runs single-threaded on the CUDA host-callback thread --
246 # straight on the decode critical path (~0.3ms/layer at H=4096, every worker and
247 # the GPU waiting on it). When a GPU is present we run the numerically identical
248 # round-trip as a captured GPU kernel BEFORE the D2H (see decode_submit) and tell
249 # the C++ side to skip its own. Measured on DeepSeek-V4-Flash bs=1 decode:
250 # 12.85 -> 15.65 tok/s, output bit-identical (tests/moe/test_dsfp4_prequant.py).
251 # the W4A8 activation round-trip block follows the checkpoint's fp8 block (128 on V4, 32 on V4.1)
252 blocks = {options.get("act_block") for name, _, options in formats if name == "ds_fp4"}
253 assert None not in blocks, "ds_fp4 experts need the act_block option, the checkpoint's fp8 activation quant block"
254 # the GPU round-trip rewrites the whole input, so only when every layer is ds_fp4 with one block
255 self._gpu_prequant = names == ["ds_fp4"] and len(blocks) == 1 and device.type == "cuda" and _DSFP4_GPU_PREQUANT
256 self._act_block = next(iter(blocks)) if self._gpu_prequant else None
257 for name, _, options in formats:
258 if name == "ds_fp4":
259 options["input_prequant"] = int(self._gpu_prequant)
260 if self._gpu_prequant:
261 logger.info_rank0(
262 "ds_fp4: input FP8 round-trip moved to the GPU "
263 "(bit-identical grid; the CPU-side scalar round-trip is skipped)"
264 )

266 # Decide the flag handshake up front (env + device + a functional stream-memop
267 # probe): its coordinator needs a core of its own, which the auto thread sizing
268 # below reserves (a coordinator time-slicing against the GEMV workers measurably
269 # destabilizes throughput on fully-subscribed boxes).
270 self._flag_sync = _FLAG_SYNC and device.type == "cuda"
271 if self._flag_sync and not _cpu_moe.memops_probe(torch.cuda.current_stream().cuda_stream):
272 logger.info_rank0(
273 "cpu-moe flag handshake unavailable: stream memory operations failed the "
274 "startup probe; using the cudaLaunchHostFunc sync"
275 )
276 self._flag_sync = False
277 # ROCm 7.14 returns wrong results when it replays host-function nodes, so there only the doorbell is graph-safe.
278 self._graph_capture_safe = self._flag_sync or _cpu_moe.host_func_capture_safe()

286 nthreads -= 1
287 core_ids = core_ids[:-1]
288 self._coord_core = coord_core

313 swiglu_limit=float(swiglu_limit) if swiglu_limit is not None else float("inf"),

321 self.core_ids = core_ids
322 self.isa = self._ext.isa_name()

335 # Flag-based handshake: the extension owns the ready/done words (gpu_sync.h) and a
336 # persistent CPU coordinator that polls the doorbell, runs the task it names on the
337 # pool, and sets done for the GPU. The WATCHDOG thread marks the executor unhealthy
338 # when a doorbell stays unanswered (dead coordinator): it poisons done to unblock the
339 # stream and raise_if_unhealthy() turns the step into a loud error instead of silent
340 # stale output (flag_sync itself was decided above, before thread sizing).
341 self._unhealthy = False
342 self._kernel_io = False
343 if self._flag_sync:
344 self._ext.start_flag_coordinator(self._coord_core)

348 if reason:
349 logger.warning_rank0(f"cpu-moe kernel IO unavailable ({reason}); using D2H/H2D copies around the doorbell")
350 else:
351 self._ext.enable_kernel_io()
352 self._kernel_io = True
353 self._watchdog_stop = False

355 # method would strong-reference the executor forever (the loop never ends
356 # on its own), pinning the C++ worker pool and the pinned banks against GC
357 # in build-many-engines scenarios. NB: the stop flag / weakref death is
358 # observed only between 2 s sleeps, so teardown of the THREAD can lag up to
359 # ~2 s -- it is a daemon, so neither GC of the executor (weakref breaks the
360 # cycle) nor process exit waits on it.
361 self._watchdog = threading.Thread(
362 target=_watchdog_main,
363 args=(weakref.ref(self),),
364 name="cpu-moe-flag-watchdog",
365 daemon=True,
366 )

381 def _make_table(self, layers: list[torch.Tensor]) -> torch.Tensor:
382 """Build a CPU int64 tensor of per-layer base addresses for one bank.
383
384 ``layers`` is one ``[num_experts, ...]`` tensor per layer (the per-layer host
385 bank contract). The C++ side stores this table's pointer and indexes
386 ``tbl[layer_id]`` at call time; both the table and the layer tensors are kept
387 on ``self._banks`` (GC guard) so the raw pointers stay valid for the
388 executor's lifetime.
389 """
390 assert len(layers) == self.num_layers, (len(layers), self.num_layers)
391 table = torch.tensor([t.data_ptr() for t in layers], dtype=torch.int64)
392 self._banks.append(table)
393 self._banks.extend(layers)
394 return table

396 def _tables(self, **roles: list[torch.Tensor]) -> dict[str, int]:
397 """Per-layer table addresses keyed by canonical bank role."""
398 return {role: self._make_table(layers).data_ptr() for role, layers in roles.items()}

400 def _resolve_fp8_block_banks(self, banks: dict, rep: int = 0) -> tuple[dict, dict, tuple[int, int]]:
401 """Block-FP8 schema: row-major e4m3 + one bf16 scale per 128x128 block; the scale banks pad their last dim, so C++ gets the real row stride."""
402 gu, gus = banks["gate_up"], banks["gate_up_scale"]
403 dn, dns = banks["down"], banks["down_scale"]
404 assert gu[rep].element_size() == 1 and dn[rep].element_size() == 1, (gu[rep].dtype, dn[rep].dtype)
405 assert gus[rep].dtype == torch.bfloat16 and dns[rep].dtype == torch.bfloat16, (gus[rep].dtype, dns[rep].dtype)
406 I = int(gu[rep].shape[1] // 2)
407 H = int(gu[rep].shape[2])
408 assert gu[rep].shape[1] == 2 * I
409 assert tuple(dn[rep].shape[1:]) == (H, I), (dn[rep].shape, H, I)
410 assert gus[rep].shape[1] == (2 * I + 127) // 128 and gus[rep].shape[2] >= (H + 127) // 128, (gus[rep].shape, H, I)
411 assert dns[rep].shape[1] == (H + 127) // 128 and dns[rep].shape[2] >= (I + 127) // 128, (dns[rep].shape, H, I)
412 banks = self._tables(gate_up=gu, down=dn, gate_up_scale=gus, down_scale=dns)
413 options = dict(gu_scale_stride=int(gus[rep].stride(1)), dn_scale_stride=int(dns[rep].stride(1)))
414 return banks, options, (H, I)

416 def _resolve_banks(self, banks: dict, fmt: str, rep: int = 0, opts: dict | None = None) -> tuple[dict, dict, tuple[int, int]]:
417 """Return (banks, options, (H, I)) for the C++ ctor: banks maps each canonical role
418 the format reads to its per-layer table's address, options holds its integer settings.
419
420 ``banks[name]`` is a list of ``num_layers`` ``[num_experts, ...]`` tensors
421 (the per-layer host bank contract); shapes are read from the first layer so
422 per-partition (TP) sizes are exact. Every bank address is a per-layer table
423 (see ``_make_table``), not a single bank's -- the C++ side resolves
424 ``tbl[layer_id]`` per task.
425 """
426 if fmt in ("gguf", "gguf_scaled"):
427 return self._resolve_gguf_banks(banks, rep, int(opts["gate_up_type"]), int(opts["down_type"]), scaled=fmt == "gguf_scaled")
428 if fmt == "bf16":
429 gate_up = banks["gate_up"]
430 down = banks["down"]
431 if gate_up[rep].dtype != torch.bfloat16 or down[rep].dtype != torch.bfloat16:
432 raise NotImplementedError(
433 f"bf16 CPU MoE requires bf16 banks, got {gate_up[rep].dtype}/{down[rep].dtype}"
434 )
435 H = int(gate_up[rep].shape[2])
436 I = int(gate_up[rep].shape[1] // 2)
437 assert gate_up[rep].shape[1] == 2 * I
438 assert tuple(down[rep].shape[1:]) == (H, I), (down[rep].shape, H, I)
439 banks = self._tables(gate_up=gate_up, down=down)
440 return banks, {}, (H, I)
441
442 if fmt == "mxfp4_triton":
443 return self._resolve_mxfp4_banks(banks, rep)
444
445 if fmt == "ds_fp4":
446 return self._resolve_dsfp4_banks(banks, rep)
447
448 if fmt == "fp8_block":
449 return self._resolve_fp8_block_banks(banks, rep)
450
451 # nvfp4: packed e2m1 (2/byte) + fp8-e4m3 per-16 block scales + fp16 row globals.
452 gup, gus, gug = banks["gate_up"], banks["gate_up_scale"], banks["gate_up_global"]
453 dnp, dns, dng = banks["down"], banks["down_scale"], banks["down_global"]
454 assert gup[rep].dtype == torch.uint8 and dnp[rep].dtype == torch.uint8, (gup[rep].dtype, dnp[rep].dtype)
455 assert gus[rep].element_size() == 1 and dns[rep].element_size() == 1, "block scales must be 1 byte"
456 assert gug[rep].dtype == torch.float16 and dng[rep].dtype == torch.float16, (gug[rep].dtype, dng[rep].dtype)
457 I = int(gup[rep].shape[1] // 2)
458 H = int(gup[rep].shape[2] * 2)
459 assert gup[rep].shape[1] == 2 * I
460 assert H % 16 == 0 and I % 16 == 0, (H, I)
461 assert tuple(dnp[rep].shape[1:]) == (H, I // 2), (dnp[rep].shape, H, I)
462 assert tuple(gus[rep].shape[1:]) == (2 * I, H // 16), (gus[rep].shape, I, H)
463 assert tuple(dns[rep].shape[1:]) == (H, I // 16), (dns[rep].shape, H, I)
464 assert tuple(gug[rep].shape[1:]) == (2 * I,) and tuple(dng[rep].shape[1:]) == (H,)
465 banks = self._tables(gate_up=gup, down=dnp, gate_up_scale=gus, gate_up_global=gug, down_scale=dns, down_global=dng)
466 return banks, {}, (H, I)

468 def _resolve_gguf_banks(self, banks: dict, rep: int, gate_up_type: int, down_type: int, scaled: bool = False) -> tuple[dict, dict, tuple[int, int]]:
469 """GGUF rows read in place: gate_up [E, 2I, row_bytes(H)] (gate rows, then up rows) and down [E, H, row_bytes(I)], one ggml type each; float rows keep their dtype.
470
471 ``scaled`` adds global [E, 4] fp32, each expert's (gate, up, down, 1) NVFP4 weight_scale_2."""
472 from freetoken.layers.quantization.ggml_types import float_dtype, row_bytes, type_name
473
474 gate_up, down = banks["gate_up"], banks["down"]
475 H, I = int(down[rep].shape[1]), int(gate_up[rep].shape[1] // 2)
476 for role, t, ggml_type, k in (("gate_up", gate_up[rep], gate_up_type, H), ("down", down[rep], down_type, I)):
477 dtype = float_dtype(ggml_type) if float_dtype(ggml_type) is not None else torch.uint8
478 if t.dtype != dtype or t.shape[-1] * t.element_size() != row_bytes(ggml_type, k):
479 raise ValueError(f"MoE layer {rep} {role} bank {tuple(t.shape)} {t.dtype} is not {type_name(ggml_type)} rows of {k} values")
480 types = {"gate_up_type": gate_up_type, "down_type": down_type}
481 if not scaled:
482 return self._tables(gate_up=gate_up, down=down), types, (H, I)
483 scales = banks["global"]
484 if scales[rep].dtype != torch.float32 or tuple(scales[rep].shape) != (gate_up[rep].shape[0], 4) or not scales[rep].is_contiguous():
485 raise ValueError(f"MoE layer {rep} global bank {tuple(scales[rep].shape)} {scales[rep].dtype} is not contiguous [experts, 4] fp32")
486 return self._tables(gate_up=gate_up, down=down, **{"global": scales}), types, (H, I)

488 def _resolve_mxfp4_banks(self, banks: dict, rep: int = 0) -> tuple[dict, dict, tuple[int, int]]:
489 """gpt-oss mxfp4 ``mxfp4_triton`` schema: transposed split-K blocks/scales
490 (N innermost) + per-output-row biases. The C++ kernel streams K and
491 accumulates a contiguous N-block, so the GPU-tiled layout is read in place
492 (no repack, no extra host memory). Block scales are e8m0 (1 byte / 32 K)."""
493 gub, gus, gob = banks["gate_up"], banks["gate_up_scale"], banks["gate_up_bias"]
494 dnb, dns, dob = banks["down"], banks["down_scale"], banks["down_bias"]
495 assert gub[rep].dtype == torch.uint8 and dnb[rep].dtype == torch.uint8, (gub[rep].dtype, dnb[rep].dtype)
496 assert gus[rep].dtype == torch.uint8 and dns[rep].dtype == torch.uint8, (gus[rep].dtype, dns[rep].dtype)
497 assert gob[rep].dtype == torch.bfloat16 and dob[rep].dtype == torch.bfloat16, (gob[rep].dtype, dob[rep].dtype)
498 # gate_up_blocks [E, H//2, 2I]; down_blocks [E, I//2, H]
499 H = int(gub[rep].shape[1] * 2)
500 I = int(gub[rep].shape[2] // 2)
501 assert gub[rep].shape[2] == 2 * I
502 assert H % 32 == 0 and I % 32 == 0, (H, I)
503 assert tuple(dnb[rep].shape[1:]) == (I // 2, H), (dnb[rep].shape, H, I)
504 assert tuple(gus[rep].shape[1:]) == (H // 32, 2 * I), (gus[rep].shape, H, I)
505 assert tuple(dns[rep].shape[1:]) == (I // 32, H), (dns[rep].shape, H, I)
506 assert tuple(gob[rep].shape[1:]) == (2 * I,) and tuple(dob[rep].shape[1:]) == (H,)
507 banks = self._tables(gate_up=gub, down=dnb, gate_up_scale=gus, down_scale=dns, gate_up_bias=gob, down_bias=dob)
508 return banks, {}, (H, I)

510 def _resolve_dsfp4_banks(self, banks: dict, rep: int = 0) -> tuple[dict, dict, tuple[int, int]]:
511 """DeepSeek-V4 ``ds_fp4`` schema: row-major e2m1 (2/byte) + e8m0 per-32 block
512 scales, no global, no bias. Layout matches nvfp4 (K contiguous per output row),
513 so the C++ GEMV reads it in place. The kernel additionally FP8-round-trips the
514 activations (the checkpoint's fp8 block) to match the DeepSeek W4A8 reference."""
515 gup, gus = banks["gate_up"], banks["gate_up_scale"]
516 dnp, dns = banks["down"], banks["down_scale"]
517 assert gup[rep].dtype == torch.uint8 and dnp[rep].dtype == torch.uint8, (gup[rep].dtype, dnp[rep].dtype)
518 assert gus[rep].element_size() == 1 and dns[rep].element_size() == 1, "block scales must be 1 byte"
519 I = int(gup[rep].shape[1] // 2)
520 H = int(gup[rep].shape[2] * 2)
521 assert gup[rep].shape[1] == 2 * I
522 assert H % 32 == 0 and I % 32 == 0, (H, I) # e8m0 scales per 32 along K
523 assert tuple(dnp[rep].shape[1:]) == (H, I // 2), (dnp[rep].shape, H, I)
524 assert tuple(gus[rep].shape[1:]) == (2 * I, H // 32), (gus[rep].shape, I, H)
525 assert tuple(dns[rep].shape[1:]) == (H, I // 32), (dns[rep].shape, H, I)
526 banks = self._tables(gate_up=gup, down=dnp, gate_up_scale=gus, down_scale=dns)
527 return banks, {}, (H, I)

529 def decode(
530 self,
531 layer_id: int,
532 hidden_states: torch.Tensor,
533 topk_weights: torch.Tensor,
534 topk_ids: torch.Tensor,
535 ) -> torch.Tensor:
536 """One MoE layer of decode on the CPU. Returns a GPU [bs, H] tensor.
537
538 All ops go on the current CUDA stream so the whole thing is captured into
539 the active CUDA graph (the two host nodes carry the data dependency on the
540 pinned buffers, which hold this step's real routing on replay)."""
541 pending = self.decode_submit(layer_id, hidden_states, topk_weights, topk_ids)
542 return self.decode_sync(pending)
543
544 def decode_submit(
545 self,
546 layer_id: int,
547 hidden_states: torch.Tensor,
548 topk_weights: torch.Tensor,
549 topk_ids: torch.Tensor,
550 ) -> tuple:
551 """Issue the D2H copies + the CPU-pool submit host node (or the publish kernel), then return without
552 waiting. Lets a caller (the hybrid backend) enqueue GPU work between this and
553 :meth:`decode_sync` so the CPU compute overlaps the GPU GEMM / PCIe fetch.
554
555 ``topk_ids`` may carry ``-1`` entries (the C++ kernel skips them), so the CPU
556 computes only the routes assigned to it. Returns an opaque handle to pass to
557 :meth:`decode_sync`. The output tensor is allocated here so it stays live (and
558 distinct from the interleaved GPU work) across the overlap window."""
559 bs = hidden_states.shape[0]
560 if bs > self.max_tokens:
561 raise ValueError(f"CPU MoE decode batch {bs} exceeds the executor's max_tokens {self.max_tokens}")
562 if not self._graph_capture_safe and torch.cuda.is_current_stream_capturing():
563 raise RuntimeError(
564 "CPU MoE decode cannot be captured into a CUDA graph here: the stream-memop "
565 "doorbell is off or failed its startup probe, and the host-function sync is not "
566 "replay-safe on this runtime; run with --cuda-graph-max-bs 0"
567 )
568
569 if self._gpu_prequant:
570 # DSV4: apply the reference FP8 round-trip on the GPU (the same kernel the
571 # GPU W4A8 path uses -> bit-identical grid) so the CPU side reads
572 # pre-quantized activations and skips its serial scalar pass.
573 from freetoken.kernel.triton.dsv4.fp8_linear import act_quant_fp8_roundtrip
574
575 hidden_states = act_quant_fp8_roundtrip(hidden_states, block=self._act_block)
576
577 stream = torch.cuda.current_stream().cuda_stream
578 if self._kernel_io and hidden_states.dtype == torch.bfloat16:
579 x = hidden_states.contiguous()
580 ids = topk_ids.to(torch.int32).contiguous()
581 w = topk_weights.to(torch.float32).contiguous()
582 assert x.shape == (bs, self.H) and ids.shape == w.shape == (bs, self.top_k), (x.shape, ids.shape, w.shape)
583 self._ext.flag_publish(stream, x.data_ptr(), ids.data_ptr(), w.data_ptr(), layer_id, bs)
584 return (bs, torch.empty_like(x), True)
585
586 # D2H: ship this step's activations + routing to pinned host memory.
587 io = self._io
588 io["x"][:bs].copy_(hidden_states, non_blocking=True)
589 io["ids"][:bs].copy_(topk_ids.to(torch.int32), non_blocking=True)
590 io["w"][:bs].copy_(topk_weights.to(torch.float32), non_blocking=True)
591
592 out = torch.empty_like(hidden_states)
593 if self._flag_sync:
594 # Front-end memops: reset done, then ring the doorbell with (layer, bs).
595 # No kernel launched; no host-func round trip.
596 self._ext.flag_signal(stream, layer_id, bs)
597 else:
598 self._ext.submit_with_cuda_stream(stream, layer_id, bs)
599 return (bs, out, False)
600
601 def decode_sync(self, pending: tuple, add_to: torch.Tensor | None = None) -> torch.Tensor:
602 """Issue the CPU-pool sync + the H2D result copy for a prior :meth:`decode_submit`,
603 and return the GPU output tensor. With flag-sync the wait is a front-end stream
604 memop on done (set by the CPU coordinator); otherwise a cudaLaunchHostFunc.
605 With ``add_to``, return ``add_to`` + the CPU output; the kernel path adds in place into ``add_to``."""
606 bs, out, kernel_io = pending
607 stream = torch.cuda.current_stream().cuda_stream
608 if self._flag_sync:
609 # Front-end WAIT(done >= 1): blocks this stream's later nodes without
610 # occupying an SM, so GPU utilization stays truthful during the CPU window.
611 self._ext.flag_wait(stream)
612 else:
613 self._ext.sync_with_cuda_stream(stream)
614 if kernel_io:
615 if add_to is not None and add_to.dtype == torch.bfloat16 and add_to.is_contiguous() and add_to.shape == out.shape:
616 self._ext.flag_combine(stream, add_to.data_ptr(), bs, True)
617 return add_to
618 self._ext.flag_combine(stream, out.data_ptr(), bs, False)
619 return out if add_to is None else add_to + out
620 out.copy_(self._io["y"][:bs], non_blocking=True)
621 return out if add_to is None else add_to + out

623 def _watchdog_tick(self, suspect: dict) -> None:
624 """One watchdog sampling round (called every 2 s by ``_watchdog_main``).
625
626 The doorbell is only declared dead when THREE things hold across >=10 s: it is
627 still pending (ready set, done==0), it was already pending when first suspected, and
628 the coordinator has served NOTHING since (flag_served unchanged). The served-count
629 criterion kills the false-positive window: two point samples can land on
630 (independent, us-scale) pending windows under heavy external load, but a
631 coordinator that made progress in between is alive by definition. ``suspect``
632 holds (first_seen, served_at_first_sight) and persists across ticks."""
633 if not self._ext.flag_pending():
634 suspect.clear()
635 return
636 now = time.monotonic()
637 served = self._ext.flag_served()
638 if not suspect or suspect["served"] != served:
639 suspect.update(first_seen=now, served=served) # new suspect, or alive-but-loaded: rearm
640 return
641 if now - suspect["first_seen"] < 10.0:
642 return
643 logger.error(
644 "cpu-moe flag watchdog: doorbell unanswered for >10s with no coordinator "
645 "progress (wedged/dead); poisoning done and failing the next step"
646 )
647 self._unhealthy = True
648 self._ext.poison_flag() # after the mark: unblock the stream into a checked failure
649 suspect.clear()
650
651 def raise_if_unhealthy(self) -> None:
652 """Raise if the flag watchdog fired (a doorbell stayed unanswered because the
653 coordinator never responded). Called by the engine once per forward -- a single
654 flag read -- so a dead coordinator surfaces as a loud error on the next step
655 instead of silently shipping stale expert outputs."""
656 if self._unhealthy:
657 raise RuntimeError(
658 "CPU MoE flag-handshake watchdog fired: a decode step's doorbell was "
659 "never answered by the coordinator thread (its outputs cannot be "
660 "trusted). This indicates a wedged/killed coordinator; restart the "
661 "engine, or set FREETOKEN_CPU_MOE_FLAG_SYNC=0 to use the "
662 "cudaLaunchHostFunc sync."
663 )

666def _watchdog_main(executor_ref) -> None:
667 """Watchdog daemon body: weakref-deref per tick so the thread never keeps a dead
668 executor alive (see the start site in ``CpuMoeExecutor.__init__``)."""
669 suspect: dict = {}
670 while True:
671 time.sleep(2.0)
672 executor = executor_ref()
673 if executor is None or executor._watchdog_stop:
674 return
675 try:
676 executor._watchdog_tick(suspect)
677 finally:
678 del executor # drop the strong ref before the next sleep
