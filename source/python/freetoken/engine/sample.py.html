55@dataclass
56class Sampler:
57 device: torch.device
58 vocab_size: int

60 def prepare(self, batch: Batch) -> BatchSamplingArgs:
61 params = [r.sampling_params for r in batch.reqs]
62 is_greedy = [p.is_greedy for p in params]
63 penalties = []
64 for row, req in enumerate(batch.reqs):
65 p = req.sampling_params
66 if not (p.presence_penalty or p.frequency_penalty) or not req.can_decode:
67 continue
68 if req.output_token_counts is None:
69 req.output_token_counts = torch.zeros(
70 self.vocab_size, dtype=torch.int32, device=self.device
71 )
72 penalties.append(
73 (row, req.output_token_counts, p.presence_penalty, p.frequency_penalty)
74 )
75 if all(is_greedy):
76 return BatchSamplingArgs(temperatures=None, penalties=penalties)
77
78 MIN_P = MIN_T = 1e-6
79 # Greedy outputs are selected explicitly in sample(); use neutral sampling
80 # parameters for those rows instead of approximating argmax at low temperature.
81 ts = [1.0 if g else max(p.temperature, MIN_T) for p, g in zip(params, is_greedy)]
82 top_ks = [
83 p.top_k if not g and p.top_k >= 1 else self.vocab_size
84 for p, g in zip(params, is_greedy)
85 ]
86 top_ps = [
87 1.0 if g else min(max(p.top_p, MIN_P), 1.0)
88 for p, g in zip(params, is_greedy)
89 ]
90 temperatures = make_device_tensor(ts, torch.float32, self.device)
91 top_k, top_p = None, None
92 if any(k != self.vocab_size for k in top_ks):
93 top_k = make_device_tensor(top_ks, torch.int32, self.device)
94 if any(p < 1.0 for p in top_ps):
95 top_p = make_device_tensor(top_ps, torch.float32, self.device)
96 greedy_mask = (
97 make_device_tensor(is_greedy, torch.bool, self.device) if any(is_greedy) else None
98 )
99 return BatchSamplingArgs(
100 temperatures, top_k=top_k, top_p=top_p, greedy_mask=greedy_mask, penalties=penalties
101 )

103 @nvtx_annotate("Sampler")
104 def sample(self, logits: torch.Tensor, args: BatchSamplingArgs) -> torch.Tensor:
105 with torch.cuda.nvtx.range("Sampler"):
106 if args.penalties:
107 logits = logits.float().clone()
108 for row, counts, presence, frequency in args.penalties:
109 logits[row] -= frequency * counts + presence * (counts > 0)
110 if args.temperatures is None: # greedy sampling
111 tokens = torch.argmax(logits, dim=-1)
112 else:
113 tokens = sample_impl(logits.float(), args.temperatures, args.top_k, args.top_p)
114 if args.greedy_mask is not None:
115 # Mixed batches still run probability sampling for all rows, but
116 # greedy rows must follow argmax's deterministic tie-breaking.
117 greedy_tokens = torch.argmax(logits, dim=-1).to(tokens.dtype)
118 tokens = torch.where(args.greedy_mask, greedy_tokens, tokens)
119 # Update on the sampling stream: overlapped scheduling can prepare the next
120 # batch before the previous token reaches Req.input_ids on the CPU.
121 for row, counts, _, _ in args.penalties:
122 counts.scatter_add_(
123 0, tokens[row : row + 1].long(), counts.new_ones(1)
124 )
125 return tokens
