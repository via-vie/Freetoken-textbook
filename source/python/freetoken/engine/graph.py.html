25@dataclass
26class GraphCaptureBuffer:
27 input_ids: torch.Tensor
28 out_loc: torch.Tensor
29 positions: torch.Tensor
30 # [3, bs] t/h/w rope positions; allocated only for mrope models (else None).
31 mrope_positions: torch.Tensor | None
32 logits: torch.Tensor
33 table_idx: torch.Tensor # per-request slot id for GatedDeltaNet state gather/scatter
34 # Decode GDN query indptr = arange(bs+1); a constant per captured bs, filled once.
35 fla_cu_seqlens: torch.Tensor

37 @classmethod
38 def init(
39 cls, bs: int, vocab_size: int, device: torch.device, mrope: bool = False
40 ) -> GraphCaptureBuffer:
41 return GraphCaptureBuffer(
42 input_ids=torch.zeros(bs, dtype=torch.int32, device=device),
43 out_loc=torch.zeros(bs, dtype=torch.int32, device=device),
44 positions=torch.zeros(bs, dtype=torch.int32, device=device),
45 mrope_positions=(
46 torch.zeros(3, bs, dtype=torch.int32, device=device) if mrope else None
47 ),
48 logits=torch.empty(bs, vocab_size, dtype=torch.float32, device=device),
49 table_idx=torch.zeros(bs, dtype=torch.int32, device=device),
50 fla_cu_seqlens=torch.arange(bs + 1, dtype=torch.int32, device=device),
51 )

53 def set_batch(self, batch: Batch) -> None:
54 from freetoken.attention.linear import FLAMetadata
55
56 _slice = slice(batch.padded_size)
57 bs = batch.padded_size
58 batch.input_ids = self.input_ids[_slice]
59 batch.out_loc = self.out_loc[_slice]
60 batch.positions = self.positions[_slice]
61 if self.mrope_positions is not None:
62 batch.mrope_positions = self.mrope_positions[:, _slice]
63 batch.linear_table_idx = self.table_idx[_slice]
64 # Decode GDN metadata reads the persistent cu_seqlens (constant arange) and the
65 # persistent table_idx slot map, so the captured kernels see stable addresses.
66 batch.fla_metadata = FLAMetadata(
67 cu_seqlens=self.fla_cu_seqlens[: bs + 1], cache_indices=self.table_idx[_slice]
68 )

70 def copy_from(self, batch: Batch) -> None:
71 _slice = slice(batch.padded_size)
72 self.input_ids[_slice] = batch.input_ids
73 if batch.out_loc is not None:
74 self.out_loc[_slice] = batch.out_loc
75 self.positions[_slice] = batch.positions
76 if self.mrope_positions is not None:
77 self.mrope_positions[:, _slice] = batch.mrope_positions
78 if batch.linear_table_idx is not None:
79 self.table_idx[_slice] = batch.linear_table_idx

82def _determine_cuda_graph_bs(
83 cuda_graph_bs: List[int] | None,
84 cuda_graph_max_bs: int | None,
85 free_memory: int,
86) -> List[int]:
87 if cuda_graph_bs is not None:
88 return cuda_graph_bs
89
90 free_memory_gb = free_memory / (1 << 30)
91 if cuda_graph_max_bs is None:
92 if free_memory_gb > 80: # H200
93 cuda_graph_max_bs = 256
94 else:
95 cuda_graph_max_bs = 160
96
97 if cuda_graph_max_bs < 1:
98 return []
99
100 candidates = [1, 2, 4] + list(range(8, cuda_graph_max_bs + 1, 8))
101 return [bs for bs in candidates if bs <= cuda_graph_max_bs]

118def _upload_graph(graph: torch.cuda.CUDAGraph, stream: torch.cuda.Stream) -> None:
119 """Upload the graph at capture time.
120
121 Otherwise the first replay pushes every node behind the graph's own stream WAIT, and a large graph then blocks cudaGraphLaunch until the WAIT is released. PLE wait-sync releases it only after the launch returns, so that deadlocks.
122 """
123 upload = _cu_graph_upload()
124 if upload is None:
125 return
126 rc = upload(graph.raw_cuda_graph_exec(), stream.cuda_stream)
127 if rc != 0:
128 raise RuntimeError(f"cuGraphUpload failed with CUresult {rc}")
129 stream.synchronize()

132class GraphRunner:
133 def __init__(
134 self,
135 stream: torch.cuda.Stream,
136 device: torch.device,
137 model: BaseLLMModel,
138 attn_backend: BaseAttnBackend,
139 cuda_graph_bs: List[int] | None,
140 cuda_graph_max_bs: int | None,
141 free_memory: int,
142 max_seq_len: int,
143 vocab_size: int,
144 dummy_req: Req,
145 moe_offload_cache: OffloadMoeCache | None = None,
146 mrope: bool = False,
147 ) -> None:
148 cuda_graph_bs = _determine_cuda_graph_bs(
149 cuda_graph_bs=cuda_graph_bs,
150 cuda_graph_max_bs=cuda_graph_max_bs,
151 free_memory=free_memory,
152 )
153 self.attn_backend = attn_backend
154 self.max_graph_bs = max(cuda_graph_bs) if cuda_graph_bs else 0
155 self.graph_bs_list = sorted(cuda_graph_bs)
156 self.dummy_req = dummy_req
157 self.moe_offload_cache = moe_offload_cache
158 self.mrope = mrope
159 self.stream = stream
160 self.device = device
161 self._capture_graphs(max_seq_len, vocab_size, model)
162
163 def _reset_moe_offload_cache(self) -> None:
164 if self.moe_offload_cache is not None:
165 self.moe_offload_cache.reset()

167 def _capture_graphs(self, max_seq_len: int, vocab_size: int, model: BaseLLMModel):
168 # Mark the post-weights "warmup" phase for /health: this stretch (graph capture — or the
169 # remaining readiness work when graphs are disabled) moves no bytes, so without this the
170 # loader would sit at 100% (last byte bar) until the ready ack. total=0 ⇒ the desktop
171 # reads it as an indeterminate phase and animates the bar. Must precede the
172 # graphs-disabled early return so that config gets the phase too.
173 emit_progress("Capturing CUDA graphs / warming up", 0, 0)
174 self.graph_map: Dict[int, torch.cuda.CUDAGraph] = {}
175 if self.max_graph_bs == 0:
176 return logger.info_rank0("CUDA graph is disabled.")
177
178 self.attn_backend.init_capture_graph(max_seq_len=max_seq_len, bs_list=self.graph_bs_list)
179
180 torch.cuda.synchronize(self.device)
181 torch.cuda.empty_cache()
182 torch.cuda.reset_peak_memory_stats(self.device)
183
184 logger.info_rank0(f"Start capturing CUDA graphs with sizes: {self.graph_bs_list}")
185 free_memory = get_free_memory(self.device)
186 logger.info_rank0(f"Free GPU memory before capturing CUDA graphs: {mem_GB(free_memory)}")
187
188 self.buffer = GraphCaptureBuffer.init(
189 self.max_graph_bs, vocab_size, self.device, mrope=self.mrope
190 )
191 self._reset_moe_offload_cache()
192
193 pbar = tqdm(
194 sorted(self.graph_bs_list, reverse=True),
195 desc="Preparing for capturing CUDA graphs...",
196 unit="batch",
197 disable=not get_tp_info().is_primary(), # disable for non-primary ranks
198 )
199 pool = None
200 for bs in pbar:
201 free_memory = get_free_memory(self.device)
202 pbar.desc = f"Capturing graphs: bs = {bs:<3} | avail_mem = {mem_GB(free_memory)}"
203 pbar.refresh()
204 graph = torch.cuda.CUDAGraph()
205 batch = Batch(reqs=[self.dummy_req] * bs, phase="decode")
206 batch.padded_reqs = batch.reqs
207 self.attn_backend.prepare_for_capture(batch)
208 self.buffer.set_batch(batch)
209 # capture on the dummy linear-state slot so GatedDeltaNet gather/scatter
210 # touches scratch (real slot indices are written by copy_from on replay). Hybrid-
211 # radix decouples the GDN slot from table_idx -> use the GDN padding slot.
212 dummy_slot = (self.dummy_req.linear_slot_idx
213 if self.dummy_req.linear_slot_idx is not None
214 else self.dummy_req.table_idx)
215 self.buffer.table_idx[:bs].fill_(dummy_slot)
216 with get_global_ctx().forward_batch(batch):
217 self.buffer.logits[:bs] = model.forward()
218 # Keep the offload cache warmed for capture. Resetting here forces
219 # CUDA graph capture to replay cold-cache expert copies.
220 with torch.cuda.graph(graph, pool=pool, stream=self.stream):
221 self.buffer.logits[:bs] = model.forward()
222 _upload_graph(graph, self.stream)
223 self._reset_moe_offload_cache()
224 if pool is None:
225 pool = graph.pool() # reuse cuda graph handle to reduce memory
226 self.graph_map[bs] = graph
227
228 self._reset_moe_offload_cache()
229 free_memory = get_free_memory(self.device)
230 logger.info_rank0(f"Free GPU memory after capturing CUDA graphs: {mem_GB(free_memory)}")
231
232 def can_use_cuda_graph(self, batch: Batch) -> bool:
233 return batch.is_decode and batch.size <= self.max_graph_bs
234
235 def replay(self, batch: Batch) -> torch.Tensor:
236 assert self.can_use_cuda_graph(batch)
237 self.buffer.copy_from(batch)
238 g = self.graph_map[batch.padded_size]
239 self.attn_backend.prepare_for_replay(batch)
240 g.replay()
241 return self.buffer.logits[: batch.size]
242
243 def pad_batch(self, batch: Batch) -> None:
244 padded_size = ( # choose the first available batch size
245 next(bs for bs in self.graph_bs_list if bs >= batch.size)
246 if self.can_use_cuda_graph(batch)
247 else batch.size
248 )
249 batch.padded_reqs = batch.reqs + [self.dummy_req] * (padded_size - batch.size)

251 # NOTE: This must be called before freeing NCCL resources to prevent program hang
252 def destroy_cuda_graphs(self) -> None:
253 # Drop the CUDAGraph objects (and the shared mempool they hold) AND the static
254 # GraphCaptureBuffer tensors ([max_bs, vocab] logits + input/out_loc/positions/...).
255 # Dropping the references is the load-bearing step; without it a runtime rebuild's
256 # free-before-alloc cannot reclaim this GPU memory. empty_cache() is left to the
257 # caller / next capture (GraphRunner._capture_graphs already runs it).
258 self.graph_map = {}
259 self.buffer = None
260 gc.collect()
