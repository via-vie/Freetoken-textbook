17def layer_slot_bytes(sources: dict[str, "list[torch.Tensor]"]) -> list[int]:
18 """Bytes one expert slot of each layer occupies on GPU: its row bytes summed over all banks.
19
20 Each bank source is per-layer ``[num_experts, *row_shape]`` tensors and is
21 already TP-sharded upstream, so the per-row byte count is the per-rank slot
22 size.
23 """
24 # marlin/b12x gate_up/down alpha scales are fixed [L*E] residency (do not scale
25 # with cache_size), so they are intentionally excluded from the per-slot growth term.
26 # tensor[0].numel() is the per-row element count (one expert slot); see the matching
27 # slot-byte idiom in kvcache/linear_state_pool.py and kvcache/dsv4/v4_pool.py.
28 num_layers = len(next(iter(sources.values())))
29 return [sum(t[l][0].numel() * t[l].element_size() for t in sources.values()) for l in range(num_layers)]

32def expert_bytes_per_slot(sources: dict[str, "list[torch.Tensor]"]) -> int:
33 """Bytes one expert slot occupies on GPU, rounded up to the mean over layers when their bank layouts differ."""
34 per_layer = layer_slot_bytes(sources)
35 return -(-sum(per_layer) // len(per_layer))

38def net_cache_budget_bytes(
39 memory_ratio: float, baseline_free: int, weights_bytes: int, fixed_cache_size: int
40) -> int:
41 """Net GPU bytes available for the MoE + KV pools: ``memory_ratio`` of the pre-model
42 baseline minus weights and fixed (non-paged) cache. The ``(1-memory_ratio)`` remainder
43 is the CUDA-graph/activation headroom. Single source of truth for startup auto-sizing
44 and the runtime-rebuild fit check."""
45 return int(memory_ratio * baseline_free) - weights_bytes - fixed_cache_size

48def required_bytes(
49 moe_cache_size: int, num_pages: int, per_expert_bytes: int, cache_per_page: int
50) -> int:
51 """GPU bytes a ``(moe_cache_size, num_pages)`` geometry occupies (MoE slots + KV pages)."""
52 return moe_cache_size * per_expert_bytes + num_pages * cache_per_page

55def plan_cache_budget(
56 budget_bytes: int,
57 per_expert_bytes: int,
58 cache_per_page: int,
59 num_experts: int,
60 total_experts: int,
61 prefill_overlap: bool,
62 kv_reserve_pages: int,
63 max_slots: int,
64) -> tuple[int, int, bool]:
65 """Split ``budget_bytes`` MoE-first into (moe_cache_size, num_pages, prefill_overlap).
66
67 ``budget_bytes`` is the net pool for MoE cache + KV cache (caller already subtracted
68 weights + fixed_cache_size; the (1-memory_ratio) remainder is the graph headroom).
69 Experts greedily fill the budget after reserving ``kv_reserve_pages`` for KV, clamped
70 to ``[floor, min(total_experts, max_slots)]`` (floor is ``2*num_experts`` when prefill
71 overlap is feasible else ``num_experts``); KV pages take whatever remains.
72 """
73 assert per_expert_bytes > 0, "per_expert_bytes must be positive"
74 assert cache_per_page > 0, "cache_per_page must be positive (owned-KV models unsupported here)"
75
76 hi = min(total_experts, max_slots)
77 # Prefill overlap borrows two full expert-layer buffers, so it needs >= 2*num_experts
78 # slots; disable it (and lower the floor) if the cap cannot fit that.
79 overlap = prefill_overlap and hi >= 2 * num_experts
80 lo = 2 * num_experts if overlap else num_experts
81 assert hi >= lo, f"slot cap {hi} below the minimum {lo} slots"
82
83 kv_reserve_bytes = kv_reserve_pages * cache_per_page
84 # MoE-priority: reserve KV first, then experts greedily take the remaining budget.
85 raw = (budget_bytes - kv_reserve_bytes) // per_expert_bytes
86 moe_cache_size = max(lo, min(raw, hi))
87 # A tiny budget may have forced moe_cache_size below 2*num_experts even with overlap on.
88 overlap = overlap and moe_cache_size >= 2 * num_experts
89
90 remaining = budget_bytes - moe_cache_size * per_expert_bytes
91 num_pages = max(remaining // cache_per_page, kv_reserve_pages)
92 # A tiny budget can floor num_pages at kv_reserve_pages even when ``remaining`` is below
93 # the reserve (or negative), yielding a plan that exceeds budget_bytes. Reject here so
94 # --moe-cache-auto fails in arithmetic instead of OOMing in a later CUDA allocation.
95 total = moe_cache_size * per_expert_bytes + num_pages * cache_per_page
96 assert total <= budget_bytes, (
97 f"cache budget too small: minimum plan (moe={moe_cache_size} slots, "
98 f"kv={num_pages} pages) needs {total} B > budget {budget_bytes} B "
99 "(raise memory_ratio, lower kv_reserve_tokens, or free GPU memory)"
100 )
101 assert num_pages > 1, "not enough memory for KV cache after MoE allocation"
102 return moe_cache_size, num_pages, overlap

105def resolve_moe_cache_auto(
106 *,
107 baseline_free: int,
108 weights_bytes: int,
109 memory_ratio: float,
110 cache_per_page: int,
111 fixed_cache_size: int,
112 per_expert_bytes: int,
113 num_experts: int,
114 total_experts: int,
115 prefill_overlap: bool,
116 kv_reserve_tokens: int,
117 page_size: int,
118 max_slots: int | None = None,
119) -> tuple[int, int, bool]:
120 """Resolve --moe-cache-auto into (moe_cache_size, num_pages, prefill_overlap).
121
122 ``max_slots`` is the expert kernel's addressable slot limit; the plan never exceeds it.
123
124 Applies memory_ratio to the persisted pre-model baseline exactly once, then defers
125 the MoE-vs-KV split to plan_cache_budget. The (1-memory_ratio) remainder is the
126 CUDA-graph/activation headroom (not subtracted here).
127 """
128 budget_bytes = net_cache_budget_bytes(memory_ratio, baseline_free, weights_bytes, fixed_cache_size)
129 max_slots = total_experts if max_slots is None else min(max_slots, total_experts)
130 kv_reserve_pages = div_ceil(kv_reserve_tokens, page_size)
131 return plan_cache_budget(
132 budget_bytes=budget_bytes,
133 per_expert_bytes=per_expert_bytes,
134 cache_per_page=cache_per_page,
135 num_experts=num_experts,
136 total_experts=total_experts,
137 prefill_overlap=prefill_overlap,
138 kv_reserve_pages=kv_reserve_pages,
139 max_slots=max_slots,
140 )
