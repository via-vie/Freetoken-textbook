43def _clamp_decode_batch_to_pools(config: EngineConfig, pool_sizes: list[int]) -> None:
44 """Clamp the decode batch so every GPU-decoded layer's routed experts fit in its pool."""
45 num_experts = config.model_config.num_experts
46 small = [s for s in pool_sizes if s < num_experts]
47 if not small:
48 return
49 limit = min(small) // config.model_config.num_experts_per_tok
50 if config.max_running_req > limit:
51 logger.warning_rank0(
52 f"the smallest MoE slot pool has {min(small)} slots; lowering max_running_req "
53 f"{config.max_running_req} -> {limit} so a decode step fits in it"
54 )
55 object.__setattr__(config, "max_running_req", limit)
56 if config.cuda_graph_max_bs is None or config.cuda_graph_max_bs > limit:
57 object.__setattr__(config, "cuda_graph_max_bs", limit)
58 if config.cuda_graph_bs is not None:
59 object.__setattr__(config, "cuda_graph_bs", [bs for bs in config.cuda_graph_bs if bs <= limit])

98def _startup_kv_budget(memory_ratio: float, init_free_memory: int, new_free_memory: int) -> int:
99 """Bytes available to the KV pool at startup: ratio-scaled pre-load free memory minus
100 what the resident model consumed. Kept as a pure function so the composition with the
101 pool families' ``solve_num_pages`` stays CPU-testable."""
102 return int(memory_ratio * init_free_memory) - (init_free_memory - new_free_memory)

105def _page_table_width(max_seq_len: int, page_size: int) -> int:
106 """Column count for the page table. ``_write_page_table`` writes WHOLE trailing pages, so the
107 highest column touched is ``align_ceil(max_seq_len, page_size) - 1`` -- which the 32-alignment
108 alone does not cover once page_size > 32 (an unaligned --max-seq-len-override on DSV4's P=128
109 or trtllm's forced 64 would index past the row)."""
110 return align_ceil(align_ceil(max_seq_len, page_size), 32)

149def _resolve_auto_attention_backend(required: frozenset[AttnType]) -> str:
150 """First candidate (in per-type priority order) whose arch condition holds,
151 whose packages are installed, and whose every comma part serves ALL required
152 types. Reproduces the historical hardware tree for FULL-only models:
153 sm_100 -> trtllm, sm_90+sgl_kernel -> "fa,fi", flashinfer -> fi, else triton."""
154 candidates: list[tuple[str, bool]] = []
155 if AttnType.DSV4 in required:
156 candidates.append(("dsv4_sparse", True))
157 if AttnType.DSV41 in required:
158 candidates.append(("dsv41_sparse", True))
159 if required & {AttnType.MLA, AttnType.DSA}:
160 candidates.append(("dsa", True))
161 if AttnType.BSA in required:
162 candidates.append(("m3_sparse", True))
163 if AttnType.QSA in required:
164 candidates.append(("qsa_sparse", True))
165 if AttnType.SWA in required:
166 candidates.append(("triton", True))
167 if AttnType.FULL in required:
168 candidates += [
169 ("trtllm", is_sm100_family()),
170 ("fa,fi", is_sm90_family()),
171 ("fi", True),
172 ("triton", True),
173 ]
174 for name, arch_ok in candidates:
175 if not arch_ok:
176 continue
177 if not _backend_parts_serve(name, required):
178 continue
179 if not _backend_requirements_met(name):
180 continue
181 return name
182 raise RuntimeError(
183 "No attention backend can serve attention types "
184 f"{sorted(t.value for t in required)} on this machine."
185 )

188def _validate_attention_backend_choice(config, override, required: frozenset[AttnType]) -> None:
189 """Config-time type x backend capability check for the resolved (or explicit)
190 backend string: every comma part must serve every required type and have its
191 packages/arch available. Replaces the per-model gates; in particular this is
192 where a DSV4 or MLA checkpoint rejects a generic backend before weights load,
193 and where a generic model rejects dsa/dsv4_sparse."""
194 from freetoken.attention import validate_attn_backend
195 from freetoken.kernel.backend import is_rocm
196
197 # Name membership first (ArgumentTypeError listing the supported names): the CLI already
198 # ran this, but the programmatic EngineConfig path reaches here unvalidated and would
199 # otherwise die on a bare KeyError from the info lookup below.
200 validate_attn_backend(config.attention_backend, allow_auto=False)
201
202 model_config = config.model_config
203 backend_parts = [p.strip() for p in config.attention_backend.split(",")]
204 for part in backend_parts:
205 info = attention_backend_info(part)
206 missing = required - info.supported_types
207 if missing:
208 valid = [
209 name
210 for name in (
211 "fa", "fi", "trtllm", "triton", "dsa", "dsv4_sparse", "m3_sparse",
212 "qsa_sparse",
213 )
214 if required <= attention_backend_info(name).supported_types
215 ]
216 missing_names = "/".join(sorted(t.value for t in missing))
217 raise ValueError(
218 f"{getattr(model_config, 'model_type', 'model')} uses {missing_names} "
219 f"attention, which backend {part!r} does not support; valid backends: "
220 f"{', '.join(valid)} (or auto), got {config.attention_backend!r}."
221 )
222 if AttnType.SWA in required and not info.consumes_attn_spec:
223 # SWA models drive window/sinks/sm_scale through the per-call AttentionSpec;
224 # a backend that drops it would attend with the wrong window silently.
225 raise ValueError(
226 f"backend {part!r} does not consume the per-call AttentionSpec that "
227 f"SWA models require, got {config.attention_backend!r}."
228 )
229
230 # An explicitly-selected backend may require a package that isn't installed. Auto
231 # never resolves to one of these when its package is missing, so this only fires for
232 # explicit --attention-backend choices.
233 for part in backend_parts:
234 info = attention_backend_info(part)
235 if info.requires_flashinfer and not _flashinfer_available():
236 if is_rocm():
237 raise RuntimeError(
238 f"Attention backend {part!r} uses CUDA-only FlashInfer kernels "
239 "unavailable on ROCm; use --attention-backend auto or triton."
240 )
241 raise RuntimeError(
242 f"Attention backend {config.attention_backend!r} requires flashinfer, which is "
243 "not installed. Install it with `pip install 'freetoken[fi]'` (or "
244 "'freetoken[accel]'), or use --attention-backend triton."
245 )
246 if info.requires_sgl_kernel and not _sgl_flash_attn_available():
247 if is_rocm():
248 raise RuntimeError(
249 f"Attention backend {part!r} uses CUDA-only sgl_kernel kernels "
250 "unavailable on ROCm; use --attention-backend auto or triton."
251 )
252 raise RuntimeError(
253 f"Attention backend {config.attention_backend!r} requires sgl_kernel, which is "
254 "not installed. Install it with `pip install 'freetoken[sgl]'` (or "
255 "'freetoken[accel]'), or use --attention-backend triton."
256 )
257 if info.requires_sm100 and not is_sm100_family():
258 raise RuntimeError(
259 f"Attention backend {config.attention_backend!r} requires a compute capability "
260 "10.x GPU: flashinfer's trtllm-gen kernels ship sm_100a/103a cubins only. "
261 "Use --attention-backend fi (or triton) instead."
262 )
263
264 if required & {AttnType.MLA, AttnType.DSA}:
265 # Plain MLA/DSA runs on page_size 1; the kpool indexer layout needs 64.
266 _kpool_ratio = max(
267 (s.index_ratio for s in model_config.kv_cache_group_specs() if s.mla),
268 default=1,
269 )
270 want_page = 64 if _kpool_ratio > 1 else 1
271 if config.page_size != want_page:
272 logger.warning_rank0(
273 f"Page size {config.page_size} is auto-adjusted to {want_page} "
274 f"for latent-KV attention."
275 )
276 override("page_size", want_page)
277
278 for part in backend_parts:
279 info = attention_backend_info(part)
280 if info.page_sizes is not None and config.page_size not in info.page_sizes:
281 override("page_size", info.page_sizes[-1])
282 logger.warning_rank0(
283 f"Page size is overridden to {info.page_sizes[-1]} for the {part} backend"
284 )

287def _make_dummy_weight_state_dict(
288 model_state: Dict[str, torch.Tensor],
289 *,
290 device: torch.device,
291) -> Dict[str, torch.Tensor]:
292 state_dict: Dict[str, torch.Tensor] = {}
293 fp8_dtypes = (torch.float8_e4m3fn, torch.float8_e5m2)
294 for key, param in model_state.items():
295 if param.dtype in fp8_dtypes:
296 # torch.randn is not implemented for fp8; fill via a uint8 view with small
297 # codes (avoid NaN/inf fp8 encodings). Lets dummy-weight startup work for
298 # block-fp8 models (the dense fp8 linears are fp8 regardless of moe_strategy).
299 t = torch.empty(param.shape, dtype=param.dtype, device=device)
300 t.view(torch.uint8).random_(0, 16)
301 state_dict[key] = t
302 elif param.dtype.is_floating_point or param.dtype.is_complex:
303 state_dict[key] = torch.randn(param.shape, dtype=param.dtype, device=device)
304 elif param.dtype == torch.uint8 and key.endswith("weight_scale_inv"):
305 # MXFP8 e8m0 exponent codes: 127 encodes scale 1.0; zeros would collapse
306 # every scale to 2^-127 and zero the model. Scoped BY NAME: other uint8
307 # buffers are packed payloads whose bytes mean something else entirely
308 # (GGUF blocks embed fp16 scales -- 0x7F7F is fp16 NaN), so they
309 # keep the benign all-zeros fill below.
310 state_dict[key] = torch.full(param.shape, 127, dtype=param.dtype, device=device)
311 else:
312 state_dict[key] = torch.zeros(param.shape, dtype=param.dtype, device=device)
313 return state_dict

320def _is_resource_failure(exc: Exception) -> bool:
321 if isinstance(exc, (torch.OutOfMemoryError, MemoryError, PinFailed)):
322 return True
323 # an anonymous mmap that does not fit raises ENOMEM, not MemoryError
324 if isinstance(exc, OSError) and exc.errno == errno.ENOMEM:
325 return True
326 # torch has no type for a failed CPU allocation
327 return isinstance(exc, RuntimeError) and "DefaultCPUAllocator" in str(exc)

341def _materialize_loaded_weight_state_dict(

343 weights: Iterable[Tuple[str, torch.Tensor]],
344 *,
345 device: torch.device,
346) -> Dict[str, torch.Tensor]:
347 state_dict: Dict[str, torch.Tensor] = {}
348 for key, weight in weights:

366class Engine:
367 def __init__(self, config: EngineConfig):
368 assert not torch.cuda.is_initialized()
369 set_tp_info(rank=config.tp_info.rank, size=config.tp_info.size)
370 set_quant_backend(_adjust_ftw_quant_backend(config.model_path, QuantBackend.parse(config.quant_backend)))
371 _ensure_expandable_segments() # before the first CUDA allocation below
372
373 from freetoken.gpu_select import bind_assigned_gpu
374
375 self.device = bind_assigned_gpu(config.tp_info.rank)
376 _adjust_config(config)
377 torch.manual_seed(42)
378 self.stream = torch.cuda.Stream()
379 torch.cuda.set_stream(self.stream)
380 self.dtype = config.dtype
381 self.config = config # retained for runtime cache rebuild (rebuild_runtime_cache)
382 # KV pool family fixed at construction from the model config: its classmethods own the
383 # page-token geometry and cost arithmetic the engine needs BEFORE the pool exists
384 # (num_pages sizing, --moe-cache-auto); the instance owns rebuild/validation after.
385 self._pool_cls = resolve_pool_class(config.model_config)
386 self.ctx = Context(config.page_size)
387 set_global_ctx(self.ctx)
388
389 self.tp_cpu_group = self._init_communication(config)
390 free_min, free_max = self._sync_get_memory()
391 init_free_memory = free_max # startup KV sizing keeps cross-rank MAX (unchanged)
392 self._baseline_free = free_min # rebuild baseline: cross-rank MIN, deterministic across ranks
393 logger.info_rank0(f"Free memory before loading model: {mem_GB(init_free_memory)}")
394
395 # ======================= Model initialization ========================
396 set_rope_device(self.device)
397 with torch.device("meta"), torch_dtype(config.dtype):
398 self.model = create_model(config.model_config)
399 self._load_weights(config)
400 if config.active_encoders:
401 from freetoken.models.blocks import SupportsMultimodal
402
403 if not isinstance(self.model, SupportsMultimodal):
404 raise TypeError(
405 f"{type(self.model).__name__} has encoders registered but lacks the SupportsMultimodal hooks; "
406 "run with --text-model-only"
407 )
408 # before the residency snapshot, so streamed blocks are not charged as resident weights
409 self.model.place_encoder_weights(config.mm.encoder_weights)
410 post_weights_free = self._sync_get_memory()[0]
411 self._weights_bytes = self._baseline_free - post_weights_free
412 # Pool-budget baseline for the desktop cache sliders: free VRAM after the weights are
413 # resident but before ANY runtime cache pool (MoE expert cache below, KV pages, GDN
414 # state) is allocated. This is the stable "if all free VRAM went to one pool" budget —
415 # unlike a query-time mem_get_info it doesn't drift with allocator caching, CUDA
416 # graphs, or other processes. Cross-rank MIN, deterministic across ranks.
417 self._post_weights_free = post_weights_free
418 self.moe_offload_cache = None
419 self.cpu_moe_executor = None
420 # Host-side auxiliary stores (qwen4_exp's pinned PLE table): after the weights so a
421 # load failure is not masked, before the MoE offload cache so the bank residency
422 # planning sees the pin quota the table already spent.
423 self._host_tables_bytes = 0
424 if hasattr(self.model, "load_host_tables"):
425 with _weight_load_context():
426 self._host_tables_bytes = int(self.model.load_host_tables(config) or 0)
427 if is_offload_moe_strategy(config.moe_strategy):
428 self._init_offload_moe_cache(config)
429 if hasattr(self.model, "prepare_for_runtime"):
430 self.model.prepare_for_runtime()
431 self.encoder_cache = None
432 self.mm_processor = None
433 if config.active_encoders:
434 from freetoken.mm.encoder_cache import EncoderCache
435 from freetoken.mm.processor import get_mm_processor
436
437 self.mm_processor = get_mm_processor(config.model_path, config.mm)
438 self.encoder_cache = EncoderCache(storage=config.mm.embed_cache_device)
439 logger.info_rank0(
440 f"Multimodal enabled: {type(self.mm_processor).__name__}, encoders "
441 f"{[e.kind for e in config.active_encoders]} on {config.mm.encoder_weights}, serving {sorted(config.served_modalities)}"
442 )
443 self._warmup_encoders()
444 elif any(getattr(config.hf_config, key, None) is not None for key in ENCODER_SECTIONS):
445 logger.info_rank0(
446 "Multimodal disabled: --text-model-only"
447 if config.mm.text_model_only
448 else "Multimodal disabled: --mm-disable"
449 if config.mm.disabled_encoders
450 else "Multimodal disabled: no encoder registered for this architecture"
451 )
452
453 # ======================= KV cache initialization ========================
454 new_free = self._sync_get_memory()[1]
455 # The engine measures the budget and settles the sibling GDN state pool's bytes
456 # off it; the KV pool family owns every geometry-specific formula behind the rest.
457 available_memory = _startup_kv_budget(config.memory_ratio, init_free_memory, new_free)
458 available_memory -= state_pool_bytes(config)
459 self.num_pages = self._pool_cls.solve_num_pages(config, available_memory)
460 num_tokens = self.num_pages * config.page_size
461 self.ctx.kv_cache = self.kv_cache = create_kv_pool(
462 config, self.num_pages, device=self.device, dtype=self.dtype
463 )
464
465 # ======================= Linear (GatedDeltaNet) state initialization ========================
466 linear_group = config.model_config.linear_attention_group()
467 if linear_group is not None:
468 from freetoken.kvcache.linear_state_pool import LinearStatePool
469
470 self.linear_state_pool = LinearStatePool(
471 group=linear_group,
472 num_slots=_linear_pool_num_slots(config),
473 dtype=self.dtype,
474 device=self.device,
475 tp_size=config.tp_info.size,
476 slot_states=config.model_config.slot_states,
477 )
478 self.ctx.linear_state_pool = self.linear_state_pool
479 else:
480 self.linear_state_pool = None
481
482 # ======================= Page table initialization ========================
483 # NOTE: 1. aligned to 128 bytes; 2. store raw locations instead of pages
484 self.max_seq_len = min(config.max_seq_len, num_tokens)
485 aligned_max_seq_len = _page_table_width(self.max_seq_len, config.page_size)
486 self.ctx.page_table = self.page_table = torch.zeros( # + 1 for dummy request
487 (config.max_running_req + 1, aligned_max_seq_len),
488 dtype=torch.int32,
489 device=self.device,
490 )
491 # Pools routed by the shared table but deriving reads through their own mappings (DSV4)
492 # re-point here (and again on any table realloc). The graph-input snapshot that reads
493 # through them belongs to the attention backend, built later in init_capture_graph.
494 self.kv_cache.attach_page_table(self.page_table)
495
496 # ======================= Attention backend initialization ========================
497 self.ctx.attn_backend = self.attn_backend = create_attention_backend(
498 config.attention_backend, config.model_config
499 )
500
501 # ======================= Sampler initialization ========================
502 self.sampler = Sampler(self.device, config.model_config.vocab_size)
503
504 post_free_memory = self._sync_get_memory()[0]
505 logger.info_rank0(f"Free memory after initialization: {mem_GB(post_free_memory)}")
506
507 # ======================= Graph capture initialization ========================
508 self.dummy_req = Req(
509 input_ids=torch.tensor([0], dtype=torch.int32, device="cpu"),
510 table_idx=config.max_running_req,
511 cached_len=0,
512 output_len=1,
513 uid=-1,
514 sampling_params=None, # type: ignore
515 cache_handle=None, # type: ignore
516 )
517 # padded/dummy rows index the GDN padding slot (0) so gather/scatter hits scratch.
518 if self.linear_state_pool is not None:
519 self.dummy_req.linear_slot_idx = self.linear_state_pool.padding_slot
520 self.page_table[self.dummy_req.table_idx].fill_(num_tokens) # point to dummy page
521 self.graph_runner = GraphRunner(
522 stream=self.stream,
523 device=self.device,
524 model=self.model,
525 attn_backend=self.attn_backend,
526 cuda_graph_bs=config.cuda_graph_bs,
527 cuda_graph_max_bs=config.cuda_graph_max_bs,
528 free_memory=init_free_memory,
529 max_seq_len=aligned_max_seq_len,

623 """Encode the chunk's cache-miss items and gather its embedding rows into batch.mm_embeds, right before the LM forward."""
624 cache = self.encoder_cache

674 def _init_offload_moe_cache(self, config: EngineConfig) -> OffloadMoeCache:
675 from freetoken.moe.expert_banks import is_mixed
676
677 method = shared_offload_method(self.model)
678 num_moe_layers = config.model_config.num_moe_layers
679 # experts of several formats: one method per MoE layer, one slot pool per format
680 layer_methods = offload_layer_methods(self.model)
681 if not is_mixed(layer_methods):
682 layer_methods = None
683 elif config.moe_strategy == "cpu" or config.moe_cpu_layers:
684 raise ValueError(
685 "this model's experts come in several formats, which are served from per-format GPU slot pools; "
686 "they do not combine with --moe-strategy cpu or --moe-cpu-layers (use offload or hybrid)"
687 )
688 cpu_layer_ids = _resolve_cpu_layers(config, num_moe_layers, reserved=self._host_tables_bytes, method=method)
689 _check_pin_budget(config, reserved=self._host_tables_bytes, method=method, layer_methods=layer_methods)
690 # the kernels were picked for model_config.decode_target; --moe-cpu-layers auto may still find that every bank fits the pin budget
691 decode_target = config.model_config.decode_target
692 if decode_target == "cpu" and not cpu_layer_ids:
693 decode_target = "gpu"
694 # split residency: where pinning is quota-capped (_pin_budget_bytes), pin only the GPU layers' banks and mlock the CPU layers'
695 # uncapped hosts keep every bank pinned (CPU decode reads them the same; overlap prefill stays on)
696 # not applied to plain --moe-strategy cpu; all-locked under a cap = --moe-strategy offload --moe-cpu-layers 1.0
697 split_residency = (
698 bool(cpu_layer_ids)
699 and config.moe_strategy in ("offload", "hybrid")
700 and _pin_budget_bytes(self._host_tables_bytes) is not None
701 )
702 if config.moe_strategy == "cpu" and not split_residency:
703 # cpu mode pins every bank for the prefill double buffer; over the pin cap that dies in cudaHostRegister, so lock everything instead
704 from freetoken.moe.expert_banks import bank_bytes_estimate, ftw_bank_bytes
705
706 budget = _pin_budget_bytes(self._host_tables_bytes)
707 bank_bytes = None
708 if budget is not None:
709 bank_bytes = ftw_bank_bytes(config.model_path) or bank_bytes_estimate(config.model_config, method)
710 if bank_bytes and bank_bytes > budget:
711 split_residency = True
712 logger.info_rank0(
713 f"--moe-strategy cpu: banks {bank_bytes / 2**30:.2f} GiB exceed the "
714 f"pin budget; OS-locking all layers instead of pinning"
715 )
716 if split_residency and config.moe_prefill_overlap:
717 # locked (unregistered) layers cannot feed the async pinned H2D double buffer; their prefill is a synchronous pageable copy via materialize
718 logger.info_rank0(
719 "--moe-cpu-layers split residency: disabling MoE prefill overlap "
720 "(locked layers prefill via synchronous pageable copies)"
721 )
722 object.__setattr__(config, "moe_prefill_overlap", False)
723 # Fast path: an FTW checkpoint loads its repacked banks directly.
724 # Slow path: load_expert_banks auto-picks parallel vs serial baseline by
725 # expert-tensor granularity. Both pin-after-fill.
726 # --expert-load: serial/parallel force the read; auto (None) lets load_expert_banks
727 # pick (parallel for scattered experts, with a low-RAM fallback to serial).
728 expert_parallel = {"serial": False, "parallel": True}.get(config.expert_load, None)
729 requested_residency = None
730 if split_residency:
731 from freetoken.moe.host_banks import HostResidency
732
733 requested_residency = [
734 HostResidency.LOCKED.value if i in cpu_layer_ids
735 else HostResidency.PINNED.value
736 for i in range(config.model_config.num_moe_layers)
737 ]
738 # read before the expert load, so a bad profile fails in seconds
739 expert_counts = resolve_profile(
740 config.moe_cache_split, config.model_path,
741 config.model_config.num_moe_layers, config.model_config.num_experts,
742 )
743 try:
744 with _weight_load_context():
745 banks = load_expert_banks(
746 config.model_path,
747 config.model_config,
748 method=method,
749 device=self.device,
750 dummy=config.use_dummy_weight,
751 parallel=expert_parallel,
752 layer_residency=requested_residency,
753 layer_methods=layer_methods,
754 )
755 except PinFailed as exc:
756 raise RuntimeError(f"{exc}; {_pin_hint(self._host_tables_bytes)}") from exc
757 from freetoken.engine.cache_budget import expert_bytes_per_slot, layer_slot_bytes
758
759 self._moe_layer_slot_bytes = layer_slot_bytes(banks.sources)
760 self._moe_pool_groups = group_layers(config.moe_cache_scope, bank_layouts(banks.sources, banks.layer_tags), self._moe_layer_slot_bytes)
761 if len(self._moe_pool_groups) > 1 and cpu_layer_ids:
762 raise ValueError(
763 "several MoE slot pools (--moe-cache-scope per-layer, or experts in several quant formats) need every MoE "
764 "layer to decode from the GPU slot cache; they do not combine with --moe-strategy cpu or --moe-cpu-layers"
765 )
766 self._moe_expert_counts = expert_counts if len(self._moe_pool_groups) > 1 else None
767 if len(self._moe_pool_groups) == 1 and config.moe_cache_split != "equal":
768 logger.warning_rank0(
769 f"--moe-cache-split {config.moe_cache_split} is ignored: all MoE layers share one slot pool "
770 f"(--moe-cache-scope per-layer gives each layer its own)"
771 )
772 self._moe_slot_limit = _slot_limit(layer_methods) if layer_methods is not None else (method.slot_limit() if method is not None else None)

776 if config.moe_cache_auto:

780 if config.num_page_override is None:
781 # Honor the plan's KV half too: MoE slots and KV pages were solved
782 # against ONE budget (ratio x baseline - weights), so both must come
783 # from it. Re-solving pages later from a fresh free-memory reading
784 # double-counts everything allocated since the weights measurement
785 # (this expert cache, the CPU-executor GPU buffers, allocator
786 # slack) and goes negative whenever the expert fill is exact --

791 f"num_pages={pages} (prefill_overlap={overlap})"
792 )

796 # per_slot is a mean over layers, so with pools of different slot sizes the split can exceed the budget: shrink until it fits

798 size -= 1

801 object.__setattr__(config, "moe_prefill_overlap", overlap)

803 f"--moe-cache-mem {mem_GB(config.moe_cache_mem)} resolved moe_cache_size={size} "

805 )

809 logger.info_rank0(
810 f"MoE slot pools: {config.moe_cache_scope}, {len(pools)} pools of {min(pool_sizes)}-{max(pool_sizes)} "
811 f"slots ({config.moe_cache_split})"
812 )
813 if decode_target == "gpu":

816 if layer_methods is not None:

819 max_slots = self._moe_slot_limit
820 elif method is not None:

829 f"binds {method.kind} / {method.kernel.name}; reconvert or select that kernel"
830 )
831 # GGUF layers keep their own ggml types, so the cache reads each layer's layout off its banks
832 layout = None if method.kind is QuantKind.GGUF else method.layout()
833 max_slots = method.slot_limit()
834 cache = OffloadMoeCache(
835 # Models with leading dense layers (GLM-4) only have experts on the MoE
836 # layers; num_moe_layers == num_layers when first_k_dense_replace == 0.
837 num_layers=config.model_config.num_moe_layers,
838 num_experts=config.model_config.num_experts,
839 cache_size=config.moe_cache_size,
840 device=self.device,
841 cache_policy=config.moe_cache_policy,
842 prefill_overlap=config.moe_prefill_overlap,

852 )

854 cache.cpu_layer_ids = cpu_layer_ids
855 cache.set_bank_sources(banks.sources, layer_residency=banks.layer_residency, layer_tags=banks.layer_tags, row_tail_bytes=banks.row_tail_bytes)

880 def _resolve_hybrid_fetch(self, config: EngineConfig, cache) -> None:
881 """Resolve --moe-hybrid-max-fetch -1 (auto) into a bandwidth-matched fetch fraction.
882
883 Perfect fetch/compute overlap wants fetched : cpu-computed misses = pcie_bw :
884 (cpu_bw - pcie_bw), i.e. fetching a pcie_bw / cpu_bw fraction of each decode
885 step's misses -- both sides then finish together instead of one idling. The
886 achieved bandwidths come from the cached `ft bench bw` profile, from the row of each
887 MoE layer's kind (format, options, expert shape, top_k, CPU worker threads); a layer
888 without a row keeps the fixed cap of 1.
889 """
890 if config.moe_hybrid_max_fetch >= 0:
891 return # explicit fixed cap
892 from freetoken.moe.bench_profile import fetch_fraction, layer_kind_key, load_layer_kinds
893 from freetoken.moe.cpu_executor import _format_label
894
895 cache.hybrid_max_fetch = 1
896 methods = offload_layer_methods(self.model)
897 if methods is None:
898 logger.warning_rank0("--moe-hybrid-max-fetch auto: the MoE layers bind no expert method to look up in the `ft bench bw` profile; using a fixed fetch cap of 1")
899 return
900 gpu_name, gpu_uuid = _profile_gpu(self.device.index)
901 rows = load_layer_kinds(gpu_name=gpu_name, gpu_uuid=gpu_uuid) or {}
902 executor = self.cpu_moe_executor
903 keys, labels = [], []
904 for m in methods:
905 fmt, opts = m.cpu_spec()
906 c = m.cfg
907 keys.append(layer_kind_key(fmt, opts, c.hidden, c.local_intermediate, c.top_k, executor.num_threads))
908 labels.append(f"{_format_label(fmt, tuple(sorted(opts.items())))} {c.hidden}x{c.local_intermediate}")
909 fractions = [fetch_fraction(rows.get(key)) for key in keys]
910 cache.hybrid_fetch_fraction = [f or 0.0 for f in fractions]
911 groups = collections.Counter(zip(labels, fractions))
912 fetched = [f"{f:.1%} on {n} {label} layers" for (label, f), n in groups.items() if f is not None]
913 if fetched:
914 logger.info_rank0(
915 "--moe-hybrid-max-fetch auto: each decode step fetches this share of a layer's expert misses over PCIe "
916 f"(benched PCIe/CPU bandwidth ratio) and computes the rest on the CPU: {'; '.join(fetched)}"
917 )
918 if missing := sorted({label for label, f in groups if f is None}):
919 capped = sum(n for (_, f), n in groups.items() if f is None)
920 logger.warning_rank0(
921 f"--moe-hybrid-max-fetch auto: no `ft bench bw` row for {capped} MoE layers ({', '.join(missing)}) at "
922 f"{executor.num_threads} CPU threads; using a fixed fetch cap of 1 there. `ft bench bw --model <checkpoint>` benches each layer kind"
923 )
924 benched_isas = {rows[key].get("cpu_moe_isa") for key, f in zip(keys, fractions) if f is not None} - {None}
925 if other := sorted(benched_isas - set(executor.isa.split(","))):
926 logger.warning_rank0(
927 f"--moe-hybrid-max-fetch auto: some `ft bench bw` rows were benched on CPU kernels {other}, but this executor runs "
928 f"{executor.isa}; using them anyway"
929 )

967 def _sync_get_memory(self) -> Tuple[int, int]:
968 """Get the min and max free memory across TP ranks."""
969 torch.cuda.synchronize(self.device)
970 torch.cuda.empty_cache()
971 torch.cuda.reset_peak_memory_stats(self.device)
972 free_memory = get_free_memory(self.device)
973 free_mem_tensor = torch.tensor([free_memory, -free_memory], device="cpu", dtype=torch.int64)
974 torch.distributed.all_reduce(
975 free_mem_tensor, op=torch.distributed.ReduceOp.MIN, group=self.tp_cpu_group
976 )
977 min_free_memory = int(free_mem_tensor[0].item())
978 max_free_memory = -int(free_mem_tensor[1].item())
979 if max_free_memory - min_free_memory > 2 * 1024 * 1024 * 1024:
980 logger.error(
981 f"Memory across TP ranks are imbalanced:"
982 f" min {mem_GB(min_free_memory)}, max {mem_GB(max_free_memory)}"
983 )
984 raise RuntimeError("Memory across TP ranks are imbalanced")
985
986 return min_free_memory, max_free_memory

988 def _target_moe_and_expert_bytes(self, moe_cache_size: int | None, pool_sizes: list[int] | None) -> tuple[int, int]:
989 cache = self.moe_offload_cache
990 if cache is None:
991 return moe_cache_size or 0, 0
992 if moe_cache_size is None:
993 target_moe = cache.cache_size
994 nbytes = sum(c.numel() * c.element_size() for _, c in cache.banks)
995 else:
996 target_moe = moe_cache_size
997 nbytes = self._moe_pool_bytes(pool_sizes)
998 # pools of several bank layouts have different slot sizes; round up so target * bytes covers them
999 return target_moe, -(-nbytes // target_moe)

1003 # the model's per-access pool property all keep pointing at THIS pool, which frees its old

1007 if self.kv_cache.needs_rebind_on_rebuild:
1008 self.model.mark_for_rebind()
1009 self.kv_cache.rebuild_from_config(config, num_pages, num_swa_pages=num_swa_pages)

1028 @torch.inference_mode()
1029 def rebuild_runtime_cache(
1030 self,
1031 *,
1032 moe_cache_size: int | None = None,
1033 num_pages: int | None = None,
1034 num_mamba_slots: int | None = None,
1035 num_swa_pages: int | None = None,
1036 ) -> None:
1037 """Idle-only in-place resize of the MoE slot cache, KV page pool, GDN (mamba) state pool,
1038 and/or the window pool (num_swa_pages: an absolute pinned window), followed by CUDA-graph
1039 re-capture. Does NOT reload weights or host expert banks. The caller (scheduler) must
1040 guarantee no in-flight prefill/decode.
1041 """
1042 config = self.config
1043 if (moe_cache_size is None and num_pages is None and num_mamba_slots is None
1044 and num_swa_pages is None):
1045 return
1046
1047 # 0a. Geometry prevalidation BEFORE any destructive free. An invalid target (moe
1048 # slots on a model with no offload cache, moe below num_experts / above the
1049 # marlin cap, non-positive pages, or too few GDN slots to run) must reject
1050 # recoverably with the old cache intact -- NOT after teardown, which would
1051 # leave the server unable to serve. These checks are model-agnostic.
1052 if moe_cache_size is not None:
1053 if self.moe_offload_cache is None:
1054 raise CacheRebuildRejected(
1055 "moe_cache_size requested but this model has no MoE offload cache"
1056 )
1057 try:
1058 # a rebuild does not lower max_running_req again, so with GPU decode every pool must hold a full batch; hybrid caps its fetches instead
1059 batch = config.max_running_req if self.moe_offload_cache.decode_target == "gpu" else 1
1060 moe_pool_sizes = split_slots(moe_cache_size, self._moe_pools(batch), self._moe_expert_counts)
1061 self.moe_offload_cache.validate_rebuild(moe_cache_size, moe_pool_sizes)
1062 except ValueError as e:
1063 raise CacheRebuildRejected(str(e)) from e
1064 if num_pages is not None and num_pages <= 0:
1065 raise CacheRebuildRejected(f"num_pages must be positive, got {num_pages}")
1066 if num_mamba_slots is not None:
1067 if self.linear_state_pool is None:
1068 raise CacheRebuildRejected(
1069 "num_mamba_slots requested but this model has no GDN state pool"
1070 )
1071 # num_mamba_slots is the USABLE slot count (what the user sets and the status bar
1072 # shows); the pool also reserves a padding sink (slot 0), so the physical pool is
1073 # num_mamba_slots + 1. _linear_pool_min_slots is the physical floor -> usable - 1.
1074 min_usable = _linear_pool_min_slots(config) - 1
1075 if num_mamba_slots < min_usable:
1076 raise CacheRebuildRejected(
1077 f"num_mamba_slots {num_mamba_slots} is below the minimum {min_usable} "
1078 f"(non-evictable working set for max_running_req={config.max_running_req}) "
1079 f"needed to run; admission would deadlock"
1080 )
1081 if num_swa_pages is not None:
1082 # An absolute window pin for the radix-SWA window pool (Gemma) or the DSV4 window tier;
1083 # meaningless for dense/MHA models and the naive SWA path (concurrency x window).
1084 if not _supports_swa_ratio(config):
1085 raise CacheRebuildRejected(
1086 "num_swa_pages requested but this model has no window pool "
1087 "(needs DSV4 or a sliding-window model with --cache-type radix)"
1088 )
1089 if num_swa_pages <= 0:
1090 raise CacheRebuildRejected(
1091 f"num_swa_pages must be positive, got {num_swa_pages}"
1092 )
1093
1094 # 0b. Pool-family budget fit-check BEFORE any destructive free: an unfit geometry
1095 # must reject (recoverable) so the old caches stay intact and serving continues,
1096 # rather than freeing and then OOMing into permanent failure. The engine supplies
1097 # the memory account; the pool answers whether its target geometry fits.
1098 target_moe, per_expert_bytes = self._target_moe_and_expert_bytes(
1099 moe_cache_size, moe_pool_sizes if moe_cache_size is not None else None
1100 )
1101 # Price the sibling GDN state pool at ITS target (physical slots = usable + padding
1102 # sink) and hand the bytes in -- the KV pool only budgets its own tiers.
1103 target_mamba = (
1104 num_mamba_slots + 1
1105 if num_mamba_slots is not None
1106 else (self.linear_state_pool.num_slots if self.linear_state_pool is not None else None)
1107 )
1108 self.kv_cache.validate_rebuild(
1109 config, num_pages=num_pages,
1110 num_swa_pages=num_swa_pages, target_moe=target_moe,
1111 per_expert_bytes=per_expert_bytes, baseline_free=self._baseline_free,
1112 weights_bytes=self._weights_bytes, current_num_pages=self.num_pages,
1113 extra_fixed_bytes=(
1114 state_pool_bytes(config, target_mamba) if target_mamba is not None else 0
1115 ),
1116 extra_note=(
1117 f", mamba={target_mamba - 1} slots" if target_mamba is not None else ""
1118 ),
1119 )
1120
1121 torch.cuda.synchronize(self.device)
1122 # Preserve the CUDA-graph batch-size set resolved at startup. The auto heuristic keys
1123 # off free memory, which is far smaller now that the caches are resident (post-cache
1124 # free << startup pre-load free), so re-deriving it here would silently drop large
1125 # batch sizes after the first rebuild. Reusing the already-resolved list keeps the
1126 # captured coverage identical (the fit-check above guarantees the graph headroom fits).
1127 prior_graph_bs = self.graph_runner.graph_bs_list
1128 # Point of no return for the scheduler's rollback logic: from here the live graphs and
1129 # pools start being freed. A failure BEFORE this flag flips leaves the engine serving
1130 # untouched (no rollback needed); after it, only a rebuild restores service.
1131 self.rebuild_teardown_started = True
1132 # 1. Tear down CUDA graphs + backend capture scratch (free-before-alloc).
1133 self.attn_backend.reset_capture()
1134 self.graph_runner.destroy_cuda_graphs()
1135 # 2. Resize caches in place (each frees its old GPU tensors before allocating).
1136 # Pin the new window first (validated above) so any KV-pool rebuild below sizes the window
1137 # to it (_dsv4_pool_sizes / _swa_paged_num_tokens read config.swa_num_pages_override).
1138 # frozen EngineConfig — mutate in place like the moe_cache_size path; `config.x = y` raises
1139 # FrozenInstanceError, which here aborts the rebuild after the CUDA graphs are gone (→ 503).
1140 if num_swa_pages is not None:
1141 object.__setattr__(config, "swa_num_pages_override", num_swa_pages)
1142 if moe_cache_size is not None:
1143 assert self.moe_offload_cache is not None, "no MoE offload cache to resize"
1144 self.moe_offload_cache.rebuild(moe_cache_size, moe_pool_sizes)
1145 if num_pages is not None:
1146 # sets self.num_pages (rebuilds KV + window)
1147 self._resize_kv_pool(config, num_pages, num_swa_pages)
1148 elif num_swa_pages is not None:
1149 # Window-only change: no page-count change, but re-derive the window pool at the new
1150 # pin against the CURRENT page count. This re-allocs the same-size full pool and
1151 # the resized window, both inside the pool's own rebuild_from_config.
1152 self._resize_kv_pool(config, self.num_pages, num_swa_pages)
1153 if num_mamba_slots is not None:
1154 # Reallocate the GDN state pool (frees old tensors first). Must sit between graph
1155 # teardown and re-capture so the recaptured graphs bind the new state tensors.
1156 # +1 for the reserved padding sink: num_mamba_slots is the usable count.
1157 self.linear_state_pool.rebuild(num_mamba_slots + 1)
1158 # 3. Refresh max_seq_len (+ generic page table) for the new token budget.
1159 self._refresh_seq_state(config)
1160 aligned_max_seq_len = _page_table_width(self.max_seq_len, config.page_size)
1161 # 4. Re-capture CUDA graphs against the new tensors (reset_capture above re-armed
1162 # the backend; _sync_get_memory empties the cache so freed memory is reclaimed).
1163 gc.collect()
1164 free_min = self._sync_get_memory()[0]
1165 self.graph_runner = GraphRunner(
1166 stream=self.stream,
1167 device=self.device,
1168 model=self.model,
1169 attn_backend=self.attn_backend,
1170 cuda_graph_bs=prior_graph_bs, # reuse the startup-resolved set (see above)
1171 cuda_graph_max_bs=config.cuda_graph_max_bs,
1172 free_memory=free_min,
1173 max_seq_len=aligned_max_seq_len,
1174 vocab_size=config.model_config.vocab_size,
1175 dummy_req=self.dummy_req,
1176 moe_offload_cache=self.moe_offload_cache,
1177 mrope=config.model_config.model_is_mrope,
1178 )

1180 def forward_batch(self, batch: Batch, args: BatchSamplingArgs) -> ForwardOutput:
1181 assert torch.cuda.current_stream() == self.stream
1182 if batch.mm_gather_plan:
1183 self._run_mm_encoder(batch)
1184 use_graph = self.graph_runner.can_use_cuda_graph(batch)
1185 with self.ctx.forward_batch(batch), self.model.forward_host_ctx(batch, use_graph):
1186 logits = self.graph_runner.replay(batch) if use_graph else self.model.forward()
1187 if self.cpu_moe_executor is not None:
1188 # One pinned read: surfaces a fired flag-handshake watchdog (dead coordinator
1189 # -> stale expert outputs) as a loud error instead of silent corruption.
1190 self.cpu_moe_executor.raise_if_unhealthy()
1191
1192 for req in batch.reqs:
1193 req.complete_one()
1194
1195 batch_logits = logits[: batch.size]
1196 next_tokens_gpu = self.sampler.sample(batch_logits, args).to(torch.int32)
1197 next_tokens_cpu = next_tokens_gpu.to("cpu", non_blocking=True)
1198 copy_done_event = torch.cuda.Event()
1199 copy_done_event.record(self.stream)
1200 return ForwardOutput(next_tokens_gpu, next_tokens_cpu, copy_done_event)

1202 @torch.inference_mode()
1203 def _warmup_prefill(self) -> None:
1204 """Compile the Triton prefill path before the first real request.
1205
1206 Decode CUDA graph capture warms the decode path, but the first prefill
1207 can still pay Triton/cublas setup costs. Use the dummy request row and
1208 restore it afterwards so padded decode graph replay keeps using the
1209 dedicated dummy KV slot.
1210 """
1211 if self.max_seq_len < 2:
1212 return
1213
1214 warmup_lens = [min(80, self.max_seq_len)]
1215 if self.max_seq_len >= 128:
1216 warmup_lens.append(128)
1217 warmup_lens = sorted({length for length in warmup_lens if length >= 2})
1218 if not warmup_lens:
1219 return
1220
1221 dummy_row = self.page_table[self.dummy_req.table_idx]
1222 dummy_slot = int(dummy_row[0].item())
1223 started = torch.cuda.Event(enable_timing=True)
1224 ended = torch.cuda.Event(enable_timing=True)
1225 started.record(self.stream)
1226 try:
1227 for length in warmup_lens:
1228 dummy_row[:length] = torch.arange(
1229 length, dtype=torch.int32, device=self.device
1230 )
1231 warm_req = Req(
1232 input_ids=torch.zeros(length, dtype=torch.int32, device="cpu"),
1233 table_idx=self.dummy_req.table_idx,
1234 cached_len=0,
1235 output_len=1,
1236 uid=-1,
1237 sampling_params=None, # type: ignore[arg-type]
1238 cache_handle=None, # type: ignore[arg-type]
1239 )
1240 batch = Batch(reqs=[warm_req], phase="prefill")
1241 batch.padded_reqs = batch.reqs
1242 batch.input_ids = torch.zeros(length, dtype=torch.int32, device=self.device)
1243 batch.positions = torch.arange(length, dtype=torch.int32, device=self.device)
1244 if self.config.model_config.model_is_mrope:
1245 batch.mrope_positions = (
1246 batch.positions.unsqueeze(0).expand(3, -1).contiguous()
1247 )
1248 batch.out_loc = dummy_row[:length]
1249 self.attn_backend.prepare_metadata(batch)
1250 with self.ctx.forward_batch(batch):
1251 self.model.forward()
1252 finally:
1253 dummy_row.fill_(dummy_slot)
1254 if self.moe_offload_cache is not None:
1255 self.moe_offload_cache.reset()
1256 ended.record(self.stream)
1257 torch.cuda.synchronize(self.device)
1258 logger.info_rank0(
1259 f"Prefill warmup complete for lengths {warmup_lens} "
1260 f"in {started.elapsed_time(ended) / 1000.0:.3f} s"
1261 )

1308def _ensure_expandable_segments() -> None:
1309 """Default the CUDA allocator to expandable segments.
1310
1311 The motivating case is the offload prefill, which repeatedly dequantizes
1312 variable-sized NVFP4 expert blocks to BF16 (a different size per layer as the

1314 allocator fragments badly -- reserved memory can balloon far past the actual peak

1323 """
1324 if os.environ.get("PYTORCH_ALLOC_CONF") or os.environ.get("PYTORCH_CUDA_ALLOC_CONF"):
1325 return
1326 try:
1327 torch.cuda.memory._set_allocator_settings("expandable_segments:True")
1328 except Exception as exc: # pragma: no cover - depends on torch build
1329 logger.info_rank0(f"Could not enable expandable_segments ({exc}); continuing")
1330 return
1331 logger.info_rank0("Enabled expandable_segments (override via PYTORCH_ALLOC_CONF)")

1447 raise ValueError(f"--moe-cpu-layers count {k} must be in [0, {num_moe_layers}]")

1449 # k == 0 yields an empty range, hence an empty set).

1510def _pin_budget_bytes(reserved: int = 0) -> int | None:
1511 """Bytes this process can still safely cudaHostRegister, or None when the platform does not cap pinning (plain Linux).

1514 if env := os.environ.get("FREETOKEN_PIN_BUDGET_GB"):
1515 cap = int(float(env) * 2**30)
1516 elif not hasattr(os, "uname") or "microsoft" not in os.uname().release.lower(): # WSL kernel tag
1517 return None
1518 else:
1519 cap = int(os.sysconf("SC_PHYS_PAGES") * os.sysconf("SC_PAGE_SIZE") * 0.4)

1551def _auto_cpu_layers(config: EngineConfig, num_moe_layers: int, *, reserved: int = 0, method=None) -> frozenset[int]:
1552 """Pick CPU (locked) MoE layers for ``--moe-cpu-layers auto``: none while the banks fit the pin budget.
1553
1554 Locks just enough head+tail layers: per-layer decode miss rates are U-shaped, so the ends are the cheapest to move off the slot cache."""
1555 bank_bytes = _bank_bytes(config, method)
1556 if not bank_bytes:
1557 return frozenset()
1558 budget = _pin_budget_bytes(reserved)
1559 if budget is None or bank_bytes <= budget:
1560 return frozenset()
1561 if not _cpu_moe_executor_viable(config.model_config):
1562 raise ValueError(
1563 f"--moe-cpu-layers auto: banks {bank_bytes / 2**30:.2f} GiB exceed the "
1564 f"pin budget {budget / 2**30:.2f} GiB, but the CPU MoE executor cannot "
1565 f"serve this model (see --moe-strategy cpu requirements)"
1566 )
1567 n = min(num_moe_layers, math.ceil(num_moe_layers * (1 - budget / bank_bytes)))
1568 head = (n + 1) // 2
1569 ids = frozenset(range(head)) | frozenset(range(num_moe_layers - (n - head), num_moe_layers))
1570 logger.info_rank0(
1571 f"--moe-cpu-layers auto: banks {bank_bytes / 2**30:.2f} GiB > pin budget "
1572 f"{budget / 2**30:.2f} GiB; locking {n} head+tail MoE layers for CPU decode "
1573 f"({sorted(ids)})"
1574 )
1575 return ids
